# Air production decisions - one per aircraft line the AI can open. The should_build_* decisions
# say whether a line is wanted; the should_open_*_line decisions at the end of this file add the
# industry band and the line exclusions and are read DIRECTLY by the ai_strategy blocks of
# WA_AI_PRODUCTION_DEFAULT_air.txt and by WA_aircraft_production_strategies (the ~2-day pulse of
# events/WA_AI_misc.txt, which only keeps the purge books now). Tech availability is
# WA_AI_PRODUCTION_has_worthwhile_* (WA_AI_PRODUCTION_air_tech.txt, GENERATED from
# tools/air_tech_registry.json); identity is WA_AI_CONFIG_AIRFORCE_*; a force cap is a deployed
# park size, never an industry reading.

########################## shared observations ##########################

# A country that must field its own air arm: a major, or a country with no major ally to lean on.
WA_AI_PRODUCTION_has_no_major_air_patron = {
	OR = {
		is_major = yes
		NOT = {
			any_allied_country = {
				is_major = yes
			}
		}
	}
}

# [land-fighter-park] Land fighter park below the cap that keeps a fighter line open (mass
# producers run double). Reads WA_AI_PRODUCTION_land_fighter_park, the sum of the small_fighter
# and small_fighter_multirole archetypes written at the top of WA_aircraft_production_strategies,
# and NOT `has_deployed_air_force_size type = fighter`: that role bucket also holds the
# interceptor and carrier-fighter archetypes, whose lines have their own bars.
# Assumes the pulse wrote the variable first - every reader is inside that same effect.
# [air-budget] every park cap below is a Schmitt pair: closes at the cap, reopens at the
# `_reopen` bar (90 %), the "was open" side read from the archetype's purge book. Without it a
# mature park sitting at its cap flips the line on every pulse.
WA_AI_PRODUCTION_fighter_line_was_open = {
	OR = {
		has_country_flag = WA_AI_PRODUCTION_AIR_open_small_fighter
		has_country_flag = WA_AI_PRODUCTION_AIR_open_small_fighter_multirole
	}
}

WA_AI_PRODUCTION_fighter_park_is_below_cap = {
	if = {
		limit = { WA_AI_CONFIG_AIRFORCE_is_mass_producer = yes }
		if = {
			limit = { WA_AI_PRODUCTION_fighter_line_was_open = yes }
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap_mass }
		}
		else = {
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap_mass_reopen }
		}
	}
	else = {
		if = {
			limit = { WA_AI_PRODUCTION_fighter_line_was_open = yes }
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap }
		}
		else = {
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap_reopen }
		}
	}
}

WA_AI_PRODUCTION_is_soviet_union_gone = {
	OR = {
		has_global_flag = soviet_union_defeated
		SOV = {
			OR = {
				has_capitulated = yes
				exists = no
			}
		}
	}
}

# [air-budget] wants_* is the want with that arm's park CAP taken out, and
# should_build_* = wants_* + the cap - the same answer as before for every reader. The split
# exists so a line closed by its cap alone can be told apart from one closed by a want (tech,
# patron, archetype, Battle of Britain, bomber bases): the first is MAINTAINED at a small factory
# floor so the park keeps modernising, the second is denied outright. See the MAINTENANCE section
# at the end of this file. Assumes every cap term of an arm lives in its *_park_is_below_cap
# trigger - a cap left inside a wants_* would make that arm unmaintainable, silently.

	WA_AI_PRODUCTION_has_worthwhile_fighter = yes

WA_AI_PRODUCTION_interceptor_park_is_below_cap = {
	if = {
		limit = {
			OR = {
				WA_AI_PRODUCTION_should_build_fighters = yes
				WA_AI_PRODUCTION_should_build_multiroles = yes
			}
		}
		if = {
			limit = { has_country_flag = WA_AI_PRODUCTION_AIR_open_small_fighter_interceptor }
			check_variable = { num_deployed_planes_with_type@small_fighter_interceptor_airframe < constant:wa_ai_production.air.interceptor_cap }
		}
		else = {
			check_variable = { num_deployed_planes_with_type@small_fighter_interceptor_airframe < constant:wa_ai_production.air.interceptor_cap_reopen }
		}
	}
	else = {
		# deliberately unpaired: with both land fighter lines shut the land park only creeps up at
		# the maintenance floor and never falls back through this bar, so it cannot flap
		check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap }
	}
}

# The defender's emergency fighter programme: every other air line waits while it runs.
WA_AI_PRODUCTION_should_build_bob_fighters = {
	WA_AI_CONFIG_AIRFORCE_is_battle_of_britain_defender = yes
	NOT = { country_exists = REN }
	NOT = { GER = { has_war_with = SOV } }
	has_global_flag = fall_of_france
	# an absent park reads 0 = "below cap" = programme ON, which closes every bomber line; the pulse
	# writes the park, so before its first pass (fresh load, human->AI switch) the answer is unknown
	has_variable = WA_AI_PRODUCTION_land_fighter_park
	# Schmitt pair the other way round (the programme CLOSES lines): engaged while the park is
	# under the cap once running, engages only under the reopen bar when not - read through the
	# flag the pulse sets from this very trigger
	if = {
		limit = { has_country_flag = WA_bob_production_flag }
		OR = {
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.bob_fighter_cap }

# [air-budget] Three states per line instead of two:
#   OPEN        want holds, park below its cap  - weight, category factor, archetype factor, floor 1
#   MAINTAINED  want holds, park AT its cap     - archetype factor and a 2 / 5 factory floor only:
#                                                 no unit_ratio weight, no category factor, no purge
#   CLOSED      the want itself is false        - the CANCEL block's -1000, purge fires
# The middle state exists for MODERNISATION: at zero factories a mature park flies its 1936
# airframes to the end of the campaign, because nothing ever replaces them. It also keeps the
# engine's production line alive, so a park that later falls back under its reopen bar ramps from
# the efficiency it had rather than from the engine base (MEASURED 9.00 on two campaign
# reopenings - documentation/WA_AI_AIR_PRODUCTION.md).
#
# should_open_*_line is UNTOUCHED and still drives the purge books, the type gates and the monthly
# weights, so the open path behaves exactly as before. The ai_strategy line blocks and their
# CANCEL complements read should_keep_*_line = open OR maintained.
#
# Two invariants keep the states disjoint inside one engine role:
#   - a line is maintained only while NO line of its type is open (should_fund_<type>_type = no),
#     so an open line never shares its role with a maintained one - this is what stops the land
#     CAS floor being served twice when the fighter cap closes the lowered line;
#   - inside the maintained set the open set's exclusion order is repeated on should_maintain_*,
#     so the three fighter lines cannot all claim the role at once.
# For a type with a single line the two reduce to the same term and only NOT should_open_* is
# written. Assumes should_fund_<type>_type stays the plain "any line of this type is open" gate;
# a term of its own there would break the first invariant silently.
#
# The medium bombers are the one deliberate exception: fast, strike and tactical are not exclusive
# when open either, so a country running all three archetypes maintains all three and its
# tactical_bomber floor is the sum. That is the intent - three airframes need more than one.

# [air-budget] The carrier roles keep their OWN maintenance floor, deliberately smaller than
# the land 2 / 5 and with no carrier-major term: 1 factory. The deck-derived ladder of
# WA_AI_PRODUCTION_DEFAULT_cv_plane.txt (1, +4 for ten decks) stays gated on the line being OPEN.
# A min-factories floor is need-blind and additive, so it goes on building after the engine's own
# deck demand has reached zero; the saturation brake was written to take that ladder away, and its
# bars sit far ABOVE the park caps for a big fleet (cv_fighter_cap 2000 vs a 3510 bar at 20-39
# hulls). Carrying 5 need-blind factories through that band is the over-production the brake
# exists to stop, so the maintained fleet gets one factory and the same brake on top.
# Fact is gone when: a cap and its role's saturation bar cross (bar below cap), which would make
# the brake the binding constraint again and this floor unreachable.

WA_AI_PRODUCTION_should_maintain_cv_fighter_type = {
	WA_AI_PRODUCTION_needs_cv_planes = yes
	WA_AI_PRODUCTION_should_maintain_cv_fighter_line = yes
}
