# Air production decisions - one per aircraft line the AI can open. Read by
# WA_aircraft_production_strategies (WA_production_strategy_effects.txt, ~2-day event in
# events/WA_AI_misc.txt), which sets the WORLD_ABLAZE_PRODUCTION_* flags the air strategies enable on.
# Tech availability is WA_AI_PRODUCTION_has_*_tech (WA_AI_PRODUCTION_air_tech.txt), which also
# owns the AI's "not worth a factory yet" rungs; identity is WA_AI_CONFIG_AIRFORCE_* and
# WA_AI_CONFIG_TECH_TREE_*; a force cap is a deployed park size, never an industry reading.

########################## shared observations ##########################

# A country that must field its own air arm: a major, or a country with no major ally to lean on.
WA_AI_PRODUCTION_has_no_major_air_patron = {
	OR = {
		is_major = yes
		NOT = {
			any_allied_country = {
				is_major = yes
			}
		}
	}
}

# [land-fighter-park] Land fighter park below the cap that keeps a fighter line open (mass
# producers run double). Reads WA_AI_PRODUCTION_land_fighter_park, the sum of the small_fighter
# and small_fighter_multirole archetypes written at the top of WA_aircraft_production_strategies,
# and NOT `has_deployed_air_force_size type = fighter`: that role bucket also holds the
# interceptor and carrier-fighter archetypes, whose lines have their own bars.
# Assumes the pulse wrote the variable first - every reader is inside that same effect.
WA_AI_PRODUCTION_fighter_park_is_below_cap = {
	if = {
		limit = { WA_AI_CONFIG_AIRFORCE_is_mass_producer = yes }
		check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap_mass }
	}
	else = {
		check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap }
	}
}

WA_AI_PRODUCTION_is_soviet_union_gone = {
	OR = {
		has_global_flag = soviet_union_defeated
		SOV = {
			OR = {
				has_capitulated = yes
				exists = no
			}
		}
	}
}

# The bomber arm has nowhere to fly from: a country that needs British bases, with the SOV gone
# while Britain is in its emergency fighter programme or evacuated - or sitting on homeland defence.
# [commonwealth-handoff] catastrophes only: a successful landing in France is never a reason to
# stop (see the [commonwealth-handoff] comment in WA_aircraft_production_strategies).
WA_AI_PRODUCTION_has_lost_its_bomber_bases = {
	WA_AI_CONFIG_AIRFORCE_needs_british_bases = yes
	OR = {
		has_idea = USA_homeland_defense
		AND = {
			WA_AI_PRODUCTION_is_soviet_union_gone = yes
			OR = {
				ENG = { has_country_flag = WA_bob_production_flag }
				has_global_flag = uk_evacuated
			}
		}
	}
}

	WA_AI_PRODUCTION_fighter_park_is_below_cap = yes

WA_AI_PRODUCTION_should_build_fighters_and_multiroles = {
	WA_AI_PRODUCTION_should_build_fighters = yes
	WA_AI_PRODUCTION_should_build_multiroles = yes
}

WA_AI_PRODUCTION_should_build_interceptors = {
	WA_AI_PRODUCTION_has_interceptor_tech = yes
	if = {
		limit = {
			OR = {
				WA_AI_PRODUCTION_should_build_fighters = yes
				WA_AI_PRODUCTION_should_build_multiroles = yes
			}
		}
		check_variable = { num_deployed_planes_with_type@small_fighter_interceptor_airframe < constant:wa_ai_production.air.interceptor_cap }
	}
	else = {
		check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.fighter_cap }
	}
}

WA_AI_PRODUCTION_should_build_heavy_fighters = {
	WA_AI_PRODUCTION_has_worthwhile_heavy_fighter = yes
	NOT = { WA_AI_CONFIG_TECH_TREE_is_italian = yes }
	NOT = { WA_AI_CONFIG_TECH_TREE_is_german = yes }
	NOT = { WA_AI_CONFIG_TECH_TREE_is_soviet = yes }
	check_variable = { num_deployed_planes_with_type@medium_fighter_airframe < constant:wa_ai_production.air.heavy_fighter_cap }
	OR = {
		AND = {
			has_war = yes
			NOT = {
				any_home_area_neighbor_country = {
					has_war_with = ROOT
				}
			}
		}
		has_war_with = ENG
		has_war_with = USA
		WA_AI_CONFIG_AIRFORCE_keeps_heavy_fighters_in_peace = yes
	}
	NOT = { WA_AI_PRODUCTION_has_lost_its_bomber_bases = yes }
	WA_AI_PRODUCTION_has_no_major_air_patron = yes
}

WA_AI_PRODUCTION_should_build_attackers = {

########################## close support ##########################

WA_AI_PRODUCTION_should_build_cas = {
	WA_AI_PRODUCTION_has_worthwhile_cas = yes
	OR = {
		WA_AI_CONFIG_AIRFORCE_cas_is_strong = yes
		WA_AI_PRODUCTION_has_multirole_tech = no
	}
	NOT = { WA_AI_CONFIG_AIRFORCE_is_carrier_strike_power = yes }
	NOT = { WA_AI_CONFIG_AIRFORCE_cas_is_weak = yes }
	# [land-cas-own-bar] Its own archetype park, not the engine `type = cas` role: that role also
	# holds cv_small_bomber_airframe, so a carrier close-support arm counted against the LAND bar.
	# Assumes small_bomber_airframe is the only land close-support archetype
	# (common/units/equipment/plane_airframes.txt).
	check_variable = { num_deployed_planes_with_type@small_bomber_airframe < constant:wa_ai_production.air.cas_cap }
	WA_AI_PRODUCTION_has_no_major_air_patron = yes
}

WA_AI_PRODUCTION_should_build_cv_cas = {
	WA_AI_CONFIG_AIRFORCE_is_carrier_strike_power = yes
	WA_AI_PRODUCTION_has_worthwhile_cv_cas = yes
	# [cv-cas-own-bar] Its own archetype park, not the engine `type = cas` role: that role also
	# holds the LAND cas archetype, so the shared 1600 bar always bound before this line's own
	# 2500 and left cv_cas_cap dead. Assumes cv_small_bomber_airframe is the only carrier
	# close-support archetype (common/units/equipment/plane_airframes.txt).
	check_variable = { num_deployed_planes_with_type@cv_small_bomber_airframe < constant:wa_ai_production.air.cv_cas_cap }
	WA_AI_PRODUCTION_has_no_major_air_patron = yes
}

# [medium-bomber-one-park] The single park the fast, strike and tactical lines share, because the
# engine gives their three archetypes one role (type = tactical_bomber). One bar for the three,
# flat for every country - see medium_bomber_cap.

	has_deployed_air_force_size = { size < constant:wa_ai_production.air.medium_bomber_cap type = tactical_bomber }

WA_AI_PRODUCTION_should_build_naval_bombers = {
	WA_AI_PRODUCTION_has_naval_bomber_tech = yes
	any_owned_state = {
		is_coastal = yes
	}
	OR = {
		WA_AI_CONFIG_AIRFORCE_is_maritime_air_power = yes
		AND = {
			OR = {
				has_war_with = JAP
				has_war_with = ENG
				has_war_with = USA
			}
			surrender_progress < constant:wa_ai_production.air.max_surrender
			alliance_strength_ratio > constant:wa_ai_production.air.min_alliance_strength
		}
	}
	if = {
		limit = { WA_AI_CONFIG_AIRFORCE_is_carrier_strike_power = yes }
		if = {
			limit = { WA_AI_PRODUCTION_has_late_cv_naval_bomber = yes }
			has_deployed_air_force_size = { size < constant:wa_ai_production.air.naval_cap_carrier_late type = naval_bomber }
		}
		else = {
			has_deployed_air_force_size = { size < constant:wa_ai_production.air.naval_cap_carrier_early type = naval_bomber }
		}
	}
	else = {
		has_deployed_air_force_size = { size < constant:wa_ai_production.air.naval_cap type = naval_bomber }
	}
	if = {
		limit = { WA_AI_CONFIG_AIRFORCE_drops_naval_air_in_eastern_war = yes }
		NOT = { has_war_with = SOV }
	}
	WA_AI_PRODUCTION_has_no_major_air_patron = yes
}

WA_AI_PRODUCTION_should_build_strategic_bombers = {
	WA_AI_PRODUCTION_has_worthwhile_strategic_bomber = yes
	WA_AI_CONFIG_AIRFORCE_builds_strategic_bombers = yes
	surrender_progress < constant:wa_ai_production.air.max_surrender
	OR = {
		num_of_military_factories > constant:wa_ai_production.industry.tier_supermajor_mils
		WA_AI_CONFIG_AIRFORCE_is_strategic_bomber_junior = yes
	}
	NOT = { WA_AI_PRODUCTION_has_lost_its_bomber_bases = yes }
	OR = {
		AND = {
			NOT = { WA_AI_CONFIG_AIRFORCE_is_mass_producer = yes }
			has_deployed_air_force_size = { size < constant:wa_ai_production.air.strategic_cap type = strategic_bomber }
		}
		AND = {
			WA_AI_CONFIG_AIRFORCE_is_mass_producer = yes
			has_deployed_air_force_size = { size < constant:wa_ai_production.air.strategic_cap_mass type = strategic_bomber }
		}
		AND = {
			WA_AI_CONFIG_AIRFORCE_has_delayed_strategic_arm = yes
			WA_AI_CONFIG_after_1944 = yes
			has_deployed_air_force_size = { size < constant:wa_ai_production.air.strategic_cap_late_surge type = strategic_bomber }
			NOT = { has_global_flag = soviet_union_defeated }
		}
	}
	OR = {
		WA_AI_CONFIG_AIRFORCE_is_strategic_bomber_junior = yes
		WA_AI_PRODUCTION_has_no_major_air_patron = yes
	}
}

# World observation in GER scope (any caller): the air battle over Britain is on. Read by the
# GER weekly raid switch in 100_wa_on_actions.txt.
WA_AI_PRODUCTION_is_battle_of_britain_active = {
	GER = {
		has_war_with = ENG
		NOT = { has_war_with = SOV }
		NOT = { has_country_flag = sealion_active }
		has_global_flag = fall_of_france
		WA_AI_CONFIG_before_1940_11 = yes
	}
}

# The defender's emergency fighter programme: every other air line waits while it runs.
WA_AI_PRODUCTION_should_build_bob_fighters = {
	WA_AI_CONFIG_AIRFORCE_is_battle_of_britain_defender = yes
	NOT = { country_exists = REN }
	NOT = { GER = { has_war_with = SOV } }
	has_global_flag = fall_of_france
	OR = {
		check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.bob_fighter_cap }
		AND = {
			WA_AI_PRODUCTION_is_soviet_union_gone = yes
			check_variable = { WA_AI_PRODUCTION_land_fighter_park < constant:wa_ai_production.air.bob_fighter_cap_sov_gone }
		}
	}
}
