# gate of USA_stockpile_more_for_lend_lease
WA_AI_PRODUCTION_USA_should_stockpile_for_lend_lease = {
		NOT = {
			OR = {
				has_global_flag = soviet_union_defeated
				has_idea = USA_homeland_defense
				SOV = {
					OR = {
						has_capitulated = yes
						exists = no
					}
				}
			}
		}
		NOT = {
			OR = {
				has_equipment = { support_equipment > constant:wa_ai_lend_lease.donor.usa_stockpile_ceiling }
				has_equipment = { infantry_equipment > constant:wa_ai_lend_lease.donor.usa_stockpile_ceiling }
			}
		}
		OR = {
			AND = {
				SOV = {
					has_war_with = GER
					NOT = { has_war_with = USA }
				}
			}
			AND = {
				ENG = {
					has_war_with = GER
					NOT = { has_war_with = USA }
				}
			}
		}
}
