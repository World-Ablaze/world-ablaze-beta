# WA Historical AI Difficulty: preserve the sequence of WW2 events/campaigns. This is NOT vanilla's
# Historical AI Focuses setup axis, whose trigger is `is_historical_focus_on`.
WA_AI_DIFFICULTY_is_historical = { # 0-2
	difficulty < 3
}

WA_AI_DIFFICULTY_is_competitive = { # 3-4
	difficulty > 2
}

WA_AI_CONFIG_cheats_enabled = { # 2 or 4
	OR = {
		WA_AI_DIFFICULTY_is_historical_hard = yes
		WA_AI_DIFFICULTY_is_competitive_hard = yes
	}
}

WA_AI_DIFFICULTY_is_historical_easy = { # 0
	difficulty < 1
}

	difficulty > 1
	difficulty < 3
}

	difficulty > 0
	difficulty < 2
}

WA_AI_DIFFICULTY_is_competitive_normal = { # 3
	difficulty > 2
	difficulty < 4
}

# [rk-no-divisions] Reichskommissariat archetype, keyed on the autonomy state so every creation
# path (GER decisions, ITA decisions, ENG dominion conversion, toolpack) is covered without a tag
# list; assumes the id in common/autonomous_states/reichskommissariat.txt stays
# autonomy_reichskommissariat. The reichsprotectorate autonomy (RCZ) deliberately does NOT pass:
# its spawned garrison force is a recorded design (events/WA_AI_GER.txt ger_armor.999) - RCZ only
# enters here if a decision later moves it to the reichskommissariat autonomy.
# ITS (Italian East Africa) is exempt by owner order 2026-08-27: its Ascari colonial divisions
# are a fed design (ITA sends it equipment and units, common/decisions/ITA.txt establish_ITS).
WA_AI_CONFIG_is_reichskommissariat = {
	has_autonomy_state = autonomy_reichskommissariat
	NOT = { original_tag = ITS }
}

# [reserve-quality] Big-industry countries whose land-equipment output is committed elsewhere
# (air/sea production): a reserve dump would field divisions no stock can equip, so their bank
# stays shut while their war is overseas-only. Assumes the tag's MIL count overstates its LAND
# output; if its production mix changes by design, remove it here.
WA_AI_CONFIG_is_reserve_materiel_limited = {
	original_tag = ENG
}

WA_AI_CONFIG_is_in_allies = {
	OR = {
		# [atlantic-naval] (2026-08-14): `is_in_faction_with = USA` was listed twice. Harmless
		# to the engine - an OR of a term with itself - but this trigger gates a
		# large share of the Allied AI, so a reader checking it should not have to
		# wonder whether the third line was meant to be a third country.
		is_in_faction_with = ENG
		is_in_faction_with = USA
	}
}

	WA_AI_CONFIG_is_major_country = yes
	OR = {
		WA_AI_DIFFICULTY_is_historical = no
		AND = {
			WA_AI_DIFFICULTY_is_historical = yes
			# Allow capital building if game goes past 1948
			# or before treaty is dissolved so dockyards are put on starting production
			OR = {
				date > 1948.1.1

					target = { ENG FRA GER ITA JAP SOV USA}
					has_naval_treaty_trigger = yes
				}
			}
		}
	}
}

# [equipment-selection] SOV added. Routing a submarine navy through this trigger moves its production
# from the naval_submarine role to the cruiser_submarine role: WA_AI_PRODUCTION_focus_on_submarines
# (WA_AI_PRODUCTION_navy.txt:22) excludes members of this list, so the country picks up
# WA_DEFAULT_production_navy_main_focus_on_cruiser_submarines instead, which takes
# naval_submarine to 0 (base 25 - 25) and cruiser_submarine to 100.
# For SOV that is the only lever that reaches the defect R30/[equipment-selection] could not: the coastal
# Malyutka (sov_submarine_hull_3, range 1640 / str 6) is the highest-priority REACHABLE design
# of the naval_submarine role, its tech is a 1936 START tech so research suppression is
# impossible, and both the design layer ([uk-air-basing]) and production_upgrade_desire_offset ([equipment-selection])
# were tried and moved nothing. Emptying the role is what remains.
# Preconditions checked before adding SOV (all already true, nothing else was needed):
#   - designs: SOV_naval_cruiser_submarine exists (ai_equipment/SOV_naval.txt), 5 hulls
#   - techs: SOV starts with sov_cruiser_submarine_1..4 researched (history/countries/SOV:153-156)
#   - fleet: 19 cruiser submarines in the 1936 OOB, so the role is not created from nothing
#   - missions: ConvoyRaiding_2 (ai_navy/taskforce/generic_taskforce_templates.txt) is a
#     cruiser_submarine-composition task force, so the boats built here get put to sea
# Side effect, accepted: WA_AI_RESEARCH_needs_cruiser_submarines (WA_AI_RESEARCH_naval.txt:120)
# reads this trigger, so SOV now weights cruiser-sub research. Only sov_cruiser_submarine_5
# (Zulu, 1948) is left to research, so the practical effect is nil.
WA_AI_CONFIG_focus_on_cruiser_submarines = {
	OR = {
		original_tag = GER
		original_tag = SOV
	}
}

# True when the country's army XP income is structurally penalised.
# Reads the aggregate country modifier, so it covers national spirits, dynamic
# modifiers and advisors alike, and it closes again by itself once the penalty is
# lifted. Used to keep "nice to have" template roles shut for countries that cannot
# fund a second design on top of their line infantry.
# Catches SOV (purge + red army), CHI/PRC, MAN, JAP, NZL, TUR, BUL, BEL.
# Does NOT catch a normal GER/ENG/USA/FRA run - their negative XP spirits are
# civil-war or transient-only.
WA_AI_CONFIG_has_penalised_army_xp_gain = {
	OR = {
		check_variable = {
			var = modifier@experience_gain_army_factor
			value = -0.05
			compare = less_than
		}
		check_variable = {
			var = modifier@experience_gain_factor
			value = -0.05
			compare = less_than
		}
	}
}

# Can this country field motorised support companies rather than horse-drawn ones?
# Read by 18 sites, not only the templates: the scripted division creator's equipment
# requirements, the reserve-division template, and the infantry / marines / mountaineer target
# templates. Two ways in:
#  - by tag: the four big motorised armies, plus the Commonwealth dominions. The dominions are
#    named because no industrial threshold separates them from a Latin American minor early on,
#    and they must stay motorised from 1936.
#  - by earned latch: WA_AI_TEMPLATES_motorized_support_earned, set once and never cleared by
#    WA_AI_TEMPLATES_update_motorized_support_latch (WA_AI_TEMPLATES_effects.txt). This replaces a
#    blanket `is_in_faction_with USA/ENG`, which handed motorised support to every Allied minor on
#    paper, however small.
#
# The latch is one-way ON PURPOSE. The readers cannot tolerate a value that flips: every read site
# above selects WHICH ai_template target is enabled, and each time the enabled target changes the
# engine hands the role to a fresh copy and decommissions the template currently holding it
# (lessons-log.md, "flag-gated ai_template targets re-run the decommission pass"). A bare threshold
# recomputed monthly would do that on factory construction, bomb damage, state loss, faction change
# and exile. If you ever add a release condition here, that lesson is the one to read first.
WA_AI_CONFIG_DIVISIONS_can_motorize_support = {
	OR = {
		original_tag = USA
		original_tag = ENG
		original_tag = FRA
		original_tag = SOV
		original_tag = CAN
		original_tag = AST
		original_tag = NZL
		original_tag = SAF
		original_tag = RAJ
		has_country_flag = WA_AI_TEMPLATES_motorized_support_earned
	}
}
