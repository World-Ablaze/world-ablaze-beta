# [sf-doctrine] Availability gates for the special-forces subdoctrines.
# Copied from the `allow` blocks of the legacy roots in common/technologies/special_forces_doctrine.txt,
# with one deliberate addition: `tech_special_forces`, the generic infantry tree's single
# special-forces unlock. The legacy lists named only the twelve national trees, so a country on the
# generic tree could field the battalions but never take the doctrine. Assumes the national trees
# keep their `<tree>_tech_mountaineers` / `<tree>_marines` / `<tree>_paratroopers` names; a rename
# there silently narrows these gates.

WA_SF_DOCTRINE_has_mountaineers_tech = {
	OR = {
		has_tech = tech_special_forces
		has_tech = ger_tech_mountaineers
		has_tech = eng_tech_mountaineers
		has_tech = sov_tech_mountaineers
		has_tech = usa_tech_mountaineers
		has_tech = fra_tech_mountaineers
		has_tech = jap_tech_mountaineers
		has_tech = ita_tech_mountaineers
		has_tech = fin_tech_mountaineers
		has_tech = hun_tech_mountaineers
		has_tech = cze_tech_mountaineers
		has_tech = pol_tech_mountaineers
		has_tech = swe_tech_mountaineers
	}
}
