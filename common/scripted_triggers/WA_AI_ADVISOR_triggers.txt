# [theorist-hiring] the engine's advisor AI leaves a stable class of countries advisor-less for
# entire campaigns even with candidates hirable and PP at the cap; this gate assumes natural
# engine hires happen EARLY, so the 1938 delay only catches the class the engine never serves.
# History and measurements: WORK.md subject theorist-hiring.
@WA_AI_ADVISOR_pp_floor = 200

WA_AI_ADVISOR_should_hire_army_theorist = {
	is_ai = yes
	has_capitulated = no
	date > 1938.1.1
	has_political_power > @WA_AI_ADVISOR_pp_floor
	amount_taken_ideas = {
		amount < 1
		slots = { army_theorist }
	}
	any_character = {
		has_advisor_role = army_theorist
		is_hired_as_advisor = no
	}
}
