# ============================================================================
# AI Research Needs Triggers for Tanks and Mechanized
# ============================================================================
# These triggers determine when the AI should prioritize researching tank/mech techs.
# They are based on strategic desire/config, NOT on whether equipment is already unlocked,
# to avoid circular logic (AI needs research to unlock, but needs unlock to research).
#
# [techtree-capability] Each variant trigger below asks THREE separate questions, one per layer:
#   WA_AI_TEMPLATES_use_armor_templates      does this country field armour at all
#   WA_AI_CONFIG_DIVISIONS_use_<variant>     do we WANT this variant (doctrine switch)
#   WA_AI_TECHTREE_has_branch_<cap>          does this country's TREE contain the branch
# The third one is new. Every trigger here used to carry the comment "has X tech tree" while
# asking only the first two - so the tree question was answered by hand-written tag lists inside
# WA_AI_CONFIG (light TD = "ITA JAP GER", mechanized SPAA = "SOV"). Those lists are gone; the
# question is now asked against common/technology_tags/00_technology.txt, tag OR tree flag, so a
# country that ADOPTED another tree (common/decisions/_unique_technologies_adoption.txt sets
# <x>_technologies_tree_flag) is answered on the tree it actually owns.
# ============================================================================

WA_AI_RESEARCH_needs_light_armor = {
	# AI wants to use armored divisions at all
	WA_AI_TEMPLATES_use_armor_templates = yes
}

WA_AI_RESEARCH_needs_medium_armor = {
	WA_AI_TEMPLATES_use_armor_templates = yes
	OR = {
		# Focused on medium armor
		WA_AI_CONFIG_DIVISIONS_focus_on_medium_armor = yes
		# Focused on heavy armor (often requires medium research path)
		WA_AI_CONFIG_DIVISIONS_focus_on_heavy_armor = yes
		# After 1941, all armor-using nations should research mediums
		WA_AI_CONFIG_after_1941 = yes
	}
}

WA_AI_RESEARCH_needs_heavy_armor = {
	WA_AI_TEMPLATES_use_armor_templates = yes
	WA_AI_CONFIG_DIVISIONS_focus_on_heavy_armor = yes
}

# [modern-chassis-tier] Same shape as needs_medium_armor, because the modern chassis IS the last
# generation of the medium class. The old `date > 1945.1.1` sat 28 months behind the template
# layer's own gate, so no AI ever researched a modern chassis inside a normal campaign and every
# medium army finished the war on its 1943 tank. The binding date is now the per-tech ai_will_do
# window (1943, or 1942 with free research slots), which is the intent everywhere else here.
WA_AI_RESEARCH_needs_modern_armor = {
	WA_AI_TEMPLATES_use_armor_templates = yes
	OR = {
		WA_AI_CONFIG_DIVISIONS_focus_on_medium_armor = yes
		WA_AI_CONFIG_DIVISIONS_focus_on_heavy_armor = yes
	}
	WA_AI_CONFIG_after_global_war_begins = yes
}

# [raj-trucks] two independent truck demands, either one justifies the tech: the transport under
# motorised SUPPORT companies (~220 per division, WA_AI_DIVISION_CREATOR_effects.txt), and supply-hub
# motorization, which a horse army needs just as much (500 trucks per hub,
# NDefines.NSupply.SUPPLY_HUB_FULL_MOTORIZATION_TRUCK_COST; the engine AI motorizes on its own below
# NDefines.NAI.DIVISION_SUPPLY_RATIO_TO_MOTORIZE). The factory floor keeps the truck line payable.
WA_AI_RESEARCH_needs_trucks = {
	OR = {
		WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
		WA_AI_RESEARCH_needs_mechanized = yes
		WA_AI_TEMPLATES_use_armor_templates = yes
		WA_AI_TEMPLATES_use_motorized_templates = yes
		WA_AI_CONFIG_is_major_country = yes
		num_of_military_factories > 10
	}
}

WA_AI_RESEARCH_needs_mechanized = {
	OR = {
		# Nations configured to use mechanized
		WA_AI_CONFIG_DIVISIONS_use_mechanized_divisions = yes
		# Nations focusing on medium/heavy armor should also get mechs for combined arms
		AND = {
			WA_AI_TEMPLATES_use_armor_templates = yes
			OR = {
				WA_AI_CONFIG_DIVISIONS_focus_on_medium_armor = yes
				WA_AI_CONFIG_DIVISIONS_focus_on_heavy_armor = yes
			}
		}
		# Late game, larger nations should research mechanized
		AND = {
			num_of_military_factories > 100
			WA_AI_CONFIG_after_1941 = yes
		}
	}
}
