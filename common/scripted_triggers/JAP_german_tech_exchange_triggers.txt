# German technology exchange (JAP_gte_technology_exchange decision).
# A route point counts as held when its controller is Japan, Germany, or anyone in either faction.
# The three JAP_gte_debug_* global flags let the debug decisions mark a whole group as held,
# because set_state_controller silently fails on states of countries Japan is not at war with.

JAP_gte_state_held = {
	CONTROLLER = {
		OR = {
			tag = JAP
			is_in_faction_with = JAP
			tag = GER
			is_in_faction_with = GER
		}
	}
}
