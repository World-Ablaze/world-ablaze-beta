# Capability triggers read by common/ai_navy templates. COUNTRY scope.

# [can-transit-attrition] the frigate mass at which a DD-heavy navy's destroyer
# convoy-escort templates hand over to the frigate force (owner ruling: 100). Single
# declaration - the USA/CAN taskforce files read this trigger. Assumes has_navy_size
# counts "owned" closely enough near the mark (repair/reserve scope undocumented).
WA_AI_NAVY_frigate_mass_reached = {
	has_navy_size = {
		size > 99
		unit = frigate
	}
}

# ============================================================================
# NAVY ARCHETYPE - composed from the WA_AI_CONFIG identities
# ============================================================================

WA_AI_NAVY_is_escort_navy = {
	AND = {
		WA_AI_COUNTRY_is_minor = yes
		WA_AI_CONFIG_is_in_allies = yes
	}
}

# Does this country fund a convoy-escort arm at all - either as a small Allied escort navy, or as
# an ocean-spanning power that funds one on its own account.
WA_AI_NAVY_is_escort_focused = {
	OR = {
		WA_AI_NAVY_is_escort_navy = yes
		WA_AI_CONFIG_is_convoy_escort_power = yes
	}
}
