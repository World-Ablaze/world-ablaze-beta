# ROOT = LEASER, THIS = TARGET
# conditions for a country to be lend-leased
WA_AI_lend_lease = {

	has_global_flag = WA_AI_lend_lease

	#####################################################################################################
	#	LEASER
	#####################################################################################################

	ROOT = {
		has_capitulated = no

		# (a WA_AI_supports_@<target> escape-hatch flag used to sit in this OR; it was never
		# set anywhere in the mod, so the branch was dead - removed)
		OR = {
			AND = {
				WA_AI_special_lend_lease_rules_SENDER = yes

				NOT = { surrender_progress > 0.05 }
			}
		}

		### neutral countries ll to neighbors only

		if = { limit = { WA_AI_neutral_country = yes }

			is_neighbor_of = PREV
			has_opinion = { target = PREV value > 25 }
		}

		### limits just any country from using lend-lease, such as random minors in south america

		if = {
			limit = {
				NOT = {
					WA_AI_major_country = yes
					is_major = yes
					is_in_faction = yes
					has_war = yes
				}
			}

			OR = {
				AND = {
					is_neighbor_of = PREV
					has_opinion = { target = PREV value > 50 }
				}

				### SPECIAL RULES

				AND = {
					has_government = fascism
					PREV = { original_tag = SPR WA_AI_fascist_nation = yes has_civil_war = yes }
				}
			}
		}

		### rules for subjects

		if = { limit = { is_subject = yes }

			OR = {
				is_subject_of = PREV # is subject of the target
				PREV = { any_enemy_country = { has_war_with = ROOT } } # common enemy
			}
		}

		### when should it lend lease a country depending on ideology

		# democratic countries

		if = {
			limit = {
				if = { limit = { is_in_faction = no }
					WA_AI_democratic_nation = yes
				}
				else = { # behave according to the faction leader's ideology
					OR = {
						AND = {
							is_faction_leader = yes
							WA_AI_democratic_nation = yes
						}
						var:faction_leader = {
							WA_AI_democratic_nation = yes
						}
					}
				}
			}

			PREV = {
				OR = {
					AND = {
						WA_AI_democratic_nation = yes

						# Non-intervention: democracies stay out of other countries' civil wars
						# unless the enemy side is itself fascist- or communist-aligned. This is
						# the Spanish Civil War case - the Republic's enemy rules as neutrality,
						# so ENG/FRA/USA no longer drip-feed the Republicans while the
						# Non-Intervention Committee is supposedly in force. A democratic side
						# fighting openly fascist or communist rebels still gets aid.
						OR = {
							has_civil_war = no

							any_enemy_country = {
								OR = {
									WA_AI_fascist_nation = yes
									WA_AI_communist_nation = yes
								}
							}
						}
					}

					AND = {
						WA_AI_communist_nation = yes

						has_offensive_war = no
					}

					AND = {
						WA_AI_nonaligned_nation = yes

						has_offensive_war = no
					}

					is_in_faction_with = ROOT

					has_war_together_with = ROOT
				}
			}
		}

		# communist countries

		else_if = {
			limit = {
				if = { limit = { is_in_faction = no }
					WA_AI_communist_nation = yes
				}
				else = { # behave according to the faction leader's ideology
					OR = {
						AND = {
							is_faction_leader = yes
							WA_AI_communist_nation = yes
						}
						var:faction_leader = {
							WA_AI_communist_nation = yes
						}
					}
				}
			}

			PREV = {
				OR = {
					WA_AI_communist_nation = yes

					AND = {
						WA_AI_democratic_nation = yes

						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					AND = {
						WA_AI_nonaligned_nation = yes

						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					is_in_faction_with = ROOT

					has_war_together_with = ROOT
				}
			}
		}

		# fascist countries

		else_if = {
			limit = {
				if = { limit = { is_in_faction = no }
					WA_AI_fascist_nation = yes
				}
				else = { # behave according to the faction leader's ideology
					OR = {
						AND = {
							is_faction_leader = yes
							WA_AI_fascist_nation = yes
						}
						var:faction_leader = {
							WA_AI_fascist_nation = yes
						}
					}
				}
			}

			PREV = {
				OR = {
					WA_AI_fascist_nation = yes

					AND = {
						WA_AI_nonaligned_nation = yes

						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					# Fascist sponsors back the anti-leftist side of a civil war (Spanish Civil
					# War pattern: Nationalist Spain rules as neutrality, so the fascist-target
					# leg above never matches it). Purely dynamic gate - any nonaligned civil-war
					# belligerent whose enemy is democratic- or communist-aligned qualifies, in
					# any timeline.
					AND = {
						WA_AI_nonaligned_nation = yes
						has_civil_war = yes

						any_enemy_country = {
							OR = {
								WA_AI_democratic_nation = yes
								WA_AI_communist_nation = yes
							}
						}
					}

					is_in_faction_with = ROOT

					has_war_together_with = ROOT
				}
			}
		}

		# nonaligned countries

		else_if = {
			limit = {
				if = { limit = { is_in_faction = no }
					WA_AI_nonaligned_nation = yes
				}
				else = { # behave according to the faction leader's ideology
					OR = {
						AND = {
							is_faction_leader = yes
							WA_AI_nonaligned_nation = yes
						}
						var:faction_leader = {
							WA_AI_nonaligned_nation = yes
						}
					}
				}
			}

			PREV = {
				OR = {
					AND = { WA_AI_democratic_nation = yes
						has_offensive_war = no
						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					AND = { WA_AI_fascist_nation = yes
						has_offensive_war = no
						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					AND = { WA_AI_communist_nation = yes
						has_offensive_war = no
						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					AND = { WA_AI_nonaligned_nation = yes
						has_offensive_war = no
						any_country = {
							OR = { tag = PREV is_in_faction_with = PREV is_subject_of = PREV has_war_together_with = PREV }
							any_neighbor_country = { OR = { tag = ROOT is_in_faction_with = ROOT is_subject_of = ROOT has_war_together_with = ROOT } }
						}
					}

					is_in_faction_with = ROOT

					has_war_together_with = ROOT
				}
			}
		}
	}

	#####################################################################################################
	#	TARGET
	#####################################################################################################

	THIS = {
		NOT = { tag = ROOT }

		has_war = yes NOT = { has_war_with = ROOT }

		set_temp_variable = { t1 = THIS.id }

		# (a WA_AI_receives_support_@<target> escape-hatch flag used to sit in this OR; it was
		# never set anywhere in the mod, so the branch was dead - removed)
		OR = {
			AND = {
				WA_AI_special_lend_lease_rules_TARGET = yes

				### why should it receive lend lease

				OR = {
					any_enemy_country = { has_war_with = ROOT } # enemy of my enemy is my friend

					ROOT = { has_opinion = { target = PREV value > 25 } } # friendly countries

					is_in_faction_with = ROOT

					has_war_together_with = ROOT

					### support wars against other ideologies

					AND = {
						any_enemy_country = { WA_AI_fascist_nation = yes }
						ROOT = { NOT = { WA_AI_fascist_nation = yes } }
					}
					AND = {
						any_enemy_country = { WA_AI_communist_nation = yes }
						ROOT = { NOT = { WA_AI_communist_nation = yes } }
					}
					AND = {
						any_enemy_country = { WA_AI_democratic_nation = yes }
						ROOT = { NOT = { WA_AI_democratic_nation = yes } }
					}

					### shared ideology

					ROOT = {
						NOT = { has_government = neutrality }
						WA_AI_PREV_has_same_ideology_with_THIS = yes
					}
				}

				###

				if = { limit = { has_capitulated = yes }

					has_army_manpower = { size > 100000 }
					has_manpower > 100000
				}

				### why should it NOT lend lease

				NOT = {
					AND = { # the enemy of this country has a NAP with ROOT or faction member
						NOT = { any_enemy_country = { has_war_with = ROOT } }

						any_enemy_country = {
							OR = {
								has_non_aggression_pact_with = ROOT

								any_country = { OR = { is_subject_of = ROOT ROOT = { is_subject_of = PREV } is_in_faction_with = ROOT } has_non_aggression_pact_with = PREV }
							}
						}
					}
				}
			}
		}

		### if a land route isnt available, check that the target has convoys

		if = {
			limit = {
				NOT = { has_country_flag = WA_AI_lend_lease_from_@ROOT } # only check when starting lend-lease

				NOT = {
					any_home_area_neighbor_country = {
						NOT = { has_war_with = ROOT has_war_with = var:ROOT.t1 tag = var:ROOT.t1 }

						OR = {
							tag = ROOT
							any_home_area_neighbor_country = {
								NOT = { has_war_with = ROOT has_war_with = var:ROOT.t1 tag = var:ROOT.t1 }

								OR = {
									tag = ROOT
									any_home_area_neighbor_country = { tag = ROOT }
								}
							}
						}
					}
				}
			}

			stockpile_ratio = { archetype = convoy ratio > 50 }
		}
	}
}

WA_AI_special_lend_lease_rules_SENDER = {

	# GER should focus on stockpiling for itself - UNLESS it is sitting on a clear exportable
	# surplus. Without the escape this gate held for the whole war (SOV alive, communist, not
	# allied) while GER hoarded 330k rifles next to allies running an equipment debt.
	if = {
		limit = {
			original_tag = GER
			has_war = yes
		}

		OR = {
			WA_AI_LEND_LEASE_has_exportable_surplus = yes
			SOV = { has_capitulated = yes }
			SOV = { exists = no }
			SOV = { is_subject = yes }
			is_in_faction_with = SOV
			SOV = { NOT = { has_government = communism } }
		}
	}

	if = { # build up its army first
		limit = {
			original_tag = GER
			has_war = no
		}

		has_army_manpower = { size > 550000 }
	}

	# SOV should focus on stockpiling for itself - same surplus escape as GER above
	if = {
		limit = {
			original_tag = SOV
			has_war = yes
		}

		OR = {
			WA_AI_LEND_LEASE_has_exportable_surplus = yes
			GER = { has_capitulated = yes }
			GER = { exists = no }
			GER = { is_subject = yes }
			is_in_faction_with = GER
			GER = { NOT = { has_government = fascism } }
		}
	}

	if = { limit = { original_tag = JAP }
		date > 1938.1.1
	}

	if = { limit = { original_tag = CHI }
		has_equipment = { infantry_equipment > 50000 }
	}
}

WA_AI_special_lend_lease_rules_TARGET = {

	# axis shouldn't lend lease japan
	if = {
		limit = {
			ROOT = {
				is_in_faction_with = GER
				has_war = yes
				capital_scope = { NOT = { is_on_continent = asia is_on_continent = oceania } }
			}
		}

		capital_scope = { NOT = { is_on_continent = asia is_on_continent = oceania } } # dont ll to asia
	}

	# japan shouldn't lend lease europe
	if = {
		limit = {
			ROOT = {
				OR = { original_tag = JAP is_subject_of = JAP is_in_faction_with = JAP }
				capital_scope = { OR = { is_on_continent = asia is_on_continent = oceania } }
			}
		}

		OR = {
			capital_scope = { OR = { is_on_continent = asia is_on_continent = oceania } }
			is_neighbor_of = ROOT # land border
		}
	}

	# make things easer for AI GER vs SOV
	if = {
		limit = {
			THIS = { original_tag = SOV }
			GER = { is_ai = yes }
		}

		OR = {
			surrender_progress > 0.0
			ROOT = { has_government = communism }
		}
	}

	# SOV should focus on stockpiling for itself - same surplus escape as the SENDER side
	if = {
		limit = {
			ROOT = { original_tag = SOV }
			THIS = { NOT = { original_tag = SPR original_tag = CHI } }
			NOT = { THIS = { has_war_together_with = ROOT } }
		}

		OR = {
			ROOT = { WA_AI_LEND_LEASE_has_exportable_surplus = yes }
			GER = { has_capitulated = yes }
			GER = { exists = no }
			GER = { is_subject = yes }
			is_in_faction_with = GER
			GER = { NOT = { has_government = fascism } }
		}
	}

	# make things easier for AI JAP vs CHI
	if = {
		limit = {
			THIS = { has_war_with = JAP OR = { original_tag = CHI original_tag = GUC original_tag = GXC original_tag = YUN original_tag = SIC original_tag = XIK original_tag = TUG original_tag = XSM original_tag = SHA original_tag = HPC original_tag = SHX original_tag = PRC original_tag = NEA } }
			JAP = { is_ai = yes }
			NOT = { THIS = { has_war_together_with = ROOT } }
		}

		#date < 1940.1.1

		OR = {
			AND = { ROOT = { tag = USA } CHI = { surrender_progress > 0.5 } }
			AND = { ROOT = { is_in_faction_with = SOV } CHI = { surrender_progress > 0.0 } }
			CHI = { surrender_progress > 0.75 }
			CHI = { is_subject = yes }
		}
	}

	# make things easier for AI GER vs POL
	if = {
		limit = {
			THIS = { original_tag = POL }
			GER = { is_ai = yes }
		}

		OR = {
			NOT = { is_in_faction_with = ENG }
			date > 1941.1.1
		}
	}

	# USA doesnt need lend-lease
	if = {
		limit = {
			THIS = { original_tag = USA }
		}

		is_major = no
	}

	if = {
		limit = {
			THIS = { is_in_faction_with = ENG }
			ROOT = { is_in_faction_with = ENG }
		}

		OR = {
			NOT = { any_enemy_country = { has_navy_size = { size > 40 unit = submarine } } }
			AND = {
				THIS = { tag = ENG }
				ROOT = { tag = USA }
			}
		}
	}
}
######################################################
#	LEND-LEASE SURPLUS RELIEF - CAPABILITY TRIGGERS ([lend-lease-relief], 2026-08-16)
#	Capability questions, not tag questions: any country in any timeline qualifies purely
#	on its stockpile, war state and geography.
#
#	One trigger PAIR per equipment archetype, generated from one shape - the numbers live in
#	common/script_constants/wa_ai_lend_lease.txt (constant:wa_ai_lend_lease.<row>.*), the
#	archetype token is the only literal. Archetypes are COMPLEMENTARY (a WA division needs
#	rifles AND heavy infantry equipment AND support ...), so no trigger ever sums two of them
#	and a country short of one gets that one (modder ruling 2026-08-16, campaign a232d96c /
#	checklist R7b: the old rifle-only gate could not see GER's heavy-infantry surplus).
#	Reads `num_equipment@<archetype>` = the FREE stockpile (same quantity has_equipment reads),
#	through check_variable so the bars can be script constants (validated context).
#
#	Per-archetype split history: support tracks only 5-25% of infantry (campaign c9ab1062),
#	so a single "inf AND sup" gate never passed; the infantry donor bar was lowered 60k -> 35k
#	after 9be92c89 (no Axis donor ever crossed 60k while four allies ran zero-rifle stockpiles);
#	the support starving line was raised 2k -> 4k after 9be92c89 (the 1200/week drip parked
#	HUN/ROM/BUL/ITA in the 2-4k band). Those numbers are unchanged in the constants file.
######################################################
# THIS = potential donor. Surplus of infantry_equipment above its floor.

	has_capitulated = no
	check_variable = { num_equipment@infantry_equipment > constant:wa_ai_lend_lease.infantry.donor_floor }
}

# Maritime convoy relief. Capacity and need are dynamic: 200 spare convoys is adequate for a
# <=20-dockyard recipient, not for a 100-dockyard oceanic power. These four mutually-exclusive
# bands are shared by recipients, donors and the production trigger.
WA_AI_LEND_LEASE_convoy_scale_small = {
	NOT = { num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_small }
}
WA_AI_LEND_LEASE_convoy_scale_medium = {
	num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_small
	NOT = { num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_medium }
}
WA_AI_LEND_LEASE_convoy_scale_large = {
	num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_medium
	NOT = { num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_large }
}
WA_AI_LEND_LEASE_convoy_scale_huge = {
	num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_large
}

# A country with a coast can consume convoy hulls even if it currently owns no dockyard.
WA_AI_LEND_LEASE_can_use_convoys = {
	OR = {
		num_of_naval_factories > 0
		any_owned_state = { is_coastal = yes }
	}
}

WA_AI_LEND_LEASE_is_starving_convoys = {
	is_ai = yes
	has_war = yes
	WA_AI_LEND_LEASE_can_use_convoys = yes
	OR = {
		# A major's worldwide army and lend-lease network can be large despite a small
		# domestic dockyard base. This is max(dockyard band, major floor), not a tag rule.
		AND = {
			is_major = yes
			check_variable = { num_equipment@convoy < constant:wa_ai_lend_lease.convoy.reserve_major_floor }
		}
		AND = {
			WA_AI_LEND_LEASE_convoy_scale_small = yes
			check_variable = { num_equipment@convoy < constant:wa_ai_lend_lease.convoy.reserve_small }
		}
		AND = {
			WA_AI_LEND_LEASE_convoy_scale_medium = yes
			check_variable = { num_equipment@convoy < constant:wa_ai_lend_lease.convoy.reserve_medium }
		}
		AND = {
			WA_AI_LEND_LEASE_convoy_scale_large = yes
			check_variable = { num_equipment@convoy < constant:wa_ai_lend_lease.convoy.reserve_large }
		}
		AND = {
			WA_AI_LEND_LEASE_convoy_scale_huge = yes
			check_variable = { num_equipment@convoy < constant:wa_ai_lend_lease.convoy.reserve_huge }
		}
	}
}

# THIS = convoy recipient, PREV = donor. Unlike the nine overland legs, this branch is
# deliberately maritime and never pathfinds. Same faction, a common enemy, or the existing
# explicit lend-lease target flag is enough; both sides must be living AIs with maritime use.
WA_AI_LEND_LEASE_convoy_pair_is_valid = {
	exists = yes
	is_ai = yes
	NOT = { tag = PREV }
	has_capitulated = no
	WA_AI_LEND_LEASE_can_use_convoys = yes
	PREV = {
		exists = yes
		is_ai = yes
		has_capitulated = no
		WA_AI_LEND_LEASE_can_use_convoys = yes
	}
	OR = {
		is_in_faction_with = PREV
		has_war_together_with = PREV
		has_country_flag = WA_AI_lend_lease_from_@PREV
	}
}

# THIS = candidate recipient, PREV = potential donor. TRUE when the donor can serve at least

# run for every country every week); the path is tested once, on the picked donor, in the

# THIS = candidate recipient, PREV = potential donor.
# PREV = { is_ai = yes } is load-bearing: under the pull model the donor scope is not
# behind any is_ai gate, and an AI must never strip a human player's stockpile.
# The WA_AI_lend_lease_from_@ flag path reuses the pair the ordinary lend-lease
# targeting system already chose (USA->ENG 1940-41 style, pre-faction).
# LAND ACCESS ([lend-lease-relief]) is NOT here - it is a state A* run once per picked donor in
# WA_AI_LEND_LEASE_request_surplus_relief; this trigger is the cheap half of the pair test.
WA_AI_LEND_LEASE_relief_pair_is_valid = {
	NOT = { tag = PREV }
	has_capitulated = no
	PREV = {
		is_ai = yes
		has_capitulated = no
	}
	OR = {
		# A common enemy suffices - requiring same-faction excluded co-belligerents
		# (FIN starved beside a GER surplus, CHI beside SOV's 101k, campaign 9be92c89)
		has_war_together_with = PREV
		has_country_flag = WA_AI_lend_lease_from_@PREV
	}
}
