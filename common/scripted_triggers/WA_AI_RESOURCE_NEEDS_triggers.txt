# NOTE (audit 2026-08-11): WA_AI_resource_<r>_shortage_months is only written for
# rubber, steel and aluminium (WA_AI_CONSTRUCTION_PRIORITY_strategies.txt). The
# shortage-months leg of the other reactive triggers below reads a variable that
# is never written (always 0 = false); those triggers work through WA_AI_needs_<r>
# alone (written by WA_AI_check_resource_needs, WA_AI_misc_effects.txt, on a ~2-day
# event - NOT monthly: the ramp to 3 and back takes about six days, so a monthly save
# can miss whole cycles).

# Check if AI has any shortage of a resource (OR: critical need OR 3+ months history)
# Coal is strategic (every synthetic refinery burns it): prospect PROACTIVELY
# while the effective balance is under +100, not only once it is negative.
# resource@coal is net-available minus unmet demand (see wa-savegame-analysis).
# What resource@coal measures: extraction X, minus the trade law's export share
# (already gone, not counted here), minus own consumption. The engine trade AI
# only imports when that number is NEGATIVE, to bring it back to ~0 - it never
# builds a reserve. So "< 100" is a genuine proactive bar: it fires for a
# producer whose retained surplus is under 100 (and for an importer parked at
# ~0), and stops for a producer keeping 100+ after export share and use. GER
# late 1943 (+15 000) is far above it. See the lessons log entry "resource@X
# and the engine trade AI" before changing the threshold.
# The gauge (resource@) carries the peacetime/proactive case; the WA_AI_needs
# counter (0..3, ~2-day ramp) carries the wartime shortage case.
# Since 2026-08-15 the prospecting ai_will_do blocks have base weight 0 and this
# trigger is the only thing that gives the coal decisions any weight for ROOT's
# own sake, so a country sitting on +15 000 coal (GER, late 1943) no longer
# re-prospects its states just because civs and PP are free.
# Two live legs, not three: WA_AI_resource_coal_shortage_months has no writer in the
# repo. A country flag of the same name as the cooperative trigger is written at
# WA_AI_misc_effects.txt:646 and read nowhere; the scripted trigger shadows it.
WA_AI_should_prospect_resource_coal = {
	OR = {
		check_variable = { resource@coal < 100 }
		check_variable = { WA_AI_needs_coal > 1 }
		check_variable = { WA_AI_resource_coal_shortage_months > 0 }
	}
	WA_AI_CONFIG_prospect_resource_disabled = no
}

WA_AI_should_prospect_resource_oil = {
	OR = {
		# WA_AI_needs_oil is written since [overextension-brake] (oil arm added 2026-08-11 to

		# matching coal). WA_AI_resource_oil_shortage_months is STILL never written
		# (always 0) - this trigger works through WA_AI_needs_oil alone.
		check_variable = { WA_AI_needs_oil > 1 }
		check_variable = { WA_AI_resource_oil_shortage_months > 0 }
	}

# Cooperative Layer: Check if allies have critical resource shortages
###############################################################################
# Each leg asks two things: does an ally have UNMET need, and could ROOT's extraction actually
# reach it.
# [coal-prospect-loop] the ally term reads the ally's effective balance (resource@<r> < 0: demand
# its imports do not cover), not WA_AI_needs_<r> alone. That counter ratchets to 3 on import
# VOLUME (WA_AI_check_resource_needs), so an importer the trade AI parks at ~0 reads "critical"
# for the whole war and a supplier prospects for allies it already supplies. If an ally can hold
# a positive balance while genuinely short (a subject stripped by its overlord), this under-fires:
# retune the ally term here, never in the writer - its import leg has readers of its own.
#
# ROOT can help in exactly two situations, and the trigger tests both:
#   it is short of the resource itself, or its OFFERED export share is already bought out, so extra
#   extraction finds a buyer. The second is computed per country in WA_AI_check_resource_needs
#   (WA_AI_misc_effects.txt) because a trigger read per decision target cannot do that arithmetic;
#   that header carries the estimator, its three biases and its falsifier.
#
# WHY NOT THE BALANCE SIGN, WHICH THIS REPLACES ([prospecting-coop], commits 2cf5f214f / ed29adbe5).
# Those headers asserted "ZERO IS NOT TUNABLE" and listed "a gate reading the offered share or the
# trade law directly" as NOT BUILT. Both are now overturned by measurement, not by preference: on
# campaign 7c7803a8, 184 of 2706 country/resource rows have their offered share bought out and 160
# of those - 87 % - still hold a POSITIVE balance. A sign test blocks every one of them. CAN sells
# all 1126.7 aluminium it offers on a +148 balance; ENG all 216 iron on +201; USA all 177.2 tungsten
# on +1. Scored against "fire if short OR sold out", the sign gate makes 166 errors, all of them
# refusals to prospect where extraction would have delivered; this pair makes 20, and blocks the
# same 15 of 16 known waste cases. The coal ceiling of 500 goes with it: the sold-out test keeps a
# selling producer (SAF at +442) and still blocks a hoarding one (GER at +13 396), which is what the
# ceiling was approximating.
#
# Evidence and probes: checklist R66. Durable rule: lessons-log, "The export share scales with
# extraction".

# Scope: country. Every call site is an ai_will_do block in
# common/decisions/_resource_prospecting.txt, so the bare check_variable reads ROOT. Keep the
# ROOT-side OR OUTSIDE any_other_country.
# [prospecting-coop-solvency] the needy ally must also be able to BUY: the engine reports spare
# civs, or its trade AI already imports the resource. Assumes engine trade preempts construction,
# so a solvent ally that wants the resource is already importing; if a solvent ally can sit at
# 0 imports with the market stocked, this over-blocks - retune here, not in the callers.
# [coal-prospect-loop] the ally-side `resource@<r> < 0` term (section header) is the ALLY's
# demand test; the ROOT-side sign test in the OR above is a different question (can ROOT's
# extraction reach anyone) and stays as R66 left it. The eight siblings carry the same ally term.
WA_AI_allies_need_coal = {
	OR = {
		check_variable = { resource@coal < 0 }
		check_variable = { WA_AI_coop_can_supply_coal = 1 }
	}
	any_other_country = {
		is_in_faction_with = ROOT
		check_variable = { WA_AI_needs_coal = 3 }
		check_variable = { resource@coal < 0 }
		OR = {
			num_of_civilian_factories_available_for_projects > 0
			check_variable = { resource_imported@coal > 0 }
		}
	}
}

# Proactive Layer: Check if AI is a strategic exporter
###############################################################################

# Strategic exporters maintain high production during peace.
# WHO is an exporter lives in WA_AI_CONFIG.txt (WA_AI_CONFIG_is_strategic_<r>_exporter);
# this file only adds the capability conditions (peace + a trade posture that
# actually sells). Read by the "Proactive: Strategic exporter boost" modifier
# that tools/migrations/prospecting/needs_aware_generator.py emits into every tungsten / rubber / oil
# prospecting decision (add = 100 on a base of 0).

WA_AI_is_strategic_tungsten_exporter = {
	has_war = no
	WA_AI_CONFIG_is_strategic_tungsten_exporter = yes
	OR = {
		has_idea = free_trade
		has_completed_focus = POR_extraction_industries  # POR focus requirement
	}
}

WA_AI_is_strategic_rubber_exporter = {
	has_war = no
	WA_AI_CONFIG_is_strategic_rubber_exporter = yes
	has_idea = free_trade
}

WA_AI_is_strategic_oil_exporter = {
	has_war = no
	WA_AI_CONFIG_is_strategic_oil_exporter = yes
	has_idea = free_trade
}

# Generic dynamic exporter: any country with Free Trade law and surplus factories
WA_AI_is_dynamic_exporter = {
	has_war = no
	has_idea = free_trade
	num_of_civilian_factories > 50
}
