# Hostile land-neighbor pressure in the home area.
# Used to keep continental countries from overcommitting to remote fronts
# when a strong border rival exists (war or pre-war).
WA_AI_MILITARY_border_hostile_pressure = {
	any_home_area_neighbor_country = {
		OR = {
			# Active border war: trigger even against slightly weaker opponents.
			AND = {
				has_war_with = ROOT
				strength_ratio = { tag = ROOT ratio > 0.55 }
			}

			# Pre-war standoff: only trigger against clearly threatening neighbors.
			AND = {
				ROOT = { NOT = { has_war_with = PREV } }
				NOT = { is_in_faction_with = ROOT }
				NOT = { is_subject_of = ROOT }
				ROOT = { NOT = { is_subject_of = PREV } }
				NOT = { has_war_together_with = ROOT }
				NOT = { has_government = ROOT }
				strength_ratio = { tag = ROOT ratio > 0.8 }
			}
		}
	}
}

# Force-balance verdict for an offensive toward the Egyptian anchor of the North-African front.
# ROOT = the country contemplating the offensive (called from ai_strategy enable and event triggers,
# where ROOT is that country). Generic by design: any attacker pushing toward Egypt can consume it,
# and the enemy is whoever actually controls the Egypt-side anchor states - never a hardcoded tag.
# True when at least one war enemy controlling East Egypt (same anchor set as
# WA_AI_MILITARY_ENG_holds_east_egypt) either:
#   (a) has a published weekly posture verdict > 0 vs us (WA_AI_MILITARY_posture_effects.txt -
#       carries the front-local division-count branch, the equipment/manpower floors, and the
#       air-inferiority veto), or
#   (b) never receives a verdict at all (the posture calculus skips enemies that are neither major
#       nor >4 controlled states; pushes against such minors stay unconditional per
#       WA_AI_MILITARY_SYSTEM.md section 9).
# When the balance turns, the weekly verdict drops to 0, this trigger goes false, and the consumer
# aborts - the front falls back to whatever defensive behaviour is otherwise active. The hard-brake
# case (posture_vs_<TAG> lingering > 0 while the brake is on) is covered by the priority-500/10000
# emergency stops in WA_AI_MILITARY_DEFAULT_FRONT_control.txt, which outrank every consumer of this
# trigger. Stale-after-peace verdicts are inert here: any_enemy_country only scopes countries we are
# currently at war with.
WA_AI_MILITARY_north_africa_offensive_viable = {
	any_enemy_country = {
		has_capitulated = no
		OR = {
			controls_state = 446
			controls_state = 447
			controls_state = 453
		}
		OR = {
			# Enemy too small for a posture verdict: no variable is ever published, attack unconditionally.
			AND = {
				is_major = no
				NOT = { num_of_controlled_states > 4 }
			}
			ROOT = { check_variable = { WA_AI_MILITARY_posture_vs_@PREV > 0 } }
		}
	}
}

#
# ALL BEHAVIOURAL SWITCHES FOR THE ALLIED BOMBING CAMPAIGN LIVE HERE. Change the campaign by
# moving the four thresholds below or the ring membership further down - never by editing the
# strategy blocks (same contract as WA_AI_AIFC_triggers.txt and WA_AI_LANDING_triggers.txt).
#
# WHY A LADDER
# Before this, "does the coalition bomb Germany" was decided by a date pair (a -500,000 blackout
# until 1941.10.1, then a -40,000 suppression until a 1944.2.1 hard backstop) plus a USA-only tech
# gate. That is a historical-script assumption: in a game where the Allies arm early, late, or
# never, the dates fire anyway and the air campaign is either grounded while a raid force sits
# idle, or released while there is nothing to release. The question the dates were proxying for
# is a capability question - "does our side field enough heavy bombers to raid the Reich and
# absorb the losses" - so the ladder asks it directly.
#
# THRESHOLDS (deployed strategic_bomber wings, WA force scale ~1/3 vanilla)
# Grounded in campaign 973154a7 (see commit 439c4106d): ENG and USA both cross 450 during late
# 1942 / early 1943 and never fall back under 300; ENG peaks ~750 then attrites to ~550 by late
# 1944. Vanilla's 900/1500/2100 numbers are unreachable at this scale and are the reason the
# original blocks never released.
@WA_AI_AIR_BOMBER_NUCLEUS = 299			# at/below this there is no bomber arm at all
@WA_AI_AIR_BOMBER_RAID_FORCE = 450		# a raid force able to reach Germany proper
@WA_AI_AIR_BOMBER_DEEP_REACH = 700		# ...and to reach Austria / Prussia (the old B-17 range gate)
@WA_AI_AIR_BOMBER_DEEP_REACH_OFF = 900	# deep-target avoidance fully retired
#
# HANDOVER (Fix 71). The four bars above only ever LIFT avoidance. Nothing then pulls the bomber
# arm ONTO the Reich: with every rung aborted the net on 6/7/8/294/38/296 is exactly 0, and the
# engine's own front-following terms decide targets instead. Campaign 5078fe10 is the proof - the
# ladder released fully between 1943.12 and 1944.4 and never re-armed, yet in the 22 months that
# followed the RAF flew every single strategic-bombing sortie over France, Italy and the Balkans
# and not one over a German region, while German factory levels rose from 453 to 602. The pull
# below closes that gap: it turns ON at @WA_AI_AIR_BOMBER_RAID_FORCE, the same bar the near-ring
# suppression turns OFF at, so the two compose into a handover instead of leaving a hole.
#
# @WA_AI_AIR_BOMBER_OWN_ARM gates the pull on ROOT's OWN bomber count, unlike the coalition-wide
# bars above. strategic_air_importance suppresses and boosts EVERY mission in a region, so a
# fighter-only air force given the pull would send its fighters to Germany instead of its front.
# A country with no bomber arm of its own must stay out of it.
@WA_AI_AIR_BOMBER_OWN_ARM = 99			# ROOT's own deployed heavies before it joins the raid
#
# Coalition-wide by design: the metric is "any one partner fields this much", not ROOT's own arm.
# CAN fields zero strategic bombers for an entire campaign; without the any_allied_country branch
# it would go on suppressing the target regions its own faction is raiding. A country in no
# faction sees only itself, which is the correct setup-agnostic floor.
