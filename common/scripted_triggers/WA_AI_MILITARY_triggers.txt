# Hostile land-neighbor pressure in the home area.
# Used to keep continental countries from overcommitting to remote fronts
# when a strong border rival exists (war or pre-war).
WA_AI_MILITARY_border_hostile_pressure = {
	any_home_area_neighbor_country = {
		OR = {
			# Active border war: trigger even against slightly weaker opponents.
			AND = {
				has_war_with = ROOT
				strength_ratio = { tag = ROOT ratio > 0.55 }
			}

			# Pre-war standoff: only trigger against clearly threatening neighbors.
			AND = {
				ROOT = { NOT = { has_war_with = PREV } }
				NOT = { is_in_faction_with = ROOT }
				NOT = { is_subject_of = ROOT }
				ROOT = { NOT = { is_subject_of = PREV } }
				NOT = { has_war_together_with = ROOT }
				NOT = { has_government = ROOT }
				strength_ratio = { tag = ROOT ratio > 0.8 }
			}
		}
	}
}

# Force-balance verdict for an offensive toward the Egyptian anchor of the North-African front.
# ROOT = the country contemplating the offensive (called from ai_strategy enable and event triggers,
# where ROOT is that country). Generic by design: any attacker pushing toward Egypt can consume it,
# and the enemy is whoever actually controls the Egypt-side anchor states - never a hardcoded tag.
# True when at least one war enemy controlling East Egypt (same anchor set as
# WA_AI_MILITARY_ENG_holds_east_egypt) either:
#   (a) has a published weekly posture verdict > 0 vs us (WA_AI_MILITARY_posture_effects.txt -
#       carries the front-local division-count branch, the equipment/manpower floors, and the
#       air-inferiority veto), or
#   (b) never receives a verdict at all (the posture calculus skips enemies that are neither major
#       nor >4 controlled states; pushes against such minors stay unconditional per
#       WA_AI_MILITARY_SYSTEM.md section 9).
# When the balance turns, the weekly verdict drops to 0, this trigger goes false, and the consumer
# aborts - the front falls back to whatever defensive behaviour is otherwise active. The hard-brake
# case (posture_vs_<TAG> lingering > 0 while the brake is on) is covered by the priority-500/10000
# emergency stops in WA_AI_MILITARY_DEFAULT_FRONT_control.txt, which outrank every consumer of this
# trigger. Stale-after-peace verdicts are inert here: any_enemy_country only scopes countries we are
# currently at war with.
WA_AI_MILITARY_north_africa_offensive_viable = {
	any_enemy_country = {
		has_capitulated = no
		OR = {
			controls_state = 446
			controls_state = 447
			controls_state = 453
		}
		OR = {
			# Enemy too small for a posture verdict: no variable is ever published, attack unconditionally.
			AND = {
				is_major = no
				NOT = { num_of_controlled_states > 4 }
			}
			ROOT = { check_variable = { WA_AI_MILITARY_posture_vs_@PREV > 0 } }
		}
	}
}
