# Conditions on the Imperial Diet, for focuses and decisions. Majority = more than half the seats, two thirds =
# 267 of 400 peers and 311 of 466 representatives. Variables in common/scripted_effects/JAP_diet_effects.txt.

JAP_peers_majority_trigger = {
	custom_trigger_tooltip = {
		tooltip = JAP_peers_majority_tt
		check_variable = { JAP_peers_support > 200 }
	}
}

JAP_representatives_majority_trigger = {
	custom_trigger_tooltip = {
		tooltip = JAP_representatives_majority_tt
		check_variable = { JAP_representatives_support > 233 }
	}
}

JAP_peers_two_thirds_trigger = {
	custom_trigger_tooltip = {
		tooltip = JAP_peers_two_thirds_tt
		check_variable = { JAP_peers_support > 266 }
	}
}

JAP_representatives_two_thirds_trigger = {
	custom_trigger_tooltip = {
		tooltip = JAP_representatives_two_thirds_tt
		check_variable = { JAP_representatives_support > 310 }
	}
}

JAP_diet_majority_trigger = {
	JAP_peers_majority_trigger = yes
	JAP_representatives_majority_trigger = yes
}

JAP_diet_two_thirds_trigger = {
	JAP_peers_two_thirds_trigger = yes
	JAP_representatives_two_thirds_trigger = yes
}
