# ============================================================================
# GROUND PRODUCTION TRIGGERS by Aigle2
# ============================================================================
# These triggers are used to determine the production of ground equipment.
# All triggers for production are done here using the PRODUCTION suffix
# so that conditions in ai_strategy files are one-liners.
#
# Ground equipment includes:
# - Infantry equipment (regular and heavy)
# - Artillery
# - Anti-tank equipment (regular and heavy)
# - Anti-air equipment (regular and heavy)
# - Support equipment
# - Trucks (motorized equipment)
# - Trains
# ============================================================================

# ============================================================================
# MASTER SWITCH
# ============================================================================
# This trigger determines if a country uses the default ground production system.
# It references the CONFIG trigger which contains the actual country tags.
# ============================================================================

WA_AI_PRODUCTION_ground_is_enabled = {
	WA_AI_CONFIG_uses_default_ground_production = yes
}

# ============================================================================
# INFANTRY EQUIPMENT TRIGGERS
# ============================================================================

WA_AI_PRODUCTION_build_infantry_base = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
}

WA_AI_PRODUCTION_build_heavy_infantry = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	WA_AI_TEMPLATES_use_heavy_infantry = yes
}

WA_AI_PRODUCTION_build_infantry_stockpile_low = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	OR = {
		has_equipment = { infantry_equipment < 10000 }
		has_equipment = { support_equipment < 5000 }
	}
}

# Reserve effectively empty while at war - the "ITA runs an equipment debt from 1941" case.
# The low tier's floors (5 inf / 2 support factories) are no-ops for a major already running
# 50 factories on infantry; this tier adds real pressure when the buffer is actually gone.
WA_AI_PRODUCTION_build_infantry_stockpile_critical = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	has_war = yes
	OR = {
		has_equipment = { infantry_equipment < 2000 }
		has_equipment = { support_equipment < 1000 }
	}
}

WA_AI_PRODUCTION_build_artillery = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
}

WA_AI_PRODUCTION_build_artillery_major = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	WA_AI_CONFIG_is_major_country = yes
}

# [minor-gun-floor] OBSERVATION - my armies REQUIRE guns, as an establishment (engine variable,
# thousands). Same demand reading the maintenance floors use, and the reason this floor cannot
# reserve a factory for a country whose templates carry no gun. Gone when a country whose
# templates clearly field artillery reads a target of 0 in the console harness.
WA_AI_PRODUCTION_has_artillery_demand = {
	check_variable = { num_target_equipment_in_armies_k@artillery_equipment > constant:wa_ai_production.maintenance.target_min_k }
}

# [minor-gun-floor] DECISION - a non-major with a real industry keeps one factory on guns. Exclusive
# with the majors' floor above by construction, and it is the ONLY floor on this gun since the
# POL/HUN/SWE type-`artillery` one was removed with it. Rests on F1 of
# documentation/WA_AI_MAINTENANCE_FLOOR_PROPOSAL.md (does a need-blind floor actually take
# factories) - ASSUMED, discharged by `event wa_maint.3 <TAG>` then the production panel.
WA_AI_PRODUCTION_should_floor_artillery_minor = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	NOT = { WA_AI_CONFIG_is_major_country = yes }
	num_of_military_factories > constant:wa_ai_production.industry.tier_small_mils
	WA_AI_PRODUCTION_has_artillery_demand = yes
}

# [minor-gun-floor] DECISION - anti-tank had NO floor in this tier, only a +15 production factor,
# and a factor multiplies a perceived need that sits near zero for a towed gun: the line opens and
# waits at the bottom of the queue. Majors are included - they had no floor here either.
# SUMS with WA_AI_PRODUCTION_DEFAULT_maintenance.txt on purpose, and the arithmetic is: 1 here,
# +1 while short (> 20 mils), +2 more when deep (> 49) since `deepen` implies `maintain` -> 2 while
# short, 4 deep. Same relation the infantry stockpile tiers already have. Rests on F1 of
# documentation/WA_AI_MAINTENANCE_FLOOR_PROPOSAL.md, ASSUMED, see the artillery twin above.

	WA_AI_PRODUCTION_ground_is_enabled = yes
	num_of_military_factories > constant:wa_ai_production.industry.tier_small_mils

WA_AI_PRODUCTION_build_trucks = {

# [raj-trucks] the truck floor is tiered by industry, like the amphibious floors. A flat
# min-10-factories is most of a 12-factory MIC, and trucks are now researched by horse armies too
# (supply-hub motorization). 500 trucks = one fully motorized hub
# (NDefines.NSupply.SUPPLY_HUB_FULL_MOTORIZATION_TRUCK_COST).
# [uk-truck-supply] the stock bar is 1500 = wa_ai_lend_lease.motorized.starve (registry-linked):
# the floor must not switch off while the country still counts as truck-starving. The old 1000
# cutoff bought two motorized hubs and died, which is the ENG "supply fulfillment < 50% in Africa"
# symptom.
# [truck-floor-ladder] The industry bands of the truck floor, factored out so the arm trigger and
# its hold partner cannot drift apart. `_top` is the deep tier's last band (supermajor AND
# superpower together); every other band is exclusive of its neighbours by construction.
WA_AI_PRODUCTION_trucks_industry_small = {
	NOT = { num_of_military_factories > constant:wa_ai_production.industry.tier_medium_mils }
}

WA_AI_PRODUCTION_trucks_industry_medium = {
	num_of_military_factories > constant:wa_ai_production.industry.tier_medium_mils
	NOT = { num_of_military_factories > constant:wa_ai_production.industry.tier_large_mils }
}

WA_AI_PRODUCTION_trucks_industry_large = {
	num_of_military_factories > constant:wa_ai_production.industry.tier_large_mils
	NOT = { num_of_military_factories > constant:wa_ai_production.industry.tier_major_mils }
}

WA_AI_PRODUCTION_trucks_industry_major = {
	num_of_military_factories > constant:wa_ai_production.industry.tier_major_mils
	NOT = { num_of_military_factories > constant:wa_ai_production.industry.tier_supermajor_mils }
}

WA_AI_PRODUCTION_trucks_industry_supermajor = {
	num_of_military_factories > constant:wa_ai_production.industry.tier_supermajor_mils

# [truck-floor-ladder] SIX EXCLUSIVE industry bands, monotone the way the mechanized floors are:
# what a truck-starving economy owes scales with the industry that has to refill the starve bar,
# because a hub costs 500 trucks whatever the country's size. A value that stopped growing above
# the large tier made a 700-factory major owe exactly what a 50-factory one owes.
# arm / hold are a SCHMITT PAIR and must never read the same bar: arm at
# wa_ai_lend_lease.motorized.starve, hold until wa_ai_production.trucks.release_stock. The blocks
# use `abort = { NOT = { hold } }`, never `abort_when_not_enabled`, or the pair does nothing.
# Assumes the archetype's max_military_factories (75, common/units/equipment/motorized.txt) stays
# above the deepest band+deep total (45); the fact is gone the day that cap drops.
WA_AI_PRODUCTION_should_floor_trucks_small = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	has_equipment = { motorized_equipment < constant:wa_ai_lend_lease.motorized.starve } # direct read - was a registry mirror
	WA_AI_PRODUCTION_trucks_industry_small = yes
}

WA_AI_PRODUCTION_should_hold_floor_trucks_small = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	has_equipment = { motorized_equipment < constant:wa_ai_production.trucks.release_stock }
	WA_AI_PRODUCTION_trucks_industry_small = yes
}

# [train-variant-choice] Which train VARIANT the lines run on. Cheap = Simplified / War Austerity
# (train_equipment_2 / _3) while the free stockpile is at or under the reserve; Armored
# (train_equipment_4) only above it. Exact complements - never both, never neither - so they carry
# NO other term: a stockpile bar is not a "uses default ground production" question, and the armored
# one is also the can_be_produced gate of train_equipment_4 (a false extra term = a silent AI ban).
# Assumes num_equipment@train_equipment is the free stockpile (the quantity has_equipment reads).
WA_AI_PRODUCTION_should_build_cheap_trains = {
	NOT = { check_variable = { num_equipment@train_equipment > constant:wa_ai_production.trains.armored_reserve } }
}

WA_AI_PRODUCTION_should_build_armored_trains = {
	check_variable = { num_equipment@train_equipment > constant:wa_ai_production.trains.armored_reserve }
}
