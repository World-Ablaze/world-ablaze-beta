# [mot-field-hospital] Does this country field a MOTORISED field hospital inside an otherwise
# horse-drawn line? Read only by the horse leaves of the infantry / mountaineer / marine
# calculators, which add +100 to reach the mirror target. Flag-only, never a live threshold: the
# value picks WHICH ai_template is enabled, so a gate that flickers re-runs the engine's template
# decommission pass. Set by WA_AI_TEMPLATES_update_motorized_hospital_latch.
WA_AI_TEMPLATES_can_motorize_field_hospital = {
	has_country_flag = WA_AI_TEMPLATES_motorized_hospital_earned
}

# [suppression-templates] occupation garrisons are only worth designing when there is (or can soon
# be) something to occupy: at war, or holding non-core ground - `always = yes` had every neutral
# minor spending its army XP on a cavalry garrison template it fielded to the front instead.
WA_AI_TEMPLATES_use_suppression_templates = {
	OR = {
		has_war = yes
		any_controlled_state = {
			NOT = { is_core_of = ROOT }
		}
		# latch: once the target was assigned, keep it - clearing the flag mid-campaign re-runs
		# the engine's template decommission pass over whatever holds the role (lessons log).
		has_country_flag = WA_SUPPRESSION_TEMPLATE
	}
}

	NOT = { WA_AI_CONFIG_DIVISIONS_force_mountainneers_instead_of_marines = yes }

# If you ever want to bring more countries into armour/motorised/mechanised production, LOWER the
# num_of_military_factories thresholds below. Do not remove the OR branches, and do not replace a gate
# with `always = yes` - these gates are read well beyond production:
#
#	use_armor_templates       -> 8 research triggers in WA_AI_RESEARCH_tanks.txt
#	use_mechanized_templates  -> template design in WA_AI_TEMPLATES_effects.txt, WA_AI_PRODUCTION_tanks.txt
#	use_motorized_templates   -> template design in WA_AI_TEMPLATES_effects.txt
#
# Making any of them unconditionally true would put every country in the world on tank research.
#
# Likewise, do NOT widen the WA_AI_CONFIG_DIVISIONS_use_*_divisions triggers: those are country identity
# markers that drive army HQ templates (uses_motorized_hq -> history/general/taog_hq_template.txt) and
# research weighting. Tune the thresholds here instead.
#
# EXCEPTION - motorised DIVISIONS have no threshold to tune: use_motorized_templates below is pure
# CONFIG identity ([mot-divisions-config-only]). Bring a country in by marking it in CONFIG.
############################################################################################################

# [mot-divisions-config-only] Motorised DIVISIONS are a CONFIG identity, never an industrial consequence:
# the only way in is WA_AI_CONFIG_DIVISIONS_use_motorized_divisions. Trucks elsewhere are untouched -
# motorised SUPPORT runs off WA_AI_CONFIG_DIVISIONS_can_motorize_support, the motorised battalions inside
# tank divisions off WA_AI_TEMPLATES_use_armor_templates, supply-hub motorization off the engine.
# The role_ratio share reads this same trigger, so target and share open and close together.
WA_AI_TEMPLATES_use_motorized_templates = {
	WA_AI_TEMPLATES_has_motorized_unlocked = yes
	WA_AI_CONFIG_DIVISIONS_use_motorized_divisions = yes
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
}

# [mot-tech-folder] "This country has trucks unlocked at all". The OR is mandatory, not defensive:
# `motorised_infantry` sits in minor_armour_folder, and the 11 countries with a NATIONAL armour folder
# hold a <tag>_motorised_infantry instead and can never research the generic one. On the bare generic
# form this was false forever for GER/SOV/USA/ENG/FRA/ITA/JAP/CZE/HUN/POL/SWE - the tell is a major with
# no WA_MOTORIZED_TEMPLATE flag. HOL is absent on purpose: its tech is commented out in armor_hol.txt.
WA_AI_TEMPLATES_has_motorized_unlocked = {
	OR = {
		has_tech = motorised_infantry
		has_tech = cze_motorised_infantry
		has_tech = eng_motorised_infantry
		has_tech = fra_motorised_infantry
		has_tech = ger_motorised_infantry
		has_tech = hun_motorised_infantry
		has_tech = ita_motorised_infantry
		has_tech = jap_motorised_infantry
		has_tech = pol_motorised_infantry
		has_tech = sov_motorised_infantry
		has_tech = swe_motorised_infantry
		has_tech = usa_motorised_infantry
	}
}

# [mech-window] "This country mechanizes at all" - tech, industry, design affordability, NO timing
# term. This is the whole of what use_mechanized_templates was before the window was added.
# Read by the three WA_AI_PRODUCTION_mech_min_factories_* FLOORS (WA_AI_PRODUCTION_tanks.txt), which
# must NOT see the window: a few factories build a mech buffer DURING it so the flip does not land on
# a stock of 0. The production WEIGHTS stay on the windowed form - weight late, floor early.
# ([mot-divisions-config-only] dropped the second reader, use_motorized_templates.)
WA_AI_TEMPLATES_mechanization_line_open = {
	WA_AI_TEMPLATES_has_mechanized_unlocked = yes
	OR = {
		WA_AI_CONFIG_DIVISIONS_use_mechanized_divisions = yes
		num_of_military_factories > 100
	}
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
	# A country with a penalised army XP income cannot afford a mechanized design
	# alongside its infantry and armor designs before the war economy matures.
	# Without this, SOV picked up WA_MECHANIZED_TEMPLATE in 1939 off the mil-factory
	# branch and spent XP it needed for its rifle division on a mech design it never
	# fielded.
	OR = {
		WA_AI_CONFIG_after_global_war_begins = yes
		NOT = { WA_AI_TEMPLATES_has_penalised_army_xp_gain = yes }
	}
}

# True when the country's army XP income is structurally penalised.
# Reads the aggregate country modifier, so it covers national spirits, dynamic
# modifiers and advisors alike, and it closes again by itself once the penalty is
# lifted. Used to keep "nice to have" template roles shut for countries that cannot
# fund a second design on top of their line infantry.
# Catches SOV (purge + red army), CHI/PRC, MAN, JAP, NZL, TUR, BUL, BEL.
# Does NOT catch a normal GER/ENG/USA/FRA run - their negative XP spirits are
# civil-war or transient-only.
WA_AI_TEMPLATES_has_penalised_army_xp_gain = {
	OR = {
		check_variable = {
			var = modifier@experience_gain_army_factor
			value = -0.05
			compare = less_than
		}
		check_variable = {
			var = modifier@experience_gain_factor
			value = -0.05
			compare = less_than
		}
	}
}

# [mech-window] "May this country's armour templates carry mechanized line battalions NOW". The
# INDUSTRIAL path is timing-gated, the identity path is not (see the exemption note on
# WA_AI_CONFIG_DIVISIONS_use_mechanized_divisions). Flipping this swaps every motorised line
# battalion in the light / medium / heavy ladders for a mechanized one across the whole fielded
# force at once - the templates carry can_upgrade_in_field - so GER, which clears 100 military
# factories on day one (104 in 1936), mechanized the month its 1938 tech landed. The gate is read
# through its latch, never live: WA_AI_TEMPLATES_update_mechanization_latch.
WA_AI_TEMPLATES_use_mechanized_templates = {
	WA_AI_TEMPLATES_mechanization_line_open = yes
	OR = {
		WA_AI_CONFIG_DIVISIONS_use_mechanized_divisions = yes
		has_country_flag = WA_AI_TEMPLATES_mechanization_earned
	}
}

# [mech-divisions-usa-only] Owner rule 2026-08-28: only the expeditionary mechanized major fields
# mechanized DIVISIONS. Deliberately NARROWER than WA_AI_TEMPLATES_use_mechanized_templates, which
# stays open for everyone and must not be touched: 44 call sites in WA_AI_TEMPLATES_effects.txt read
# it to pick the MECHANIZED variant of every armour template, and the mechanized_equipment factory
# floors hang off it. Narrowing that one would strip the mech battalions out of GER/SOV tank
# divisions - the exact opposite of the rule. This trigger gates only the mech division target
# template and its role_ratio share.
WA_AI_TEMPLATES_use_mechanized_division_templates = {
	WA_AI_TEMPLATES_use_mechanized_templates = yes
	WA_AI_CONFIG_DIVISIONS_is_expeditionary_mechanized_major = yes
}

# [historical-tank-park] The ordinary handoff closes this role only after the mission's live run
# state ends. Before the handoff, the role remains available so a country without a medium target
# cannot lose all armour templates.
# [armor-class-handoff] This trigger LOST a tag term when WA_AI_CONFIG_TEMPLATES_focus_on_light_armor
# was deleted, so the run is open to any country holding a light-support chassis, not only the five
# tags that term listed. That widening is the intent already stated by
# WA_AI_TEMPLATES_light_support_armor_owns_light_role below: the chassis techs are country-specific
# (sov_light_tank_support_chassis_*), and a second country researching one takes the light slot the
# same way. If a generic support chassis is ever added, this becomes a real behaviour change.
WA_AI_TEMPLATES_use_light_support_armor_templates = {
	OR = {
		AND = {
			WA_AI_TEMPLATES_use_armor_templates = yes
			WA_AI_TEMPLATES_has_light_support_armor_unlocked = yes
			WA_AI_CONFIG_before_global_war_begins = yes
			NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
			OR = {
				NOT = { WA_AI_TEMPLATES_switch_from_light_to_medium_armor = yes }
				WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
			}
		}
		AND = {
			WA_AI_CONFIG_pursues_historical_tank_park = yes
			WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
			has_country_flag = WA_AI_TEMPLATES_sov_light_support_temporary_latched
		}
	}
}

# templates declare role = light_armor, so without this the engine arbitrates two enabled templates

# 1942 closes the light slot outright instead of falling back to the plain light template - which
# is what lets the engine retire the light-support divisions (a template captured by a role the AI
# wants zero of is decommissioned: WA_AI_TEMPLATES_garrison.txt header).

	WA_AI_TEMPLATES_has_light_support_armor_unlocked = yes

# [armor-class-handoff] A HANDOFF, not a cliff: this closes the light template, so it may not fire
# before the medium class can actually open. `has_medium_armor_unlocked` is that guarantee - without
# it a country whose medium tech never arrives loses light at the era boundary and holds no armour
# template at all. Assumes the medium ladder admits everyone past the boundary
# (WA_AI_CONFIG_TEMPLATES_admits_medium_armor); if that fallback is ever removed, this term
# stops being sufficient and the gap returns.
WA_AI_TEMPLATES_switch_from_light_to_medium_armor = {
	WA_AI_TEMPLATES_use_armor_templates = yes

	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
}

# [armor-class-handoff] The DIVISION half of the handoff: the boundary closes the light class,
# but the fielded light divisions still need a path INTO the medium class. This window keeps the
# LIGHT_MEDIUM transition templates selectable after the switch (their replace_with flips each
# division onto the medium template at 0.8 match), so conversion replaces abandonment. Assumes
# the engine still upgrades a role whose role_ratio want is 0; false = divisions freeze as before.
WA_AI_TEMPLATES_should_convert_light_to_medium_armor = {
	WA_AI_TEMPLATES_switch_from_light_to_medium_armor = yes
	WA_AI_TEMPLATES_has_light_armor_unlocked = yes
	WA_AI_TEMPLATES_use_medium_armor_templates = yes
	NOT = { WA_AI_TEMPLATES_light_support_armor_owns_light_role = yes }
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
}

# [armor-class-handoff] The medium role's SELECTED template sits in the 6/3/6 inf-support
# family (values 6105-6110 all share that line composition). Read by the light ladder's MIS

# [historical-tank-park] The light-support twin of the conversion window above (which excludes
# the light-support country by design): when the finite run ends, the fielded park converts into
# the medium class instead of decommissioning. The latch flag is the memory that this country ran
# the park; the majority term retires the window together with the park itself. Same engine
# assumption as the light window: the engine still upgrades a role whose role_ratio want is 0.
WA_AI_TEMPLATES_should_convert_light_support_to_medium_armor = {
	# [light-support-conversion] Two memories of a run: the composition latch (15000/15001 path)
	# and the temporary-corps latch (15006 path). The temporary path never writes the first one,
	# so without the second the window was unreachable for the historical park.
	OR = {
		has_country_flag = WA_AI_TEMPLATES_light_support_composition_latched
		has_country_flag = WA_AI_TEMPLATES_sov_light_support_temporary_latched
	}
	NOT = { WA_AI_TEMPLATES_use_light_support_armor_templates = yes }
	WA_AI_TEMPLATES_use_medium_armor_templates = yes
	# Park still fielded. The temporary corps fields 6 support / 6 mot shapes where the majority
	# vote ties, so that path retires on the containing test instead (same-role templates only,
	# no infantry template carries a support battalion after 1941).
	OR = {
		has_template_ai_majority_unit = light_support_armor_battalion_line
		AND = {
			has_country_flag = WA_AI_TEMPLATES_sov_light_support_temporary_latched
			has_template_containing_unit = light_support_armor_battalion_line
		}
	}
	# The temporary corps stays on its mission target while the mission counts it.
	NOT = {
		AND = {
			has_country_flag = WA_AI_TEMPLATES_sov_light_support_temporary_latched
			WA_AI_PRODUCTION_historical_tank_park_mission_is_active = yes
		}
	}
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
}

# Gates the infantry / marines / suppression target templates
# (WA_AI_TEMPLATES_calculate_all_templates). Without it a country holds no WA_*_TEMPLATE flag,
# the only enabled ai_template is WA_infantry_fallback whose ROLE-level upgrade_prio is 0, and
# its 1936 templates never change again - measured on 320 of 334 countries in campaign 2f8cbd51.
#
# [templates-admission] admission gate into the AI template system. Gate on the hidden TECH each
# country-tree design focus sets, never the paired spirit (strippable by
# WA_AI_TEMPLATES_remove_wrong_army_spirits) and never a focus list (misses trees, e.g. RAJ).
# Shared-tree entries grant NO free-design spirit on purpose (it would buff human players).
# The deadline flag admits countries whose design focus is unreachable - they pay full XP.
WA_AI_TEMPLATES_has_infantry_focus_completed = {
	OR = {
		has_idea = GER_okw # RCZ garrison country (ger_armor.999) - no focus tree, joins the template system via the OKW idea

		has_tech = infantry_modernization_tech # set by the free-design focus of every country tree

		# Shared / generic focus trees - target template only, no free-design spirit.
		has_completed_focus = army_effort # generic.txt - 49 countries had it in campaign 2f8cbd51
		has_completed_focus = SOUTH_AMERICA_expand_army # south_america_generic.txt - 19 countries
		has_completed_focus = NORDIC_nordic_army # nordic_shared.txt - joint tree, never reached in that campaign
		has_completed_focus = BALTIC_arm_the_nation # baltic_shared.txt - same, kept so the path exists off the historical rails

		has_country_flag = WA_AI_TEMPLATES_design_deadline_passed
	}
}

	}
}

WA_AI_TEMPLATES_has_required_spirits = {
	# Check army_spirit - must be the SPECIFIC ideology-correct one
	OR = {
		AND = {
			has_government = democratic
			has_idea = relief_of_command_spirit
		}
		AND = {
			OR = {
				has_government = fascism
				has_government = neutrality
			}
			has_idea = professional_officer_corps_spirit
		}
		# Italy fallback: accept state_serves_the_military until reforms are done
		AND = {
			tag = ITA
			has_government = fascism
			NOT = { has_country_flag = adequate_army }
			has_idea = state_serves_the_military_spirit
		}
		AND = {
			has_government = communism
			has_idea = ideological_loyalty_spirit
		}
	}
	# Check academy_spirit - ANY academy spirit is valid
	WA_AI_TEMPLATES_has_any_academy_spirit = yes
}
