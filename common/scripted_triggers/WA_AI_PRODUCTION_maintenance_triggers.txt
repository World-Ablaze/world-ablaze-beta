# ============================================================================
# MAINTENANCE FLOORS - keeping a fielded park supplied when nothing else does
# ============================================================================
# [maintenance-floor] A small factory floor so a park that the AI has stopped expanding still
# receives new equipment, and therefore still MODERNISES: at zero factories a mature park drives
# the chassis and fires the guns it had when the line went quiet, for the rest of the campaign.
# Design and measurements: documentation/WA_AI_MAINTENANCE_FLOOR_PROPOSAL.md.
#
# TWO REASONS A PARK GOES UNSUPPLIED, and they need different gates:
#   ARMOUR   surplus can collapse engine demand even while the class remains fielded. Every tank
#            archetype - the four main gun classes AND every variant chassis - therefore keeps a
#            flat five-factory floor while armies require it.
#   GROUND   nothing closes anything - anti-tank, AA, heavy AA, heavy and pack artillery have no
#            line to close and, before this file, no factory floor anywhere in the DEFAULT tier,
#            so the engine's need heuristic alone decided whether they were built. Their gate is
#            the shortage and nothing else. Amtrac is the exception: its line DOES close (peace),
#            so it carries a closure term in addition to the shortage gate.
#
# THE SUPPORT-EQUIPMENT SHORTAGE IS A RATIO, never an absolute stock: free spares over what the armies REQUIRE
# (`num_target_equipment_in_armies_k@<archetype>`, engine variable, documented in the 1.19.2
# install). It scales itself with army size, so no per-archetype stock bar has to be guessed. The
# `if / else = { always = no }` wrapper is the division-by-zero guard and is the shape Expert AI 5.0
# uses for the same computation (EAI_PROD_EQUIP_REDUCE_infantry_equipment).
#
# For support equipment, `_short` ARMS the floor and `_supplied` RELEASES it, at two different bars - a Schmitt pair read by
# the `enable` and `abort` blocks of common/ai_strategy/WA_AI_PRODUCTION_DEFAULT_maintenance.txt.
# `abort` is NOT the negation of `enable`: it must also fire when a closure term comes back, or a
# reopened line would keep its floor. That is why every `_should_hold_*` repeats the closure terms.
#
# [armor-maintenance] Owner ruling 2026-09-19 supersedes the old category-push exclusion: the
# factor raises demand, while this floor guarantees a minimum after captured or inherited stocks
# suppress that demand. Mine covers the old double-count objection because a minimum does not add
# five factories above an allocation already at five or more; these blocks are the only chassis
# floors in the DEFAULT tier.
#
# ALSO FLOORED ELSEWHERE, and floors on one id SUM. WA_AI_PRODUCTION_lend_lease.txt floors the
# `anti_tank`, `anti_air` and `artillery` types on donors: those countries maintain deeper, and the
# harness prints it. [minor-gun-floor] WA_AI_PRODUCTION_DEFAULT_ground.txt carries a BASELINE floor
# of 1 on `anti_tank_equipment` (>= 10 mils, templates demand it) under these shortage ones, so the
# anti-tank totals are 2 while short and 4 deep, not 1 and 3.
# No `equipment_production_min_factories` on the `armor` TYPE exists anywhere: `armor` is
# category-push-only, per [armor-prod-category].
# ============================================================================

##################### ARMOUR CHASSIS - PRODUCTION PERMISSION #####################

# OBSERVATION - the engine will actually open a tank line for me.
# [armor-maintenance] `can_be_produced` on every chassis archetype of
# common/units/equipment/tank_chassis.txt bans the line outright under either treaty idea
# (MEASURED: 35 of the 36 archetypes carry the term). A floor is need-blind and cannot override a
# production ban, so without this term a treaty-bound country reserves factories for a line that
# never opens. The fact is gone when the treaty ideas stop appearing in those `can_be_produced`
# blocks - regrep before trusting this trigger.
WA_AI_PRODUCTION_can_produce_armour_chassis = {
	NOT = { has_idea = BUL_treaty_of_neuilly_sur_seine }
	NOT = { has_idea = HUN_treaty_of_trianon }
}

##################### light_tank (light_tank_chassis) #####################

	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_PRODUCTION_can_produce_armour_chassis = yes
	NOT = { has_country_flag = AI_purge_light_tank_chassis_production }
	num_of_military_factories > constant:wa_ai_production.industry.wartime_min_mils

# archetype: tank destroyer, SP artillery, SP rocket, assault gun, SPAA, infantry-support and
# support tank, plus the super-heavy family, the amphibious tank and the two armoured cars.

# so the light_tank_chassis floor above never reached it and the line ran on perceived need alone.
# Each gate is self-limiting: `num_target_equipment_in_armies_k@<arch>` is 0 for a country whose
# templates never ask for that chassis, so no floor is owed and none arms. The fact is gone when a
# country whose divisions clearly carry the variant reads require=0 in the console harness.

##################### amtrac (amphibious_mechanized_equipment) #####################

# OBSERVATION - free spares under the ARM bar, as a share of what the armies require.
WA_AI_PRODUCTION_is_amtrac_short = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies_k@amphibious_mechanized_equipment > constant:wa_ai_production.maintenance.target_min_k } }
		set_temp_variable = { _mnt_r_amtrac = num_equipment@amphibious_mechanized_equipment }
		divide_temp_variable = { _mnt_r_amtrac = 1000 }
		divide_temp_variable = { _mnt_r_amtrac = num_target_equipment_in_armies_k@amphibious_mechanized_equipment }
		check_variable = { _mnt_r_amtrac < constant:wa_ai_production.maintenance.shortage_enable }
	}
	else = { always = no }
}

# OBSERVATION - spares back above the RELEASE bar. The other half of the Schmitt pair.
WA_AI_PRODUCTION_is_amtrac_supplied = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies_k@amphibious_mechanized_equipment > constant:wa_ai_production.maintenance.target_min_k } }
		set_temp_variable = { _mnt_r_amtrac = num_equipment@amphibious_mechanized_equipment }
		divide_temp_variable = { _mnt_r_amtrac = 1000 }
		divide_temp_variable = { _mnt_r_amtrac = num_target_equipment_in_armies_k@amphibious_mechanized_equipment }
		check_variable = { _mnt_r_amtrac > constant:wa_ai_production.maintenance.shortage_abort }
	}
	else = { always = no }
}

# DECISION - arm the floor.
WA_AI_PRODUCTION_should_maintain_amtrac = {
	WA_AI_PRODUCTION_ground_is_enabled = yes
	NOT = { WA_AI_PRODUCTION_build_amphibious_mech = yes }
	num_of_military_factories > constant:wa_ai_production.industry.wartime_min_mils
	WA_AI_PRODUCTION_is_amtrac_short = yes
}
