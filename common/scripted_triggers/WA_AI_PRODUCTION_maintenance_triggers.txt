# ============================================================================
# MAINTENANCE FLOORS - keeping a fielded park supplied when nothing else does
# ============================================================================
# [maintenance-floor] A small factory floor so a park that the AI has stopped expanding still
# receives new equipment, and therefore still MODERNISES: at zero factories a mature park drives
# the chassis and fires the guns it had when the line went quiet, for the rest of the campaign.
# Design and measurements: documentation/WA_AI_MAINTENANCE_FLOOR_PROPOSAL.md.
#
# TWO REASONS A PARK GOES UNSUPPLIED, and they need different gates:
#   ARMOUR   WA itself closes the lines (the class ladder, the era boundary, a fielded cap). The
#            four chassis keys below therefore carry the negation of their own build decision.
#   GROUND   nothing closes anything - anti-tank, AA, heavy AA, heavy and pack artillery have no
#            line to close and, before this file, no factory floor anywhere in the DEFAULT tier,
#            so the engine's need heuristic alone decided whether they were built. Their gate is
#            the shortage and nothing else. Amtrac is the exception: its line DOES close (peace),
#            so it carries a closure term like the armour keys.
#
# THE SHORTAGE IS A RATIO, never an absolute stock: free spares over what the armies REQUIRE
# (`num_target_equipment_in_armies_k@<archetype>`, engine variable, documented in the 1.19.2
# install). It scales itself with army size, so no per-archetype stock bar has to be guessed. The
# `if / else = { always = no }` wrapper is the division-by-zero guard and is the shape Expert AI 5.0
# uses for the same computation (EAI_PROD_EQUIP_REDUCE_infantry_equipment).
#
# `_short` ARMS the floor, `_supplied` RELEASES it, at two different bars - a Schmitt pair read by
# the `enable` and `abort` blocks of common/ai_strategy/WA_AI_PRODUCTION_DEFAULT_maintenance.txt.
# `abort` is NOT the negation of `enable`: it must also fire when a closure term comes back, or a
# reopened line would keep its floor. That is why every `_should_hold_*` repeats the closure terms.
#
# WHY THE CATEGORY PUSH IS NEGATED ON THE ARMOUR KEYS. `WA_AI_PRODUCTION_armor_category_push`
# raises the engine's perceived NEED for armour; a floor FORCES allocation regardless of need. The
# two together double-count, which is why the per-chassis floors were dropped once before
# (`[armor-prod-category]`, WA_AI_PRODUCTION_DEFAULT_tanks.txt). Negating it makes them mutually
# exclusive by construction: this floor can only run while that lever has nothing to multiply.
#
# ALSO FLOORED ELSEWHERE, and floors on one id SUM: `minor_unit_production` (POL/HUN/SWE) and the
# CZE historical plan floor the `armor` TYPE; WA_AI_PRODUCTION_lend_lease.txt floors the `anti_tank`,
# `anti_air` and `artillery` types on donors. Those countries maintain deeper. The harness prints it.
# ============================================================================

##################### light_tank (light_tank_chassis) #####################

# OBSERVATION - free spares under the ARM bar, as a share of what the armies require.
WA_AI_PRODUCTION_is_light_tank_short = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies_k@light_tank_chassis > constant:wa_ai_production.maintenance.target_min_k } }
		set_temp_variable = { _mnt_r_light_tank = num_equipment@light_tank_chassis }
		divide_temp_variable = { _mnt_r_light_tank = 1000 }
		divide_temp_variable = { _mnt_r_light_tank = num_target_equipment_in_armies_k@light_tank_chassis }
		check_variable = { _mnt_r_light_tank < constant:wa_ai_production.maintenance.shortage_enable }
	}
	else = { always = no }
}

# OBSERVATION - spares back above the RELEASE bar. The other half of the Schmitt pair.
WA_AI_PRODUCTION_is_light_tank_supplied = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies_k@light_tank_chassis > constant:wa_ai_production.maintenance.target_min_k } }
		set_temp_variable = { _mnt_r_light_tank = num_equipment@light_tank_chassis }
		divide_temp_variable = { _mnt_r_light_tank = 1000 }
		divide_temp_variable = { _mnt_r_light_tank = num_target_equipment_in_armies_k@light_tank_chassis }
		check_variable = { _mnt_r_light_tank > constant:wa_ai_production.maintenance.shortage_abort }
	}
	else = { always = no }
}

# DECISION - arm the floor.
WA_AI_PRODUCTION_should_maintain_light_tank = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	NOT = { WA_AI_PRODUCTION_armor_category_push = yes }
	NOT = { WA_AI_PRODUCTION_build_light_armor = yes }
	num_of_military_factories > constant:wa_ai_production.industry.wartime_min_mils
	WA_AI_PRODUCTION_is_light_tank_short = yes
}

# DECISION - hold it. Read NEGATED by the ai_strategy `abort` block, so it repeats every term of
# the enable: a closure term coming back must retire the floor, not only a refilled stockpile.
WA_AI_PRODUCTION_should_hold_maintain_light_tank = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	NOT = { WA_AI_PRODUCTION_armor_category_push = yes }
	NOT = { WA_AI_PRODUCTION_build_light_armor = yes }
	num_of_military_factories > constant:wa_ai_production.industry.wartime_min_mils
	NOT = { WA_AI_PRODUCTION_is_light_tank_supplied = yes }
}

# DECISION - the deep half of the pair, above the large industry band.
WA_AI_PRODUCTION_should_deepen_light_tank_maintenance = {
	WA_AI_PRODUCTION_should_maintain_light_tank = yes
	num_of_military_factories > constant:wa_ai_production.industry.tier_large_mils
}

WA_AI_PRODUCTION_should_hold_deepen_light_tank_maintenance = {
	WA_AI_PRODUCTION_should_hold_maintain_light_tank = yes
	num_of_military_factories > constant:wa_ai_production.industry.tier_large_mils
}
