WA_AI_PRODUCTION_is_using_default_system = {
	always = yes
}

# A large naval producer becomes the coalition's convoy arsenal while at least one eligible
# AI partner is below that partner's dynamic convoy reserve. THIS = producer; inside
# any_country, THIS = recipient and PREV = producer, matching convoy_pair_is_valid's contract.
# [convoys] the coalition must actually depend on sea lanes before a major spends dockyards
# building hulls for it. WA_AI_PRODUCTION_coalition_sea_share is the share of the SIDE's industry
# (civ + mil + naval factories) that this country cannot reach OVERLAND, recomputed monthly by
# WA_AI_PRODUCTION_update_coalition_sea_weight (WA_production_strategy_effects.txt).
#
# WEIGHTED, not existential (user ruling 2026-08-20): "does the side span two continents" is
# overturned by any minor - the Axis of campaign 3d68a183 already reached Africa through ITS and
# ITL, two colonial subjects with negligible industry. The share asks how much of the alliance's
# WEIGHT is across water.
#
# LAND ACCESS, not continents ([convoys], same day, before any campaign ran the first version):
# continents do not measure water in either direction - London and Berlin are both `europe` with the
# Channel between them, and Berlin reaches Cairo overland across two of them.
#
# This gate also decides the stock brake: WA_DEFAULT_production_convoy_stop_MAJORS carries
# NOT = { this trigger } in both enable and abort, so a land coalition's arsenal never opens AND
# its >1000 brake arms normally, while a maritime coalition keeps both the boost and the exemption.
WA_AI_PRODUCTION_should_build_convoy_relief = {
	WA_AI_PRODUCTION_is_using_default_system = yes
	is_major = yes
	num_of_naval_factories > constant:wa_ai_lend_lease.convoy.dockyards_large
	check_variable = { WA_AI_PRODUCTION_coalition_sea_share > constant:wa_ai_lend_lease.convoy.coalition_sea_share_bar }
	any_country = {
		WA_AI_LEND_LEASE_is_starving_convoys = yes
		WA_AI_LEND_LEASE_convoy_pair_is_valid = yes
	}
}

# [convoys] (2026-08-21, campaign 8c0fea4c, checklist R87) - a minor whose dockyards have genuinely
# nothing to build. Owner ruling: build convoys by default in that case.
#
# READ THIS BEFORE REUSING THE MEASUREMENT. The campaign showed 36 idle dockyard levels across the
# five Commonwealth dominions at 1941.6 (RAJ 1 active factory of 8 levels, AST 1/6, CAN 1/16,
# NZL 1/7, SAF 1/4, against ENG 81/88). The FIRST version of this fix blamed the convoy ladder in
# WA_AI_PRODUCTION_DEFAULT_navy.txt - tier 0 stopping at 200 convoys, tiers 1-3 needing 20
# shipyards, stop_MINORS braking above 300. **That was wrong**, and the closure test is what killed
# it: every minor on the GENERIC naval tech tree faces the identical ladder and saturates its yards
# anyway - SWE 17 factories on 16 dockyard levels, TUR 21/20, ARG 6/5, BRA 6/5. The real cause was
# that the dominions run the BRITISH tech tree and had no matching ship DESIGN ([commonwealth-handoff],
# common/ai_equipment/ENG_naval.txt). Do not re-derive the ladder story from the same numbers.
#
# What survives is the narrow, real case this trigger now names: a minor that is not a naval power,
# is not the majors' coalition arsenal, is at war and has yards - once its escort programme is
# satisfied, its spare dockyards should make convoys for the side rather than nothing. The payload
# is `equipment_production_surplus_management` alone, which by construction only spends what
# nothing else wants; the forced `equipment_production_min_factories` of the first version was
# removed with [commonwealth-handoff] because it would have outbid the escorts.
#
# Also gates the [convoys] exemption inside WA_DEFAULT_production_convoy_stop_MINORS, so a country in
# this state is not simultaneously told to stop building convoys.
# Assumes: a country that is not a major naval power has no warship programme large enough to want
# every slipway. Gone when: WA gives minors a real capital-ship role.
WA_AI_PRODUCTION_dockyards_have_nothing_to_build = {
	WA_AI_PRODUCTION_is_using_default_system = yes
	has_war = yes
	num_of_naval_factories > 3
	NOT = { WA_AI_MILITARY_has_ocean_going_fleet = yes }
	NOT = { WA_AI_PRODUCTION_should_build_convoy_relief = yes }
}

# Decision: may this major lay down capital ships now? Historical mode holds capitals back for
# the AI Capital Ship System (events/WA_AI_Capitals.txt) until the treaty era ends - by treaty
# status (live), or by the calendar fallback WA_AI_CONFIG_naval_treaty_era_expired.
WA_AI_PRODUCTION_should_build_capitals = {
	WA_AI_CONFIG_is_major_country = yes
	OR = {
		WA_AI_DIFFICULTY_is_historical = no
		AND = {
			WA_AI_DIFFICULTY_is_historical = yes
			OR = {
				WA_AI_CONFIG_naval_treaty_era_expired = yes
				any_country_of = {
					target = { ENG FRA GER ITA JAP SOV USA}
					has_naval_treaty_trigger = yes
				}
			}
		}
	}
}
