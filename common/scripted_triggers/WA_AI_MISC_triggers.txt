# true neutrals
WA_AI_neutral_country = {
	NOT = { has_war = yes }
	OR = { has_government = neutrality has_government = democratic }
	OR = {
		tag = BEL
		tag = SWI
		tag = SWE
		tag = IRE
		tag = TIB
		tag = NEP
		tag = BHU
		tag = POR
	}
}

# triggers the AI to build more units when threatened
WA_AI_mobilize_army = {
	NOT = { has_war = yes WA_AI_CONFIG_is_industrial_power = yes }

	OR = {
		has_country_flag = WA_AI_mobilize_army

		any_neighbor_country = {

			capital_scope = { has_state_flag = WA_AI_original_capital_@PREV }

			any_of_scopes = { array = controlled_states
				is_in_home_area = yes
				any_neighbor_state = {
					is_controlled_by = ROOT
					is_in_home_area = yes
				}
			}

			OR = {
				has_added_tension_amount > 10
				has_war = yes
				is_justifying_wargoal_against = ROOT
				has_wargoal_against = ROOT
				any_other_country = {
					OR = {
						is_justifying_wargoal_against = PREV
						has_wargoal_against = PREV
					}
				}
			}
		}

		NOT = {
			OR = {
				check_variable = { ROOT.reserves = 10 }
				check_variable = { ROOT.reserves = 20 }
				check_variable = { ROOT.reserves = 30 }
				check_variable = { ROOT.reserves = 40 }
				check_variable = { ROOT.reserves = 50 }
				check_variable = { ROOT.reserves = 60 }
				check_variable = { ROOT.reserves = 70 }
				check_variable = { ROOT.reserves = 80 }
				check_variable = { ROOT.reserves = 90 }
				check_variable = { ROOT.reserves = 100 }
			}
		}

		### special cases

		original_tag = CHI
	}
}

WA_AI_defend_core = {
	OR = {
		surrender_progress > 0.1

		AND = { # guard cairo, singapore
			OR = { original_tag = EGY original_tag = MAL original_tag = UKE }
			is_in_faction_with = ENG
		}

		AND = { # keep units on the east front
			OR = { tag = ROM tag = HUN tag = FIN }
			is_neighbor_of = SOV
			has_war_with = SOV
		}
	}
}

WA_AI_pp_advisor_prio = {
	if = {
		limit = {
			NOT = {
				tag = CHI
			}
		}
		WA_AI_CONFIG_before_1938 = yes
	}
	if = {
		limit = {
			tag = CHI
		}
		date < 1937.6.1
	}
	has_allowed_idea_with_traits  = { idea = silent_workhorse limit = 1 }
	NOT = { has_idea_with_trait = silent_workhorse }
}

WA_wait_for_barb = {
	if = {
		limit = {
			SOV = { is_ai = yes }
		}
		OR = { 
			has_war = yes
			AND = {
				has_war = no
				date > 1940.10.1
				date < 1941.2.1
			}
			AND = {
				has_war = no
				date > 1941.10.1
				date < 1942.2.1
			}
			AND = {
				has_war = no
				date > 1942.10.1
				date < 1943.2.1
			}
			date > 1944.1.1
			NOT = {
				GER = {
					focus_progress  = {
						focus = GER_eastern_lebensraum
						progress > 0
					}
				}
			}
			NOT = {
				GER = { has_completed_focus = GER_eastern_lebensraum }
			}
		}
	}
}
