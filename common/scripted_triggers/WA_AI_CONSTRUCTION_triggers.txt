### available construction slots

# [refineries] per-building state caps for the shared-slot buildings ([pc-queue] added CIC/MIC/NIC).
# Protects: `free_building_slots = { building = X size > 0 }` reports only the SHARED pool, never
# X's own `level_cap.state_max` - past the cap a state stays "available" forever,
# `add_building_construction` silently no-ops, and because that state also stays #1 in
# WA_AI_shared_slot_scores the whole building type stalls country-wide.
# Test = `built < cap` AND `committed < cap`, committed = built + queued, held as the state variable
# WA_AI_committed_<X> by WA_AI_add_<X> (building_adders). Rejected alternatives: a built-level-only
# test reads "free" for all the months a full queue takes to finish; tracking queued separately needs
# a reconcile cadence, while the SUM needs none (a completion just moves a level queued -> built).
# `committed` is trusted only while the state flag WA_AI_committed_<X>_recent is live (TTL in
# building_adders): once it lapses nothing can still be queued, `built < cap` alone decides, and the
# next add rebases the sum - the only path by which a CANCELLED construction is forgotten, since
# script cannot see the construction queue.
# MUST MATCH `level_cap.state_max` in common/buildings/00_buildings.txt.
@WA_AI_REF_STATE_MAX = 6
@WA_AI_SR_STATE_MAX = 15
@WA_AI_HSR_STATE_MAX = 15
@WA_AI_AR_STATE_MAX = 15
@WA_AI_HAR_STATE_MAX = 15
# [pc-queue] the three factory types join the same standard (industrial_complex / arms_factory / dockyard).
@WA_AI_CIC_STATE_MAX = 45
@WA_AI_MIC_STATE_MAX = 45
@WA_AI_NIC_STATE_MAX = 45
# [silo-breadth] fuel_silo joins the same standard - it shares slots and has the lowest cap of all,
# so the uncapped test stalled the type country-wide the moment the top-scored state held 6.
@WA_AI_SILO_STATE_MAX = 6

# THIS = state, ROOT = builder. [na-corridor] is this state on the CONTESTED EDGE of its theatre -
# close enough to a front that an air base built here can reach it?
# Protects: an air theatre is a geographic block (north_africa = Morocco..Egypt, 32 states) and
# WA_AI_build_theatre_air_bases selects inside it on air-base level alone, so without this it parks
# projects in the deep rear while front-line states run over capacity.
# Two rings, not one: ring 1 (states touching enemy ground) can be two states on a narrow front,
# both possibly ineligible, and the caller's fallback then hands the pulse back to the rear anyway;
# ring 2 adds the immediate rear that still bases planes over the front, and leaves Morocco (3+ rings
# from Tunisia) out, which is the whole point. Controller geography only - no tag, date or faction
# term; `has_war_with = ROOT` is the same enemy test the AIR theatre-contest triggers use
# (WA_AI_MILITARY_triggers.txt), so "contested" and "near the contested edge" cannot drift apart.
# Cost: ASSUMED that PDXScript evaluates an AND block in order and stops at the first false, so the
# caller placing this LAST in its limit block runs it on a handful of states per rung. If that is
# wrong it is two neighbour rings over the whole map per rung per pulse per builder; the tell is a
# frame-time regression tracking the theatre-air pulse, and the fix is to precompute the rings into a
# state array, not to reorder the block.
# Fact is gone when: an air theatre is redefined small enough that every member state is within two
# rings of a front - then this trigger is true everywhere and selects nothing.
WA_AI_PC_state_is_near_contested_edge = {
	OR = {
		any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
		any_neighbor_state = {
			any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
		}
	}
}

### [pc-queue] per-line depth cap for the breadth-first selection (WA_AI_CONSTRUCTION_queue_functions.txt).
# THIS = state. True when fewer than @WA_AI_QUEUE_DEPTH_<X> levels of X are queued-but-unbuilt here,
# read as WA_AI_committed_<X> - building_level@X ([refineries] sum); a lapsed TTL flag means the sum
# is untrusted, so depth reads 0. Depth is a PREFERENCE in the selection (tiers A/B before C/D),
# never a wall - a burst larger than K x depth keeps spreading. It is the "I know when this line
# finishes" knob: a line finishes within depth x T(one level at MAX_CIV_FACTORIES_PER_LINE civs).
# Civ count does not change per-line speed, only K (number of lines) scales with civs - which is why
# depth is a constant and K is not.
@WA_AI_QUEUE_DEPTH_CIC = 3
@WA_AI_QUEUE_DEPTH_MIC = 3
@WA_AI_QUEUE_DEPTH_NIC = 3
@WA_AI_QUEUE_DEPTH_REF = 3
@WA_AI_QUEUE_DEPTH_SR = 3
@WA_AI_QUEUE_DEPTH_HSR = 3
@WA_AI_QUEUE_DEPTH_AR = 3
@WA_AI_QUEUE_DEPTH_HAR = 3
# [silo-breadth] cap 6 with depth 3 = at most two picks per state before tier C; keeps a 20-silo
# burst spreading over >= 4 states instead of two.
@WA_AI_QUEUE_DEPTH_SILO = 3

### priority construction AI

# [refineries] there is deliberately NO affordability gate on PC admission: admission is free, the
# weekly WA_AI_PC_assign_factories funds from the sorted queue, each strategy's _project_queue_max
# bounds the clutter and the 30-week stall sweep drops what never funds. Do NOT re-add one - the
# deleted WA_AI_PC_can_afford_project was dead on saturated majors and redundant on slack ones.
# The floored gates below read the allocator's own numbers - constant:wa_ai_pc.alloc.fraction /
# .stable_base_fraction / .max_civs_per_project / .max_surrender in common/script_constants/wa_ai_pc.txt,
# the SAME declaration WA_AI_CONSTRUCTION_PRIORITY_core.txt reads.

# THIS = country. "This economy is still standing" - the ONE rule behind every PC admission and
# allocation gate (allocator floor in WA_AI_CONSTRUCTION_PRIORITY_core.txt, the three floored civ
# gates below, the refinery / MIC->CIC branches in WA_AI_CONSTRUCTION_PRIORITY_strategies.txt).
# Deliberately WITHOUT a civ-count term: the callers that need "and big enough" say so themselves
# (constant:wa_ai_pc.alloc.min_economy_civs, the floored gates' own thresholds). The railway system
# keeps its own constant:wa_ai_railway.eligibility.max_surrender - separate control panel by design.
WA_AI_PC_economy_alive = {
	has_capitulated = no
	surrender_progress < constant:wa_ai_pc.alloc.max_surrender
}

# THIS = country. Stable civilian-factory availability for PC strategy gates.
# num_of_civilian_factories_available_for_projects reads ~0 whenever the vanilla construction queue
# is saturated - a self-referential collapse (the PC's own allocation is part of what saturates it).
# Each gate passes when EITHER the engine reports real spare civs above the threshold, OR the stable
# floored base (constant:wa_ai_pc.alloc.stable_base_fraction * .alloc.fraction of TOTAL civs - the
# pool WA_AI_PC_assign_factories will actually allocate from) clears the same threshold. Three named
# variants because PDXScript triggers cannot take a threshold argument without meta_trigger - keep
# the bodies in sync.
WA_AI_PC_has_project_civs_16 = {	# passes at 16+ civs
	set_temp_variable = { _pc_civ_floor_ = num_of_civilian_factories }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.stable_base_fraction }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.fraction }
	OR = {
		num_of_civilian_factories_available_for_projects > 15
		AND = {
			WA_AI_PC_economy_alive = yes
			check_variable = { _pc_civ_floor_ > 15 }
		}
	}
}

WA_AI_PC_has_project_civs_20 = {	# passes at 20+ civs
	set_temp_variable = { _pc_civ_floor_ = num_of_civilian_factories }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.stable_base_fraction }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.fraction }

# THIS = STATE, PREV = BUILDER, _project_building_type, _project_province_id
WA_AI_PC_can_build_project = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = {
			is_subject_of = ROOT
		}
	}

	free_building_slots = { building = infrastructure size < 10 include_locked = no } # not impassable

	# Check not already building the same type here (uses per-country array)
	NOT = { WA_AI_PC_has_same_type_project_here = yes }

	WA_AI_PC_building_slot_available_for_type = yes
}

# THIS = STATE, PREV = BUILDER, _project_building_type, _project_province_id
# [uk-air-basing] does ANY builder already have a project of this type here? Was ROOT-only:
# WA_AI_PC_projects_in_state_of_@ROOT is keyed by builder, so for the building types with
# allied_build = yes (air_base, infrastructure) a host nation and its ally each saw an empty array
# for the other and both queued the same state. The second term is the builder-agnostic counter
# maintained by WA_AI_PC_start_project / WA_AI_PC_end_project_by_id. The ROOT-array term is KEPT
# deliberately: on a save resumed from an older build that counter reads 0 for every in-flight
# project, and without it the guard would regress to nothing until the legacy queue drains.
WA_AI_PC_has_same_type_project_here = {
	OR = {
		any_of = { array = WA_AI_PC_projects_in_state_of_@ROOT value = v_proj
			check_variable = { ROOT.WA_AI_PC_building_type^v_proj = _project_building_type }
		}
		check_variable = { WA_AI_PC_state_type_projects^_project_building_type > 0 }
	}
}

### [equipment-selection] air-base priority band gate
# THIS/ROOT = builder country. TRUE when this country - or a faction partner it can build for
# (air_base has allied_build = yes) - fields an air force large enough that BASING is a plausible
# constraint on deploying it, rather than airframes or pilots.
# No capacity term of its own, deliberately: every air-base project in the PC queue was created by a
# strategy that ALREADY proved a capacity deficit (WA_AI_build_uk_air_hosting_capacity's hosting gate,
# WA_AI_build_theatre_air_bases' side-wide per-theatre gate), and re-deriving it here would re-run a
# faction-wide 12-airframe sum per candidate state. The project's existence IS the deficit evidence.
# has_deployed_air_force_size is the engine's own counter - num_deployed_planes is not readable as a
# set_temp_variable rvalue, which is why both strategies inline 12 per-airframe sums and this gate
# uses the trigger form instead.
# Thresholds are absolute, on WA's air scale (~1/3 vanilla): no major clears them before ~1940, so
# early-war construction ordering is unchanged and the blast radius is confined to the mid/late war.
WA_AI_PC_air_force_needs_basing_priority = {
	OR = {
		has_deployed_air_force_size = { size > constant:wa_ai_pc.air_prio.own_planes }
		any_allied_country = {
			has_deployed_air_force_size = { size > constant:wa_ai_pc.air_prio.ally_planes }
		}
	}
}

# THIS = STATE, PREV = BUILDER, _project_building_type
# Checks if building slot is available for the specified building type
WA_AI_PC_building_slot_available_for_type = {

	if = { limit = { check_variable = { _project_building_type = 1 } }

		free_building_slots = { building = infrastructure size > 0 include_locked = no }

# THIS = state. Central and southern England states expected to host the faction's aviation (within
# fighter range of northern France / Low Countries).
# [equipment-selection] THE ORDER OF THIS LIST IS THE BUILD PRIORITY - south-east first. It is the
# authoritative geography statement, mirrored id-for-id and rank-for-rank by
# WA_AI_uk_air_hosting_priority_init (common/scripted_effects/WA_AI_CONSTRUCTION_PRIORITY_strategies.txt),
# which loads it into global.WA_AI_uk_air_priority_states. The list genuinely has to exist twice:
# @ constants are file-scoped, and a state-scope membership test cannot be built from a plain-id
# array (comparing THIS.id against a stored plain id never matches - see the .id scope-encoding entry
# in wa-lessons-learned). MEMBERSHIP stays owned HERE: the selector re-tests this trigger on every
# array candidate and falls back to an unordered scan over it, so drift costs a state its RANK, never
# its eligibility. Change one, change the other.
# Ranking method: mean pixel distance from each state's provinces (generated WA_AI_MAP province
# coordinates) to the Low Countries / Northern France / Paris arc - mean, not nearest, because these
# fields must cover the whole theatre, not one landing beach. Two ranks are judgement, not arithmetic:
# 127 Sussex outranks 961 South London on nearest-field distance and coastal position (the forward
# Channel fighter position); 123 Cornwall is LAST because its seaward frontage is Biscay and Brittany,
# not the Channel narrows - a Coastal Command position, not a Northern-France cover position.
# Adding states raises the structural ceiling (11 x constant:wa_ai_pc.air.uk_target_level_cap), so
# states that hold ENG air bases but are absent here (122 Wales, 858 Berkshire, 130 Yorkshire) are
# left out until the basing OUTCOME justifies more capacity; Berkshire is the first candidate.
WA_AI_uk_air_hosting_state = {
	OR = {
		state = 857 # rank  1 - Kent
		state = 127 # rank  2 - Sussex
		state = 961 # rank  3 - South London
		state = 125 # rank  4 - Essex
		state = 126 # rank  5 - North London
		state = 860 # rank  6 - Norfolk
		state = 859 # rank  7 - Dorset
		state = 338 # rank  8 - Gloucestershire
		state = 129 # rank  9 - Leicestershire
		state = 128 # rank 10 - Warwickshire
		state = 123 # rank 11 - Cornwall
	}
}

# THIS = state, ROOT = builder country. Input: _project_building_type (= 2).
# [equipment-selection] the "can this builder put another air-base level here" test, extracted so the
# ordered south-first walk and its unordered fallback in WA_AI_build_uk_air_hosting_capacity cannot
# drift apart. See the [uk-air-basing] comment below for why the allied branch must repeat the host's
# free-slot test.
WA_AI_uk_air_site_buildable = {
	OR = {
		AND = {
			is_controlled_by = ROOT
			WA_AI_available_AIR = yes
			WA_AI_PC_can_build_project = yes
		}
		AND = {
			NOT = { is_controlled_by = ROOT }
			WA_AI_available_AIR_allied = yes
			NOT = { WA_AI_PC_has_same_type_project_here = yes }
			# [uk-air-basing] the ally must pass the SAME free-slot test the host nation gets via
			# WA_AI_PC_can_build_project above. Without it an allied builder can queue a
			# 2000-IC air base into a state with no free shared slot, fund it for weeks, and
			# then have WA_AI_PC_complete_project_by_id refuse to spawn the building (its own
			# WA_AI_PC_building_slot_available gate) - the whole build silently discarded. Air
			# bases compete for state slots with the arms factories ENG builds by the hundred
			# in exactly these states, so a full state is the normal case, not an edge case.
			# Same trigger, same _project_building_type = 2, so queue-time and completion-time
			# agree.
			WA_AI_PC_building_slot_available_for_type = yes
		}
	}
}

############################################################################################################
#	[na-corridor] coalition logistics - who may build rail/ports on whose soil
#
#	The PC *executor* has always accepted allied ground: WA_AI_PC_start_project passes on
#	CONTROLLER = { is_in_faction_with = ROOT } (PRIORITY_core.txt), the controller test in
#	WA_AI_PC_assign_factories and the completion path do the same, supply_node / rail_way / naval_base
#	are all allied_build = yes, and provincial pathfinder type 2 walks allied provinces. Only the
#	*selectors* were ROOT-or-subject gated, so a corridor that ended up on an ally's soil had no
#	builder at all.
#
#	The ally leg is deliberately NOT "any faction member". It fires only where the controller cannot
#	run the railway system for itself, which keeps two healthy allies out of each other's rail networks.
#
#	The predicates below answer TWO different questions and must never be merged: PAYMENT ("does this
#	ally need ME to pay for its logistics?") and PERMISSION ("may I build here?", which the engine
#	answers more widely - WA_AI_PC_state_controller_allows_admission below). A corridor hop needs
#	permission on both nodes and payment on at least one; a healthy ally's INTERIOR hops therefore
#	still have no payer but itself, so no partner builds them.
############################################################################################################

# Railway eligibility thresholds and the per-enemy route budget are read here as
# constant:wa_ai_railway.eligibility.* / routes.max_per_enemy (common/script_constants/wa_ai_railway.txt),
# one declaration shared with the on_action and railway_core - nothing to keep in sync by hand.

# THIS = country. Is this country able to run the railway system for itself?
# [pc-queue] this IS the on_weekly railway eligibility gate, lifted verbatim out of
# WA_AI_misc_on_actions.txt so that the on_action and [na-corridor]'s ally leg cannot drift apart.
# `has_capitulated` is deliberately NOT a term: it answers "did this government fall", not "can this
# country build". A capitulated tag keeps its territory, its faction and its factories; the capability
# terms below are what decide whether rail is worth building.
WA_AI_PC_country_can_build_own_logistics = {
	# Higher peacetime industrial threshold - war uses a lower one inside WA_AI_PC_railway
	OR = {
		has_war = yes
		num_of_civilian_factories > constant:wa_ai_railway.eligibility.min_civs_peace
	}
	# Skip minor countries unless at war or rich
	OR = {
		is_major = yes
		has_war = yes
		num_of_civilian_factories > constant:wa_ai_railway.eligibility.minor_civ_threshold
	}
	# Minimum state count - skip tiny nations
	num_of_controlled_states > constant:wa_ai_railway.eligibility.min_states
	# Skip countries losing badly - UNLESS the war economy still works.
	# surrender_progress measures VP loss, not capability: a major can cross the bar while
	# holding ~180 states and 300+ divisions.
	OR = {
		surrender_progress < constant:wa_ai_railway.eligibility.max_surrender
		AND = {
			num_of_civilian_factories > constant:wa_ai_railway.eligibility.min_civs_peace
			num_of_controlled_states > constant:wa_ai_railway.eligibility.recovery_min_states
		}
	}
}


# THIS = country. [na-corridor] AUTHORITY vs FUNDING.
# WA_AI_PC_country_can_build_own_logistics is a SIZE test: at war it asks only for > 5 controlled
# states and a surrender bar, with NO civ-factory term at all. The railway run it authorises applies
# its own floor one level down (railway_core: min_civs, min_civs_war while at war). Everything in the
# band "at war, > 5 states, under the wartime floor" therefore reads CAPABLE, queues routes it can
# never fund, and - the ally leg being the complement of the authority test - locked every faction
# partner out of its ground at the same time. This trigger is the funding half.
# Reads num_of_civilian_factories, the RAW count, deliberately: num_of_available_civilian_factories
# collapses to ~0 as soon as the vanilla construction queue saturates, which would classify healthy
# majors as unfundable (lessons log, the available-factories gotcha). The peacetime term matches the
# authority test's own bar so the two agree at peace.
# NAME CAVEAT: this is authority AND funding, not funding alone - it also fails on the surrender bar
# and the > 5 controlled states, so a 200-civ ally pinned above max_surrender reads "cannot fund".
WA_AI_PC_country_can_fund_own_logistics = {
	WA_AI_PC_country_can_build_own_logistics = yes
	OR = {
		AND = {
			has_war = yes
			num_of_civilian_factories > constant:wa_ai_railway.eligibility.min_civs_war
		}
		AND = {
			has_war = no
			num_of_civilian_factories > constant:wa_ai_railway.eligibility.min_civs_peace
		}
	}
}

# THIS = country. [na-corridor] THE ALLY LEG'S GATE: an ally ROOT may build logistics FOR is one that
# cannot PAY for them, not merely one too small to be allowed to plan them. Strict complement of the
# funding test, so the two can never disagree. A partner above the wartime civ floor stays excluded -
# that is what keeps two healthy allies out of each other's rail networks; what changes is that a
# belligerent BELOW that floor is visible to its allies as what it is.
WA_AI_PC_country_cannot_fund_own_logistics = {
	NOT = { WA_AI_PC_country_can_fund_own_logistics = yes }
}

# THIS = state, ROOT = builder country. True when ROOT may queue logistics construction here:
# ROOT-controlled, subject-controlled, or [na-corridor]'s ally leg - a faction ally that cannot PAY
# for its own logistics, not merely one too small to be allowed to plan them (trigger pair above).
WA_AI_PC_can_build_logistics_here = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = { is_subject_of = ROOT }
		AND = {
			CONTROLLER = { is_in_faction_with = ROOT }
			CONTROLLER = { WA_AI_PC_country_cannot_fund_own_logistics = yes }
		}
	}
}

# THIS = country, ROOT = builder country. [na-corridor] countries whose controlled states ROOT may
# treat as its own when selecting logistics targets: itself, its subjects, and faction allies that
# cannot FUND their own logistics. Country-scoped twin of WA_AI_PC_can_build_logistics_here, for the
# selectors that iterate `every_country = { limit = { ... } every_controlled_state = { ... } }` rather
# than states. Also what WA_AI_PC_corridor_node_is_my_charge reads.
WA_AI_PC_is_logistics_build_partner = {
	exists = yes
	OR = {
		tag = ROOT
		is_subject_of = ROOT
		AND = {
			is_in_faction_with = ROOT
			WA_AI_PC_country_cannot_fund_own_logistics = yes
		}
	}
}

############################################################################################################
#	[na-corridor] the PC ADMISSION controller gate, named.
#
#	THIS = state, ROOT = builder country. True when this state's controller lets ROOT queue a building
#	here. Extracted VERBATIM from WA_AI_PC_start_project (WA_AI_CONSTRUCTION_PRIORITY_core.txt,
#	"Validate location is controlled"), which is now its only gameplay reader; behaviour unchanged.
#
#	Protects: it is the LAST word on whether a queued project is legal. A selector that admits ground
#	this refuses produces silent, traceless failure - the project is simply never queued, and the
#	request is re-made and re-pathfound every 4 weeks, refused without a mark.
#
#	Assumes: THREE terms, no capability test - an ally that CAN build for itself still hosts our
#	construction. Corollary, and the reason this is named: selector <= admission <= validity.
#	  - selector  = WA_AI_PC_is_corridor_side_holder below (province-scoped, same three terms)
#	  - admission = this trigger (state-scoped)
#	  - validity  = the lane / air-lane / fill / completion tests in PRIORITY_core.txt, which each
#	                carry a FOURTH term, `ROOT = { is_subject_of = PREV }` (an overlord's ground is
#	                valid for its subject).
#	The four validity sites are DELIBERATELY NOT this trigger and must not be "synced" to it: doing so
#	would narrow the lane, the fill and the sweep. Widening admission to four terms instead would let
#	projects be queued that the engine then refuses.
#	Fact is gone when: this gate gains or loses a term. Then the corridor selector below and the
#	WA_TLM_r103_corridor_blocked_n probe both need the same move, in the same commit.
############################################################################################################

WA_AI_PC_state_controller_allows_admission = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = { is_in_faction_with = ROOT }
		CONTROLLER = { is_subject_of = ROOT }
	}
}

### [na-corridor] - theatre corridors (WA_AI_PC_railway_corridor_pass, railway_core; node data in
### WA_AI_CONSTRUCTION_PRIORITY_corridor_data.txt)

# THIS = country, ROOT = builder country. [na-corridor] PERMISSION: is this holder on ROOT's own side,
# so that ground it controls is ground ROOT may build corridor logistics on? Country-scoped mirror of
# WA_AI_PC_state_controller_allows_admission - the same three terms in the same order, because a
# selector wider than admission requests hops the executor then refuses in silence, and a selector
# narrower than admission refuses hops the executor would have accepted. Deliberately NOT
# WA_AI_PC_is_logistics_build_partner: that answers the PAYMENT question, asked separately by
# WA_AI_PC_corridor_node_is_my_charge below.
# THIRD READER: WA_AI_PC_corridor_compute_sizing (railway_helpers) walks every_country on this trigger
# to total OUR SIDE's divisions in the corridor states - a DEMAND question, deliberately the same
# predicate: the holders whose ground we may build on are the ones whose armies eat that ground's
# supply, and letting the two drift would size the corridor for a different coalition than it serves.
# Fact is gone when: the admission gate's term list changes - move both together. A term added or
# removed here moves the SELECTOR, the blocked_n probe AND every corridor rail target, in that order
# of visibility - the first two fail loudly, the third is only a level that is quietly wrong.
WA_AI_PC_is_corridor_side_holder = {
	exists = yes
	OR = {
		tag = ROOT
		is_subject_of = ROOT
		is_in_faction_with = ROOT
	}
}

# THIS = ROOT = builder country. Input: _corridor_prov (temp) = a corridor node province id.
# PERMISSION: true when the node is ground ROOT MAY build logistics on - held by ROOT, a subject of
# ROOT, or any faction ally (WA_AI_PC_is_corridor_side_holder). PROVINCE control, never state control:
# on a contested front the two diverge (an enemy-controlled state can hold our provinces inside it,
# railway_strategies.txt [na-corridor] notes). There is no province-scoped CONTROLLER, hence the
# any_country walk; 20 nodes every 4 weeks per builder is negligible. The PAYMENT half lives next
# door - testing permission and payment as ONE test leaves a hop between a healthy ally's node and
# ours with no builder anywhere.
WA_AI_PC_corridor_node_is_ours = {
	meta_trigger = {
		text = {
			any_country = {
				WA_AI_PC_is_corridor_side_holder = yes
				controls_province = [x]
			}
		}
		x = "[?_corridor_prov]"
	}
}

# THIS = ROOT = builder country. Input: _corridor_prov (temp). PAYMENT: true when this node is one
# ROOT should pay to develop - its own ground, a subject's, or a faction ally's that cannot fund its
# own logistics (WA_AI_PC_is_logistics_build_partner).
# A hop needs permission on BOTH ends and payment on AT LEAST ONE. That is what keeps "two healthy
# allies out of each other's rail networks" literally true: a healthy ally's INTERIOR hops (both ends
# its own) still have no payer but itself, so no partner touches them; only a hop we already have a
# foot on crosses the line. It also bounds duplicate spend to hops that STRADDLE two partners' ground,
# and keeps an out-of-theatre faction member (JAP, which passes the North Africa theatre gate the
# moment ENG holds Egypt) from entering the pass at all - permission on ally nodes, charge on none.
# The WA_TLM_r103_corridor_orphan gauge reads "this hop belongs to someone else": both ends ours and
# neither our charge means the ends are held by partners that genuinely CAN fund, so each has a foot
# on it. A transient positive is normal; orphan high AND the hop still unbuilt on the map means the
# owner is not building it either - a real defect of the owner's pass (theatre gate, civ floor,
# queue), not of this rule.
WA_AI_PC_corridor_node_is_my_charge = {
	meta_trigger = {
		text = {
			any_country = {
				WA_AI_PC_is_logistics_build_partner = yes
				controls_province = [x]
			}
		}
		x = "[?_corridor_prov]"
	}
}

# THIS = ROOT = builder country. Input: _corridor_prov. True when the node is held by someone at
# war with ROOT - the corridor pass reads this over the node list to pick the war (rail_war) or the
# preparation (rail_prewar) priority band, so a peaceful owner of Libya preps at 500 and the same
# owner at war with the holder of the next node builds at 1000. Province-level for the same reason
# as above.
WA_AI_PC_corridor_node_is_hostile = {
	meta_trigger = {
		text = {
			any_country = {
				has_war_with = ROOT
				controls_province = [x]
			}
		}
		x = "[?_corridor_prov]"
	}
}

# THIS = ROOT = builder country. Input: _corridor_prov. True when the node's province is OWNED by a
# country at war with ROOT - i.e. whoever holds it stands on conquered enemy soil. This is the
# corridor's definition of "the attacker" for forward ports (user ruling 2026-08-16: the attacker is
# the faction attacking Libya, never the defender on its own ground): ENG holding Ajdabiya (owner ITL,
# at war with ENG) is the attacker; ITA/ITL holding it is not; a German holding it is not (owner ITL is
# its ally); the Axis holding an Egyptian node owned by UKE is - symmetric. Owner is read from the
# province's state (province owner = state owner).
WA_AI_PC_corridor_node_is_conquered = {
	check_variable = { global.WA_AI_MAP_province_state_id^_corridor_prov > 0 }
	var:global.WA_AI_MAP_province_state_id^_corridor_prov = {
		OWNER = { has_war_with = ROOT }
	}
}

# THIS = ROOT = builder country. Input: _corridor_hostile_n_ (temp) = how many corridor nodes are
# held by an enemy of ROOT (computed by the pass over the node list). When does the North African
# corridor pass act at all?
#  - contested: the theatre is contested for ROOT (delegates to the AIR trigger, which is generic -
#    an enemy-controlled AND an own-side-controlled state among Morocco..Egypt, no tags);
#  - preparation: NO corridor node is enemy-held. The pass then only acts where ROOT / a subject /
#    a partner holds two consecutive nodes, i.e. "the owner of Libya / Egypt / Tunisia prepares its
#    corridor". Keyed on the nodes rather than on has_war = no because ITA is at war in 1936
#    (Ethiopia) and would otherwise never prepare Libya.
# No gap on either path: a neutral ITA keeps preparing under leg 2, an Axis Egypt keeps building
# toward Alexandria under leg 1. Between the two (at war in the theatre but holding no own-side
# state) the pass is silent: nothing to connect.
WA_AI_PC_corridor_theatre_live_north_africa = {
	OR = {
		check_variable = { _corridor_hostile_n_ = 0 }
		WA_AI_MILITARY_AIR_theatre_contested_north_africa = yes
	}
}

# THIS = state, ROOT = builder country. Cheap membership test for the land-war railway strategy:
# is this state a frontline worth a route against the enemy currently being processed?
# Reads the chain-global temps the strategy sets per enemy (_routes_for_this_enemy, the per-enemy
# route budget; _current_enemy_tag; _check_railway_level, the level at which a frontline is "done").
# [na-corridor] extracted so all three candidate populations - ROOT's states, subject states and
# dependent-ally states - are filtered identically before the shared body runs.
WA_AI_PC_railway_land_frontline_candidate = {
	### Skip if we already have enough routes for this enemy
	check_variable = { _routes_for_this_enemy < constant:wa_ai_railway.routes.max_per_enemy }
	WA_AI_PC_railway_state_is_frontline_hub = yes
}

# [rail-spine-tree] the frontline test WITHOUT the per-enemy route budget: the spine gate counts every
# frontline hub of the enemy, not only those a route slot is left for - counted after the budget,
# ">= min_hubs" could only ever read "budget reached". THIS = state; reads _current_enemy_tag and
# _check_railway_level (the level at which a frontline is "done").
WA_AI_PC_railway_state_is_frontline_hub = {
	# State borders this specific enemy (or the enemy's puppets)
	any_neighbor_state = {
		OR = {
			is_controlled_by = var:_current_enemy_tag
			controller = { is_subject_of = var:_current_enemy_tag }
		}
	}
	# State has a supply hub (proper check)
	WA_AI_PC_state_has_supply_hub = yes
	# Skip states already at the family's "done" level (native check - includes focus/event built)
	NOT = { WA_AI_PC_state_has_railway_at_level = yes }
}

# [rail-spine-tree] a frontline hub state ALREADY at the family's done level: it draws no route, but the
# trunk under it still caps it, so the spine counts it as a served hub. THIS = state; reads
# _current_enemy_tag and _check_railway_level.
WA_AI_PC_railway_state_is_served_frontline_hub = {
	any_neighbor_state = {
		OR = {
			is_controlled_by = var:_current_enemy_tag
			controller = { is_subject_of = var:_current_enemy_tag }
		}
	}
	WA_AI_PC_state_has_supply_hub = yes
	WA_AI_PC_state_has_railway_at_level = yes
}

# [rail-spine-tree] may this state host the RAILHEAD (the trunk's end) against the enemy being processed?
# A hub state that is NOT a frontline of that enemy. The landmass test is the caller's (it needs the
# state id). THIS = state; reads _current_enemy_tag.
WA_AI_PC_railway_state_is_railhead_candidate = {
	WA_AI_PC_state_has_supply_hub = yes
	NOT = {
		any_neighbor_state = {
			OR = {
				is_controlled_by = var:_current_enemy_tag
				controller = { is_subject_of = var:_current_enemy_tag }
			}
		}
	}
}
