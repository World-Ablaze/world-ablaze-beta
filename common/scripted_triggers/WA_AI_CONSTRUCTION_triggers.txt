### priority construction AI

WA_AI_PC_can_afford_project = {

	@AI_MAX_FRACTION_OF_FACTORIES_TO_ASSIGN_ON_PROJECTS_TOTAL = 0.35
	@AI_MIN_NUM_OF_FACTORIES_TO_ASSIGN_ON_PROJECT = 20
	# must match @AI_PC_STABLE_BASE_FRACTION in common/scripted_effects/WA_AI_CONSTRUCTION_PRIORITY_core.txt
	@AI_PC_STABLE_BASE_FRACTION = 0.30

	# check that the ai can afford to assign factories to a new project
	set_temp_variable = { max_factories_on_projects = ROOT.num_of_civilian_factories_available_for_projects }
	add_to_temp_variable = { max_factories_on_projects = ROOT.WA_AI_PC_assigned_factories_total }
	multiply_temp_variable = { max_factories_on_projects = @AI_MAX_FRACTION_OF_FACTORIES_TO_ASSIGN_ON_PROJECTS_TOTAL }

	set_temp_variable = { min_factories_on_projects = ROOT.WA_AI_PC_assigned_factories_total }
	add_to_temp_variable = { min_factories_on_projects = @AI_MIN_NUM_OF_FACTORIES_TO_ASSIGN_ON_PROJECT }

	# Fix: mirror the stable-base floor used by WA_AI_PC_assign_factories. Without it this gate
	# repeats the self-referential expression (available_for_projects reads 0 once the vanilla
	# queue saturates) and stays closed forever after the allocation has decayed - even when
	# assign_factories would happily fund a new project from the floored base.
	set_temp_variable = { _pc_floor_max = ROOT.num_of_civilian_factories }
	multiply_temp_variable = { _pc_floor_max = @AI_PC_STABLE_BASE_FRACTION }
	multiply_temp_variable = { _pc_floor_max = @AI_MAX_FRACTION_OF_FACTORIES_TO_ASSIGN_ON_PROJECTS_TOTAL }

	OR = {
		check_variable = { min_factories_on_projects < max_factories_on_projects }
		AND = {
			ROOT = {
				num_of_civilian_factories > 19
				has_capitulated = no
				surrender_progress < 0.3
			}
			check_variable = { min_factories_on_projects < _pc_floor_max }
		}
	}
}

# THIS = country. Stable civilian-factory availability for PC strategy gates.
# num_of_civilian_factories_available_for_projects reads ~0 whenever the vanilla
# construction queue is saturated - the same self-referential collapse fixed in
# WA_AI_PC_can_afford_project above. Each gate passes when EITHER the engine
# reports real spare civs above the threshold, OR the stable floored base
# (@AI_PC_STABLE_BASE_FRACTION * @AI_MAX_FRACTION_OF_FACTORIES_TO_ASSIGN_ON_PROJECTS_TOTAL
# = 10.5% of TOTAL civs, the pool WA_AI_PC_assign_factories will actually
# allocate from) clears the same threshold. Two named variants because PDXScript
# triggers cannot take a threshold argument without meta_trigger - keep both
# bodies in sync.
WA_AI_PC_has_project_civs_16 = {	# passes at 16+ civs (replaces bare "> 15" gates)
	set_temp_variable = { _pc_civ_floor_ = num_of_civilian_factories }
	multiply_temp_variable = { _pc_civ_floor_ = @AI_PC_STABLE_BASE_FRACTION }
	multiply_temp_variable = { _pc_civ_floor_ = @AI_MAX_FRACTION_OF_FACTORIES_TO_ASSIGN_ON_PROJECTS_TOTAL }
	OR = {
		num_of_civilian_factories_available_for_projects > 15
		AND = {
			has_capitulated = no
			surrender_progress < 0.3
			check_variable = { _pc_civ_floor_ > 15 }
		}
	}
}

WA_AI_PC_has_project_civs_20 = {	# passes at 20+ civs (replaces bare "NOT ... < 20" gates)
	set_temp_variable = { _pc_civ_floor_ = num_of_civilian_factories }

}

# THIS = STATE, PREV = BUILDER, _project_building_type, _project_province_id
WA_AI_PC_can_build_project = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = {
			is_subject_of = ROOT
		}
	}

	free_building_slots = { building = infrastructure size < 10 include_locked = no } # not impassable

	# Check not already building the same type here (uses per-country array)
	NOT = { WA_AI_PC_has_same_type_project_here = yes }

	WA_AI_PC_building_slot_available_for_type = yes
}

# THIS = STATE, PREV = BUILDER, _project_building_type, _project_province_id
# Checks if ANY builder already has a project of the same type in this state.
# Fix 46: was ROOT-only. WA_AI_PC_projects_in_state_of_@ROOT is keyed by builder, so for the
# building types with allied_build = yes (air_base, infrastructure) the host nation and its
# ally each saw an empty array for the other and both queued the same state - campaign
# 9a4cd657 1944.7 had ENG and USA simultaneously funding an air base in state 126 and in
# state 860, and the UK-hosting capacity gate then counted the same state twice as "claimed".
# The second term is the builder-agnostic counter maintained by WA_AI_PC_start_project /
# WA_AI_PC_end_project_by_id. The original ROOT-array term is KEPT deliberately: on a save
# resumed from a pre-Fix-46 build the counter starts at 0 for every in-flight project, and
# without it the guard would regress to nothing until the whole legacy queue drains.
WA_AI_PC_has_same_type_project_here = {
	OR = {
		any_of = { array = WA_AI_PC_projects_in_state_of_@ROOT value = v_proj
			check_variable = { ROOT.WA_AI_PC_building_type^v_proj = _project_building_type }
		}
		check_variable = { WA_AI_PC_state_type_projects^_project_building_type > 0 }
	}
}

### Fix 47: air-base priority band gate
# THIS/ROOT = builder country. TRUE when this country - or a faction partner it can build
# for (air_base has allied_build = yes) - fields an air force large enough that BASING is a
# plausible constraint on deploying it, rather than airframes or pilots.
#
# Why a bare size test is the right gate, with no capacity term of its own: every air-base
# project in the PC queue was created by a strategy that ALREADY proved a capacity deficit
# (WA_AI_build_uk_air_hosting_capacity's hosting-capacity gate, WA_AI_build_theatre_air_bases'
# side-wide per-theatre gate). Re-deriving the deficit here would re-run a faction-wide
# 12-airframe sum inside a trigger evaluated per candidate state. The project's existence is
# the deficit evidence; this trigger only asks whether the air force behind it is worth
# outranking a refinery for.
#
# has_deployed_air_force_size is the engine's own counter - num_deployed_planes is not
# readable as a set_temp_variable rvalue (campaign 5709c8b9), which is why both strategies
# inline 12 per-airframe sums, and why this gate deliberately uses the trigger form instead.
#
# Thresholds are absolute, on WA's air scale (~1/3 vanilla; the strategies' own token floor
# is 99 deployed planes). No major clears them before ~1940, so early-war construction
# ordering is unchanged and the blast radius is confined to the mid/late war, which is when
# the overstacking this fixes actually appears.
@AI_PC_AIR_PRIO_OWN_PLANES = 500		# own air force big enough to outrank the strategic band
@AI_PC_AIR_PRIO_ALLY_PLANES = 1000		# or a faction partner whose air force we host
WA_AI_PC_air_force_needs_basing_priority = {
	OR = {
		has_deployed_air_force_size = { size > @AI_PC_AIR_PRIO_OWN_PLANES }
		any_allied_country = {
			has_deployed_air_force_size = { size > @AI_PC_AIR_PRIO_ALLY_PLANES }
		}
	}
}

# THIS = STATE, PREV = BUILDER, _project_building_type
# Checks if building slot is available for the specified building type
WA_AI_PC_building_slot_available_for_type = {

	if = { limit = { check_variable = { _project_building_type = 1 } }

		free_building_slots = { building = infrastructure size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 2 } }

		free_building_slots = { building = air_base size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 3 } }

		free_building_slots = { building = anti_air_building size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 4 } }

		OR = {
			AND = { ROOT = { has_tech = radio_detection } 				radar_station < 1 }
			AND = { ROOT = { has_tech = decimetric_radar } 				radar_station < 2 }
			AND = { ROOT = { has_tech = improved_decimetric_radar } 	radar_station < 3 }
			AND = { ROOT = { has_tech = centimetric_radar } 			radar_station < 4 }
			AND = { ROOT = { has_tech = improved_centimetric_radar } 	radar_station < 5 }
			AND = { ROOT = { has_tech = advanced_centimetric_radar } 	radar_station < 6 }
		}
	}

	else_if = { limit = { check_variable = { _project_building_type = 5 } }

		free_building_slots = { building = arms_factory size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 6 } }

		free_building_slots = { building = industrial_complex size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 7 } }

		free_building_slots = { building = dockyard size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 8 } }

		# The shared-slot fallbacks below bypass the include_locked tech gate, so check the
		# unlock tech explicitly - without it a country lacking synth_oil_experiments passes
		# the slot check for a building it cannot construct.
		ROOT = { has_tech = synth_oil_experiments }
		OR = {
			free_building_slots = { building = synthetic_refinery size > 0 include_locked = no }
			free_building_slots = { building = industrial_complex size > 0 include_locked = no }
			free_building_slots = { building = arms_factory size > 0 include_locked = no }
			free_building_slots = { building = dockyard size > 0 include_locked = no }
		}
		synthetic_refinery < 3
	}

	else_if = { limit = { check_variable = { _project_building_type = 9 } }

		check_variable = { building_level@arms_factory > 0 }
	}

	else_if = { limit = { check_variable = { _project_building_type = 10 } }

		check_variable = { building_level@industrial_complex > 0 }
	}

	else_if = { limit = { check_variable = { _project_building_type = 11 } }

		free_building_slots = { building = hydro_steel_refinery size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 12 } }

		free_building_slots = { building = hydro_aluminium_refinery size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 15 } }

		free_building_slots = { building = steel_refinery size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 16 } }

		free_building_slots = { building = aluminium_refinery size > 0 include_locked = no }
	}

	# Railway type 13 - railways don't use traditional building slots
	else_if = { limit = { check_variable = { _project_building_type = 13 } }
		always = yes
	}

	# Naval base type 14 - can be built in allied territory
	else_if = { limit = { check_variable = { _project_building_type = 14 } }
		always = yes
	}
}

# Fix 41: overtake-lane revalidation - "is this starved project still worth building?"
# THIS = ROOT = builder country. Input: _reval_project_id (temp) = PC project index.
# Called by the anti-starvation lane in WA_AI_PC_assign_factories
# (WA_AI_CONSTRUCTION_PRIORITY_core.txt) before diverting factories to a project that
# has been starved >= @AI_PC_AGING_LANE_WEEKS; a FAIL cancels the project on the spot.
# Dispatches on WA_AI_PC_building_type^_reval_project_id. Every geographic branch is
# dynamic game state (control, war, contested theatres) - no historical-path assumption,
# so it behaves identically in historical and ahistorical games.
WA_AI_PC_project_still_relevant = {
	# 13/14 rail family: still ours (ROOT or ROOT subject - rails serve OUR supply net)
	# AND still near a live front (kills the Romanian-border project after the front
	# moved to Moscow; spares the still-contested Libya corridor).
	if = {
		limit = {
			OR = {
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 13 }
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 14 }
			}
		}
		var:WA_AI_PC_target_state^_reval_project_id = {
			CONTROLLER = {
				OR = {
					tag = ROOT
					is_subject_of = ROOT
				}
			}
			any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
		}
	}
	# 2 air bases: friendly ground (ROOT or faction ally - air_base has allied_build)
	# AND a reason to base there: the UK hosting belt, or membership in a theatre that
	# is currently contested for ROOT (same state/contested trigger pairs the theatre
	# air-base builder uses; the pairs must-match, WA_AI_MILITARY_triggers.txt).
	else_if = {
		limit = { check_variable = { WA_AI_PC_building_type^_reval_project_id = 2 } }
		var:WA_AI_PC_target_state^_reval_project_id = {
			CONTROLLER = {
				OR = {
					tag = ROOT
					is_in_faction_with = ROOT
				}
			}
			OR = {
				WA_AI_uk_air_hosting_state = yes
				AND = { WA_AI_MILITARY_AIR_theatre_state_north_africa = yes		ROOT = { WA_AI_MILITARY_AIR_theatre_contested_north_africa = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_italy = yes			ROOT = { WA_AI_MILITARY_AIR_theatre_contested_italy = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_western_europe = yes	ROOT = { WA_AI_MILITARY_AIR_theatre_contested_western_europe = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_scandinavia = yes		ROOT = { WA_AI_MILITARY_AIR_theatre_contested_scandinavia = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_balkans = yes			ROOT = { WA_AI_MILITARY_AIR_theatre_contested_balkans = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_eastern_europe = yes	ROOT = { WA_AI_MILITARY_AIR_theatre_contested_eastern_europe = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_middle_east = yes		ROOT = { WA_AI_MILITARY_AIR_theatre_contested_middle_east = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_burma_india = yes		ROOT = { WA_AI_MILITARY_AIR_theatre_contested_burma_india = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_southeast_asia = yes	ROOT = { WA_AI_MILITARY_AIR_theatre_contested_southeast_asia = yes } }
				AND = { WA_AI_MILITARY_AIR_theatre_state_pacific_islands = yes	ROOT = { WA_AI_MILITARY_AIR_theatre_contested_pacific_islands = yes } }
			}
		}
	}
	# 8 synthetic refinery: rubber still needed (>= 2 of 3 on the monthly needs counter,
	# WA_AI_misc_effects.txt) - never geographic; a refinery pays off wherever it stands.
	else_if = {
		limit = { check_variable = { WA_AI_PC_building_type^_reval_project_id = 8 } }
		check_variable = { WA_AI_needs_rubber > 1 }
	}
	# 11/15 steel mills (hydro/plain): steel still needed - never geographic.
	else_if = {
		limit = {
			OR = {
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 11 }
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 15 }
			}
		}
		check_variable = { WA_AI_needs_steel > 1 }
	}
	# 12/16 aluminium smelters (hydro/plain): aluminium still needed - never geographic.
	else_if = {
		limit = {
			OR = {
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 12 }
				check_variable = { WA_AI_PC_building_type^_reval_project_id = 16 }
			}
		}
		check_variable = { WA_AI_needs_aluminium > 1 }
	}
	# default (1 infra, 3 AA, 4 radar, 5/6/7 factories, 9/10 conversions): the target
	# state is still on our side - the minimum bar for any building to ever spawn.
	# A project with target_state 0 passes: we cannot judge what we cannot locate, and
	# the orphan-cleanup pass in WA_AI_PC_update_project_progress owns that case.
	else = {
		if = {
			limit = { check_variable = { WA_AI_PC_target_state^_reval_project_id > 0 } }
			var:WA_AI_PC_target_state^_reval_project_id = {
				CONTROLLER = {
					OR = {
						tag = ROOT
						is_in_faction_with = ROOT
						is_subject_of = ROOT
					}
				}
			}
		}
	}
}
