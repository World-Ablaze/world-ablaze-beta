### available construction slots

# Fix 81: per-building state caps for the shared-slot buildings (refineries; Fix 88 added CIC/MIC/NIC).
# `free_building_slots = { building = X size > 0 }` only reports the SHARED pool;
# it does not consult X's own `level_cap.state_max`. A state with more free
# shared slots than the cap therefore stays "available" forever once the cap is
# hit, `add_building_construction` silently no-ops, and because that state also
# stays #1 in WA_AI_shared_slot_scores the whole building type stalls country-wide.
# The check is `built < cap` AND `committed < cap`, where committed = built + queued is the
# state variable WA_AI_committed_<X> maintained by WA_AI_add_<X> (building_adders). The SUM is
# what makes this cheap: a completion moves one level from queued to built and leaves the sum
# unchanged, so nothing has to be reconciled on any cadence (cut 2 tracked queued + last_level
# and reconciled from the level delta at C.10 pulses - a state whose line finished between
# pulses read stale, and a country whose lane sits behind WA_AI_construction_timer never
# reconciled at all). A built-level-only check is NOT enough either: the level rises only when
# a refinery FINISHES (months), so a state stays below the cap on built level while its queue
# is already full and every pulse in between keeps feeding it.
# `committed` is trusted only while the state flag WA_AI_committed_<X>_recent is live (TTL in
# building_adders): once no add has happened for that long, nothing can still be queued, so
# `built < cap` alone decides and the next add rebases the sum. That is the only path by which a
# CANCELLED construction is forgotten - script cannot see the construction queue.
# MUST MATCH `level_cap.state_max` in common/buildings/00_buildings.txt.
@WA_AI_REF_STATE_MAX = 6
@WA_AI_SR_STATE_MAX = 15
@WA_AI_HSR_STATE_MAX = 15
@WA_AI_AR_STATE_MAX = 15
@WA_AI_HAR_STATE_MAX = 15

@WA_AI_CIC_STATE_MAX = 45
@WA_AI_MIC_STATE_MAX = 45
@WA_AI_NIC_STATE_MAX = 45

### Fix 88: per-line depth cap for the breadth-first selection (WA_AI_CONSTRUCTION_queue_functions.txt).

# state, read as WA_AI_committed_<X> - building_level@X (Fix 81 sum). A lapsed TTL flag means the sum is
# no longer trusted, so depth is treated as 0. Depth is a PREFERENCE in the selection (tiers A/B before
# C/D), never a wall - a burst larger than K x depth keeps spreading, and the tier-D fallback is the

# depth x T(one level at MAX_CIV_FACTORIES_PER_LINE civs); civ count does not change per-line speed,

# WA_AI_PC_can_afford_project was DELETED here on 2026-08-16 (Fix 90b, campaign af003548). It was
# the affordability half of the PRE-2026-01 engine's admission gate (the other half was the
# `active_projects < 5` slot test of the fixed 5-slot project array): back then admitting a
# project COMMITTED its civs on the spot (WA_AI_PC_start_project computed them from the same
# formula), so "assigned + 20 < 0.35 x (engine_available + assigned)" was a real question. The
# dynamic queue made admission free - WA_AI_PC_assign_factories funds from the sorted queue every
# week, each strategy's _project_queue_max bounds the clutter, the Fix 41/78 lanes serve
# starvation and the 30-week sweep drops what never funds - so the trigger only ever re-asked,
# one week early, the question the fill answers. It had been repaired twice (the 0.30 floor
# mirror, the 0.35 -> 0.40 resync) to keep it from closing forever on saturated queues, and its
# last three readers were the same story: dead on saturated majors, redundant on slack ones.
# The MIC->CIC conversion legs it braked now carry an explicit budget instead
# (constant:wa_ai_pc.qmax.mic_to_cic in WA_AI_CONSTRUCTION_queue_functions.txt), and the C.10 vanilla-queue
# infra function no longer double-admits into the PC (WA_AI_resource_extraction owns that).
# The floored gates below read the allocator's own numbers - constant:wa_ai_pc.alloc.fraction /
# .stable_base_fraction / .max_civs_per_project / .max_surrender in common/script_constants/wa_ai_pc.txt,
# the SAME declaration core.txt reads (script constants since 2026-08-16; the 0.35-vs-0.40 drift of
# Fix 90 is structurally impossible now).

# THIS = country. "This economy is still standing" - the ONE rule behind every PC admission and
# allocation gate that used to spell `has_capitulated = no  surrender_progress < 0.3` by hand
# (allocator floor in WA_AI_CONSTRUCTION_PRIORITY_core.txt, the three floored civ gates below, the

WA_AI_PC_economy_alive = {
	has_capitulated = no
	surrender_progress < constant:wa_ai_pc.alloc.max_surrender
}

# THIS = country. Stable civilian-factory availability for PC strategy gates.
# num_of_civilian_factories_available_for_projects reads ~0 whenever the vanilla
# construction queue is saturated - a self-referential collapse (the PC's own
# allocation is part of what saturates it). Each gate passes when EITHER the engine
# reports real spare civs above the threshold, OR the stable floored base
# (constant:wa_ai_pc.alloc.stable_base_fraction * constant:wa_ai_pc.alloc.fraction
# = 12% of TOTAL civs since Fix 90 - was 10.5% at 0.35 - the pool WA_AI_PC_assign_factories will actually
# allocate from) clears the same threshold. Three named variants because PDXScript
# triggers cannot take a threshold argument without meta_trigger - keep the
# bodies in sync. Fix 90b (2026-08-16) retired the last bare engine-token gates in the
# PC strategies (radar > 15, islands > 19, resource extraction > 30) onto these.
WA_AI_PC_has_project_civs_16 = {	# passes at 16+ civs (replaces bare "> 15" gates)
	set_temp_variable = { _pc_civ_floor_ = num_of_civilian_factories }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.stable_base_fraction }
	multiply_temp_variable = { _pc_civ_floor_ = constant:wa_ai_pc.alloc.fraction }
	OR = {
		num_of_civilian_factories_available_for_projects > 15
		AND = {
			WA_AI_PC_economy_alive = yes
			check_variable = { _pc_civ_floor_ > 15 }
		}
	}
}

WA_AI_PC_can_build_project = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = {
			is_subject_of = ROOT
		}
	}

# THIS = STATE, PREV = BUILDER, _project_building_type, _project_province_id
# Checks if ANY builder already has a project of the same type in this state.
# Fix 46: was ROOT-only. WA_AI_PC_projects_in_state_of_@ROOT is keyed by builder, so for the
# building types with allied_build = yes (air_base, infrastructure) the host nation and its
# ally each saw an empty array for the other and both queued the same state - campaign
# 9a4cd657 1944.7 had ENG and USA simultaneously funding an air base in state 126 and in
# state 860, and the UK-hosting capacity gate then counted the same state twice as "claimed".
# The second term is the builder-agnostic counter maintained by WA_AI_PC_start_project /
# WA_AI_PC_end_project_by_id. The original ROOT-array term is KEPT deliberately: on a save
# resumed from a pre-Fix-46 build the counter starts at 0 for every in-flight project, and
# without it the guard would regress to nothing until the whole legacy queue drains.
WA_AI_PC_has_same_type_project_here = {
	OR = {
		any_of = { array = WA_AI_PC_projects_in_state_of_@ROOT value = v_proj
			check_variable = { ROOT.WA_AI_PC_building_type^v_proj = _project_building_type }
		}
		check_variable = { WA_AI_PC_state_type_projects^_project_building_type > 0 }
	}
}

### Fix 47: air-base priority band gate
# THIS/ROOT = builder country. TRUE when this country - or a faction partner it can build
# for (air_base has allied_build = yes) - fields an air force large enough that BASING is a
# plausible constraint on deploying it, rather than airframes or pilots.
#
# Why a bare size test is the right gate, with no capacity term of its own: every air-base
# project in the PC queue was created by a strategy that ALREADY proved a capacity deficit
# (WA_AI_build_uk_air_hosting_capacity's hosting-capacity gate, WA_AI_build_theatre_air_bases'
# side-wide per-theatre gate). Re-deriving the deficit here would re-run a faction-wide
# 12-airframe sum inside a trigger evaluated per candidate state. The project's existence is
# the deficit evidence; this trigger only asks whether the air force behind it is worth
# outranking a refinery for.
#
# has_deployed_air_force_size is the engine's own counter - num_deployed_planes is not
# readable as a set_temp_variable rvalue (campaign 5709c8b9), which is why both strategies
# inline 12 per-airframe sums, and why this gate deliberately uses the trigger form instead.
#
# Thresholds are absolute, on WA's air scale (~1/3 vanilla; the strategies' own token floor
# is 99 deployed planes). No major clears them before ~1940, so early-war construction
# ordering is unchanged and the blast radius is confined to the mid/late war, which is when
# the overstacking this fixes actually appears.
WA_AI_PC_air_force_needs_basing_priority = {
	OR = {
		has_deployed_air_force_size = { size > constant:wa_ai_pc.air_prio.own_planes }
		any_allied_country = {
			has_deployed_air_force_size = { size > constant:wa_ai_pc.air_prio.ally_planes }
		}
	}
}

# THIS = STATE, PREV = BUILDER, _project_building_type
# Checks if building slot is available for the specified building type
WA_AI_PC_building_slot_available_for_type = {

	if = { limit = { check_variable = { _project_building_type = 1 } }

		free_building_slots = { building = infrastructure size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 2 } }

		free_building_slots = { building = air_base size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 3 } }

		free_building_slots = { building = anti_air_building size > 0 include_locked = no }
	}

	else_if = { limit = { check_variable = { _project_building_type = 4 } }

		OR = {
			AND = { ROOT = { has_tech = radio_detection } 				radar_station < 1 }
			AND = { ROOT = { has_tech = decimetric_radar } 				radar_station < 2 }
			AND = { ROOT = { has_tech = improved_decimetric_radar } 	radar_station < 3 }
			AND = { ROOT = { has_tech = centimetric_radar } 			radar_station < 4 }
			AND = { ROOT = { has_tech = improved_centimetric_radar } 	radar_station < 5 }

# THIS = state. Central and southern England states expected to host the
# faction's aviation (within fighter range of northern France / Low Countries).
#
# Fix 51: THE ORDER OF THIS LIST IS THE BUILD PRIORITY - south-east first. It is the
# authoritative geography statement, and it is mirrored id-for-id (and rank-for-rank)
# by WA_AI_uk_air_hosting_priority_init in
# common/scripted_effects/WA_AI_CONSTRUCTION_PRIORITY_strategies.txt, which loads it
# into global.WA_AI_uk_air_priority_states. The list genuinely has to exist twice:
# @ constants are file-scoped, and a state-scope membership test cannot be built from
# a plain-id array (comparing THIS.id against a stored plain id never matches - see
# the .id scope-encoding entry in wa-lessons-learned). MEMBERSHIP stays owned here:
# the selector re-tests this trigger on every array candidate and falls back to an
# unordered scan over it, so if the two ever drift, a state can lose its RANK but
# never its eligibility. Change one, change the other.
#
# Ranking method (verified 2026-08-14, not inherited): mean pixel distance from each
# state's provinces (generated WA_AI_MAP province coordinates) to the target set the
# header names - Vlaanderen/Holland/Nord-Pas-de-Calais, Picardy/Upper+Lower Normandy/
# Brittany, Ile de France/Paris. Mean, not nearest: these fields have to cover the
# whole Northern-France + Low-Countries theatre, not one landing beach.
#   857 101.8 | 961 110.0 | 127 114.9 | 125 113.1 | 126 113.5 | 860 117.6
#   859 137.3 | 338 133.4 | 129 133.3 | 128 143.6 | 123 151.8
# Two ranks are judgement, not arithmetic:
#  - 127 Sussex is placed above 961 South London despite a 5px worse mean (inside the
#    noise of a province-centroid measure) because it is the better NEAREST field
#    (56.2 vs 58.3) and is coastal - the forward Channel fighter position.
#  - 123 Cornwall is placed LAST although it is the second most southerly state by
#    latitude. Its seaward frontage is Biscay and Brittany, not the Channel narrows:
#    its distance to the Calais/Belgium arc is 118.8, second-worst of the eleven and
#    worse than Warwickshire's 100.8. It is a Coastal Command position, not a
#    Northern-France cover position.
# States NOT in this list that hold ENG air bases in campaigns (122 Wales, 858
# Berkshire, 130 Yorkshire) are left out deliberately - adding states raises the
# structural ceiling (11 x constant:wa_ai_pc.air.uk_target_level_cap), and checklist R8 says not to do
# that until the basing OUTCOME (R15) justifies more capacity. Berkshire measures
# better than Gloucestershire (122.4 vs 133.4) and is the first candidate if that
# ever changes.
WA_AI_uk_air_hosting_state = {
	OR = {
		state = 857 # rank  1 - Kent
		state = 127 # rank  2 - Sussex
		state = 961 # rank  3 - South London
		state = 125 # rank  4 - Essex
		state = 126 # rank  5 - North London
		state = 860 # rank  6 - Norfolk
		state = 859 # rank  7 - Dorset
		state = 338 # rank  8 - Gloucestershire
		state = 129 # rank  9 - Leicestershire
		state = 128 # rank 10 - Warwickshire
		state = 123 # rank 11 - Cornwall
	}
}

# Railway eligibility thresholds. @ constants are file-scoped; before Fix 75 these lived in
# BOTH common/on_actions/WA_AI_misc_on_actions.txt (as an inline limit block) and
# WA_AI_CONSTRUCTION_PRIORITY_railway_core.txt, with a "must match" comment holding them together.
# The on_action copy is now gone - it calls the trigger below instead - so only railway_core.txt
# (authoritative: its constants block is the railway control panel the docs point at) and this
# file carry them. REGISTRY: tools/constants_registry.json (railway_eligibility_* and
# railway_max_routes_per_enemy) - `python tools/check_constants.py` verifies the copies agree.
@WA_AI_PC_railway_MIN_CIVS_PEACE = 75
@WA_AI_PC_railway_MINOR_CIV_THRESHOLD = 50
@WA_AI_PC_railway_MIN_STATES = 5
@WA_AI_PC_railway_MAX_SURRENDER = 0.3
@WA_AI_PC_railway_RECOVERY_MIN_STATES = 20
# Per-enemy route budget read by WA_AI_PC_railway_land_frontline_candidate below. Until 2026-08-16
# that test carried the bare literal 4 with a comment naming this constant, while the two
# declarations (railway_core, railway_strategies) had no reader at all.
@WA_AI_PC_railway_MAX_ROUTES_PER_ENEMY = 4

# THIS = country. Is this country able to run the railway system for itself?
#
# Fix 75: this IS the on_weekly railway eligibility gate, lifted verbatim out of
# WA_AI_misc_on_actions.txt so that the on_action and Fix 74's ally leg cannot drift apart - they
# were two hand-maintained copies of the same judgement, and R46 already listed that as a
# regression risk.
#
# `has_capitulated` is deliberately NOT a term here. It answers "did this government fall", not
# "can this country build" - the same category error R27 records for surrender_progress. A
# capitulated tag keeps its territory, its factions and its factories; what decides whether rail is
# worth building is the capability terms below. FRA on f9321934 fails this trigger anyway (21 civs
# against a 75-civ recovery hatch it needs because its surrender_progress is pinned high), which is
# why Fix 74's ally leg still fires for it and ENG/USA build the Tunisian rail.
WA_AI_PC_country_can_build_own_logistics = {
	# Higher peacetime industrial threshold - war uses a lower one inside WA_AI_PC_railway
	OR = {
		has_war = yes
		num_of_civilian_factories > @WA_AI_PC_railway_MIN_CIVS_PEACE
	}
	# Skip minor countries unless at war or rich
	OR = {
		is_major = yes
		has_war = yes
		num_of_civilian_factories > @WA_AI_PC_railway_MINOR_CIV_THRESHOLD
	}
	# Minimum state count - skip tiny nations
	num_of_controlled_states > @WA_AI_PC_railway_MIN_STATES
	# Skip countries losing badly - UNLESS the war economy still works.
	# surrender_progress measures VP loss, not capability: SOV crossed 30% in Oct 1942 with ~180
	# states and 300+ divisions and was locked out of railway building for the entire eastern war.
	OR = {
		surrender_progress < @WA_AI_PC_railway_MAX_SURRENDER
		AND = {
			num_of_civilian_factories > @WA_AI_PC_railway_MIN_CIVS_PEACE
			num_of_controlled_states > @WA_AI_PC_railway_RECOVERY_MIN_STATES
		}
	}
}

# THIS = country. Negation, for Fix 74's ally leg: an ally ROOT may build logistics FOR is one that
# cannot build them itself. Defined as the strict complement so the two can never disagree.
WA_AI_PC_country_cannot_build_own_logistics = {
	NOT = { WA_AI_PC_country_can_build_own_logistics = yes }
}

# THIS = state, ROOT = builder country. True when ROOT may queue logistics construction here.
# Superset of the two pre-Fix-74 tests: ROOT-controlled (the old ROOT loop) and subject-controlled
# (Fix 27's puppet loop), plus the new ally leg.
WA_AI_PC_can_build_logistics_here = {
	OR = {
		is_controlled_by = ROOT
		CONTROLLER = { is_subject_of = ROOT }
		AND = {
			CONTROLLER = { is_in_faction_with = ROOT }
			CONTROLLER = { WA_AI_PC_country_cannot_build_own_logistics = yes }
		}
	}
}

# THIS = country, ROOT = builder country. Countries whose controlled states ROOT may treat as its
# own when selecting logistics targets: itself, its subjects, and faction allies that cannot build
# for themselves. Country-scoped twin of WA_AI_PC_can_build_logistics_here, for the selectors that
# iterate `every_country = { limit = { ... } every_controlled_state = { ... } }` rather than states.
WA_AI_PC_is_logistics_build_partner = {
	exists = yes
	OR = {
		tag = ROOT
		is_subject_of = ROOT
		AND = {
			is_in_faction_with = ROOT
			WA_AI_PC_country_cannot_build_own_logistics = yes
		}
	}
}

# THIS = state, ROOT = builder country. Cheap membership test for the land-war railway strategy:
# is this state a frontline worth a route against the enemy currently being processed?
# Reads the chain-global temps the strategy sets per enemy (_routes_for_this_enemy, the per-enemy
# route budget; _current_enemy_tag; _check_railway_level, the level at which a frontline is "done").
# Extracted by Fix 74 so all three candidate populations - ROOT's states, subject states and
# dependent-ally states - are filtered identically before the shared body runs.
WA_AI_PC_railway_land_frontline_candidate = {
	### Skip if we already have enough routes for this enemy
	check_variable = { _routes_for_this_enemy < @WA_AI_PC_railway_MAX_ROUTES_PER_ENEMY }
	# State borders this specific enemy (or the enemy's puppets)
	any_neighbor_state = {
		OR = {
			is_controlled_by = var:_current_enemy_tag
			controller = { is_subject_of = var:_current_enemy_tag }
		}
	}
	# State has a supply hub (proper check)
	WA_AI_PC_state_has_supply_hub = yes
	# Skip states that already have level 5 railways (native check - includes focus/event built)
	NOT = { WA_AI_PC_state_has_railway_at_level = yes }
}
