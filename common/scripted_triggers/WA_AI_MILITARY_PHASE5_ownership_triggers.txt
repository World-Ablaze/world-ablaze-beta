# WA AI Military Phase 5 - Mutual exclusion ownership triggers
############################################################################################################
# These triggers express "this country owns the COUNTRY-layer ai_strategy block for
# the given Exclusive (type, key) pair", per the precedence rule documented in
# documentation/WA_AI_MILITARY_SYSTEM.md section 6 (Country > Faction > Default for
# Exclusive types).
#
# Faction-layer and Region-layer ai_strategy blocks that write the same (type, key) gate
# their `enable` block on `NOT = { WA_AI_MILITARY_country_owns_<slug> = yes }` so that
# country-specific overrides are not silently masked by faction defaults.
#
# This file is generated by tools-style scripting (see commit message); when adding a
# new Country-layer Exclusive ai_strategy block that overrides an existing Faction or
# Region default, add the country tag to the relevant trigger here.
#
# DO NOT add country tags to triggers in this file unless a Country-layer ai_strategy
# block actually writes the corresponding (type, key) - the trigger encodes ownership,
# not intention.
############################################################################################################

# [atlantic-naval] (2026-08-14): strike_force_home_base is Exclusive per region, and the
# Faction layer starts writing the western-Atlantic anchorages in
# WA_AI_NAVAL_FACTION_ALLIES_atlantic_home_base_western. The USA claims both
# regions in WA_AI_NAVAL_COUNTRY_USA_protect_eastern_waters (together
# with 170 Florida Coast, which the Faction layer does not touch), so the
# ownership is recorded here and the Faction block yields to it.
#
# These two are deliberately NOT the bare `tag = USA` that the rest of this file
# uses, and the difference matters. Elsewhere the Country-layer block is
# unconditional, so the tag alone is a faithful statement of ownership. Here the
# owning block is CONDITIONAL - it enables only once Britain has fallen or the USA
# is fighting the Axis outside the Allies - and a bare tag would therefore lock the
# USA out of the Faction anchorages in every game where those conditions never
# occur, i.e. the historical case this fix exists to serve. That is precisely the
# failure [atlantic-naval] diagnosed (checklist R36): a geopolitical tag list used as an
# exclusion, disabling a plan for the one navy able to run it. So the trigger
# mirrors the owning block's own enable and is true only while that block is live.
#
# If protect_eastern_Waters' enable is ever edited, edit these in the same commit.

# strike_force_home_base id = 55 (Newfoundland Sea)
WA_AI_MILITARY_country_owns_sfhb_55 = {
	original_tag = USA
	WA_AI_MILITARY_USA_protect_eastern_waters_active = yes
}

# strike_force_home_base id = 54 (Eastern Seaboard)
WA_AI_MILITARY_country_owns_sfhb_54 = {
	original_tag = USA
	WA_AI_MILITARY_USA_protect_eastern_waters_active = yes
}

# Mirror of WA_AI_NAVAL_COUNTRY_USA_protect_eastern_waters' enable block.
# Country-scope. Kept as one trigger so the two ownership slugs above cannot drift
# apart from each other.
WA_AI_MILITARY_USA_protect_eastern_waters_active = {
	OR = {
		AND = {
			is_in_faction_with = ENG
			has_global_flag = ENG_fall_of_the_uk
		}
		has_war_with = ENG
		AND = {
			NOT = { is_in_faction_with = ENG }
			has_war_with = GER
		}
		AND = {
			NOT = { is_in_faction_with = ENG }
			has_war_with = ITA
		}
		AND = {
			NOT = { is_in_faction_with = ENG }
			has_war_with = FRA
		}
	}
}
