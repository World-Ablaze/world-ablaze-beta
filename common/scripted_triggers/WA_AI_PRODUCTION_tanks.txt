# ============================================================================
# TANKS PRODUCTION TRIGGERS by Aigle2
# ============================================================================
# These triggers are used to determine the production of tank equipment.
# All triggers for production are done here using the PRODUCTION suffix
# so that conditions in ai_strategy files are one-liners.
#
# Tank equipment includes:
# - Light armor (tanks, TD, SPG, SPAA)
# - Medium armor (tanks, TD, SPG, SPAA)
# - Heavy armor (tanks, TD, SPG, SPAA)
# - Modern armor (tanks, TD, SPG, SPAA)
# - Mechanized equipment (APC, TD, AA, Artillery)
# ============================================================================

# ============================================================================
# MASTER SWITCH
# ============================================================================
# This trigger determines if a country uses the default tanks production system.
# It references the CONFIG trigger which contains the actual country tags.
# ============================================================================

WA_AI_PRODUCTION_tanks_is_enabled = {
	WA_AI_CONFIG_uses_default_tanks_production = yes
}

# ============================================================================
# LIGHT ARMOR TRIGGERS
# ============================================================================

WA_AI_PRODUCTION_build_light_armor = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	NOT = { WA_AI_PRODUCTION_light_armor_at_fielded_cap = yes }

		WA_AI_TEMPLATES_use_light_armor_templates = yes
		WA_AI_TEMPLATES_use_medium_armor_templates = yes

		WA_AI_TEMPLATES_use_modern_armor = yes

}

# [armor-role-budget] The light-support chassis line. Owner rule 2026-08-28: build until the fielded
# cap, then stop - the cap belongs HERE and not on the role, because a role the AI wants zero of has
# its divisions decommissioned (WA_AI_TEMPLATES_garrison.txt header) and the owner wants them kept
# until 1942. Assumes light_tank_support_chassis is the archetype the battalion needs
# (common/units/armor_support.txt); if the battalion is ever re-based on another chassis this
# trigger measures the wrong stock and silently never caps.
WA_AI_PRODUCTION_build_light_support_armor = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_use_light_support_armor_templates = yes
	NOT = { WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = yes }
}

# One question: are enough light-support chassis already in the armies. FIELDED, not stockpiled -
# losses move it down and can reopen the line, which is intended while the run is open.
WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = {
	check_variable = { num_equipment_in_armies@light_tank_support_chassis > constant:wa_ai_production.army_composition.light_support_armor_fielded_cap }
}

# [historical-tank-park] The Soviet mission also requires plain light chassis. This cap is scoped
# to its owner because other countries use light chassis as normal components and must not inherit
# the mission's 5k stop condition.
WA_AI_PRODUCTION_light_armor_at_fielded_cap = {
	WA_AI_CONFIG_pursues_historical_tank_park = yes
	check_variable = { num_equipment_in_armies@light_tank_chassis > constant:wa_ai_production.army_composition.light_armor_fielded_cap }
}

# The off switch: the country CAN build the chassis but should not - capped, or past 1942 with the
# role gone. Without it the post-1942 decommission returns chassis to the stockpile, the fielded
# count drops back under the cap, and nothing else would stop the line restarting for no role.
WA_AI_PRODUCTION_light_support_armor_line_off = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_has_light_support_armor_unlocked = yes
	NOT = { WA_AI_PRODUCTION_build_light_support_armor = yes }
}

# [mech-window] The three equipment_variant_production_factor gates below keep reading
# WA_AI_TEMPLATES_use_mechanized_templates - the WINDOWED form. A production WEIGHT has nothing to
# act on until the templates demand mech, and switching it on early is the reported symptom itself
# (the AI moving its lines onto mechanized). Only the three mech_min_factories_* FLOORS further
# down read the ungated WA_AI_TEMPLATES_mechanization_line_open, so a few factories build a buffer
# during the window and the flip does not land on a stock of 0. Weight late, floor early.

# [armor-prod-category] Gate of the armour CATEGORY demand push. One trigger, no industrial
# bands: a production factor scales the engine's own perceived need, so it needs none of the
# size tiering a need-blind floor required. Asks only whether the country fields armour at all.
WA_AI_PRODUCTION_armor_category_push = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_use_armor_templates = yes
}

# ============================================================================
# FACTORY FLOOR BANDS - MECHANIZED ONLY. The tank chassis bands were removed with their
# floors; the armor category factor is the tank lever now. Bands sit well below a healthy
# major's organic allocation, so they no-op except on a country the demand heuristic starves.
# ============================================================================

WA_AI_PRODUCTION_mech_min_factories_small = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes

	num_of_military_factories < 150
