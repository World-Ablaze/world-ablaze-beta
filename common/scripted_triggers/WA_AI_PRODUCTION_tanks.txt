# ============================================================================
# TANKS PRODUCTION TRIGGERS by Aigle2
# ============================================================================
# These triggers are used to determine the production of tank equipment.
# All triggers for production are done here using the PRODUCTION suffix
# so that conditions in ai_strategy files are one-liners.
#
# Tank equipment includes:
# - Light armor (tanks, TD, SPG, SPAA)
# - Medium armor (tanks, TD, SPG, SPAA)
# - Heavy armor (tanks, TD, SPG, SPAA)
# - Modern armor (tanks, TD, SPG, SPAA)
# - Mechanized equipment (APC, TD, AA, Artillery)
# ============================================================================

# ============================================================================
# MASTER SWITCH
# ============================================================================
# This trigger determines if a country uses the default tanks production system.
# It references the CONFIG trigger which contains the actual country tags.
# ============================================================================

WA_AI_PRODUCTION_tanks_is_enabled = {
	WA_AI_CONFIG_uses_default_tanks_production = yes
}

# [armor-role-budget] The light-support chassis line. Owner rule 2026-08-28: build until the fielded
# cap, then stop - the cap belongs HERE and not on the role, because a role the AI wants zero of has
# its divisions decommissioned (WA_AI_TEMPLATES_garrison.txt header) and the owner wants them kept
# until 1942. Assumes light_tank_support_chassis is the archetype the battalion needs
# (common/units/armor_support.txt); if the battalion is ever re-based on another chassis this
# trigger measures the wrong stock and silently never caps.
WA_AI_PRODUCTION_build_light_support_armor = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_use_light_support_armor_templates = yes
	NOT = { WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = yes }
}

# One question: are enough light-support chassis already in the armies. FIELDED, not stockpiled -
# losses move it down and can reopen the line, which is intended while the run is open.
WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = {
	check_variable = { num_equipment_in_armies@light_tank_support_chassis > constant:wa_ai_production.army_composition.light_support_armor_fielded_cap }
}

# [historical-tank-park] The Soviet mission also requires plain light chassis. This cap is scoped
# to its owner because other countries use light chassis as normal components and must not inherit
# the mission's 5k stop condition.
WA_AI_PRODUCTION_light_armor_at_fielded_cap = {
	WA_AI_CONFIG_pursues_historical_tank_park = yes
	check_variable = { num_equipment_in_armies@light_tank_chassis > constant:wa_ai_production.army_composition.light_armor_fielded_cap }
}

# The off switch: the country CAN build the chassis but should not - capped, or past 1942 with the

# [modern-switch-amorce] THREE disjoint states for the medium MAIN GUN weight, and the class
# weight follows the chassis the country is GROWING - never the one it currently fields.
#   1. window shut          -> full class weight, medium is the main battle tank
#   2. priming (this one)   -> NEGATIVE: the country is buying a modern stock, and medium is what
#                              was stopping it. MEASURED 1943.5 (GER, test_bascule / bascule2):
#                              the 30-factory priming floor was REQUESTED and 0 assigned, because
#                              medium held +150. A need-blind min_factories floor does NOT preempt
#                              a saturated pool - the competitor has to come down, which is the
#                              owner ruling of 2026-09-20.
#   3. switched             -> the legacy weight of a chassis that is no longer the main gun
# The park is not starved in state 2: its demand is unchanged and its demand-driven maintenance
# floor (WA_AI_PRODUCTION_should_maintain_medium_tank) is need-blind and untouched by a factor.
# The fact is gone when a save inside the window shows the modern line still at 0 assigned.
WA_AI_PRODUCTION_should_damp_medium_main_for_priming = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_use_medium_armor_templates = yes
	WA_AI_TEMPLATES_modern_line_open = yes
	NOT = { WA_AI_TEMPLATES_modern_chassis_owns_medium_role = yes }
}

# [modern-switch-amorce] The class weight belongs to whichever chassis the park FIELDS - not to
# the one the role targets. Before the switch, and all through the conversion, that is
# medium_tank_chassis; once most of the park's main-gun demand is modern, medium is a LEGACY
# line - still wanted (light-support family, 1936 OOB divisions) but no longer the main battle
# tank. Leaving the full class weight on it is what kept the AI's factories on medium after the
# switch; dropping it AT the switch would instead starve a park that has not converted yet.
# Both chassis carry 150 during the conversion window, on purpose: un-converted divisions need
# medium and converted ones need modern, both at once.
# ONLY the main gun chassis moves: the same target keeps 6 medium_support_armor_battalion_line,
# i.e. 150 medium_tank_support_chassis per division, so every VARIANT factor is left alone and
# the [armor-prod-category] parity rule holds for them. The fact is gone when a post-switch save
# shows medium_tank_chassis lines outbidding modern ones while the park is already modern.
WA_AI_PRODUCTION_build_medium_armor = {
	WA_AI_PRODUCTION_tanks_is_enabled = yes
	WA_AI_TEMPLATES_use_medium_armor_templates = yes
	NOT = { WA_AI_TEMPLATES_modern_line_open = yes }
	NOT = { WA_AI_TEMPLATES_modern_chassis_owns_medium_role = yes }
}

# [modern-switch-amorce] PRIMING line for the modern chassis. The maintenance floor
# (WA_AI_PRODUCTION_should_maintain_modern_tank) reads a demand that only exists once a template
# mounts the battalion, so it cannot arm before the switch and the switch landed on a stockpile
# of 0. A min_factories floor and not a production factor: a factor multiplies a NEED, and no
# template demands the chassis yet. The three bands are DISJOINT by construction (same constant
# read with `>` and `NOT >`), which matters because floors on one id SUM. The NOT term retires
# the line at the switch, where the need-driven factor and the maintenance floor take over; it
# assumes the latch is one-way, and the fact is gone when the wa_armor_budget harness shows a
# country carrying a priming floor while its latch reads 1.

# [modern-switch-amorce] CATCH-UP line: the switch retargets the whole medium park in one
# month, so the month after it the park is short by design. This is the ON/OFF half of the
# subject - it reads the DEFICIT (chassis in the armies against what those armies require),
# not a stockpile, arms a large floor while the park is hollow and releases when it is not.
# Hysteresis lives HERE and not on the template gate: a template gate that toggles
# decommissions whatever template holds the role (MEASURED, wa-lessons-learned), a factory
# floor that toggles costs nothing. It SUMS with the flat maintenance floor on the same id,
# intended - replacing losses and re-equipping a retargeted park are different problems.
# Assumes the engine's own need-driven allocation is not enough on its own; the fact is gone
# when a post-switch save shows modern lines already served above these counts.
WA_AI_PRODUCTION_is_modern_chassis_underfilled = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies@modern_tank_chassis > constant:wa_ai_production.armor.modern_switch_min_park } }
		set_temp_variable = { _mod_fill = num_equipment_in_armies@modern_tank_chassis }
		divide_temp_variable = { _mod_fill = num_target_equipment_in_armies@modern_tank_chassis }
		check_variable = { _mod_fill < constant:wa_ai_production.armor.modern_catchup_fill_enable }
	}
	else = { always = no }
}

WA_AI_PRODUCTION_is_modern_chassis_under_release_bar = {
	if = {
		limit = { check_variable = { num_target_equipment_in_armies@modern_tank_chassis > constant:wa_ai_production.armor.modern_switch_min_park } }
		set_temp_variable = { _mod_fill = num_equipment_in_armies@modern_tank_chassis }
		divide_temp_variable = { _mod_fill = num_target_equipment_in_armies@modern_tank_chassis }
		check_variable = { _mod_fill < constant:wa_ai_production.armor.modern_catchup_fill_abort }
	}
	else = { always = no }
}
