	WA_AI_CONFIG_uses_default_army_composition = yes
}

# [armor-role-budget] The four *_armor triggers below ARE the budget's slot definition: each answers
# "is this tank role open for this country", and the budget is split evenly between the open ones
# (common/scripted_effects/WA_AI_PRODUCTION_armor_budget.txt). They no longer gate an ai_strategy
# block. Assumes each slot maps to exactly one role_ratio id - light_armor covers BOTH light
# templates because WA_light_support_armor_role declares role = light_armor.

WA_AI_PRODUCTION_build_army_light_armor = {
	# [historical-tank-park] The Soviet temporary corps is recruitable from 1936, then the whole
	# light role closes once its historical park is complete. Ahistorical SOV keeps the normal path.
	NOT = {
		AND = {
			WA_AI_DIFFICULTY_is_historical = yes
			WA_AI_CONFIG_pursues_historical_tank_park = yes
			NOT = { WA_AI_PRODUCTION_should_continue_historical_tank_park = yes }
		}
	}
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	OR = {
		NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
		AND = {
			WA_AI_CONFIG_pursues_historical_tank_park = yes
			WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
			has_country_flag = WA_AI_TEMPLATES_sov_light_support_temporary_latched
		}
	}
	OR = {
		WA_AI_TEMPLATES_use_light_armor_templates = yes
		WA_AI_TEMPLATES_use_light_support_armor_templates = yes
	}
}

# The transition path is the second way into the medium slot: a country past the light-to-medium
# switch builds medium templates even before it passes WA_AI_CONFIG_TEMPLATES_admits_medium_armor.
# It used to be a SECOND static block, which double-counted medium (ENG read 40 against 10 for every
# other role); inside the budget it is one slot either way.
WA_AI_PRODUCTION_build_army_medium_armor = {
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	OR = {
		WA_AI_TEMPLATES_use_medium_armor_templates = yes
		WA_AI_TEMPLATES_switch_from_light_to_medium_armor = yes
	}
}

WA_AI_PRODUCTION_build_army_heavy_armor = {
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	WA_AI_TEMPLATES_use_heavy_armor_templates = yes
}

# [mech-divisions-usa-only] Owner rule 2026-08-28: mechanized DIVISIONS are for the expeditionary
# mechanized major and nobody else. Reads the narrow template trigger, so the role_ratio share and
# the target template open and close together - a share with no target is wasted, a target with no
# share gets its divisions decommissioned. Mechanized inside TANK divisions is untouched: that runs
# off WA_AI_TEMPLATES_use_mechanized_templates, which stays open for everyone.
WA_AI_PRODUCTION_build_army_mechanized = {
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	WA_AI_TEMPLATES_use_mechanized_division_templates = yes
}

# Expeditionary mechanized majors (currently only the USA, per
# WA_AI_CONFIG_DIVISIONS_is_expeditionary_mechanized_major: huge fuel and industry,
# no threatened land border) fight overseas, not at home. Their late-war army
# should be mostly mobile - infantry capped around 30% of the role_ratio total.
# Fires in historical games unconditionally; in non-historical games only after
# 1942 and only while no enemy actually sits on the border.
WA_AI_PRODUCTION_build_army_expeditionary_mechanization = {
	NOT = { WA_AI_CONFIG_early_game_army_expansion_override = yes }
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	# Requires the mechanized role to be active, so the extra ratio is never
	# assigned to a role the country cannot design or build yet.
	WA_AI_PRODUCTION_build_army_mechanized = yes
	WA_AI_CONFIG_DIVISIONS_is_expeditionary_mechanized_major = yes
	OR = {
		WA_AI_DIFFICULTY_is_historical = yes
		AND = {
			WA_AI_CONFIG_after_global_war_begins = yes
			NOT = { any_neighbor_country = { has_war_with = ROOT } }
		}
	}
}

# [historical-tank-park] The live run state, deliberately independent of templates: templates
# consume it to decide whether the normal light-to-medium handoff may close the light role.
WA_AI_PRODUCTION_should_continue_historical_tank_park = {
	WA_AI_DIFFICULTY_is_historical = yes
	WA_AI_CONFIG_pursues_historical_tank_park = yes
	NOT = { any_enemy_country = { is_major = yes } }
	OR = {
		NOT = { WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = yes }
		NOT = { WA_AI_PRODUCTION_light_armor_at_fielded_cap = yes }
	}
	if = {
		limit = { tag = SOV }
		has_active_mission = SOV_the_greatest_tank_army
	}
}

# [historical-tank-park] Force-build the mission for its entire live run. Template selection is
# independent state and must not withdraw the mission's 500 light-role desire mid-run. Payload is
# WA_AI_PRODUCTION_DEFAULT_historical_tank_park.
WA_AI_PRODUCTION_should_build_historical_tank_park = {
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
}

# [historical-tank-park] The 500 light-role force lasts for the live mission, not merely until
# either chassis bar is full. Keep the existing major-war exit: this is a pre-war historical push.
WA_AI_PRODUCTION_should_force_historical_tank_park = {
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	WA_AI_DIFFICULTY_is_historical = yes
	WA_AI_CONFIG_pursues_historical_tank_park = yes
	NOT = { any_enemy_country = { is_major = yes } }
	if = {
		limit = { tag = SOV }
		has_active_mission = SOV_the_greatest_tank_army
	}
}

# [light-support-conversion] Gate of the armor-budget light floor (WA_AI_ARMOR_BUDGET_reconcile
# section 2b). It shares the force-build run state so the fielded cap or major-war exit removes
# both the +500 force and the light-role floor.
WA_AI_PRODUCTION_should_floor_tank_park_budget = {
	WA_AI_PRODUCTION_army_composition_is_enabled = yes
	WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
	WA_AI_TEMPLATES_use_light_support_armor_templates = yes
}

# [division-target-scaling] One question: does this country need a bigger theoretical army than the
# global division-target weights give it, in the run-up to the war it starts. Tester report
# 2026-08-28: GER and ITA under-build until mid-1940. Payload is
# WA_AI_PRODUCTION_DEFAULT_early_war_army_target_boost. The window closes at 1940.7.1 because after
# the fall of France the war itself drives the target (threat and war factors multiply it) and the
# boost would compound with them.
WA_AI_PRODUCTION_early_war_army_target_boost = {
	WA_AI_CONFIG_MILITARY_is_axis_european_major = yes
	date < 1940.7.1
}

# One question: must this country train no new divisions at all? Payload lives in
# WA_AI_PRODUCTION_DEFAULT_trains_no_divisions (ai_strategy/..._army_composition.txt).
# Reichskommissariat branch: ITS never passes WA_AI_CONFIG_is_reichskommissariat.
# Micro-industry branch: has_civil_war exemption per the SCW ruling that brakes
# must not decide a civil war. The 5 stays while the fitness floor moved to > 10
# (owner 2026-08-27): 5-9 factories = trains freely at home, not sent abroad
# (registry group expeditionary_fitness_mil_factory_floor, advisory).
WA_AI_PRODUCTION_trains_no_divisions = {
	OR = {
		WA_AI_CONFIG_is_reichskommissariat = yes
		AND = {
			num_of_military_factories < 5
			num_divisions > 4
			has_civil_war = no
		}
	}
}
