# Continental corridor (JAP_island_defence_revision, JAP_island_garrisons in common/national_focus/japan.txt).
# JAP_debug_corridor_open (debug decision) forces both corridor triggers true.

# Japan or its subject (the "control" of the WA required_states tooltips).
JAP_corridor_our_side = {
	OR = {
		tag = JAP
		is_subject_of = JAP
	}
}

# Operation Ichigo has opened the corridor (Lay the Continental Railway): we hold every corridor state from Beijing
# to Nanning, and no enemy holds Tonkin (French Indochina may stay neutral, as it did).
JAP_continental_corridor_held = {
	custom_trigger_tooltip = {
		tooltip = JAP_continental_corridor_held_tt
		OR = {
			JAP = { has_country_flag = JAP_debug_corridor_open }
			AND = {
				608 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# Beijing
				607 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# Zhengzhou
				620 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# Wuhan
				602 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# North Hunan: Changsha, Hengyang
				599 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# North Guangxi: Guilin
				932 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# Guangxi: Liuzhou
				594 = { CONTROLLER = { JAP_corridor_our_side = yes } }	# Nanning
				671 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Tonkin: Hanoi
			}
		}
	}
}

# The corridor stays open (Carry the Southern Resources Overland) as long as no enemy holds any corridor state,
# the wording of the vanilla Chinese supply route decisions (CHI_highlight_the_hanoi_decision_tt).
JAP_continental_corridor_open = {
	OR = {
		JAP = { has_country_flag = JAP_debug_corridor_open }
		AND = {
			608 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Beijing
			607 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Zhengzhou
			620 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Wuhan
			602 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# North Hunan: Changsha, Hengyang
			599 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# North Guangxi: Guilin
			932 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Guangxi: Liuzhou
			594 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Nanning
			671 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Tonkin: Hanoi
		}
	}
}

# Far Eastern resources (JAP_far_east_resources_overland): open as long as no enemy holds Chita or Ulan-Ude, where the
# Trans-Siberian Railway enters the Far East.
JAP_far_east_route_open = {
	OR = {
		JAP = { has_country_flag = JAP_debug_corridor_open }
		AND = {
			563 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Chita
			564 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Ulan-Ude
		}
	}
}

# Indian and Burmese resources (JAP_india_resources_overland): open as long as the continental corridor is open and no
# enemy holds Bangkok, Burma or East Bengal on the Second Trunk Railway Group.
JAP_india_route_open = {
	OR = {
		JAP = { has_country_flag = JAP_debug_corridor_open }
		AND = {
			JAP_continental_corridor_open = yes
			289 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Bangkok
			288 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# Burma: Rangoon
			430 = { CONTROLLER = { NOT = { has_war_with = JAP } } }	# East Bengal: Chittagong
		}
	}
}
