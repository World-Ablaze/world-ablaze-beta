# Continental corridor (JAP_island_defence_revision, JAP_island_garrisons in common/national_focus/japan.txt).

# Japan, its subjects or its faction.
JAP_corridor_our_side = {
	OR = {
		tag = JAP
		is_subject_of = JAP
		is_in_faction_with = JAP
	}
}

# Operation Ichigo has opened the corridor: our side holds the cities from Beijing to Nanning, and no enemy holds
# Hanoi (French Indochina may stay neutral, as it did). JAP_debug_corridor_open (debug decision) forces it open.
JAP_continental_corridor_held = {
	custom_trigger_tooltip = {
		tooltip = JAP_continental_corridor_held_tt
		OR = {
			JAP = { has_country_flag = JAP_debug_corridor_open }
			AND = {
				any_country = { JAP_corridor_our_side = yes controls_province = 9843 }	# Beijing
				any_country = { JAP_corridor_our_side = yes controls_province = 9958 }	# Zhengzhou
				any_country = { JAP_corridor_our_side = yes controls_province = 4619 }	# Wuhan
				any_country = { JAP_corridor_our_side = yes controls_province = 7097 }	# Changsha
				any_country = { JAP_corridor_our_side = yes controls_province = 7502 }	# Hengyang
				any_country = { JAP_corridor_our_side = yes controls_province = 10404 }	# Guilin
				any_country = { JAP_corridor_our_side = yes controls_province = 4028 }	# Liuzhou
				any_country = { JAP_corridor_our_side = yes controls_province = 7137 }	# Nanning
				any_country = { controls_province = 4075 NOT = { has_war_with = JAP } }	# Hanoi
			}
		}
	}
}
