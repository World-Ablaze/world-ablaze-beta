####################################################################################################
#	World Ablaze AI - RAILWAY priority-construction constants (the railway 'control panel').
#	Read from script with `constant:wa_ai_railway.<group>.<key>` (HOI4 1.18 script constants). One
#	declaration visible from railway_core / helpers / strategies AND from WA_AI_CONSTRUCTION_triggers.txt
#	(eligibility filter, per-enemy route budget) - the file-scoped `@` copies that had to be kept in
#	sync by hand are gone (2026-08-16). Bands and the type id are PC-wide: constant:wa_ai_pc.prio.rail_war
#	/ .rail_prewar / constant:wa_ai_pc.type_id.rail (wa_ai_pc.txt). Cost mirrors of 00_buildings.txt are
#	tracked by tools/constants_registry.json - run `python tools/check_constants.py` after editing.
#	Full restart (not `reload`) when tuning: the game reloads this database separately from the scripts.
#	System doc: documentation/WA_AI_RAILWAY_SYSTEM.md.
####################################################################################################

wa_ai_railway = {
	schema = {
		any_key = yes
		data = {
			{
				any_key = yes
				data = fixed_point
			}
		}
	}

	# WA_AI_PC_railway run interval, in weeks (the counter is decremented by the weekly pulse).
	interval = {
		peace_weeks = 12	# ~6 months during peace
		              # was @WA_AI_PC_railway_INTERVAL_PEACE
		war_weeks   = 8	# ~3 months during war
		              # was @WA_AI_PC_railway_INTERVAL_WAR
	}

	# Which countries run the railway system - WA_AI_PC_country_can_run_railway_system (triggers file, on_weekly)
	# and the civ floor inside WA_AI_PC_railway (railway_core). Fix 43 escape hatch = recovery_min_states.
	eligibility = {
		min_civs            = 50	# base civ minimum inside the run, peacetime side of the run's own floor
		                      # was @WA_AI_PC_railway_MIN_CIVS
		min_civs_war        = 30	# wartime side of the same floor. Fix 104: was min_civs x war_civs_factor
		                      # (0.6) computed at each call site. It is an absolute now because a scripted
		                      # TRIGGER cannot multiply, and WA_AI_PC_country_can_fund_own_logistics must
		                      # read the SAME floor the run applies - a hand-synced second copy is exactly
		                      # what script constants exist to prevent.
		                      # The 0.6 ratio is NOT a rule any more and nothing enforces one: the registry
		                      # only checks equality groups, so a ratio contract here would be prose
		                      # pretending to be a mechanism. The one invariant a human must hold is
		                      # min_civs_war <= min_civs - a wartime floor above the peacetime floor would
		                      # mean a country qualifies in peace and stops qualifying when the war it was
		                      # preparing for starts.
		min_civs_peace      = 75	# higher threshold during peace
		                      # was @WA_AI_PC_railway_MIN_CIVS_PEACE
		min_states          = 5	# minimum controlled states to run
		                      # was @WA_AI_PC_railway_MIN_STATES
		max_surrender       = 0.3	# skip above this surrender progress...
		                      # was @WA_AI_PC_railway_MAX_SURRENDER
		recovery_min_states = 20	# ...unless civs > min_civs_peace and states > this (war economy intact, Fix 43)
		                      # was @WA_AI_PC_railway_RECOVERY_MIN_STATES
		minor_civ_threshold = 50	# minor nations bypass the state gate above this many civs
		                      # was @WA_AI_PC_railway_MINOR_CIV_THRESHOLD
	}

	# Route caps per execution and queue sizing. queue_full is BOTH the recalculation-skip threshold
	# (railway_core, Fix 29b: live type-13 projects only) AND the Fix 77 admission cap on queued segments
	# (railway_helpers) - they were two @ names 'equal BY DESIGN'; one key makes the design structural.
	routes = {
		max_total                    = 8	# max routes processed per run
		                               # was @WA_AI_PC_railway_MAX_ROUTES_TOTAL
		max_per_enemy                = 4	# max routes per enemy country (enforced by WA_AI_PC_railway_land_frontline_candidate)
		                               # was @WA_AI_PC_railway_MAX_ROUTES_PER_ENEMY
		queue_full                   = 12	# 'a full queue is 12' - skip recalculation at/above, and never admit past it (Fix 77)
		                               # was @WA_AI_PC_railway_QUEUE_SKIP_THRESHOLD, @WA_AI_PC_railway_MAX_QUEUED
		partial_path_priority_factor = 0.7	# partial paths (dead-ends) get 70% priority
		                               # was @WA_AI_PC_railway_PARTIAL_PATH_PRIORITY_FACTOR
		theatre_separation_distance  = 10	# helpers: distance beyond which two targets are separate theatres
		                               # was @WA_AI_PC_railway_THEATRE_SEPARATION_DISTANCE
	}

	# Fix 95 - THEATRE CORRIDORS (WA_AI_PC_railway_corridor_pass in railway_core, strategy
	# WA_AI_PC_railway_STRATEGY_theatre_corridor in railway_strategies, node data in
	# WA_AI_CONSTRUCTION_PRIORITY_corridor_data.txt). A corridor is an ordered list of named
	# province nodes; whoever holds two consecutive nodes (ROOT / subject / Fix-74 partner soil)
	# queues the rail between them plus the node's missing depot/port. Runs on its OWN interval,
	# independent of interval.war_weeks: one type-13 project = one level on one hop and the
	# province-duplicate guard admits one project per province per pass, so at 8-week cadence a
	# 20-hop route to level 1 took 24 weeks - the pass cadence, not the civ pool, was the bound.
	corridor = {
		interval_weeks     = 4	# corridor pass cadence (weeks); the counter ticks on the weekly pulse
		# Fix 107: rail_level (a flat TARGET of 2) became a floor/cap PAIR around a computed target.
		# Protects against: sizing with no reference to demand. Campaign 9d83084c measured every
		# corridor hop at exactly 2 on all 121 saves, 1940.12 to 1946.2 - the system was not blocked,
		# it had finished, and 4 + 8x2 = 20 supply was the ceiling by construction.
		# Assumes: the engine railway flow ladder 4 + 8xlevel (supply.railway_* below, registered).
		# Fact is gone when: that ladder is retuned - then both numbers below mean a different flow.
		rail_level_floor   = 2	# never size BELOW this. Keeps the change monotone: a corridor with no divisions
		                    	# on it still gets the pre-Fix-107 level, so a peacetime corridor cannot regress.
		rail_level_cap     = 4	# never size ABOVE this. Railways cannot be downgraded (LOGISTICS_MODEL section 10
		                    	# rule 3: overbuilding is permanent, underbuilding is recoverable), so the cap is the
		                    	# irreversibility budget. 4 = 36 supply. User ruling 2026-08-18, against the campaign
		                    	# target "Allied victory in North Africa in early 1943".
		# Demand model (LOGISTICS_MODEL section 11.4). Both are CALIBRATION KNOBS, not measurements.
		# Protects against: nothing yet - they are the first demand estimate this codebase has.
		# Assumes: supply_per_division is an average need per division, and that the presence index
		# WA_AI_PC_corridor_compute_sizing builds is a usable proxy for a division count (it is NOT a
		# count - it over-reads a dispersed army; see that effect's header).
		# Fact is gone when: a campaign shows rail targets systematically above or below what the
		# front actually needed - retune here, not in the effect.
		supply_per_division = 2.5	# average supply need per division. SUPPOSE, placeholder of 2026-08-17.
		target_ratio        = 0.4	# target supply ratio = SUPPLY_THRESHOLD_FOR_ARMY_ATTRITION (05_defines.lua):
		                     	# the NO-ATTRITION floor, not an optimum - combat and speed penalties still
		                     	# scale below 1.0. Sizing against 1.0 would overbuild permanently.
		# Port-vs-rail cost arbitration (LOGISTICS_MODEL section 11.7). DERIVED from the prices PC
		# ACTUALLY CHARGES, which are not the engine's:
		#   port  = constant:wa_ai_pc.cost.naval_base (10000) FLAT per level / supply.port_per_level (5)  = 2000 per point
		#   rail  = constant:wa_ai_pc.cost.railway (800) per province connection x N / railway_per_level (8) = 100 x N
		# N is the WHOLE chain from the entry port to the far node, not one hop: the chain is in SERIES,
		# so a point at the far end costs a level on EVERY connection (LOGISTICS_MODEL section 10 rule 3).
		# Break-even: 100 x N = 2000 -> N = 20.
		# DEVIATION from LOGISTICS_MODEL section 11.7, recorded: that section derived 10-14 from the ENGINE
		# price with its -556/level taper (constant:wa_ai_railway.cost.naval_base_per_level). PC is a shadow
		# system - it never pays the engine price, it pays cost.naval_base FLAT (see that key's header, Fix 72),
		# so the port is dearer than section 11.7 assumed and the break-even moves out.
		# Protects against: defaulting to one lever. Below 20 the rail is cheaper per point AND ~12x faster
		# to deliver (800/F days per segment against 10000/F for one port level, and 8 rail segments may be
		# in flight at once against 1 port per province).
		# Fact is gone when: cost.naval_base, cost.railway, port_per_level or railway_per_level changes -
		# RE-DERIVE BY HAND, nothing checks it (noted on cost_naval_base in tools/constants_registry.json).
		port_break_even_segments = 20	# at or above this many provinces on the chain, raise the PORT; below, the RAIL
		queue_max          = 8	# concurrent corridor projects per builder PER BUILDING TYPE (WA_AI_PC_start_project filters
		                    	# building_type before the type_id budget): 8 rails + 8 depots + 8 ports at most, tag corridor, Fix 47 scoped
		max_routes_per_run = 14	# pathfinds per corridor pass = the whole node-pair list of the largest corridor (15 nodes ->
		                    	# 14 pairs). NOT a rotating window: the pass validates every corridor project against the
		                    	# routes it pathfound, so a window smaller than the list would cancel the in-progress
		                    	# segments of the routes it skipped. Hops are 1-5 provinces (A* cap 100), so this is cheap.
	}

	# Supply-throughput model used to size ports against railways (railway_helpers).
	supply = {
		railway_base              = 4	# railway throughput = base + per_level x level
		                            # was @WA_AI_PC_SUPPLY_RAILWAY_BASE
		railway_per_level         = 8	# L1=12, L2=20, L3=28, L4=36, L5=44
		                            # was @WA_AI_PC_SUPPLY_RAILWAY_PER_LEVEL
		port_per_level            = 5	# port throughput = level x 5
		                            # was @WA_AI_PC_SUPPLY_PORT_PER_LEVEL
		port_max_useful_level     = 9	# L5 railway = 44 supply, L9 port = 45
		                            # was @WA_AI_PC_PORT_MAX_USEFUL_LEVEL
		home_port_search_distance = 5	# max BFS distance from capital
		                            # was @WA_AI_PC_HOME_PORT_SEARCH_DISTANCE
		home_port_target_supply   = 44	# target supply capacity (L5 railway)
		                            # was @WA_AI_PC_HOME_PORT_TARGET_SUPPLY
	}

	# Route SCORING costs (railway_helpers). The base prices are the PC-wide shadow prices
	# constant:wa_ai_pc.cost.railway / .naval_base (one key for scoring AND charging - Fix 73 made them agree,
	# 2026-08-16 made them one declaration); only the naval-base per-level taper and the segment estimate
	# live here. MUST MATCH common/buildings/00_buildings.txt - registered (cost_naval_base_per_level).
	cost = {
		naval_base_per_level       = -556	# naval_base.per_level_extra_cost (negative taper)
		                             # was @WA_AI_PC_NAVAL_BASE_PER_LEVEL_COST
		railway_segments_per_state = 3	# estimated railway segments per state
		                             # was @WA_AI_PC_RAILWAY_SEGMENTS_PER_STATE
	}

}
