	}
	text = {

	}
}
# LOC-KEY TRAP: defined_text resolves localization_key through the localisation
# database, so a returned string that happens to be a loc key (every full unit name
# is one, via afo_unit_l_english.yml) comes back as its localised display name and
# the engine silently drops the invalid token from the template. Every defined_text
# in this file must therefore return a fragment that is NOT a loc key (trailing
# underscore, embedded " = { ... }", or plain suffix); the meta_effect in
# WA_AI_DIVISION_CREATOR_effects.txt supplies the literal remainder of the name.

# Returns the engineer support unit name suffix: the motorized engineer is a
# battalion and the horse engineer is a company, so a [SUPPORT_MOT] token
# substitution cannot build a valid name from a single pattern (see
# common/units/support_engineer.txt). Spliced as engineer_[SUPPORT_ENGINEER].
defined_text = {
	name = WA_AI_DIVISION_GetSpawnDivisionTemplateSupportEngineer
	text = {
		trigger = {
			WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
		}
		localization_key = "mot_battalion_divisional"
	}
	text = {
		localization_key = "horse_company_divisional"
	}
}
