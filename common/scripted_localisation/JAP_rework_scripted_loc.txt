# ── 日本の派閥「態度」の段階表示(Japan-Rework) ──

# 変数 JAP_<faction>_attitude は on_actions/JAP_rework_on_actions.txt の on_startup で初期化。

defined_text = {
	name = JAP_GetKantaihaAttitudeLevel
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude > 1 }
		}
		localization_key = USA_faction_loyal_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = 1 }
		}
		localization_key = USA_faction_friendly_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = 0 }
		}
		localization_key = USA_faction_calm_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = -1 }
		}
		localization_key = USA_faction_worried_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = -2 }
		}
		localization_key = USA_faction_agitated_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = -3 }
		}
		localization_key = USA_faction_angry_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude = -4 }
		}
		localization_key = USA_faction_violent_attitude
	}
	text = {
		trigger = {
			original_tag = JAP
			check_variable = { JAP_kantaiha_attitude < -4 }
		}
		localization_key = USA_faction_rebellious_attitude
	}
}
