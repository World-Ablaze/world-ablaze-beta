# Records the day Japan went to war with the United States or Britain, which the Combined Fleet's
# Second Stage operations will count from.
on_actions = {
	on_daily_JAP = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = JAP_pacific_war_started }
					OR = {
						has_war_with = USA
						has_war_with = ENG
					}
				}
				set_country_flag = JAP_pacific_war_started
			}
		}
	}
}
