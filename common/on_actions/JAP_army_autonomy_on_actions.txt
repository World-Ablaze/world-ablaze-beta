# Starts the focuses the army takes on its own (events/JAP_army_autonomy.txt) so that they complete on
# the historical date: 70 days before it, 21 for the border fortresses and the Kantokuen.
on_actions = {
	on_daily_JAP = {
		effect = {
			# Border Dispute Guidelines, issued 25 April 1939 (not while the player is working on it by hand)
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_border_conflict_guidelines }
					NOT = { has_shine_effect_on_focus = JAP_border_conflict_guidelines }
					NOT = { has_country_flag = JAP_border_conflict_guidelines_started }
					NOT = { focus_progress = { focus = JAP_border_conflict_guidelines progress > 0 } }
					date > 1939.2.13
				}
				set_country_flag = JAP_border_conflict_guidelines_started
				country_event = { id = jap_army_auto.1 }
			}
			# Northern Indochina, crossing on 23 September 1940 (once France has fallen and while at war with China,
			# not while the player is working on it by hand)
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_northern_indochina_occupation }
					NOT = { has_shine_effect_on_focus = JAP_northern_indochina_occupation }
					NOT = { has_country_flag = JAP_northern_indochina_occupation_started }
					NOT = { focus_progress = { focus = JAP_northern_indochina_occupation progress > 0 } }
					date > 1940.7.14
					has_war_with = CHI
					OR = {
						has_global_flag = fall_of_france
						FRA = { has_capitulated = yes }
					}
					671 = {
						OWNER = {
							NOT = { tag = JAP }
							NOT = { has_war_with = JAP }
						}
					}
				}
				set_country_flag = JAP_northern_indochina_occupation_started
				country_event = { id = jap_army_auto.3 }
			}
			# Border fortresses, alongside Northern Indochina on 23 September 1940 (not while the player is
			# working on it by hand)
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_border_fortifications }
					NOT = { has_shine_effect_on_focus = JAP_border_fortifications }
					NOT = { has_country_flag = JAP_border_fortifications_started }
					NOT = { focus_progress = { focus = JAP_border_fortifications progress > 0 } }
					date > 1940.9.1
				}
				set_country_flag = JAP_border_fortifications_started
				country_event = { id = jap_army_auto.7 }
			}
			# Kantokuen, second mobilisation on 16 July 1941 (not while the player is working on it by hand)
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_kantokuen }
					NOT = { has_shine_effect_on_focus = JAP_kantokuen }
					NOT = { has_country_flag = JAP_kantokuen_started }
					NOT = { focus_progress = { focus = JAP_kantokuen progress > 0 } }
					date > 1941.6.24
					country_exists = SOV
				}
				set_country_flag = JAP_kantokuen_started
				country_event = { id = jap_army_auto.5 }
			}
		}
	}
}
