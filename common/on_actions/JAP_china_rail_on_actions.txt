on_actions = {
	on_startup = {
		effect = {
			# Numeric state scopes in a scripted effect do not resolve from the scopeless on_startup (JAP_rework_on_actions.txt),
			# hence JAP. Once only, so loading a save does not add them again.
			JAP = {
				if = {
					limit = { NOT = { has_global_flag = JAP_china_rail_set_up } }
					set_global_flag = JAP_china_rail_set_up
					JAP_china_long_supply_setup = yes
					JAP_china_hard_to_build_setup = yes
				}
			}
		}
	}
}
