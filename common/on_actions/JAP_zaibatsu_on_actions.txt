on_actions = {
	# The last levels of the zaibatsu, while hired (build_zaibatsu_tree.py): Yasuda takes 1% economic fatigue off every 120 days,
	# Nissan builds three building slots and three military factories in Manchuria every 60 days.
	on_monthly_JAP = {
		effect = {
			if = {
				limit = { has_idea = JAP_zb_yasuda_5 }
				add_to_variable = { JAP_zb_yasuda_months = 1 }
				if = {
					limit = { check_variable = { JAP_zb_yasuda_months > 3 } }
					set_variable = { JAP_zb_yasuda_months = 0 }
					economy_fatigue_level_down_1 = yes
				}
			}
			if = {
				limit = { has_idea = JAP_zb_nissan_5 }
				add_to_variable = { JAP_zb_nissan_months = 1 }
				if = {
					limit = { check_variable = { JAP_zb_nissan_months > 1 } }
					set_variable = { JAP_zb_nissan_months = 0 }
					if = {
						limit = {
							country_exists = MAN
							MAN = { is_subject_of = ROOT }
						}
						MAN = {
							random_owned_controlled_state = {
								prioritize = { 716 328 }
								add_extra_state_shared_building_slots = 3
								add_building_construction = {
									type = arms_factory
									level = 3
									instant_build = yes
								}
							}
						}
					}
					else = {
						random_owned_controlled_state = {
							prioritize = { 716 328 }
							limit = { is_core_of = MAN }
							add_extra_state_shared_building_slots = 3
							add_building_construction = {
								type = arms_factory
								level = 3
								instant_build = yes
							}
						}
					}
				}
			}
		}
	}
}
