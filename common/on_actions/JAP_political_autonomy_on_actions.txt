# Starts the focuses of the political tree that are taken for the player (events/JAP_political_autonomy.txt),
# 21 days before the historical date and only once the spine has got that far.
on_actions = {
	on_daily_JAP = {
		effect = {
			# the general election of 20 February 1936 (fired daily so EVENT_PROCESS_OFFSET = 7 does not delay it to 26 February)
			if = {
				limit = {
					NOT = { has_country_flag = JAP_diet_election_1936_fired }
					has_government = democratic
					date > 1936.2.19
					date < 1936.6.1
				}
				set_country_flag = JAP_diet_election_1936_fired
				country_event = { id = jap_diet.1 }
			}
			# the Hull Note, once at war with the United States
			if = {
				limit = {
					has_war_with = USA
					NOT = { has_completed_focus = JAP_pol_hull_note }
					NOT = { has_country_flag = JAP_pol_hull_note_taken_at_war }
				}
				set_country_flag = JAP_pol_hull_note_taken_at_war
				complete_national_focus = {
					focus = JAP_pol_hull_note
					use_side_message = yes
					originator_name = JAP_government_originator
				}
			}
			# the Hiranuma cabinet, 5 January 1939
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_hiranuma_cabinet }
					NOT = { has_shine_effect_on_focus = JAP_pol_hiranuma_cabinet }
					NOT = { has_country_flag = JAP_pol_hiranuma_cabinet_started }
					has_completed_focus = JAP_pol_no_dealings_with_the_nationalists
					date > 1938.12.14
				}
				set_country_flag = JAP_pol_hiranuma_cabinet_started
				country_event = { id = jap_pol_auto.1 }
			}
			# the Hiranuma cabinet resigns, 28 August 1939
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_complicated_and_inscrutable }
					NOT = { has_shine_effect_on_focus = JAP_pol_complicated_and_inscrutable }
					NOT = { has_country_flag = JAP_pol_complicated_and_inscrutable_started }
					has_completed_focus = JAP_pol_hiranuma_cabinet
					has_completed_focus = JAP_pol_new_order_in_east_asia    # cannot be passed over
					date > 1939.8.6
				}
				set_country_flag = JAP_pol_complicated_and_inscrutable_started
				country_event = { id = jap_pol_auto.3 }
			}
			# the Yonai cabinet, 16 January 1940
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_yonai_cabinet }
					NOT = { has_shine_effect_on_focus = JAP_pol_yonai_cabinet }
					NOT = { has_country_flag = JAP_pol_yonai_cabinet_started }
					has_completed_focus = JAP_pol_complicated_and_inscrutable
					date > 1939.12.25
				}
				set_country_flag = JAP_pol_yonai_cabinet_started
				country_event = { id = jap_pol_auto.5 }
			}
			# the turmoil of 1937, from 1 January
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_political_upheaval }
					NOT = { has_shine_effect_on_focus = JAP_pol_political_upheaval }
					NOT = { has_country_flag = JAP_pol_political_upheaval_started }
					has_completed_focus = JAP_pol_active_duty_officer_rule
					date > 1936.12.31
				}
				set_country_flag = JAP_pol_political_upheaval_started
				country_event = { id = jap_pol_auto.7 }
			}
			# the first Konoe cabinet, 4 June 1937: the day after the turmoil ends
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_first_konoe_cabinet }
					NOT = { has_shine_effect_on_focus = JAP_pol_first_konoe_cabinet }
					NOT = { has_country_flag = JAP_pol_first_konoe_cabinet_started }
					has_completed_focus = JAP_pol_political_upheaval
					date > 1936.1.1
				}
				set_country_flag = JAP_pol_first_konoe_cabinet_started
				country_event = { id = jap_pol_auto.9 }
			}
			# the Koiso cabinet, once the Tojo cabinet has fallen
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_koiso_cabinet }
					NOT = { has_shine_effect_on_focus = JAP_pol_koiso_cabinet }
					NOT = { has_country_flag = JAP_pol_koiso_cabinet_started }
					has_completed_focus = JAP_pol_tojo_cabinet
					has_country_flag = JAP_pol_tojo_cabinet_falls    # TOJO_FALLS_1127
					NOT = { has_country_flag = JAP_pol_suzuki_cabinet_started }    # SUZUKI_FROM_TOJO
					NOT = { has_country_flag = JAP_pol_army_government_started }    # KETSUGO_1127
					date > 1936.1.1
				}
				set_country_flag = JAP_pol_koiso_cabinet_started
				country_event = { id = jap_pol_auto.11 }
			}
			# the Suzuki cabinet, once Naha has fallen under the Koiso cabinet
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_suzuki_cabinet }
					NOT = { has_shine_effect_on_focus = JAP_pol_suzuki_cabinet }
					NOT = { has_country_flag = JAP_pol_suzuki_cabinet_started }
					OR = { has_completed_focus = JAP_pol_koiso_cabinet has_completed_focus = JAP_pol_tojo_cabinet }    # SUZUKI_FROM_TOJO: straight from Tojo if Naha falls under him
					NOT = { has_country_flag = JAP_pol_koiso_cabinet_in_progress_flag }
					any_enemy_country = { controls_province = 11932 }    # SUZUKI_1127: Naha
					NOT = { has_country_flag = JAP_pol_army_government_started }    # KETSUGO_1127
					date > 1936.1.1
				}
				set_country_flag = JAP_pol_suzuki_cabinet_started
				country_event = { id = jap_pol_auto.13 }
			}
			# the army takes over: the coup of the Kyujo Incident, or the home islands invaded
			if = {
				limit = {
					NOT = { has_completed_focus = JAP_pol_army_government }
					NOT = { has_shine_effect_on_focus = JAP_pol_army_government }
					NOT = { has_country_flag = JAP_pol_army_government_started }
					has_war = yes
					OR = {
						has_country_flag = JAP_kyujo_coup
						JAP_home_islands_invaded = yes
					}
					date > 1936.1.1
				}
				set_country_flag = JAP_pol_army_government_started
				country_event = { id = jap_pol_auto.15 }
			}
		}
	}
}
