############################################################################################################
#	World Ablaze AI mod - on_actions
############################################################################################################

on_actions = {
	on_startup = {
		# AI System
		effect = {

			# startup AI effects
			every_country = {
				limit = { is_ai = yes }
				country_event = { id = WA_AI_background.7 hours = 1 }
				# [mech-window] FIRST of the template latches: WA_AI_TEMPLATES_update_modern_chassis_latch
				# below reads WA_AI_TEMPLATES_use_mechanized_templates, which reads the flag this
				# one sets. Set after it, the chassis latch would arm a month late.
				WA_AI_TEMPLATES_update_mechanization_latch = yes
				# [modern-chassis-tier] before the calculate, same as the monthly pulse. No 1936
				# country can satisfy it; it is here so a mid-campaign start is not a special case.
				WA_AI_TEMPLATES_update_modern_chassis_latch = yes
				# [mot-field-hospital] same: before the calculate, and no 1936 country satisfies it.
				WA_AI_TEMPLATES_update_motorized_hospital_latch = yes
				WA_AI_TEMPLATES_calculate_templates = yes
				# [armor-role-budget] open the books at 1936 so the first monthly pulse diffs
				# against a real value instead of installing a full set. (WA_AI_PRODUCTION_armor_budget.txt)
				WA_AI_ARMOR_BUDGET_reconcile = yes
				if = {
					limit = { has_war = no }
					add_ideas = WA_AI_peace_attrition_cheat
				}
			}

			# WA_AI_set_resource_variables
			random_country = { country_event = WA_AI_C.10002 }

			# set default EAI settings
			set_global_flag = WA_AI_lend_lease

			# radar script
			every_state = { limit = { WA_AI_radar_location = yes }
				set_state_flag = WA_AI_radar_site
			}

			# save original capital location
			every_country = {
				country_event = WA_AI_background.5
			}

			# priority construction
			random_country = {
				WA_AI_PC_set_global_variables = yes
			}

			# Initialize EAI MAP system for provincial pathfinding and railway building
			random_country = {
				WA_AI_MAP_startup = yes
			}

			# [scripted-invasion-reservation] load the generated scripted-invasion calendar onto the
			# 5 invader countries and calibrate the day epoch. New-game creation only - on_startup
			# does NOT re-fire on save loads (lessons-log); arrays and epoch persist in the save,
			# pre-fix saves stay inert by design (see the effect header).
			# Effects: WA_AI_LANDING_effects.txt / WA_AI_LANDING_reservations_data.txt.
			random_country = {
				WA_AI_LANDING_init_reservations = yes
			}

			# WA_TLM telemetry: zero-init every country + start the month clock so
			# the namespace is a build fingerprint from the very first save.
			# (documentation/WA_TLM_TELEMETRY_SYSTEM.md; effects in WA_TLM_core.txt)
			random_country = {
				set_variable = { global.WA_TLM_clock = 0 }
			}
			every_country = {
				WA_TLM_init_country = yes
			}
		}

		# AI Characters
		# Pre-emptively promote historical field marshals because the AI runs out and shuffles
		effect = {
			ENG = {
				if = {
					limit = { is_ai = yes }
					ENG_bernard_montgomery = {
						promote_leader = yes
					}
					ENG_harold_alexander = {
						promote_leader = yes
					}
				}
			}
			USA = {
				if = {
					limit = { is_ai = yes }
					USA_dwight_d_eisenhower = {
						promote_leader = yes
					}
					USA_omar_bradley = {
						promote_leader = yes
					}
				}
			}
			SOV = {
				if = {
					limit = { is_ai = yes }
					# Zhukov, Timoshenko Shaposhnikov, Kulik are automatically promoted in SOV_war_heroes
				}
			}
		}

		# Misc AI
		effect = {
			every_country = {
				limit = {
					is_ai = yes
				}
				every_army_leader = {
					limit = { has_trait = corps_commander_trait }
					remove_trait = { trait = corps_commander_trait }
				}
				if = {
					limit = {
						has_character = SPR_enrique_lister
					}
					every_unit_leader = {
						limit = {
							NOT = {
								is_character = SPR_enrique_lister
							}
						}
						add_trait = { trait = ai_reassignment }
					}
					every_navy_leader = {
						add_trait = { trait = ai_admiral }
					}
				}
				else = {
					every_unit_leader = {
						add_trait = { trait = ai_reassignment }
					}
					every_navy_leader = {
						add_trait = { trait = ai_admiral }
					}
				}
			}
		}
		effect = {
			if = {
				limit = {
					is_historical_focus_on = yes
					POR = {
						is_ai = yes
					}
				}
				POR = {
					set_rule = { can_not_declare_war = yes }
				}
			}
		}
		effect = {
			if = {
				limit = {
					is_historical_focus_on = yes
					SOV = {
						is_ai = yes
					}
				}
				SOV = {
					set_rule = { can_send_volunteers = no }
				}
			}
		}
		effect = {
			if = {
				limit = {
					COS = {
						is_ai = yes
					}
				}
				COS = {
					set_rule = { can_join_factions = no }
				}
			}
		}
		effect = {
			if = {
				limit = {
					PRU = {
						is_ai = yes
					}
				}
				PRU = {
					set_rule = { can_join_factions = no }
				}
			}
		}
		effect = {
			if = {
				limit = {
					USA = { is_ai = yes }
				}
				630 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				631 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				632 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				633 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				638 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				639 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				642 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				725 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
			}
		}
		effect = {
			if = {
				limit = {
					ENG = { is_ai = yes }
				}
				118 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				116 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
			}
		}
		effect = {
			if = {
				limit = {
					JAP = { is_ai = yes }
				}
				648 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				645 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				613 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
				924 = {
					add_dynamic_modifier = { modifier = island_province_ai }
				}
			}
		}
	}
}
