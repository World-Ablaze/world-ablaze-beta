on_actions = {
	# Recalculate refineries for players right at game start instead of waiting for the first weekly tick
	on_startup = {
		effect = {
			every_country = {
				limit = { is_ai = no }
				wa_refinery_full_recalc = yes
			}
		}
	}
	# Recalculate as soon as trade (or anything else) changes the iron, bauxite or coal balance
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				wa_refinery_recalc_if_resources_changed = yes
			}
		}
	}
}
