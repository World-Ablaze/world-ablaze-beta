# Weekly update of the southern resources carried over the continental corridor (JAP_island_garrisons).
on_actions = {
	on_weekly_JAP = {
		effect = {
			if = {
				limit = {
					OR = {
						has_country_flag = JAP_corridor_active
						has_country_flag = JAP_far_east_active
						has_country_flag = JAP_india_route_active
					}
				}
				JAP_corridor_update = yes
			}
		}
	}
}
