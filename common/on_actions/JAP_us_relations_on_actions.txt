# The war with China costs Japan goodwill in the West (once).
on_actions = {
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					tag = JAP
					FROM = { tag = CHI }
					NOT = { has_country_flag = JAP_china_incident_opinion_given }
				}
				set_country_flag = JAP_china_incident_opinion_given
				every_country = { limit = { OR = { tag = ENG tag = USA tag = FRA } } add_opinion_modifier = { target = JAP modifier = JAP_china_incident_opinion } }
			}
		}
	}
}
