# Naming convention : WA_AI_TEMPLATES_GENERIC_[TEMPLATE_TYPE]_[WIDTH]_[COMPOSITION]_[MOTORIZATION]
# Compositions :
# - INF : Infantry only
# - MIX : Infantry and line support
# Motorizations :
# - MOT : Motorized
# - HRS : Horse-drawn / feet

# WARNING - these target templates must NOT look like a line infantry division.
#
# The engine assigns every existing division template the role whose ENABLED target it
# matches best, scoring (matched units - per-category oversize) / total target units.
# This role is upgrade_prio 0 and has no role_ratio share, so the AI desires zero
# divisions of it - and a template in a role the AI wants zero of is decommissioned and
# can never be recruited again.
#
# When these targets were built out of infantry_heavy_horse_battalion_line plus pack
# artillery / anti-tank regimental support, they described a rifle division better than
# the infantry targets did. SOV's "Strelkovaya Diviziya" scored 0.4167 here against
# 0.1667 on its enabled infantry target, was classified garrison, and was decommissioned
# on 1936.1.7 with 87 divisions using it - frozen at exactly 87 until 1939. ITA and FIN
# show the same symptom. ENG/FRA/USA score infantry well above garrison and are fine.
#
# Keep every target here to cheap light infantry plus military police: high suppression
# per IC, and nothing a real line division carries, so no line division can match it.
# See NDefines.NAI.GARRISON_TEMPLATE_SCORE_* - the AI wants cost/suppression to be low.
WA_garrison_role = {
	role = garrison
	upgrade_prio = {
		base = 0
		modifier = {
			add = 100
			has_army_experience > 100
		}
	}

	WA_AI_TEMPLATES_GENERIC_LIGHT_GARRISON_20_INF_HRS = {
		enable = { has_country_flag = { flag = WA_GARRISON_TEMPLATE value = 9000 } } upgrade_prio = { base = 10 }
		reinforce_prio = 1
		custom_icon = 7
		can_upgrade_in_field = { always = yes }
		upgrade_prio = { base = 10 }
		# Small-country garrison, deliberately tiny. At 10 battalions with no support this
		# still described a light-infantry line division well enough to capture one.
		target_template = {
			regiments = {
				infantry_light_horse_battalion_line = 4
			}
			support = {
				military_police_horse_company_divisional = 1
			}
		}
	}

	WA_AI_TEMPLATES_GENERIC_HEAVY_GARRISON_20_INF_HRS = {
		enable = { has_country_flag = { flag = WA_GARRISON_TEMPLATE value = 9001 } } upgrade_prio = { base = 10 }
		reinforce_prio = 1
		custom_icon = 7
		can_upgrade_in_field = { always = yes }
		upgrade_prio = { base = 10 }
		# Was 10x infantry_heavy + 3 pack artillery + 3 anti-tank regimental + 3 support
		# companies, i.e. a rifle division. See the warning at the top of this file.
		target_template = {
			regiments = {
				infantry_light_horse_battalion_line = 6
			}
			support = {
				military_police_horse_company_divisional = 1
			}
		}
	}

	WA_AI_TEMPLATES_GENERIC_HEAVY_GARRISON_30_INF_HRS = {
		enable = { has_country_flag = { flag = WA_GARRISON_TEMPLATE value = 9002 } } upgrade_prio = { base = 10 }
		reinforce_prio = 1
		custom_icon = 7
		can_upgrade_in_field = { always = yes }
		upgrade_prio = { base = 10 }
		# Was 15x infantry_heavy + 3 pack artillery + 3 anti-tank regimental + 3 support
		# companies, i.e. a rifle division. See the warning at the top of this file.
		target_template = {
			regiments = {
				infantry_light_horse_battalion_line = 8
			}
			support = {
				military_police_horse_company_divisional = 1
			}
		}
	}

}
