############################################################################################################
# WA AI Templates - ARMY HQ ROLE
#
# [hq-role-capture] Declares the HQ role vanilla's generic.txt carries, which replace_path drops.
#
# What it does NOT do: classify army-HQ divisions. MEASURED - a template only stores a `role=` when
# the AI DESIGNED it, and the AI never designs an `is_army_hq` template, so this role holds nothing
# ("There is no existing template with correct role"). A history-authored HQ is classified by an
# engine heuristic instead; the lever for that is the battalion count in
# history/general/taog_hq_template.txt, not this file.
#
# Why it is kept: vanilla declares the same role with the same zero upgrade_prio, and its purpose
# there is not documented. Removing a guard whose reason is unknown is how the garrison role was
# lost twice. It costs nothing - upgrade_prio 0 means the AI never spends XP through it.
#
# The two entries are the yes/no halves of ONE config trigger, so exactly one target is always
# enabled and no dead-role-entry fallback is needed.
############################################################################################################

WA_hq_role = {
	role = hq_role
	upgrade_prio = { base = 0 }

	WA_AI_TEMPLATES_GENERIC_HQ_MOT = {
		enable = { WA_AI_CONFIG_DIVISIONS_uses_motorized_hq = yes }
		upgrade_prio = { base = 0 }
		can_upgrade_in_field = { always = no }
		target_template = {
			regiments = {
				hq_motorized = 4
			}
			support = {
				hq_support_company = 1
			}
		}
	}

	WA_AI_TEMPLATES_GENERIC_HQ_INF = {
		enable = { WA_AI_CONFIG_DIVISIONS_uses_motorized_hq = no }
		upgrade_prio = { base = 0 }
		can_upgrade_in_field = { always = no }
		target_template = {
			regiments = {
				hq_infantry = 4
			}
			support = {
				hq_support_company = 1
			}
		}
	}
}
