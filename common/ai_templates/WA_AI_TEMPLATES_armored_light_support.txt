WA_light_support_armor_role = {
	role = light_armor
	front_role_override = offence	# Engine keys front assignment on 'role = armor' or this override
									# (defines ASSIGN_TANKS_TO_WAR_FRONT = 6.0). WA uses its own granular role
									# tokens, so without this line armour was distributed like infantry and
									# every front_armor_score entry in the mod was inert.
	upgrade_prio = {
		base = 100
		# [dead-role-entry] Twin of the guard in WA_AI_TEMPLATES_armored_light.txt: this entry's
		# enables all key on WA_LIGHT_SUPPORT_ARMOR_TEMPLATE, so without that flag the entry has
		# zero enabled targets yet still draws full lottery weight.
		modifier = {
			factor = 0
			NOT = { has_country_flag = WA_LIGHT_SUPPORT_ARMOR_TEMPLATE }
		}
	}
