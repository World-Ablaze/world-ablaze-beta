# WA AI Templates - Switzerland, militia role (Country layer)
#
# [swi-militia] Restores the `militias` role that `replace_path="common/ai_templates"` deleted with
# vanilla's `templates_SWI.txt`. Without a role-level entry the Swiss AI's own
# `ai_strategy = { type = template_prio  id = militias  value = 100000 }`
# (common/ai_strategy/SWI.txt) points at a role no file defines, so it wants 0 militia divisions.
# Assumes the Citizen Militia stays LOCKED and grows only via SWI_upgrade_template_and_divisions
# (pure militia_heavy_horse_battalion_line, no support, 2 -> 20 battalions). If that template ever
# gains support companies or a second sub-unit, this target must follow or the role stops matching.
############################################################################################################

WA_SWI_militia_role = {

	available_for = {
		SWI
	}

	role = militias

	# 0 = never spend army XP here: the template is is_locked = yes, the AI cannot edit it.
	upgrade_prio = {
		factor = 0
	}

	WA_AI_TEMPLATES_COUNTRY_SWI_MILITIA_20_HRS = {

		enable = { always = yes }

		reinforce_prio = 1

		can_upgrade_in_field = { always = no }

		upgrade_prio = {
			factor = 0
		}

		target_template = {
			regiments = {
				militia_heavy_horse_battalion_line = 20
			}
		}
	}
}
