WA_light_armor_role = {
	role = light_armor
	front_role_override = offence	# Engine keys front assignment on 'role = armor' or this override
									# (defines ASSIGN_TANKS_TO_WAR_FRONT = 6.0). WA uses its own granular role
									# tokens, so without this line armour was distributed like infantry and
									# every front_armor_score entry in the mod was inert.
	upgrade_prio = {
		base = 100
		# [dead-role-entry] Two role-level entries share role = light_armor (this file and
		# WA_AI_TEMPLATES_armored_light_support.txt) and the engine's lottery gives full weight
		# to an entry with zero enabled targets. Every enable here keys on WA_LIGHT_ARMOR_TEMPLATE,
		# so no flag means no reachable target.
		modifier = {
			factor = 0
			NOT = { has_country_flag = WA_LIGHT_ARMOR_TEMPLATE }
		}
	}
