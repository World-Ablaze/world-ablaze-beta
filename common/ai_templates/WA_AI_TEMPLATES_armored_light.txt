# - Airborne : 11000-11999
# - Mountaineer : 12000-12999
# - Rangers : 13000-13999

# Naming convention : WA_AI_TEMPLATES_GENERIC_[TEMPLATE_TYPE]_[WIDTH]_[COMPOSITION]_[MOTORIZATION]
# Compositions :
# - INF : Infantry only
# - MIX : Infantry and line support
# - MAR : Marines only
# Motorizations :
# - MOT : infantry_heavy_motorized_battalion_line
# - HRS : Horse-drawn / feet

	role = light_armor
	front_role_override = offence	# Engine keys front assignment on 'role = armor' or this override
									# (defines ASSIGN_TANKS_TO_WAR_FRONT = 6.0). WA uses its own granular role
									# tokens, so without this line armour was distributed like infantry and
									# every front_armor_score entry in the mod was inert.
	upgrade_prio = {
		base = 100
