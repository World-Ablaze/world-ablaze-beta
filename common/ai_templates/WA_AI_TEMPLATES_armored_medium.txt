# - MOT : Motorized
# - HRS : Horse-drawn / feet

WA_medium_armor_role = {
	role = medium_armor
	front_role_override = offence	# Engine keys front assignment on 'role = armor' or this override
									# (defines ASSIGN_TANKS_TO_WAR_FRONT = 6.0). WA uses its own granular role
									# tokens, so without this line armour was distributed like infantry and
									# every front_armor_score entry in the mod was inert.
	upgrade_prio = {
		base = 100
		# [dead-role-entry] The engine's role-entry lottery gives full weight to an entry whose
		# targets are all disabled, turning it into a permanent army-XP sink. Every enable here
		# keys on a medium-range WA_MEDIUM_ARMOR_TEMPLATE value, unreachable once the modern
		# latch adds +500 - and absent flag means no reachable target at all.
		modifier = {
			factor = 0
			has_country_flag = WA_AI_TEMPLATES_modern_chassis_earned
		}
		modifier = {
			factor = 0
			NOT = { has_country_flag = WA_MEDIUM_ARMOR_TEMPLATE }
		}
	}
