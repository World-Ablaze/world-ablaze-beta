# purpose: keep Allied Atlantic routing inside the northern lend-lease corridor
# range:   0 to +2000
# policy:  Additive per region
# domain:  NAVAL
#
# Fix 52 (2026-08-13, checklist R36): removed
#   NOT = { WA_AI_MILITARY_COUNTRY_owns_naval_atlantic_corridor = yes }
# from this block and from its two siblings below. That trigger resolves to a plain
# tag list (ENG / USA / GER), so the whole three-part Atlantic escort plan - pull onto
# the corridors here, push off the mid-ocean gaps in *_off_corridor_avoidance - was
# disabled for exactly the two navies able to run it. The effective recipients were
# CAN / RAJ / AST / NZL: 27 hulls in the 1942.6 save of campaign bec4d829, mostly at
# anchor, while the USN held 559 ships and put ZERO escorts in the Atlantic (its four
# escort task forces sat in the Gulf of Aden, the Solomon Sea, the Algerian coast and
# the central NE Pacific). The exclusion was not a de-duplication of Country-layer
# work: `naval_dominance` is Additive per region (WA_AI_MILITARY_SYSTEM.md), the USA
# naval file carries no naval_dominance at all, ENG carries exactly one (Alboran), and
# the only mod-wide strategy touching a north-corridor region for the USA is an
# avoid=2000 on 244. GER's presence in the excluded list is dead weight here - it can
# never pass the `allowed` gate - which is what shows the trigger to be a geopolitical
# classification rather than a "these already have their own plan" list.
# The situational trigger this enable was reaching for, WA_AI_MILITARY_FACTION_owns_
# naval_atlantic_corridor, exists and is referenced nowhere; it is deliberately left
# unused rather than swapped in, so this fix changes WHO is eligible and nothing else.
WA_AI_NAVAL_FACTION_ALLIES_atlantic_north_corridor = {
	allowed = {
		WA_AI_CONFIG_MILITARY_is_allied_naval_corridor_user = yes
	}
	enable = {
		WA_AI_MILITARY_allies_at_war_with_axis = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = naval_dominance id = 43 value = 80 }  # Western Approaches
	ai_strategy = { type = naval_dominance id = 55 value = 80 }  # Newfoundland Sea
	ai_strategy = { type = naval_dominance id = 50 value = 80 }  # Labrador Sea
	ai_strategy = { type = naval_dominance id = 247 value = 80 } # Labrador Basin
	# Fix 59 (2026-08-14): dropped `id = 36` (Greenland). Checked against
	# map/definition.csv: all 8 of its provinces are type `land`, so the region can
	# carry no naval dominance and the entry was inert. The waters it was reaching
	# for are already in this list - 44 Denmark Strait and 244 Greenland Iceland
	# Rise to the east, 50 Labrador Sea and 247 Labrador Basin to the west - so the
	# corridor loses no coverage.
	ai_strategy = { type = naval_dominance id = 243 value = 80 } # Icelandic Basin
	ai_strategy = { type = naval_dominance id = 244 value = 80 } # Greenland Iceland Rise
	ai_strategy = { type = naval_dominance id = 44 value = 80 }  # Denmark Strait

	# Fix 60 (2026-08-14, checklist R36): the pull half of the pair.
	# Until now this plan only ever PUSHED - naval_dominance says "we want presence
	# here", and atlantic_off_corridor_avoidance puts +1000 on the mid-ocean gaps -
	# but nothing pulled fleets ONTO the corridor, so the only naval_avoid_region
	# values any corridor region ever saw were the legacy Country-layer walls
	# pointing the other way (+2000 on 244 from USA_legacy_USA_smart_pacific_fleet
	# and again from ENG_legacy_ENG_atlantic_sea_zone_corridors, +2000 on 57 from
	# USA_legacy_USA_smart_atlantic_fleet). A negative naval_avoid_region is an
	# attraction in this engine and it is WA's own established idiom for it -
	# ENG_protect_home uses -10000 on the English Channel and the USA Torch blocks
	# use -2000 on the Bay of Biscay. -1000 is chosen to mirror the +1000 that
	# off_corridor_avoidance puts on the gaps, so the plan reads as one symmetric
	# pair rather than two unrelated magnitudes; this is the same "every boost
	# paired with a suppression" rule the AIFC header states, applied in reverse.
	# It does NOT by itself beat the legacy walls (244 nets +3000 for the USA until
	# those are retired) - that is the follow-up pass, and this value is sized for
	# the world after it, not against it.
	ai_strategy = { type = naval_avoid_region id = 43 value = -1000 }  # Western Approaches
	ai_strategy = { type = naval_avoid_region id = 55 value = -1000 }  # Newfoundland Sea
	ai_strategy = { type = naval_avoid_region id = 50 value = -1000 }  # Labrador Sea
	ai_strategy = { type = naval_avoid_region id = 247 value = -1000 } # Labrador Basin
	ai_strategy = { type = naval_avoid_region id = 243 value = -1000 } # Icelandic Basin
	ai_strategy = { type = naval_avoid_region id = 244 value = -1000 } # Greenland Iceland Rise
	ai_strategy = { type = naval_avoid_region id = 44 value = -1000 }  # Denmark Strait
}

# purpose: let Allied convoy-escort missions clear their score bar at all
# range:   -100 to 0
# policy:  Additive per mission
# domain:  NAVAL
#
# Fix 53a (2026-08-13, checklist R36). This line used to live inside the
# north-corridor block above as
#     ai_strategy = { type = naval_mission_threshold id = MISSION_CONVOY_ESCORT value = 5000 }
# which is sign-inverted against that block's own stated purpose. A POSITIVE
# naval_mission_threshold RAISES the bar a mission's score must clear before the AI
# will assign it - vanilla's own JAP block says so in as many words
# (`value = 150 #puts our threshold at 250`, in a block named
# conserve_fuel_for_usa_fight whose whole job is to stop escorting), and WA's
# WA_AI_NAVAL_DEFAULT_legacy_AI_naval_mission_fix drives patrol/strike NEGATIVE with
# the comment "Fixes the AI not putting out its strike force and patrols". So the
# Atlantic escort plan was ending with "and never escort": +5000 against an engine
# base of 100, i.e. 50x the type's documented ceiling (WA_AI_MILITARY_SYSTEM.md:106
# gives the range as -100 to +100), and undocumented in the block's own header, which
# declares "range: 0 to +2000 / Additive per region" and describes naval_dominance only.
# Before Fix 52 this reached CAN/RAJ/AST/NZL - the 27 hulls "mostly at anchor" in R36's
# opening reading - and Fix 52, by removing the ENG/USA/GER exclusion, handed it to the
# two navies the whole item is about. It is split out into its own block rather than
# deleted-in-place because a country-wide mission threshold is not a per-region rule and
# does not belong in a block named for a corridor.
#
# -100 is the documented floor, not a guess at a magic number: it puts the escort bar at
# 0 (base 100 - 100), i.e. any positive escort score assigns the mission, which is the
# same permissiveness WA already grants patrol and strike force. Deliberately NOT matched
# to those blocks' out-of-range -500/-1000; if escort share still fails R36's 25% leg
# after Fix 53b restores the escort SCORE terms, going further negative is the next step,

# purpose: contest the Gibraltar approaches, where the Allied southern route enters the Med
# range:   0 to 100
# policy:  Additive per region
# domain:  NAVAL
#
# Fix 60 (2026-08-14, checklist R36). The Atlantic plan stopped at the Iberian
# Coast (47). Everything the convoys do after that - Alboran, then the Algerian
# Coast, then east - was covered by exactly one thing mod-wide: an avoid = 2000 in
# spanish_waters_unsafe for when the route is already lost. There was no "hold this
# open" term anywhere, so the southern half of the escort problem had no plan at
# all, only a contingency. This block is that term, and it is the strict complement
# of spanish_waters_unsafe: that one requires the Axis to hold Spain or Morocco,
# this one requires the negation, so they can never both be live.
# Gibraltar itself is gated on the state, not on a tag or a date: if the Rock's
# controller is at war with us, the approaches are not a corridor, they are a trap.
#
# ENG note (deliberate, not an oversight): ENG's Country-layer Gibraltar block
# already writes naval_dominance id = 249 value = 100, so ENG reads 170 on Alboran
# while both are live. naval_dominance is documented Additive per region, so this
# is legal and it is directionally right - Alboran genuinely is Britain's highest

# The five blocks below were duplicated or mis-layered in the ENG and USA naval
# files. Three of them (war_against_axis, german_bight, the Baltic trap) existed
# TWICE with the same payload and near-identical enables; two more lived in the
# COUNTRY_ENG file behind multi-tag `allowed` lists, which is Faction-layer content
# wearing a Country-layer name. Promotion is per WA_AI_MILITARY_SYSTEM.md: prefer
# the highest shared layer, and never duplicate the same block at two layers.
#
# Gating uses archetype triggers, never tag lists. Where an enable carried a
# historical assumption that would silently disable the rule in an ahistorical
# game, it has been dropped and the drop is noted on the block - that is design
# principle 1 (no gaps when the script diverges), and each such change strictly
# WIDENS coverage rather than restricting it.

# purpose: keep Allied fleets out of the Italian coastal seas while Italy holds Sicily
# range:   0 to +2000
# policy:  Additive per region
# domain:  NAVAL
#
# Fix 62: promoted from WA_AI_NAVAL_COUNTRY_ENG_legacy_ENG_war_against_axis and
# WA_AI_NAVAL_COUNTRY_USA_legacy_USA_war_against_axis, which were identical down to
# the region list and the enable - only the `allowed` tag differed. That pair is

	}
	abort_when_not_enabled = yes

	ai_strategy = { type = naval_avoid_region id = 373 value = 2000 } # Ligurian Sea
	ai_strategy = { type = naval_avoid_region id = 169 value = 2000 } # Tyrrhenian Sea

# purpose: keep Allied fleets out of the German Bight while the enemy holds Hamburg
# range:   0 to +2000
# policy:  Additive per region
# domain:  NAVAL
#
# Fix 62: promoted from the ENG and USA `german_bight` blocks, same payload.
# Two enable differences resolved rather than merged:
#  - each version asked whether state 803 Hamburg was controlled by someone in
#    faction with ITSELF (`is_in_faction_with = ENG` / `= USA`). Rewritten as the
#    generic "not one of us", which is what both meant and which also covers the
#    case of an Allied member that is in neither of those two factions.
#  - the USA version added `OR = { ... date < 1942.3.1 }`, i.e. avoid it
#    unconditionally before March 1942. Dropped as redundant: before that date the
#    Germans hold Hamburg in every timeline where this block can fire at all, so
#    the ownership test already returns the same answer.
# `GER = { has_government = fascism }` is kept - it is the block's actual subject
# (a hostile Germany), not a historical-path assumption about who fights whom.
WA_AI_NAVAL_FACTION_ALLIES_german_bight = {
	allowed = {
		WA_AI_CONFIG_MILITARY_is_anglo_major = yes
	}
	enable = {
		GER = { has_government = fascism }
		NOT = {
			803 = { # Hamburg
				CONTROLLER = {
					OR = {
						tag = ROOT
						is_subject_of = ROOT
						is_in_faction_with = ROOT
					}
				}
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = naval_avoid_region id = 242 value = 2000 } # German Bight
}
