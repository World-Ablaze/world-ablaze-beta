# purpose: keep Allied Atlantic routing inside the northern lend-lease corridor
# range:   0 to +2000
# policy:  Additive per region
# domain:  NAVAL
#
# [atlantic-naval] no owner exclusion. The removed
#   NOT = { WA_AI_MILITARY_COUNTRY_owns_naval_atlantic_corridor = yes }
# resolves to a plain tag list (ENG / USA / GER), so it disabled the whole three-part Atlantic escort
# plan - pull onto the corridors here, push off the mid-ocean gaps in *_off_corridor_avoidance - for
# exactly the two navies able to run it, leaving it to the dominions. It was not a de-duplication of
# Country-layer work: naval_dominance is Additive per region (WA_AI_MILITARY_SYSTEM.md), the USA
# naval file carries no naval_dominance at all and ENG carries one (Alboran); GER cannot even pass
# the `allowed` gate, which is what shows the trigger to be a geopolitical classification rather
# than a "these already have their own plan" list.
# The situational trigger this enable was reaching for, WA_AI_MILITARY_FACTION_owns_naval_atlantic_
# corridor, exists and is referenced nowhere; it is deliberately left unused rather than swapped in,
# so the change moves WHO is eligible and nothing else.
WA_AI_NAVAL_FACTION_ALLIES_atlantic_north_corridor = {
	allowed = {
		WA_AI_CONFIG_MILITARY_is_allied_naval_corridor_user = yes
	}
	enable = {
		WA_AI_MILITARY_allies_at_war_with_axis = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = naval_dominance id = 43 value = 80 }  # Western Approaches
	ai_strategy = { type = naval_dominance id = 55 value = 80 }  # Newfoundland Sea
	ai_strategy = { type = naval_dominance id = 50 value = 80 }  # Labrador Sea
	ai_strategy = { type = naval_dominance id = 247 value = 80 } # Labrador Basin
	# [atlantic-naval] no `id = 36` (Greenland): checked against map/definition.csv, all 8 of its
	# provinces are type `land`, so the region can carry no naval dominance and the entry was inert. The
	# waters it was reaching for are already in this list - 44 to the east, 50 and 247 to the west.
	ai_strategy = { type = naval_dominance id = 243 value = 80 } # Icelandic Basin
	ai_strategy = { type = naval_dominance id = 44 value = 80 }  # Denmark Strait

	# [atlantic-naval] the PULL half of the pair. A negative naval_avoid_region is an attraction in this
	# engine and it is WA's own established idiom for it (ENG_protect_home uses -10000 on the English
	# Channel). Until this block the plan only ever PUSHED - naval_dominance says "we want presence
	# here" and off_corridor_avoidance puts +1000 on the mid-ocean gaps - while nothing pulled fleets
	# ONTO the corridor, so the only naval_avoid_region values a corridor region ever saw were legacy
	# Country-layer walls pointing the other way. -1000 mirrors that +1000 so the plan reads as one
	# symmetric pair rather than two unrelated magnitudes: the AIFC "every boost paired with a
	# suppression" rule applied in reverse. It does NOT by itself beat the legacy walls - this value is sized for the world after that pass, not
	# against it.
	ai_strategy = { type = naval_avoid_region id = 43 value = -1000 }  # Western Approaches
	ai_strategy = { type = naval_avoid_region id = 55 value = -1000 }  # Newfoundland Sea
	ai_strategy = { type = naval_avoid_region id = 50 value = -1000 }  # Labrador Sea
	ai_strategy = { type = naval_avoid_region id = 247 value = -1000 } # Labrador Basin
	ai_strategy = { type = naval_avoid_region id = 243 value = -1000 } # Icelandic Basin
	# [can-transit-attrition] 44 keeps its dominance above, so it keeps its pull: an unpaired
	# dominance is the half-pair this block's own comment warns about. Delete BOTH lines to
	# retire Denmark Strait from the corridor.
	ai_strategy = { type = naval_avoid_region id = 44 value = -1000 }  # Denmark Strait
}

# purpose: let Allied convoy-escort missions clear their score bar at all
# range:   -100 to 0
# policy:  Additive per mission
# domain:  NAVAL
#
# [atlantic-naval] split out of the north-corridor block, where this line read `value = 5000` -
# sign-inverted against that block's own stated purpose. A POSITIVE naval_mission_threshold RAISES
# the bar a mission's score must clear before the AI will assign it: vanilla's own JAP block says so
# in as many words (`value = 150 #puts our threshold at 250`, in conserve_fuel_for_usa_fight, whose
# whole job is to stop escorting), and WA's own naval_mission_fix drives patrol/strike NEGATIVE to
# make the AI put them out. So the Atlantic escort plan was ending with "and never escort", at 50x
# the type's documented ceiling (WA_AI_MILITARY_SYSTEM.md gives the range as -100 to +100).
# -100 is the documented floor, not a guess at a magic number: it puts the escort bar at 0 (base
# 100 - 100), i.e. any positive escort score assigns the mission, the same permissiveness WA already
# grants patrol and strike force. Deliberately NOT matched to those blocks' out-of-range
# -500/-1000; if escort share still fails, going further negative is the next step, and keeping this
# one in spec is what makes that a readable experiment.
# It is its own block because a country-wide mission threshold is not a per-region rule and does not
# belong in a block named for a corridor.
WA_AI_NAVAL_FACTION_ALLIES_convoy_escort_threshold = {
	allowed = {
		WA_AI_CONFIG_MILITARY_is_allied_naval_corridor_user = yes
	}
	enable = {
		WA_AI_MILITARY_allies_at_war_with_axis = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = naval_mission_threshold
		id = "MISSION_CONVOY_ESCORT"
		value = -100
	}
}
