#          home is safe. One source of +200 where three used to stack (war_against_ENG +200 on any
#          ITA/ITL-held Libyan state, north_africa_focus +200 on `date > 1937` for ever,
#          hold_africa +200 to 1941 - up to +600 on a 7-region area, E1 ceiling is +200). Fires at
#          peace too (pre-positioning the Libyan garrison, R60 corridor prep) and at war with whoever
#          holds East Egypt or Tunisia - controller-based, no ENG/FRA tag, so an ahistorical enemy
#          in Egypt gets the same effort. Ethiopia's conquest phase keeps precedence (its own +200s).

# policy:  Additive per area (sole ITA-scope +200 writer for north_africa; the R13 FRONT pair is
#          the +60 front_unit_request beside it)
# domain:  THEATRE

	allowed = {
		tag = ITA
	}
	enable = {
		WA_AI_MILITARY_libya_bridgehead_held = yes
		NOT = { WA_AI_MILITARY_italy_home_threatened = yes }
		WA_AI_MILITARY_ethiopian_war_finished = yes
		OR = {
			has_war = no
			446 = { CONTROLLER = { has_war_with = ROOT } }
			447 = { CONTROLLER = { has_war_with = ROOT } }
			453 = { CONTROLLER = { has_war_with = ROOT } }
			458 = { CONTROLLER = { has_war_with = ROOT } }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = area_priority
		id = north_africa
		value = 200
	}
}
