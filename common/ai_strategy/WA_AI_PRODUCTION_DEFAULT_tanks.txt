# ============================================================================
# ARMOR CATEGORY DEMAND
# ============================================================================
# [armor-prod-category] The one lever that covers chassis AND their specialised variants
# (assault gun, TD, SPAA, infantry support) - `armor` is an equipment CATEGORY
# (common/script_enums.txt), so it needs no per-archetype entry. It scales the engine's own
# perceived need, the same weapon the air strategies already use per category, and it stacks
# on the per-variant factors above. Assumes the tank lines are DEMAND-bound: a save showing
# tank lines running under their requested factories means the bound moved to allocation and
# this number no longer addresses it.

WA_AI_PRODUCTION_DEFAULT_armor_category_push = {
	enable = { WA_AI_PRODUCTION_armor_category_push = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = equipment_production_factor id = armor value = 60 }
}

# ============================================================================
# FACTORY FLOORS - MECHANIZED ONLY
# ============================================================================
# The role_ratio system creates mech division WANTS, but nothing below guaranteed the
# factories to fill them. Mirrors what WA_AI_PRODUCTION_DEFAULT_ground.txt already does for
# infantry (12) and trucks (10). Banded by industrial size; gated on the template trigger that
# opens the mech role.
# [armor-prod-category] Tank chassis carry NO floor: the armor category factor above is their
# only lever. A floor is need-blind (it ignores what is available), so re-adding one under the
# category factor would force allocation on top of a raised demand and double-count.

WA_AI_PRODUCTION_DEFAULT_mech_floor_small = {
	enable = { WA_AI_PRODUCTION_mech_min_factories_small = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = equipment_production_min_factories_archetype id = mechanized_equipment value = 3 }
}
