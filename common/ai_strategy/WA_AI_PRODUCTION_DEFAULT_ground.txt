# [truck-floor-ladder] The band floors, monotone over the industry tiers (3 / 6 / 10 / 16 / 22 /
# 30). Exactly one arms at a time - the bands are exclusive in WA_AI_PRODUCTION_ground.txt - and
# the deep tier below adds on top. `abort` reads the HOLD trigger, never abort_when_not_enabled:
# the arm bar and the release bar are a Schmitt pair and dropping the floor at the arm bar is what
# makes a 30-factory floor flap. Values are literals because `ai_strategy value =` cannot read a
# `constant:` (wa-constants-registry); the band BOUNDS are the shared industry constants.
# Worst arsenal share is at each band's low edge: 50 MIL -> 15 (30 %), 100 -> 24 (24 %),
# 200 -> 37 (18 %), 300 -> 45 (15 %). What the engine does when the floors together exceed the
# arsenal is ASSUMED in this repo, so the ladder is sized to stay under the band edge, not to the
# cap.
WA_AI_PRODUCTION_floor_trucks_small = {
	enable = { WA_AI_PRODUCTION_should_floor_trucks_small = yes }
	abort  = { NOT = { WA_AI_PRODUCTION_should_hold_floor_trucks_small = yes } }

	ai_strategy = { type = equipment_production_min_factories_archetype id = motorized_equipment value = 3 }
}

# [train-variant-choice] Variant choice for EXISTING train lines, all AI countries. production_upgrade_desire_offset
# takes a specific equipment id (vanilla SOV/GER/ENG use it on infantry_equipment_N); +100 = convert a line to
# it without waiting for a stockpile surplus, -100 = hold a line off it. It never chooses what a NEW line
# builds: that is the `priority` order and the AI-only can_be_produced gate on the Armored Train, both in
# common/units/equipment/trains.txt. Parent chain is _1 -> _2 -> _3 -> _4 (the AI walks it to the deepest
# producible link and never skips one - see trains.txt), so the -100/+100 on _4 act on the _3 -> _4 step.
WA_AI_PRODUCTION_build_cheap_trains = {
	enable = { WA_AI_PRODUCTION_should_build_cheap_trains = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = production_upgrade_desire_offset id = train_equipment_2 value = 100 }
	ai_strategy = { type = production_upgrade_desire_offset id = train_equipment_3 value = 100 }
	ai_strategy = { type = production_upgrade_desire_offset id = train_equipment_4 value = -100 }
}
