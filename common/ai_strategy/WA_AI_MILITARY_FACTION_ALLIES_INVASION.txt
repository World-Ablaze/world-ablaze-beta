# Phase 6 split from WA_AI_MILITARY_ALLIES_no_italy_invasion_early (INVASION domain)
# purpose: stop the bloc scattering naval invasions around Italy and the Adriatic. The third enable
#          branch (Allies hold 115 + 1032 + 156) is the anti-scatter rule and is KEPT DELIBERATELY,
#          unlike in the THEATRE twin where the same branch was removed on 2026-08-12: once a land
#          route up the peninsula exists, further amphibious operations against Italy compete with it
#          for the same divisions. The failure this prevents is measured - campaign 2b607968 at
#          1943.7.15 had 32 Allied divisions parked in fresh Sardinia / Calabria / Dodecanese
#          invasion plans while 13 fought on Sicily (see WA_AI_MILITARY_ALLIES_hold_sicily_bridge_*).
#          Commit 2e3686fa1 is explicit that the INVASION domain must keep dying at first capture.
#          What was wrong was the THEATRE twin using the same branch to cut *theatre priority*, which
#          starves the land campaign the branch assumes is happening. Suppressing redundant landings
#          and deprioritising the theatre are different decisions and no longer share a gate.
#          NOTE the payload also carries Balkan and anti-Sealion-mirror policy (invade YUG / ALB /
#          ENG): changing this enable changes those too. That coupling is why the fix went to the
#          THEATRE side only.
# range:   invade 0 to -2000 per target; invasion_unit_request -100 to +200
# policy:  Additive per target
# domain:  INVASION
WA_AI_MILITARY_ALLIES_no_italy_invasion_early_INVASION = {

	allowed = {
		WA_AI_CONFIG_MILITARY_is_western_invasion_pool = yes
	}
	enable = {
		WA_AI_MILITARY_should_allies_no_italy_invasion_early_invasion = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = invade
		id = "ITA"
		value = -2000
	}
	ai_strategy = {
		type = invade

		value = -2000
