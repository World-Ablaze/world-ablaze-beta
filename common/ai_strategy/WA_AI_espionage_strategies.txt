# [intel-adversary-targeting] Intel networks on the two strongest adversaries picked by
# WA_AI_set_intel_network_targets: two operatives on the first until its network reaches
# @INTEL_PRIMARY_NETWORK_BUILT, one after that, and one on the second. The build/keep pair must
# read the same threshold or both arm (or neither) at once - hence one constant.
@INTEL_PRIMARY_NETWORK_BUILT = 50
WA_AI_intel_network_primary_build = {
	enable = {
		has_variable = WA_AI_intel_target_1
		NOT = { has_variable = WA_AI_rescue_operative_target }
		network_strength = {
			target = var:WA_AI_intel_target_1
			value < @INTEL_PRIMARY_NETWORK_BUILT
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = operative_mission
		mission = build_intel_network
		value = 50000
		mission_target = var:WA_AI_intel_target_1
		num_operatives = 2
	}
}

WA_AI_intel_network_primary_keep = {
	enable = {
		has_variable = WA_AI_intel_target_1
		NOT = { has_variable = WA_AI_rescue_operative_target }
		NOT = {
			network_strength = {
				target = var:WA_AI_intel_target_1
				value < @INTEL_PRIMARY_NETWORK_BUILT
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = operative_mission
		mission = build_intel_network
		value = 50000
		mission_target = var:WA_AI_intel_target_1
		num_operatives = 1
	}
}
