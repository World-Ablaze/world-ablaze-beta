		# the approach - the two are gate-exclusive on the same verdict, never additive (this repo
		# sizes put_unit_buffers tiers exclusively; see the Commonwealth-handoff section of
		# documentation/WA_AI_MILITARY_SYSTEM.md, "no ratio is inferred by adding two orders").
		NOT = { WA_AI_MILITARY_gulf_approach_massed = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = put_unit_buffers
		order_id = 9206
		states = { 656 }
		ratio = 0.05
		subtract_fronts_from_need = no # pool: the Kuwait guard must survive unrelated Asian front demand
		subtract_invasions_from_need = no
	}
}
