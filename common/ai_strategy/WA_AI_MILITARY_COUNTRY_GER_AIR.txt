# purpose: keep a standing share of the Luftwaffe over the Reich once an enemy fields a bomber arm able
#          to raid it - the home-defence pull the engine's per-raid LAND_DEFENSE terms only supply
#          after the bombers arrive
# range:   +50000 per strategic region (HEAD carried +10000; the modder raised it to +50000 in the
#          working tree on 2026-08-16 before this relocation, aligning it with the DEFAULT_AIR standing
#          tier - kept at that value)
# policy:  Additive per region
# domain:  AIR
WA_AI_MILITARY_GER_AIR_reich_air_defence = {
	allowed = {
		original_tag = GER
	}
	enable = {
		WA_AI_MILITARY_AIR_actor_enabled = yes
		has_deployed_air_force_size = { size > 99 }
		WA_AI_MILITARY_AIR_enemy_fields_bomber_arm = yes			# arm above @WA_AI_AIR_BOMBER_NUCLEUS
	}
	# release only once every enemy arm is back under @WA_AI_AIR_ENEMY_BOMBER_ARM_OFF - the ON/OFF band
	# keeps a bomber count hovering at the bar from flipping the pull every 5-day air update
	abort = {
		OR = {
			NOT = { WA_AI_MILITARY_AIR_actor_enabled = yes }
			WA_AI_MILITARY_AIR_enemy_bomber_arm_gone = yes
		}
	}

	ai_strategy = { type = strategic_air_importance id = 6 value = 50000 }		# North-West Germany
	ai_strategy = { type = strategic_air_importance id = 7 value = 50000 }		# Western Germany
	ai_strategy = { type = strategic_air_importance id = 8 value = 50000 }		# Eastern Germany
}
