# purpose: stop defending the Raj's borders once it is collapsing or its sea line is cut (the
#          _defend_the_raj twin covers the viable window - keep the two conditions complementary)
# range:   +1000
# policy:  dont_defend_ally_borders per ally
# domain:  DIPLOMACY
WA_AI_MILITARY_COUNTRY_USA_DIPLOMACY_abandon_the_raj = {
	allowed = {
		tag = USA
	}
	enable = {
		has_government = democratic
		is_in_faction_with = RAJ
		is_in_faction_with = ENG
		JAP = {
			has_war_with = RAJ
		}
		OR = {
			RAJ = {
				surrender_progress > 0.2
			}
			NOT = {
				OR = {
					923 = {
						CONTROLLER = {
							OR = {
								tag = ENG
								is_in_faction_with = ENG
								is_subject_of = ENG
							}
						}
					}
					NOT = {
						AST = {
							surrender_progress > 0
						}
					}
				}
			}
		}
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = dont_defend_ally_borders
		id = "RAJ"
		value = 1000
	}
}
