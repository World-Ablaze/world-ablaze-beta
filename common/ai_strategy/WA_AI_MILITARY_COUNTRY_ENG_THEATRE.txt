# purpose: the Egyptian line - Alexandria, Marsa Matruh, the Suez Canal and Palestine held as one
#          garrison. Replaces WA_AI_MILITARY_ENG_protect_africa (deleted above) and the former
#          single-state 0.5 hoard on 447.
# pool:    subtract_fronts_from_need = no - the Nile line must hold while the Libyan front asks for
#          everything. The default (yes) is exactly what starved the deleted block: front orders in
#          the buffer's area subtract from its need, and north_africa is where ENG's land war lives.
# area:    north_africa KEPT on purpose - `area` names the ai_areas whose orders may DRAW on the
#          buffer, so the Egyptian front can still use these divisions when it attacks; only the
#          NEED is protected, by subtract_fronts_from_need. Omitting `area` would seal the garrison
#          (E4: right for a fortress point, wrong for a theatre reserve) - that is the second half of
#          the old defect, 6-10 divisions frozen on Alexandria alone. No narrower alias is possible:
#          common/ai_areas is at the boot-measured cap of 72 definitions on HOI4 1.19.2.
# ratio:   0.25, the E4 normal-garrison bar. The former 0.5 carried a `# siege:` justification that
#          its own always-on enable never matched (the Fix 100 shape); at 0.25 the ladder does not
#          apply and the justification is retired rather than restated.
# order_id 11 - private pool. order_id 1 is shared by eight entries in this file and mixing a
#          non-yielding member into it is the E4b failure mode.
# domain:  THEATRE
WA_AI_MILITARY_ENG_defense_of_el_alamein = {
	allowed = {
		original_tag = ENG
	}

		}
		# our side still holds the anchor
		447 = {
			CONTROLLER = {
				OR = {
					is_in_faction_with = ENG
					is_subject_of = ENG
					tag = ENG
				}
			}
		}
		# somebody hostile holds North African / East African ground - controller geography, no tag,
		# no date, so it reads correctly in an ahistorical game

	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = put_unit_buffers

		order_id = 11
