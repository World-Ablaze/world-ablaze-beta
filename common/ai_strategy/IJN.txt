IJN_tokyo_express_bismarck_focus = {
	allowed = {
		original_tag = IJN
	}

	enable = {
		tag = IJN
		has_country_flag = IJN_te_target_bismarck
	}

	abort_when_not_enabled = yes

	# The official ai_strategy docs cap naval_dominance at 100 (a percentage); WA's only
	# use of it (ENG, Gibraltar) is exactly 100, so an out-of-range 1000 may have been
	# discarded wholesale and with it the AI's reason to patrol here at all.
	ai_strategy = {
		type = naval_dominance
		id = 84 #Bismarck Sea
		value = 100
	}

	# Negative naval_avoid_region actively PULLS patrols into a region: WA's own ENG
	# strategy pins the Home Fleet to the Channel with value = -10000.
	ai_strategy = {
		type = naval_avoid_region
		id = 84 #Bismarck Sea
		value = -10000
	}
}
