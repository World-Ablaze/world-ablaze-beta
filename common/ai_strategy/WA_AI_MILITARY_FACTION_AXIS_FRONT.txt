# purpose: make Axis fronts facing the Soviet Union execute their attack orders when the weekly
#          posture calculus says battles are winnable (WA_AI_MILITARY_posture_effects.txt). Was
#          unconditionally on, which kept a broken 1943+ Ostheer lunging into a superior Red Army;
#          with the gate, the date-phased Barbarossa blocks below (priority 0) carry the scripted
#          1941-42 opening and this block only re-fires when the front is actually winnable.
#          Stale-variable note: posture_vs_SOV lingers after a peace, so has_war_with = SOV stays here.
# range:   n/a (front_control)
# policy:  Exclusive per front - priority 300; mutually exclusive with the grind block below by
#          posture level; outranked by the level-500/10000 emergency stops in DEFAULT_FRONT_control.
# domain:  FRONT
# FIX 54 (2026-08-13): generalised off the literal tag SOV. The posture calculus publishes a verdict
# per enemy but this block could read only one of them, so Soviet-aligned minors and any ahistorical
# enemy were invisible to it. See WA_AI_MILITARY_ALLIES_exec_vs_germany in the ALLIES_FRONT file for
# the full argument and the deliberate widening it carries. Name kept for this step; stage 2 collapses
# the three faction pairs into one Default-layer pair.
WA_AI_MILITARY_AXIS_exec_vs_sov = {
	enable = {
		WA_AI_MILITARY_is_axis_member = yes
		WA_AI_MILITARY_posture_has_execute_target = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		country_trigger = {
			has_war_with = ROOT
			has_capitulated = no
			OR = {
				# Too small for a posture verdict - see WA_AI_MILITARY_posture_has_execute_target.
				AND = {
					is_major = no
					NOT = { num_of_controlled_states > 4 }
				}
				ROOT = { check_variable = { WA_AI_MILITARY_posture_vs_@PREV = 1 } }
			}
		}
		ratio = 0.2
		priority = 300
		ordertype = front
		execution_type = balanced
		execute_order = yes
	}
}


# purpose: attrition grind vs the Soviet Union - posture level 2 means the Soviets are at maximum
#          mobilization with a nearly dry manpower pool while the Axis holds a deep reserve, so
#          pressing attacks pays even at odds below the level-1 bar. Careful execution: bleed,
#          do not lunge.
# range:   n/a (front_control)
# policy:  Exclusive per front - priority 300; mutually exclusive with the exec block above by
#          posture level (the calculus emits exactly one level per enemy).
# domain:  FRONT
# FIX 54: generalised off the literal tag SOV - see the exec block above. No "too small" branch here
# on purpose: an enemy with no published verdict reads 0, never 2, so exec and grind stay mutually
# exclusive exactly as before.

		}
		ratio = 0.2
