# Fix 53a (2026-08-13, checklist R36): de-duplicated. Each id was written TWICE,
# once quoted and once bare. Both spellings are accepted by the engine (vanilla uses
# the bare form at ENG.txt:1379 and the quoted form at JAP.txt:970), and
# naval_mission_threshold is documented Additive per mission
# (WA_AI_MILITARY_SYSTEM.md:106), so the pairs SUMMED: strike force was running at
# -1000 and patrol at -2000, double the intended values and 10-20x the type's
# documented -100..+100 range. Against an engine base of 100 that put patrol's bar at
# -1900 while convoy escort's stayed at +100 - the asymmetry behind R36's "USA runs 88
# ships on patrol across its own convoy coasts against 51 on escort". The intended
# single values are kept as-is (still out of the documented range, still permissive
# enough for the block's stated purpose) so this change removes an artefact without
# also re-tuning the two missions it was aimed at.

	enable = {
		always = yes
	}
	abort_when_not_enabled = yes

	ai_strategy = {
	    type = naval_mission_threshold
	    id = "MISSION_STRIKE_FORCE"
	    value = -500
	}

	ai_strategy = {
	    type = naval_mission_threshold
	    id = "MISSION_PATROL"
	    value = -1000
	}
}
