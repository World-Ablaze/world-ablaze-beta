# ============================================================================
# CV Plane Production Strategies
# ============================================================================
# These strategies are used to control the production of cv planes
# Due to how AI calculations work in HOI4,
# unit_roles define also AI CV squadrons compositions.
#
# [equipment-selection] the min-factory floors are four per-role blocks so each floor can be
# braked independently once that role's carrier aviation has outgrown the decks. The rationale,
# the measurements and the bar derivation live in the header of
# common/scripted_triggers/WA_AI_PRODUCTION_carrier_planes.txt - read it before touching any
# number here.
# ============================================================================

# [air-budget] The carrier pool's unit_ratio (cv_fighter / cv_cas / cv_naval_bomber) is emitted
# by WA_AI_AIR_BUDGET_reconcile (WA_AI_PRODUCTION_air_budget.txt, monthly) from the open carrier
# lines - the static 150 / 100 / -1000 block that lived here is gone with it. unit_ratio only
# splits the deck-derived pool between the cv roles (documentation.info, UNIT RATIOS / AIR); the
# floors below are the volume lever and stay as they are.

# Those min factories are to force AI to modernize its planes if requirements is met.
# [equipment-selection] gated per role on that role not being saturated. When the gate closes the
# floor disappears and the role falls back to the engine's own deck-derived demand,
# which for a saturated country is already zero - that is the whole point.
WA_DEFAULT_production_carrier_planes_floor_fighter = {
    enable = {
        WA_AI_PRODUCTION_carrier_planes_floor_fighter = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = equipment_production_min_factories  id = cv_fighter	value = 2 }
}

WA_DEFAULT_production_carrier_planes_floor_nav_bomber = {
    enable = {
        WA_AI_PRODUCTION_carrier_planes_floor_nav_bomber = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = equipment_production_min_factories  id = cv_naval_bomber	value = 2 }
}

# Carrier majors (a real deck fleet to feed) get a higher floor; minors keep the
# 2-factory default above. These values are ADDITIVE with the base floor - the two
# together are what produced the flat 8 factories measured on every carrier line of
# ENG/JAP/USA at both 1944.6 and 1946.8 in campaign f9321934.
WA_DEFAULT_production_carrier_planes_major_fighter = {
    enable = {
        WA_AI_PRODUCTION_carrier_planes_major_fighter = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = equipment_production_min_factories  id = cv_fighter	value = 5 }
}

WA_DEFAULT_production_carrier_planes_major_nav_bomber = {
    enable = {
        WA_AI_PRODUCTION_carrier_planes_major_nav_bomber = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = equipment_production_min_factories  id = cv_naval_bomber	value = 5 }
}
