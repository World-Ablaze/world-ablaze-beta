# ============================================================================
# CV Plane Production Strategies
# ============================================================================
# These strategies are used to control the production of cv planes
# Due to how AI calculations work in HOI4,
# unit_roles define also AI CV squadrons compositions.
# ============================================================================

# Ratios raised from 50/50 after campaign 9be92c89: cv_fighter at unit_ratio 50 lost
# every factory-allocation contest against land fighter (300) and sat pinned at the
# 2-factory floor for four years - USA carrier fighter wings averaged 4.8/10 planes
# while its nav bombers (whose land rival is only 10) reached 14 factories and 8.0/10.
# Incremental step (150/100), not the old system's 300: re-tune after an observer run.
WA_DEFAULT_production_carrier_planes_base_ratios = {
    enable = {
        WA_AI_PRODUCTION_production_carrier_planes_base_ratios = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = unit_ratio id = cv_fighter value = 150 }
    ai_strategy = { type = unit_ratio id = cv_naval_bomber value = 100 }
    ai_strategy = { type = unit_ratio id = cv_cas value = -1000 }

    #Those min factories are to force AI to modernize its planes if requirements is met.
    ai_strategy = { type = equipment_production_min_factories  id = cv_fighter	value = 2 }
    ai_strategy = { type = equipment_production_min_factories  id = cv_naval_bomber	value = 2 }
}

# Carrier majors (a real deck fleet to feed) get a higher floor; minors keep the
# 2-factory default above.
WA_DEFAULT_production_carrier_planes_major = {
    enable = {
        WA_AI_PRODUCTION_carrier_planes_major = yes
    }
    abort_when_not_enabled = yes

    ai_strategy = { type = equipment_production_min_factories  id = cv_fighter	value = 6 }
    ai_strategy = { type = equipment_production_min_factories  id = cv_naval_bomber	value = 6 }
}
