# purpose: Soviet counterattack window (ai_barb_timer) - execute attack orders vs Germany while the
#          weekly posture calculus says battles are winnable. The posture gate replaces the old
#          inline WA_AI_fielded_eq_ratio > 0.9 check: the calculus applies the same equipment floor
#          plus manpower and fighting-strength conditions (WA_AI_MILITARY_posture_effects.txt).
#          Level 1 (execute) and level 2 (grind) both qualify - during the counterattack window the
#          balanced execution below is intended either way.
#          Stale-variable note: posture_vs_GER lingers after a peace, so has_war_with = GER stays here.
# range:   n/a (front_control)
# policy:  Exclusive per front (Country layer); outranked by the level-500/10000 emergency stops in
#          DEFAULT_FRONT_control.
# domain:  FRONT

	allowed = {
		tag = SOV

		WA_AI_MILITARY_should_sov_counterattack = yes
	}

	abort_when_not_enabled = yes

	ai_strategy = {
		type = front_control
		country_trigger = {
			OR = {
				tag = GER
				is_in_faction_with = GER
			}
		}
		ordertype = front
		execution_type = balanced
		manual_attack = yes
		execute_order = yes
	}
}
