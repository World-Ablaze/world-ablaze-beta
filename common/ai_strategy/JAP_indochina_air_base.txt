# Occupation of Northern Indochina (JAP_northern_indochina_occupation) gives Japan air base access from
# whoever owns Tonkin. While Japan holds it, that country's AI stays friendly to Japan and accepts its
# air base requests, in the same way SIA_no_european_involvement steers the Siamese AI.
JAP_indochina_keep_air_base_access = {
	allowed = {
		always = yes
	}
	enable = {
		JAP = { has_country_flag = JAP_indochina_air_base_rights }
		671 = { is_owned_by = ROOT }
		NOT = { has_war_with = JAP }
	}
	abort = {
		OR = {
			has_war_with = JAP
			NOT = { 671 = { is_owned_by = ROOT } }
		}
	}

	ai_strategy = { type = diplo_action_acceptance id = JAP target = air_base_access value = 5000 }
	ai_strategy = { type = befriend id = JAP value = 200 }
	ai_strategy = { type = support id = JAP value = 200 }
}
