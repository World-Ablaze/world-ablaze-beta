# ============================================================================
# Army Composition Strategies
# ============================================================================
# These strategies control army composition using role_ratio.
# Every block here is PURELY ADDITIVE - it reduces infantry and adds
# the same amount to a new type, so they stack while maintaining sum = 100.
#
# Base: infantry = 100
# Each modifier: infantry -= X, new_type += X (net change = 0)
#
# EXCEPTION - the tank roles are NOT here. They share one constant budget emitted at runtime by
# common/scripted_effects/WA_AI_PRODUCTION_armor_budget.txt. See ARMOR MODIFIERS below.
# ============================================================================

# ============================================================================
# BASE COMPOSITION (always active when system enabled)
# ============================================================================

WA_AI_PRODUCTION_build_army_base = {
	enable = { WA_AI_PRODUCTION_army_composition_is_enabled = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = 100 }
}

# ============================================================================
# ARMOR MODIFIERS - MOVED, do not re-add blocks here
# ============================================================================
# [armor-role-budget] The five static blocks that used to live here (light, medium, the
# light-to-medium transition, heavy, modern) each moved 10 points from infantry to one tank role,
# so the armour share was 10 x the number of open tank roles and infantry paid it once per role.
# The tank roles now share ONE constant budget emitted at runtime by
# common/scripted_effects/WA_AI_PRODUCTION_armor_budget.txt, because the per-role value is
# budget / open-roles and `ai_strategy value =` takes a literal. A tank role_ratio block added
# back in this file would stack on top of that budget and re-create the bug.
# WHICH roles are open is still decided by the WA_AI_PRODUCTION_build_army_*_armor triggers in
# common/scripted_triggers/WA_AI_PRODUCTION_army_composition.txt; the SIZE is
# constant:wa_ai_production.army_composition.armor_budget_total.

# ============================================================================
# MOBILE MODIFIERS
# ============================================================================

# Motorized: infantry -5, motorized +5
WA_AI_PRODUCTION_DEFAULT_build_army_motorized = {
	enable = { WA_AI_PRODUCTION_build_army_motorized = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = -5 }
	ai_strategy = { type = role_ratio id = motorized value = 5 }
}

# Mechanized: infantry -20, mechanized +20
# [mech-divisions-usa-only] Since 2026-08-28 this is the expeditionary mechanized major OUTSIDE its
# window, not a global base tier: WA_AI_PRODUCTION_build_army_mechanized is now gated on
# WA_AI_TEMPLATES_use_mechanized_division_templates, so every other country has no mechanized
# role_ratio at all. Its value is unchanged.
# The NOT below keeps this and the -30/+30 block mutually exclusive (stacked, USA's infantry want
# went NEGATIVE - -25% of 575 wanted divisions in campaign 9be92c89 - and it stopped training
# entirely). The exclusion lives HERE and not in the trigger because the expeditionary trigger
# itself requires WA_AI_PRODUCTION_build_army_mechanized - putting a NOT in the trigger would
# recurse.
WA_AI_PRODUCTION_DEFAULT_build_army_mechanized = {
	enable = {
		WA_AI_PRODUCTION_build_army_mechanized = yes
		NOT = { WA_AI_PRODUCTION_build_army_expeditionary_mechanization = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = -20 }
	ai_strategy = { type = role_ratio id = mechanized value = 20 }
}

# Expeditionary mechanization: infantry -30, mechanized +30. REPLACES the base
# mechanized block above (mutually exclusive), it does not stack on it.
# [armor-role-budget] Deepest reachable USA stack: 100 -25 armor budget -30 here -10 special forces
# = 35 infantry. Motorized cannot join it (its template trigger excludes mechanized) and marines and
# mountaineers are mutually exclusive, so 35 is the floor, not an estimate.
WA_AI_PRODUCTION_DEFAULT_build_army_expeditionary_mechanization = {
	enable = { WA_AI_PRODUCTION_build_army_expeditionary_mechanization = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = -30 }
	ai_strategy = { type = role_ratio id = mechanized value = 30 }
}

# Shared payload for every owner rule that forbids training divisions; WHO is suppressed lives in
# WA_AI_PRODUCTION_trains_no_divisions (scripted_triggers/WA_AI_PRODUCTION_army_composition.txt).
# role_ratio is base 100 + value (documentation.info, UNIT RATIOS / ALL BUT AIR); -1000 keeps the
# want at zero through every additive modifier above. cavalry covers the Military Police
# templates; garrison role is gone (see SUPPORT MODIFIERS above).
# [rk-no-divisions] A Reichskommissariat trains NO divisions of any role (owner rule 2026-08-27):
# its manpower and MIC belong to the overlord, so anything it trains burns overlord equipment.
# The engine may also DECOMMISSION existing RK divisions of zero-want roles (recorded mechanism,
# see WA_AI_TEMPLATES_garrison.txt header) - acceptable here, the owner rule wants them at none.
# [minor-expeditionary-fitness] A country under 5 military factories does not BUILD past 5
# divisions (owner rule, no exemptions incl. a crippled country under invasion): it cannot equip
# them. A cap, not a zero - disarmed under the cap. Overshoot bound = the training queue at the
# moment deployed crosses 4: t0 arm with Q queued (a <5-MIL conveyor holds ~1-3), t1 those Q
# deploy (ASSUMED: -500/-1000 stop new queuing, do not cancel queued), terminal <= 5+Q, flat
# after. Accepted cycle cost, in writing: if the engine decommissions zero-want-role divisions
# (WA_AI_TEMPLATES_garrison.txt header, ASSUMED for deployed units), the count can saw between
# 4 and 5+Q, burning training time and equipment each pass - if a campaign shows that sawtooth,
# the lever is a flag-latched hysteresis, not a deeper value.
WA_AI_PRODUCTION_DEFAULT_trains_no_divisions = {
	enable = { WA_AI_PRODUCTION_trains_no_divisions = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = build_army id = infantry value = -500 }
	ai_strategy = { type = role_ratio id = infantry value = -1000 }
	ai_strategy = { type = role_ratio id = cavalry value = -1000 }
	ai_strategy = { type = role_ratio id = suppression value = -1000 }
	ai_strategy = { type = role_ratio id = motorized value = -1000 }
	ai_strategy = { type = role_ratio id = mechanized value = -1000 }
	ai_strategy = { type = role_ratio id = armor value = -1000 }
	ai_strategy = { type = role_ratio id = light_armor value = -1000 }
	ai_strategy = { type = role_ratio id = medium_armor value = -1000 }
	ai_strategy = { type = role_ratio id = heavy_armor value = -1000 }
	ai_strategy = { type = role_ratio id = modern_armor value = -1000 }
	ai_strategy = { type = role_ratio id = marines value = -1000 }
	ai_strategy = { type = role_ratio id = mountaineers value = -1000 }
}
