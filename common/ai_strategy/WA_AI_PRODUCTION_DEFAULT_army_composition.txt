# ============================================================================
# Army Composition Strategies
# ============================================================================
# These strategies control army composition using role_ratio.
# All strategies are PURELY ADDITIVE - each reduces infantry and adds
# the same amount to a new type, so they stack while maintaining sum = 100.
#
# Base: infantry = 100
# Each modifier: infantry -= X, new_type += X (net change = 0)
# ============================================================================

# ============================================================================
# BASE COMPOSITION (always active when system enabled)
# ============================================================================

WA_AI_PRODUCTION_build_army_base = {
	enable = { WA_AI_PRODUCTION_army_composition_is_enabled = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = 100 }
}

# Mechanized: infantry -20, mechanized +20
# The NOT below makes this the base tier only: expeditionary majors get the -30/+30
# block instead of stacking both (stacked, USA's infantry want went NEGATIVE - -25%
# of 575 wanted divisions in campaign 9be92c89 - and it stopped training entirely).
# The exclusion lives HERE and not in the trigger because the expeditionary trigger
# itself requires WA_AI_PRODUCTION_build_army_mechanized - putting a NOT in the
# trigger would recurse.
WA_AI_PRODUCTION_DEFAULT_build_army_mechanized = {
	enable = {
		WA_AI_PRODUCTION_build_army_mechanized = yes
		NOT = { WA_AI_PRODUCTION_build_army_expeditionary_mechanization = yes }
	}
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = -20 }
	ai_strategy = { type = role_ratio id = mechanized value = 20 }
}

# Expeditionary mechanization: infantry -30, mechanized +30. REPLACES the base
# mechanized block above (mutually exclusive), it does not stack on it.
# Post-fix invariant (USA with everything active): 100 -20 medium -20 modern
# -30 here -10 marines -5 garrison = -5, recovered to +10 by the infantry floor
# below. Without modern armor: +15. Non-expeditionary majors are untouched by
# this block and by the floor (GER/SOV stay at their previous arithmetic).
WA_AI_PRODUCTION_DEFAULT_build_army_expeditionary_mechanization = {
	enable = { WA_AI_PRODUCTION_build_army_expeditionary_mechanization = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = -30 }
	ai_strategy = { type = role_ratio id = mechanized value = 30 }
}

# Infantry floor: insurance against the role_ratio residual going negative again.
# Only reachable on the deepest modifier stack (expeditionary + modern armor,
# currently USA-only), so it cannot move any other country's composition.
WA_AI_PRODUCTION_DEFAULT_build_army_infantry_floor = {
	enable = { WA_AI_PRODUCTION_build_army_infantry_floor = yes }
	abort_when_not_enabled = yes

	ai_strategy = { type = role_ratio id = infantry value = 15 }
}
