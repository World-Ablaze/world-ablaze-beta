############################################################################################################
#	WA AI LANDING - scripted-landing invasion freeze (MARKER)
############################################################################################################
#	Control panel (switches, window length, macro-theatre definition):
#	    common/scripted_triggers/WA_AI_LANDING_triggers.txt
#	Consumers (the ai_strategy blocks that do the suppressing):
#	    common/ai_strategy/WA_AI_MILITARY_DEFAULT_INVASION_landing_freeze.txt
#	Spec: documentation/WA_AI_MILITARY_SYSTEM.md sec.10
#
#	WHY A SCRIPTED MARKER EXISTS AT ALL. An ai_strategy block cannot set a flag, and the *execution* of a
#	scripted landing is not expressible as an ai_strategy `enable` - a landing's enable turning true and a
#	landing actually happening are different events (every operation in events/WA_AI_invasions.txt re-fires
#	itself every 7 days until its preconditions hold, so "enabled" is true for months before anything
#	lands). The only place that knows a landing has really happened is the effect that creates the
#	divisions on the enemy beach: WA_AI_DIVISION_spawn_invasion. That is the single shared marker point -
#	all ~90 scripted operations route through it - and it is where this system is called from.
############################################################################################################

############################################################################################################
#	WA_AI_LANDING_register_scripted_landing
#
#	SCOPE:   ROOT = THIS = the invading country (the scope WA_AI_DIVISION_spawn_invasion runs in).
#	INPUT:   _invasion_state - temp variable, state ID of the landing site. Already set by every caller;
#	         this effect is only ever called from inside spawn_invasion's own success branch.
#	OUTPUT:  a freeze flag stamped on the invader and on every AI country in its faction, for the
#	         macro-theatre the landing site sits in. WA_TLM probe counters on each stamped country.
#	         No output at all for a landing in the Americas (see the triggers file) - by design.
#
#	COST: one `every_other_country` sweep with a 3-term limit, per landing operation per target country
#	(the call site sits inside spawn_invasion's has_relation_modifier guard, so a multi-wave operation
#	such as Weserubung books ONE sweep, not six). ~200 country evaluations on a few dozen days per
#	campaign - negligible against the weekly AI work, and event-driven rather than on a pulse.
############################################################################################################
WA_AI_LANDING_register_scripted_landing = {
	if = {
		limit = { check_variable = { _invasion_state > 0 } }
		# Enter the landing state to read its continent. Nothing is written in state scope: the two
		# stamp effects below run back in country scope via PREV, so no temp variable ever has to
		# survive a scope hop.
		var:_invasion_state = {
			if = {
				limit = {
					OR = {
						is_on_continent = europe
						is_on_continent = africa
						is_on_continent = middle_east
					}
				}
				PREV = { WA_AI_LANDING_stamp_freeze_west = yes }
			}
			else_if = {
				limit = {
					OR = {
						is_on_continent = asia
						is_on_continent = australia
					}
				}
				PREV = { WA_AI_LANDING_stamp_freeze_east = yes }
			}
			# else: north_america / south_america - no scripted landing targets them and freezing a
			# real theatre off an American landing would be wrong. Deliberately no branch.
		}
	}
}

############################################################################################################
#	WA_AI_LANDING_stamp_freeze_west / _stamp_freeze_east
#
#	SCOPE: ROOT = THIS = the invading country.
#
#	THE TWO BODIES MUST MATCH except for the flag name and the two theatre-specific counter names. They
#	are duplicated rather than parameterised because the flag name is a literal token and a meta_effect
#	round-trip to inject it would cost more than it saves for two cases.
#
#	clr + set rather than a bare set: re-stamping must restart the 90-day clock deterministically, and
#	the codebase does not rely on set_country_flag refreshing an existing flag's date.
############################################################################################################
WA_AI_LANDING_stamp_freeze_west = {
	clr_country_flag = WA_AI_LANDING_freeze_west
	set_country_flag = WA_AI_LANDING_freeze_west
	# tlm: r44_freeze_n / _freeze_west_n / _freeze_first_t / _freeze_last_t - consumer: checklist R44.
	# Verified effect (doc sec.3.6 rule 1): the flag this counts is written on the line above, so the
	# counter cannot run ahead of the behaviour. _first_t is written under an `= 1` guard, so read
	# _freeze_n > 0 before trusting _first_t (a stamp at clock 0 is a legitimate 0).
	add_to_variable = { WA_TLM_r44_freeze_n = 1 }
	add_to_variable = { WA_TLM_r44_freeze_west_n = 1 }
	if = {
		limit = { check_variable = { WA_TLM_r44_freeze_n = 1 } }
		set_variable = { WA_TLM_r44_freeze_first_t = global.WA_TLM_clock }
	}
	set_variable = { WA_TLM_r44_freeze_last_t = global.WA_TLM_clock }

	every_other_country = {
		limit = {
			exists = yes
			is_ai = yes
			is_in_faction_with = ROOT
		}
		clr_country_flag = WA_AI_LANDING_freeze_west
		set_country_flag = WA_AI_LANDING_freeze_west
		# tlm: same family, on the faction member that received the freeze - consumer: checklist R44
		add_to_variable = { WA_TLM_r44_freeze_n = 1 }
		add_to_variable = { WA_TLM_r44_freeze_west_n = 1 }
		if = {
			limit = { check_variable = { WA_TLM_r44_freeze_n = 1 } }
			set_variable = { WA_TLM_r44_freeze_first_t = global.WA_TLM_clock }
		}
		set_variable = { WA_TLM_r44_freeze_last_t = global.WA_TLM_clock }
	}
}

############################################################################################################
#	[scripted-invasion-reservation] TARGET RESERVATION - the calendar's targets are off-limits BEFORE the op
############################################################################################################
#	The freeze above suppresses engine invasions AFTER a scripted landing. This system is its
#	forward-looking twin: while the scripted calendar still owes an operation against a country, the
#	invader's whole faction stops requesting divisions for engine-planned invasions of that country -
#	so the engine does not park a corps for its own Brittany landing while the calendar will open
#	Normandy itself (owner request, 2026-08-23).
#
#	PER COUNTRY, NOT PER STATE - the owner's explicit call: the waste being fixed is divisions
#	immobilised against the target COUNTRY anywhere, not just on the scripted beach.
#
#	PIECES
#	  WA_AI_LANDING_reservations_data.txt   GENERATED calendar mirror: per-invader arrays
#	                                        op_anchor (state) / op_expiry (day since 1936.1.1)
#	                                        - regenerate via tools/gen_ai_landing_reservations.py
#	  this file                             epoch init + monthly updater + stamp (below)
#	  WA_AI_LANDING_triggers.txt            the switch (WA_AI_LANDING_reservations_enabled)
#	  WA_AI_MILITARY_DEFAULT_INVASION_landing_freeze.txt   the consumer ai_strategy block
#	  documentation/WA_AI_MILITARY_SYSTEM.md sec.22        the spec
#
#	MECHANISM. Monthly, each AI invader on the historical difficulty walks its pending operations
#	(expiry day not passed). For each, the reservation target is the CURRENT CONTROLLER of the op's
#	anchor state - dynamic geography, never a stored tag, so a France that fell to GER reserves GER
#	and a Torch against an unconquered VIC reserves VIC. If that controller is at war with the
#	invader, a timed flag WA_AI_LANDING_reserved_for_<member tag> is stamped ON THE TARGET for the
#	invader and every AI member of its faction. The flag is a LEASE (@WA_AI_LANDING_RESERVATION_
#	LEASE_DAYS > the monthly pulse) renewed while any op against that target is pending, so it dies
#	by itself within one lease of the LAST scheduled operation against that country - which is the
#	requested semantics "until the date of the last scripted invasion". Precision bounds: a new
#	faction member waits <= 1 pulse for its flag; a target that changes controller keeps the stale
#	flag <= 1 lease (t0 stamp, t+31 re-stamp on the new controller, t+50 old lease dies).
############################################################################################################

# Reservation lease, in days. Must exceed the monthly pulse gap (31 days) or reservations flicker
# between pulses; the excess over 31 is the precision tail after the last scheduled operation.
@WA_AI_LANDING_RESERVATION_LEASE_DAYS = 50

############################################################################################################
#	WA_AI_LANDING_init_reservations
#
#	SCOPE: any (startup, via random_country).
#	Loads the generated calendar arrays and calibrates the day epoch:
#	global.WA_AI_LANDING_epoch = value of global.num_days at 1936.1.1, so that
#	(global.num_days - epoch) is "days since 1936.1.1" - the unit op_expiry is generated in.
#
#	RUNS AT NEW-GAME CREATION ONLY: on_startup does NOT re-fire when a savegame is loaded
#	(MEASURED - lessons-log "on_startup does not re-fire when a savegame is loaded"). That is
#	sufficient because the arrays, the epoch variable and the epoch_set flag all persist in the
#	save. ACCEPTED consequence: a campaign saved on a build WITHOUT this system never initializes
#	it - op_n stays absent, the monthly gate reads 0, the system is inert for that campaign, and
#	the absence of wa_tlm_resv_* on such a save is the fingerprint. No migration path on purpose.
#	The mis-dated-epoch scenario (init running mid-campaign and hitting the Blitzkrieg branch)
#	cannot occur for the same reason: init never runs on a loaded save.
#	global.num_days available at startup is ASSUMED (vanilla reads it mid-game only,
#	00_on_actions.txt:662); if it read 0 here, every op would sit "pending" until the boot test
#	catches it - epoch and op_expiry are both printable from the console.
############################################################################################################
WA_AI_LANDING_init_reservations = {
	WA_AI_LANDING_load_ops = yes
	if = {
		limit = { NOT = { has_global_flag = WA_AI_LANDING_epoch_set } }
		set_global_flag = WA_AI_LANDING_epoch_set
		set_variable = { global.WA_AI_LANDING_epoch = global.num_days }
		# Non-1936 bookmark correction: epoch must be backdated to 1936.1.1. The only other
		# bookmark is Blitzkrieg 1939.8.14 (common/bookmarks/blitzkrieg.txt) = 1321 days after
		# 1936.1.1 on the leap-aware calendar; +-1 day of engine leap ambiguity is immaterial
		# against the 50-day lease. A THIRD bookmark added later needs its own branch here.
		if = {
			limit = { WA_AI_CONFIG_after_1937 = yes }
			subtract_from_variable = { global.WA_AI_LANDING_epoch = 1321 }
		}
	}
}

############################################################################################################
#	WA_AI_LANDING_update_reservations
#
#	SCOPE:  ROOT = THIS = an invader country. Caller (WA_AI_misc_on_actions on_monthly) gates on
#	        is_ai + WA_AI_DIFFICULTY_is_historical + has_war + op_n > 0, so this body runs for at
#	        most the 5 calendar invaders, monthly, in wartime.
#	READS:  WA_AI_LANDING_op_anchor / op_expiry / op_n (generated), global.WA_AI_LANDING_epoch.
#	OUTPUT: reservation flags on every pending target, via WA_AI_LANDING_stamp_reservation.
############################################################################################################
WA_AI_LANDING_update_reservations = {
	set_temp_variable = { _resv_today = global.num_days }
	# subtract_from_TEMP_variable, not subtract_from_variable: the non-temp form leaves a temp
	# operand UNCHANGED (MEASURED 2026-08-23, wa_resv.1 leg A: today printed = num_days, epoch
	# never subtracted - the mixed idiom silently no-ops on the temp).
	subtract_from_temp_variable = { _resv_today = global.WA_AI_LANDING_epoch }
	for_each_loop = {
		array = WA_AI_LANDING_op_anchor
		value = _resv_anchor
		index = _resv_i
		if = {
			limit = { check_variable = { WA_AI_LANDING_op_expiry^_resv_i > _resv_today } }
			set_temp_variable = { _resv_target = 0 }
			var:_resv_anchor = {
				CONTROLLER = { set_temp_variable = { _resv_target = id } }
			}
			if = {
				limit = {
					check_variable = { _resv_target > 0 }
					var:_resv_target = { has_war_with = ROOT }
				}
				WA_AI_LANDING_stamp_reservation = yes
			}
		}
	}
}

############################################################################################################
#	WA_AI_LANDING_stamp_reservation
#
#	SCOPE:  ROOT = THIS = the invader. INPUT: _resv_target - temp, country id at war with ROOT.
#	OUTPUT: on the target, one timed flag per reserving country: WA_AI_LANDING_reserved_for_<TAG>
#	        for ROOT and every AI member of ROOT's faction. @PREV renders as the tag of the country
#	        scope surrounding the target scope (vanilla dynamic-flag idiom, see decisions/JAP.txt
#	        WTT_border_conflict_*_@FROM). clr + set rather than bare set: re-stamping must restart
#	        the lease deterministically, same rule as the freeze stamps above.
#	CONSUMER: WA_AI_MILITARY_INV_reserved_scripted_target reads the flag through @FROM in its
#	        country_trigger (scope = enemy country, FROM = the evaluating country).
############################################################################################################
WA_AI_LANDING_stamp_reservation = {
	var:_resv_target = {
		clr_country_flag = WA_AI_LANDING_reserved_for_@PREV
		set_country_flag = { flag = WA_AI_LANDING_reserved_for_@PREV value = 1 days = @WA_AI_LANDING_RESERVATION_LEASE_DAYS }
	}
	# tlm: resv_stamp_n / resv_first_t / resv_last_t - consumer: [scripted-invasion-reservation]
	# verification. Verified effect (doc sec.3.6 rule 1): the flag this counts is written on the
	# lines above. Second signal for the first campaign: the WA_AI_LANDING_reserved_for_* flags
	# themselves serialize into the save on the target country.
	add_to_variable = { WA_TLM_resv_stamp_n = 1 }
	if = {
		limit = { check_variable = { WA_TLM_resv_stamp_n = 1 } }
		set_variable = { WA_TLM_resv_first_t = global.WA_TLM_clock }
	}
	set_variable = { WA_TLM_resv_last_t = global.WA_TLM_clock }

	every_other_country = {
		limit = {
			exists = yes
			is_ai = yes
			is_in_faction_with = ROOT
		}
		var:_resv_target = {
			clr_country_flag = WA_AI_LANDING_reserved_for_@PREV
			set_country_flag = { flag = WA_AI_LANDING_reserved_for_@PREV value = 1 days = @WA_AI_LANDING_RESERVATION_LEASE_DAYS }
		}
	}
}
