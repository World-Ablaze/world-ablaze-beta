### manages lend lease diplomacy

WA_AI_update_lend_lease_targets = {

	# lend lease coordinated target by player
	if = {
		limit = {
			any_other_country = {
				WA_AI_PREV_on_same_side_with_THIS = yes
				has_country_flag = WA_AI_coordinated_lend_lease
				set_temp_variable = { coordinated_lend_lease_iniatiator = THIS.id }
			}
		}

		var:coordinated_lend_lease_iniatiator = {

			var:WA_AI_coordinated_lend_lease_target = {

				if = { limit = { NOT = { has_country_flag = WA_AI_lend_lease_from_@ROOT } }

					ROOT = {
						WA_AI_clear_lend_lease_targets = yes
						WA_AI_set_lend_lease_target = yes
					}

					if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: [This.GetName] (coordinated)" }
				}
			}
		}
	}

	# lend lease any valid targets
	else_if = {
		limit = {
			# game rules for lend lease
			any_other_country = {
				OR = {
					is_subject_of = ROOT

					has_war_together_with = ROOT

					AND = {
						has_war = yes

						ROOT = {
							OR = {
								AND = { has_government = neutrality NOT = { threat < 0.4 } }
								AND = { has_government = fascism }
								AND = { has_government = communism }
								AND = { has_government = democratic NOT = { threat < 0.5 } }
								AND = { has_government = democratic original_tag = USA has_completed_focus = USA_lend_lease_act }
							}
						}
					}
				}
			}
		}

		every_country = { limit = { WA_AI_lend_lease = yes }

			add_to_temp_array = { array = updated_lend_lease_targets value = THIS.id }
		}

		# clear invalid targets
		for_each_scope_loop = { array = WA_AI_lend_lease_targets

			if = { limit = { NOT = { is_in_array = { array = updated_lend_lease_targets value = THIS.id } } }

				add_to_array = { array = ROOT.WA_AI_remove_lend_lease value = THIS.id }
			}
		}
		for_each_scope_loop = { array = WA_AI_remove_lend_lease

			WA_AI_clear_lend_lease_target = yes
		}
		clear_array = WA_AI_remove_lend_lease

		# set lend lease targets
		for_each_scope_loop = { array = updated_lend_lease_targets

			if = { limit = { NOT = { is_in_array = { array = ROOT.WA_AI_lend_lease_targets value = THIS.id } } }

				ROOT = { WA_AI_set_lend_lease_target = yes }
			}
		}
	}
	else = {
		WA_AI_clear_lend_lease_targets = yes
	}
}

WA_AI_set_lend_lease_target = { # THIS = SENDER PREV = TARGET

	PREV = { set_country_flag = WA_AI_lend_lease_from_@PREV }
	add_to_array = { array = WA_AI_lend_lease_targets value = PREV.id }
	if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [This.GetName] | LEND-LEASE: [Prev.GetName]" }
}

WA_AI_clear_lend_lease_target = { # THIS = TARGET PREV = SENDER

	clr_country_flag = WA_AI_lend_lease_from_@PREV

	# compare tags because the id may not match after civil wars
	for_each_loop = { array = PREV.WA_AI_lend_lease_targets

		var:v = {
			if = { limit = { tag = PREV }

				remove_from_array = { array = PREV.PREV.WA_AI_lend_lease_targets index = i }
				set_temp_variable = { break = 1 }
			}
		}
	}
	set_temp_variable = { break = 0 }

	if = { limit = { PREV = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Prev.GetName] | CANCEL LEND-LEASE: [This.GetName]" }
}

WA_AI_clear_lend_lease_targets = { # THIS = SENDER

	if = { limit = { check_variable = { WA_AI_lend_lease_targets^num > 0 } }

		every_country = { limit = { has_country_flag = WA_AI_lend_lease_from_@PREV }

			WA_AI_clear_lend_lease_target = yes
		}

		clear_array = WA_AI_lend_lease_targets
	}
}
### Surplus relief - direct equipment transfer from a neighbouring ally (weekly) - [lend-lease-relief] (2026-08-16)
# ROOT = RECIPIENT (AI; the weekly call site is inside the is_ai gate).
# Pull, not push: the starving country selects ONE donor per week (random_other_country caps
# intake at one donor per recipient per week) and every send runs in the DONOR's scope with
# target = ROOT. Per-transfer re-reads of the donor's stockpile stop a donor the moment it
# drops to its own floor, even mid-tick across several recipients.
#
# The system imitates an OVERLAND truck convoy, so it fires only when a state path exists
# between the two CAPITALS that crosses nothing but territory controlled by the recipient,
# the donor, a subject of either, or a faction member / co-belligerent of the recipient
# (modder ruling 2026-08-16). The nine land-equipment legs remain overland-only. Convoys use
# the separate maritime branch below and never enter this A*. Same landmass is NOT enough. The
# path is the state-level A* (WA_AI_PATHFIND_get_path, type 3) run ONCE per picked donor,
# cached in _llr_path_ok for every leg; it is deliberately not in the pick's `limit`.
#
# Archetypes are COMPLEMENTARY (rifles AND heavy infantry AND support ...) - one leg per
# archetype, never a sum; policy per leg lives in WA_AI_LEND_LEASE_triggers.txt and the
# numbers in common/script_constants/wa_ai_lend_lease.txt. Amounts are tiered by donor
# surplus (base / x2 / x4) and clamped so the donor never dips under its floor - a 300k-rifle
# hoarder dripping 6000/week while an ally collapses is not relief.
#
# Form (validated by boot test wa_test.300 on 2026-08-16): the archetype token stays a literal
# per leg (nothing can render it from an index - scripted-loc rendering does not see the
# scope's variables inside meta_effect, and an unparsable has_equipment meta_trigger evaluates
# TRUE), while amount and target are rendered from variables (the exact form the boot test proved).
# Do not "factor" further.
#
# Telemetry (WA_TLM v18; convoy extension v28; pair matrix v32): llr_starving_n /
# llr_donor_selected_n / llr_path_refused_n on the RECIPIENT, plus llr_convoy_starving_n /
# _donor_selected_n, llr_sent_n^idx / llr_sent_amount^idx on the DONOR (idx = row order of the
# constants file), llr_recv_donor / _n / _amount pair arrays on the RECIPIENT,
# llr_first_t / llr_last_t on both - counted on the verified send only, never on entry.
WA_AI_LEND_LEASE_request_surplus_relief = {
	if = {
		limit = { WA_AI_LEND_LEASE_is_starving_any = yes }

		# tlm: llr_starving_n - weeks this recipient was short of at least one archetype (the pull
		# was entered). With donor_selected_n it says whether the lottery ever found a candidate.
		add_to_variable = { WA_TLM_llr_starving_n = 1 }

		random_other_country = {
			limit = {
				PREV = { WA_AI_LEND_LEASE_relief_pair_is_valid = yes }
				PREV = { WA_AI_LEND_LEASE_donor_can_serve = yes }
			}

			# THIS = donor, ROOT = recipient from here on
			set_temp_variable = { _llr_donor = THIS.id }
			# tlm: llr_donor_selected_n - the lottery picked a donor (path not yet tested). Standing family llr, consumer R7b.
			ROOT = { add_to_variable = { WA_TLM_llr_donor_selected_n = 1 } }

			WA_AI_LEND_LEASE_relief_land_access = yes

			if = {
				limit = { check_variable = { _llr_path_ok = 1 } }
				WA_AI_LEND_LEASE_relief_leg_infantry = yes
				WA_AI_LEND_LEASE_relief_leg_heavy_infantry = yes
				WA_AI_LEND_LEASE_relief_leg_support = yes
				WA_AI_LEND_LEASE_relief_leg_artillery = yes
				WA_AI_LEND_LEASE_relief_leg_heavy_artillery = yes
				WA_AI_LEND_LEASE_relief_leg_anti_tank = yes
				WA_AI_LEND_LEASE_relief_leg_anti_air = yes
				WA_AI_LEND_LEASE_relief_leg_motorized = yes
				WA_AI_LEND_LEASE_relief_leg_train = yes
			}
			else = {
				# tlm: llr_path_refused_n - the picked donor has no land path to us. Standing family llr, consumer R7b.
				ROOT = { add_to_variable = { WA_TLM_llr_path_refused_n = 1 } }
				if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: relief from [This.GetName] refused - no land access between capitals" }
			}
		}
	}

	# Maritime convoy relief is separate from the Fix-92 overland transfer. One donor per
	# starving recipient per weekly pulse; no A*. The leg clamps the send to the donor floor.
	if = {
		limit = { WA_AI_LEND_LEASE_is_starving_convoys = yes }
		# tlm: llr_convoy_starving_n - weekly recipient-state denominator for R56.
		add_to_variable = { WA_TLM_llr_convoy_starving_n = 1 }
		random_other_country = {
			limit = {
				PREV = { WA_AI_LEND_LEASE_convoy_pair_is_valid = yes }
				WA_AI_LEND_LEASE_has_exportable_surplus_convoys = yes
			}
			# tlm: llr_convoy_donor_selected_n - a real maritime donor was selected for R56.
			ROOT = { add_to_variable = { WA_TLM_llr_convoy_donor_selected_n = 1 } }
			WA_AI_LEND_LEASE_relief_leg_convoy = yes
		}
	}
}

# THIS = donor, ROOT = recipient. Sets _llr_path_ok = 1 when a state path exists from the
# recipient's capital state to the donor's capital state through friendly territory
# (pathfind type 3, WA_AI_pathfinding_effects.txt). Conservative: any failure (75-iteration
# cap, missing capital) reads as no access.
WA_AI_LEND_LEASE_relief_land_access = {
	set_temp_variable = { _llr_path_ok = 0 }
	set_temp_variable = { _pathfind_start = 0 }
	set_temp_variable = { _pathfind_target = 0 }
	ROOT = { capital_scope = { set_temp_variable = { _pathfind_start = THIS.id } } }
	capital_scope = { set_temp_variable = { _pathfind_target = THIS.id } }
	if = {
		limit = {
			# .id of a state scope is an engine-encoded reference (can be negative) - test against
			# the 0 we initialised, never with > 0 (lessons log)
			NOT = { check_variable = { _pathfind_start = 0 } }
			NOT = { check_variable = { _pathfind_target = 0 } }
		}
		set_temp_variable = { _pathfind_type = 3 }
		WA_AI_PATHFIND_get_path = yes
		if = { limit = { check_variable = { pathfind_success = 1 } }
			set_temp_variable = { _llr_path_ok = 1 }
		}
	}
}

# THIS = donor, ROOT = recipient. Generic tier rule shared by every leg. Inputs (temp):
# _llr_donor_stock, _llr_recipient_stock, _llr_floor, _llr_starve, _llr_base, _llr_tier2,
# _llr_tier4. Output: _llr_amount (0 = do not send). The donor never dips under its floor.
WA_AI_LEND_LEASE_relief_compute_amount = {
	set_temp_variable = { _llr_amount = 0 }
	if = {
		limit = {
			check_variable = { _llr_recipient_stock < _llr_starve }
			check_variable = { _llr_donor_stock > _llr_floor }
		}
		set_temp_variable = { _llr_amount = _llr_base }
		if = { limit = { check_variable = { _llr_donor_stock > _llr_tier4 } }
			multiply_temp_variable = { _llr_amount = 4 }
		}
		else_if = { limit = { check_variable = { _llr_donor_stock > _llr_tier2 } }
			multiply_temp_variable = { _llr_amount = 2 }
		}
		set_temp_variable = { _llr_headroom = _llr_donor_stock }
		subtract_from_temp_variable = { _llr_headroom = _llr_floor }
		if = { limit = { check_variable = { _llr_amount > _llr_headroom } }
			set_temp_variable = { _llr_amount = _llr_headroom }
		}
		# never over-relieve: fill the recipient to its starve line plus one base, no more -
		# bounds the waste on an archetype the recipient does not field to starve + base, once
		set_temp_variable = { _llr_need = _llr_starve }
		add_to_temp_variable = { _llr_need = _llr_base }
		subtract_from_temp_variable = { _llr_need = _llr_recipient_stock }
		if = { limit = { check_variable = { _llr_amount > _llr_need } }
			set_temp_variable = { _llr_amount = _llr_need }
		}
		if = { limit = { check_variable = { _llr_amount < 1 } }
			set_temp_variable = { _llr_amount = 0 }
		}
	}
}

# THIS = donor, ROOT = recipient. Telemetry + log after a VERIFIED send: the leg re-reads the
# recipient's free stock after the meta_effect (_llr_recipient_after) and this records only if it
# rose above the pre-send read (_llr_recipient_stock); a silent meta failure lands in
# llr_send_failed_n instead of inflating llr_sent_n.
# tlm: llr_sent_n^idx / llr_sent_amount^idx / llr_send_failed_n (donor) + llr_first_t / llr_last_t
# (both sides). Standing family llr, consumers R7b / R56.
WA_AI_LEND_LEASE_relief_record = {
	# A late-created tag has no startup init; an old save has arrays of size 10. Grow either
	# shape idempotently before idx 10 is written, preserving the nine existing rows.
	if = { limit = { check_variable = { WA_TLM_llr_sent_n^num < 11 } } resize_array = { array = WA_TLM_llr_sent_n size = 11 value = 0 } }
	if = { limit = { check_variable = { WA_TLM_llr_sent_amount^num < 11 } } resize_array = { array = WA_TLM_llr_sent_amount size = 11 value = 0 } }
	if = {
		limit = { NOT = { check_variable = { _llr_recipient_after > _llr_recipient_stock } } }
		add_to_variable = { WA_TLM_llr_send_failed_n = 1 }
		if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: relief archetype [?_llr_arch_idx] x [?_llr_amount] from [This.GetName] did NOT land (stock unchanged) - meta send failed?" }
	}
	else = {
	add_to_variable = { WA_TLM_llr_sent_n^_llr_arch_idx = 1 }
	add_to_variable = { WA_TLM_llr_sent_amount^_llr_arch_idx = _llr_amount }
	if = { limit = { check_variable = { WA_TLM_llr_first_t = 0 } } set_variable = { WA_TLM_llr_first_t = global.WA_TLM_clock } }
	set_variable = { WA_TLM_llr_last_t = global.WA_TLM_clock }
	# tlm: llr_recv_donor / llr_recv_n / llr_recv_amount - [lend-lease-observability] the pair
	# matrix on the RECIPIENT: parallel arrays (donor country-id / verified sends / units, all
	# archetypes incl. convoy row 10 summed), appended on the verified send only, same gate as
	# llr_sent_*. Assumes the shared temp pool across the ROOT hop (probe wa_test.301: 4242) and
	# the country-id array encoding WA_AI_lend_lease_targets already uses.
	set_temp_variable = { _llr_donor_id = THIS.id }
	ROOT = {
		if = { limit = { check_variable = { WA_TLM_llr_first_t = 0 } } set_variable = { WA_TLM_llr_first_t = global.WA_TLM_clock } }
		set_variable = { WA_TLM_llr_last_t = global.WA_TLM_clock }
		set_temp_variable = { _llr_pair_idx = -1 }
		# break hygiene: a leaked break = 1 from an earlier effect aborts a default-break loop
		set_temp_variable = { break = 0 }
		for_each_loop = { array = WA_TLM_llr_recv_donor
			if = { limit = { check_variable = { v = _llr_donor_id } }
				set_temp_variable = { _llr_pair_idx = i }
			}
		}
		if = { limit = { check_variable = { _llr_pair_idx < 0 } }
			add_to_array = { array = WA_TLM_llr_recv_donor value = _llr_donor_id }
			add_to_array = { array = WA_TLM_llr_recv_n value = 0 }
			add_to_array = { array = WA_TLM_llr_recv_amount value = 0 }
			set_temp_variable = { _llr_pair_idx = WA_TLM_llr_recv_donor^num }
			subtract_from_temp_variable = { _llr_pair_idx = 1 }
		}
		add_to_variable = { WA_TLM_llr_recv_n^_llr_pair_idx = 1 }
		add_to_variable = { WA_TLM_llr_recv_amount^_llr_pair_idx = _llr_amount }
	}
	if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: surplus relief archetype [?_llr_arch_idx] x [?_llr_amount] received from [This.GetName]" }
	}
}

# --- per-archetype legs. THIS = donor, ROOT = recipient. One shape, the archetype token and
# --- the constants row are the only differences. Add a row = copy one leg + one constants row
# --- + register the idx in WA_TLM_core (llr arrays) and the checklist probe.
# ---
# --- SCOPE, and it is unproven: each leg writes the recipient's stock inside `ROOT = { }` and
# --- reads it back HERE, in the donor's scope. That only works if a temp variable set on another
# --- scope stays readable for the rest of the effect execution; the engine documents nothing
# --- either way. Campaign 07270b64 says it does hold - llr_send_failed_n reached 5,924 on the
# --- USA, and the only path to relief_record that campaign had was the convoy leg computing a
# --- positive amount out of exactly these ROOT-written temps (the nine land legs never ran: every
# --- one of 94 recipients had llr_donor_selected_n = llr_path_refused_n). Boot probe
# --- wa_test.301 prints the answer as one number; if it prints 0, every `ROOT = { set_temp_variable }`
# --- below has to become `set_temp_variable = { x = ROOT.<dynamic variable> }`.

WA_AI_LEND_LEASE_relief_leg_infantry = {
	set_temp_variable = { _llr_arch_idx = 1 }
	set_temp_variable = { _llr_donor_stock = num_equipment@infantry_equipment }
	ROOT = { set_temp_variable = { _llr_recipient_stock = num_equipment@infantry_equipment } }
	set_temp_variable = { _llr_floor = constant:wa_ai_lend_lease.infantry.donor_floor }
	set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.infantry.starve }
	set_temp_variable = { _llr_base = constant:wa_ai_lend_lease.infantry.base }
	set_temp_variable = { _llr_tier2 = constant:wa_ai_lend_lease.infantry.tier2 }
	set_temp_variable = { _llr_tier4 = constant:wa_ai_lend_lease.infantry.tier4 }
	WA_AI_LEND_LEASE_relief_compute_amount = yes
	if = {
		limit = { check_variable = { _llr_amount > 0 } }
		meta_effect = {
			text = { send_equipment = { type = infantry_equipment amount = [AMT] target = [TGT] } }
			AMT = "[?_llr_amount]"
			TGT = "[ROOT.GetTag]"
		}
		ROOT = { set_temp_variable = { _llr_recipient_after = num_equipment@infantry_equipment } }
		WA_AI_LEND_LEASE_relief_record = yes
	}
}

# idx 10 - convoy_1. THIS = maritime donor, ROOT = recipient. Recipient need and weekly
# amount use the RECIPIENT's dockyard scale; the floor uses the DONOR's scale. Unlike the
# land legs there is no donor-stock multiplier and no starve+base overshoot.
WA_AI_LEND_LEASE_relief_leg_convoy = {
	set_temp_variable = { _llr_arch_idx = 10 }
	set_temp_variable = { _llr_donor_stock = num_equipment@convoy }
	ROOT = { set_temp_variable = { _llr_recipient_stock = num_equipment@convoy } }

	if = { limit = { WA_AI_LEND_LEASE_convoy_scale_small = yes }
		set_temp_variable = { _llr_floor = constant:wa_ai_lend_lease.convoy.donor_floor_small }
	}
	else_if = { limit = { WA_AI_LEND_LEASE_convoy_scale_medium = yes }
		set_temp_variable = { _llr_floor = constant:wa_ai_lend_lease.convoy.donor_floor_medium }
	}
	else_if = { limit = { WA_AI_LEND_LEASE_convoy_scale_large = yes }
		set_temp_variable = { _llr_floor = constant:wa_ai_lend_lease.convoy.donor_floor_large }
	}
	else = { set_temp_variable = { _llr_floor = constant:wa_ai_lend_lease.convoy.donor_floor_huge } }

	ROOT = {
		if = { limit = { WA_AI_LEND_LEASE_convoy_scale_small = yes }
			set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.convoy.reserve_small }
			set_temp_variable = { _llr_base = constant:wa_ai_lend_lease.convoy.base_small }
		}
		else_if = { limit = { WA_AI_LEND_LEASE_convoy_scale_medium = yes }
			set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.convoy.reserve_medium }
			set_temp_variable = { _llr_base = constant:wa_ai_lend_lease.convoy.base_medium }
		}
		else_if = { limit = { WA_AI_LEND_LEASE_convoy_scale_large = yes }
			set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.convoy.reserve_large }
			set_temp_variable = { _llr_base = constant:wa_ai_lend_lease.convoy.base_large }
		}
		else = {
			set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.convoy.reserve_huge }
			set_temp_variable = { _llr_base = constant:wa_ai_lend_lease.convoy.base_huge }
		}
		# Recipient target = max(dockyard-band reserve, major floor). Weekly cap stays
		# dockyard-scaled, so a low-dockyard major reaches 1000 over several bounded pulls.
		if = {
			limit = {
				is_major = yes
				check_variable = { _llr_starve < constant:wa_ai_lend_lease.convoy.reserve_major_floor }
			}
			set_temp_variable = { _llr_starve = constant:wa_ai_lend_lease.convoy.reserve_major_floor }
		}
	}

	set_temp_variable = { _llr_amount = 0 }
	if = {
		limit = {
			check_variable = { _llr_recipient_stock < _llr_starve }
			check_variable = { _llr_donor_stock > _llr_floor }
		}
		set_temp_variable = { _llr_amount = _llr_base }
		set_temp_variable = { _llr_headroom = _llr_donor_stock }
		subtract_from_temp_variable = { _llr_headroom = _llr_floor }
		if = { limit = { check_variable = { _llr_amount > _llr_headroom } }
			set_temp_variable = { _llr_amount = _llr_headroom }
		}
		set_temp_variable = { _llr_need = _llr_starve }
		subtract_from_temp_variable = { _llr_need = _llr_recipient_stock }
		if = { limit = { check_variable = { _llr_amount > _llr_need } }
			set_temp_variable = { _llr_amount = _llr_need }
		}
		if = { limit = { check_variable = { _llr_amount < 1 } }
			set_temp_variable = { _llr_amount = 0 }
		}
	}

	if = {
		limit = { check_variable = { _llr_amount > 0 } }
		# NOT send_equipment - convoys are the one cargo that effect does not move. Boot probe
		# wa_test.301 (2026-08-19) sent 50 as `convoy_1` and 50 as the `convoy` archetype in the
		# same execution in which an infantry send_equipment moved 12 000 rifles: the recipient's
		# stock stayed at 0 for both spellings. Peer evidence agrees - across 1 575 vanilla and
		# 580 Expert AI equipment-transfer blocks, ZERO send_equipment carries a convoy, while all
		# 123 convoy movements use the pair below (AST_diplomatic_events.7/8 -50/+50, MEX.txt,
		# POR.txt, NOR.txt, SEA_Philippines.txt).
		# The pair is create-and-destroy, not a transfer, so both halves must stay together.
		# `producer = PREV` stamps the donor as creator, which is what the campaign probe reads
		# out of the recipient's convoys={} block. No meta render is involved any more: the
		# engine's own `amount` accepts a variable (effects_documentation.md).
		set_temp_variable = { _llr_amount_neg = _llr_amount }
		multiply_temp_variable = { _llr_amount_neg = -1 }
		add_equipment_to_stockpile = { type = convoy amount = _llr_amount_neg }
		# BARE name, no scope prefix. The temp pool is shared for the whole effect execution
		# (probe: a temp set inside ROOT = { } reads back 4242 in the donor scope), and a
		# scope-prefixed temp read resolves to nothing: `amount = PREV._llr_amount` rendered as
		# "0 units ... removed" in the effect tooltip while `producer = PREV` resolved correctly.
		# Prefix scopes for SCOPE-OWNED things (producer, country variables), never for a temp.
		ROOT = {
			add_equipment_to_stockpile = { type = convoy amount = _llr_amount producer = PREV }
		}
		ROOT = { set_temp_variable = { _llr_recipient_after = num_equipment@convoy } }
		WA_AI_LEND_LEASE_relief_record = yes
	}
}
