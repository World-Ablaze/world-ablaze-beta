### manages lend lease diplomacy

WA_AI_update_lend_lease_targets = {

	# lend lease coordinated target by player
	if = {
		limit = {
			any_other_country = {
				WA_AI_PREV_on_same_side_with_THIS = yes
				has_country_flag = WA_AI_coordinated_lend_lease
				set_temp_variable = { coordinated_lend_lease_iniatiator = THIS.id }
			}
		}

		var:coordinated_lend_lease_iniatiator = {

			var:WA_AI_coordinated_lend_lease_target = {

				if = { limit = { NOT = { has_country_flag = WA_AI_lend_lease_from_@ROOT } }

					ROOT = {
						WA_AI_clear_lend_lease_targets = yes
						WA_AI_set_lend_lease_target = yes
					}

					if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: [This.GetName] (coordinated)" }
				}
			}
		}
	}

	# lend lease any valid targets
	else_if = {
		limit = {
			# game rules for lend lease
			any_other_country = {
				OR = {
					is_subject_of = ROOT

					has_war_together_with = ROOT

					AND = {
						has_war = yes

						ROOT = {
							OR = {
								AND = { has_government = neutrality NOT = { threat < 0.4 } }
								AND = { has_government = fascism }
								AND = { has_government = communism }
								AND = { has_government = democratic NOT = { threat < 0.5 } }
								AND = { has_government = democratic original_tag = USA has_completed_focus = USA_lend_lease_act }
							}
						}
					}
				}
			}
		}

		every_country = { limit = { WA_AI_lend_lease = yes }

			add_to_temp_array = { array = updated_lend_lease_targets value = THIS.id }
		}

		# clear invalid targets
		for_each_scope_loop = { array = WA_AI_lend_lease_targets

			if = { limit = { NOT = { is_in_array = { array = updated_lend_lease_targets value = THIS.id } } }

				add_to_array = { array = ROOT.WA_AI_remove_lend_lease value = THIS.id }
			}
		}
		for_each_scope_loop = { array = WA_AI_remove_lend_lease

			WA_AI_clear_lend_lease_target = yes
		}
		clear_array = WA_AI_remove_lend_lease

		# set lend lease targets
		for_each_scope_loop = { array = updated_lend_lease_targets

			if = { limit = { NOT = { is_in_array = { array = ROOT.WA_AI_lend_lease_targets value = THIS.id } } }

				ROOT = { WA_AI_set_lend_lease_target = yes }
			}
		}
	}
	else = {
		WA_AI_clear_lend_lease_targets = yes
	}
}

WA_AI_set_lend_lease_target = { # THIS = SENDER PREV = TARGET

	PREV = { set_country_flag = WA_AI_lend_lease_from_@PREV }
	add_to_array = { array = WA_AI_lend_lease_targets value = PREV.id }
	if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [This.GetName] | LEND-LEASE: [Prev.GetName]" }
}

WA_AI_clear_lend_lease_target = { # THIS = TARGET PREV = SENDER

	clr_country_flag = WA_AI_lend_lease_from_@PREV

	# compare tags because the id may not match after civil wars
	for_each_loop = { array = PREV.WA_AI_lend_lease_targets

		var:v = {
			if = { limit = { tag = PREV }

				remove_from_array = { array = PREV.PREV.WA_AI_lend_lease_targets index = i }
				set_temp_variable = { break = 1 }
			}
		}
	}
	set_temp_variable = { break = 0 }

	if = { limit = { PREV = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Prev.GetName] | CANCEL LEND-LEASE: [This.GetName]" }
}

WA_AI_clear_lend_lease_targets = { # THIS = SENDER

	if = { limit = { check_variable = { WA_AI_lend_lease_targets^num > 0 } }

		every_country = { limit = { has_country_flag = WA_AI_lend_lease_from_@PREV }

			WA_AI_clear_lend_lease_target = yes
		}

		clear_array = WA_AI_lend_lease_targets
	}
}
### Surplus relief - direct equipment transfer from a hoarding ally (weekly)
# ROOT = RECIPIENT (AI; the weekly call site is inside the is_ai gate).
# Pull, not push: the starving country selects ONE donor per week and the send runs in
# the donor's scope with target = ROOT - the repo's most-proven dynamic-target form
# (no meta_effect, no tag rendering, nothing that can silently fail to compile; the
# previous push version resolved a temp variable across a scope switch and used the
# unproven `equipment =` parameter, either of which can no-op silently).
# random_other_country (not every_) caps intake at one donor per recipient per week,
# and per-transfer re-reads of the donor's stockpile stop a donor the moment it drops
# below its own floor, even mid-tick across several recipients.
# Policy lives in WA_AI_LEND_LEASE_should_receive_relief_{infantry,support}
# (common/scripted_triggers/WA_AI_LEND_LEASE_triggers.txt), per-archetype so a sender
# with only rifles to spare ships only rifles. Amounts are tiered by donor surplus -
# a 300k-rifle hoarder dripping 6000/week while an ally collapses is not relief.
WA_AI_LEND_LEASE_request_surplus_relief = {
	if = {
		limit = {
			OR = {
				WA_AI_LEND_LEASE_is_starving_infantry = yes
				WA_AI_LEND_LEASE_is_starving_support = yes
			}
		}

		random_other_country = {
			limit = {
				OR = {
					PREV = { WA_AI_LEND_LEASE_should_receive_relief_infantry = yes }
					PREV = { WA_AI_LEND_LEASE_should_receive_relief_support = yes }
				}
			}

			# --- infantry leg (THIS = donor, PREV/ROOT = recipient) ---
			if = {
				limit = { PREV = { WA_AI_LEND_LEASE_should_receive_relief_infantry = yes } }

				if = {
					limit = { has_equipment = { infantry_equipment > 299999 } }
					send_equipment = { type = infantry_equipment amount = 24000 target = ROOT }
				}
				else_if = {
					limit = { has_equipment = { infantry_equipment > 149999 } }
					send_equipment = { type = infantry_equipment amount = 12000 target = ROOT }
				}
				else = {
					send_equipment = { type = infantry_equipment amount = 6000 target = ROOT }
				}
			}

			# --- support leg ---
			if = {
				limit = { PREV = { WA_AI_LEND_LEASE_should_receive_relief_support = yes } }

				if = {
					limit = { has_equipment = { support_equipment > 59999 } }
					send_equipment = { type = support_equipment amount = 4800 target = ROOT }
				}
				else_if = {
					limit = { has_equipment = { support_equipment > 29999 } }
					send_equipment = { type = support_equipment amount = 2400 target = ROOT }
				}
				else = {
					send_equipment = { type = support_equipment amount = 1200 target = ROOT }
				}
			}

			if = { limit = { ROOT = { has_country_flag = WA_AI_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | LEND-LEASE: surplus relief received from [This.GetName]" }
		}
	}
}
