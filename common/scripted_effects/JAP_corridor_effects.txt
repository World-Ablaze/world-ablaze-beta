# Resources carried home overland: three groups, each opened by its own focus.
#   south     Carry the Southern Resources Overland (JAP_island_garrisons): Dutch East Indies and Malaya
#   far_east  Carry the Far Eastern Resources Overland (JAP_far_east_resources_overland): the Soviet Far East
#   india     Carry the Indian and Burmese Resources Overland (JAP_india_resources_overland): Burma and India up to Calcutta
# Run weekly (common/on_actions/JAP_corridor_on_actions.txt) and once when each focus completes.
# While a group's route is open (its trigger in common/scripted_triggers/JAP_corridor_triggers.txt), each of its states
# held by Japan or a subject is shipped: it gets JAP_corridor_shipped (state_resources_factor = -1), and what it was
# producing just before (resource@) is added to Kanto (282) through that group's received modifier. When the route
# closes or the state is lost, the state is released and Kanto loses its share. If the route opens again, it is shipped
# again. A state belongs to one group only.
JAP_corridor_update = {

# State scope. JAP_corridor_open_t is set by JAP_corridor_update for the group being updated. So is JAP_corridor_rate_t.
JAP_corridor_state_update = {
	if = {
		limit = {
			has_state_flag = JAP_corridor_shipped_flag
			OR = {
				check_variable = { JAP_corridor_open_t = 0 }
				NOT = { CONTROLLER = { JAP_corridor_our_side = yes } }
			}
		}
		remove_dynamic_modifier = { modifier = JAP_corridor_shipped }
		clr_state_flag = JAP_corridor_shipped_flag
	}
	else_if = {
		limit = {
			NOT = { has_state_flag = JAP_corridor_shipped_flag }
			check_variable = { JAP_corridor_open_t = 1 }
			CONTROLLER = { JAP_corridor_our_side = yes }
		}
		set_variable = { JAP_corridor_oil = resource@oil }
		set_variable = { JAP_corridor_aluminium = resource@aluminium }
		set_variable = { JAP_corridor_rubber = resource@rubber }
		set_variable = { JAP_corridor_tungsten = resource@tungsten }
		set_variable = { JAP_corridor_steel = resource@steel }
		set_variable = { JAP_corridor_chromium = resource@chromium }
		set_variable = { JAP_corridor_coal = resource@coal }
		set_variable = { JAP_corridor_bauxite = resource@bauxite }
		set_variable = { JAP_corridor_iron = resource@iron }
		add_dynamic_modifier = { modifier = JAP_corridor_shipped }
		set_state_flag = JAP_corridor_shipped_flag
	}
	# what a shipped state loses follows the railway (JAP_corridor_shipped: state_resources_factor = JAP_corridor_share)
	if = {
		limit = { has_state_flag = JAP_corridor_shipped_flag }
		set_variable = { JAP_corridor_share = JAP_corridor_rate_t }
		multiply_variable = { JAP_corridor_share = -1 }
		force_update_dynamic_modifier = yes
	}
}

# the route has (has_railway_level looks at one state at a time), all of it at level 5.

	set_temp_variable = { JAP_corridor_rate_t = 0 }

		limit = {
			has_railway_level = { level = 1 state = 608 }
			has_railway_level = { level = 1 state = 607 }

			has_railway_level = { level = 1 state = 671 }
		}

	}
	if = {
		limit = {
			has_railway_level = { level = 3 state = 608 }
			has_railway_level = { level = 3 state = 607 }
			has_railway_level = { level = 3 state = 620 }
			has_railway_level = { level = 3 state = 602 }

			has_railway_level = { level = 3 state = 594 }
			has_railway_level = { level = 3 state = 671 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.6 }
	}
	if = {
		limit = {
			has_railway_level = { level = 4 state = 608 }
			has_railway_level = { level = 4 state = 607 }
			has_railway_level = { level = 4 state = 620 }
			has_railway_level = { level = 4 state = 602 }

			has_railway_level = { level = 4 state = 671 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.8 }
	}
	if = {
		limit = {
			has_railway_level = { level = 5 state = 608 }
			has_railway_level = { level = 5 state = 607 }
			has_railway_level = { level = 5 state = 620 }

		limit = { check_variable = { JAP_corridor_rate_t = 0 } }
		set_temp_variable = { JAP_corridor_open_t = 0 }
	}

# Country scope. JAP_corridor_rate_t = the share carried over this route: 0.2 for every railway level that every state of
# the route has (has_railway_level looks at one state at a time), all of it at level 5.
JAP_corridor_rate_far_east = {
	set_temp_variable = { JAP_corridor_rate_t = 0 }
	if = {
		limit = {
			has_railway_level = { level = 1 state = 563 }
			has_railway_level = { level = 1 state = 564 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.2 }
	}
	if = {
		limit = {
			has_railway_level = { level = 2 state = 563 }
			has_railway_level = { level = 2 state = 564 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.4 }
	}
	if = {
		limit = {
			has_railway_level = { level = 3 state = 563 }
			has_railway_level = { level = 3 state = 564 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.6 }
	}
	if = {
		limit = {
			has_railway_level = { level = 4 state = 563 }
			has_railway_level = { level = 4 state = 564 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 0.8 }
	}
	if = {
		limit = {
			has_railway_level = { level = 5 state = 563 }
			has_railway_level = { level = 5 state = 564 }
		}
		set_temp_variable = { JAP_corridor_rate_t = 1 }
	}
	if = {
		limit = { JAP = { has_country_flag = JAP_debug_corridor_open } }
		set_temp_variable = { JAP_corridor_rate_t = 1 }
	}
	# no railway, nothing carried
	if = {
		limit = { check_variable = { JAP_corridor_rate_t = 0 } }
		set_temp_variable = { JAP_corridor_open_t = 0 }
	}
}
