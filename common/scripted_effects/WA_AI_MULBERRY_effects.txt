############################################################################################################
#	[dday-mulberry] Mulberry placement/dismantle bodies. They live in scripted_effects - NOT in
#	the event bodies - because meta_effect is only proven in this folder; fired inline from an
#	events/ file the event executed its prefix and silently dropped the rest (harness control
#	run 2026-08-31: 101_flag set, no-target log never printed). Branches use complementary ifs,
#	never if/else, for the same reason. Scope: THIS = ROOT = the country the event fired on.
############################################################################################################

# Mulberry A body, called by WA_AI_invasions.101. Reads global.WA_AI_mulberry_a_* targets,
# writes the *_placed_* latches, schedules the dismantle and its own retry.
WA_AI_MULBERRY_run_a = {
	if = {
		limit = { NOT = { has_country_flag = WA_AI_invasions.101_flag } }
		set_country_flag = WA_AI_invasions.101_flag
	}

	set_temp_variable = { _mulb_state_ = global.WA_AI_mulberry_a_state }
	set_temp_variable = { _mulb_prov_ = global.WA_AI_mulberry_a_prov }
	set_temp_variable = { _mulb_ok_ = 0 }
	if = {
		limit = { check_variable = { _mulb_prov_ < 1 } }
		log = "[GetDateText] [Root.GetName]: WA_AI_invasions.101 fired with no Mulberry A target recorded - nothing placed"
	}
	if = {
		limit = { check_variable = { _mulb_prov_ > 0 } }
		meta_effect = {
			text = {
				if = {
					limit = { [MULB_STATE] = { is_controlled_by_ROOT_or_ally = yes } }
					set_temp_variable = { _mulb_ok_ = 1 }
				}
			}
			MULB_STATE = "[?_mulb_state_]"
		}
	}

	if = {
		limit = { check_variable = { _mulb_ok_ = 1 } }
		meta_effect = {
			text = {
				[MULB_STATE] = {
					set_building_level = {
						type = naval_base
						province = { id = [MULB_PROV] }
						level = 5
						instant_build = yes
					}
				}
			}
			MULB_STATE = "[?_mulb_state_]"
			MULB_PROV = "[?_mulb_prov_]"
		}
		set_variable = { global.WA_AI_mulberry_a_placed_state = _mulb_state_ }
		set_variable = { global.WA_AI_mulberry_a_placed_prov = _mulb_prov_ }
		country_event = { id = WA_AI_invasions.100 days = 120 }
		log = "[GetDateText] [Root.GetName]: WA_AI_invasions.101 placed Mulberry A (naval_base 5) at state [?_mulb_state_|.0] prov [?_mulb_prov_|.0]"
	}
	else_if = {
		limit = {
			check_variable = { _mulb_prov_ > 0 }
			NOT = { has_country_flag = { flag = WA_AI_invasions.101_flag days > 45 } }
		}
		country_event = { id = WA_AI_invasions.101 days = 3 }
	}
	else_if = {
		limit = { check_variable = { _mulb_prov_ > 0 } }
		# Beachhead failed: retire the un-placed target so nothing later reads it.
		set_variable = { global.WA_AI_mulberry_a_state = 0 }
		set_variable = { global.WA_AI_mulberry_a_prov = 0 }
		log = "[GetDateText] [Root.GetName]: WA_AI_invasions.101 gave up (state never taken) - Mulberry A target retired"
	}
}

# Dismantle body, called by WA_AI_invasions.100. Reads ONLY the *_placed_* latches (never the
# targets), clears both from whichever scheduler is still alive; second firing is a no-op.

		limit = { check_variable = { global.WA_AI_mulberry_a_placed_prov > 0 } }
		set_temp_variable = { _mulb_state_ = global.WA_AI_mulberry_a_placed_state }
		set_temp_variable = { _mulb_prov_ = global.WA_AI_mulberry_a_placed_prov }
		meta_effect = {
			text = {
				[MULB_STATE] = {

						type = naval_base

						level = 0

					}
				}
			}
			MULB_STATE = "[?_mulb_state_]"
			MULB_PROV = "[?_mulb_prov_]"
		}
		set_variable = { global.WA_AI_mulberry_a_placed_state = 0 }
		set_variable = { global.WA_AI_mulberry_a_placed_prov = 0 }

		limit = { check_variable = { global.WA_AI_mulberry_b_placed_prov > 0 } }
		set_temp_variable = { _mulb_state_ = global.WA_AI_mulberry_b_placed_state }
