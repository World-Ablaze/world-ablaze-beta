### Dynamic Queue System - Get new project ID
# Finds an unused project ID or creates a new one
# Output: new_project_ (temp variable)
WA_AI_PC_get_new_project_ID = {
	# Find first empty slot in target_state array (value = 0)
	set_temp_variable = { new_project_ = -1 }
	# Temp vars persist across calls within one execution chain: without this reset, the second and
	# later calls (e.g. multi-segment railway queueing) inherit _found_id = 1 from the previous call,
	# break after element 0, and append a new slot instead of reusing a free one.
	set_temp_variable = { _found_id = 0 }

	for_each_loop = { array = WA_AI_PC_target_state index = _idx value = _val break = _found_id
		if = { limit = { check_variable = { _val = 0 } }
			set_temp_variable = { new_project_ = _idx }
			set_temp_variable = { _found_id = 1 }
		}
	}

	# If no empty slot found, create new one at end
	if = { limit = { check_variable = { new_project_ = -1 } }
		set_temp_variable = { new_project_ = WA_AI_PC_target_state^num }
		add_to_array = { WA_AI_PC_target_state = 0 }
	}
}

### Dynamic Queue System - Get construction speed for a state
# THIS = state, _project_building_type >>> construction_speed_
WA_AI_PC_get_build_speed = {
	set_temp_variable = { construction_speed_ = 1 }

	# Add building type specific modifiers
	if = { limit = { check_variable = { _project_building_type = 1 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_infrastructure_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 2 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_air_base_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 3 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_anti_air_building_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 4 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_radar_station_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 5 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_arms_factory_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 6 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_industrial_complex_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 7 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_dockyard_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 8 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_synthetic_refinery_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 11 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_hydro_steel_refinery_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 12 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_hydro_aluminium_refinery_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 15 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_steel_refinery_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 16 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_aluminium_refinery_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 13 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_rail_way_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 14 } }
		add_to_temp_variable = { construction_speed_ = modifier@production_speed_naval_base_factor }
	}

	# Add general construction speed modifier
	add_to_temp_variable = { construction_speed_ = modifier@production_speed_buildings_factor }

	# Infrastructure effect for factory-type buildings (types 5-12, 15-16)
	if = { limit = { OR = { AND = { check_variable = { _project_building_type > 4 } check_variable = { _project_building_type < 13 } } check_variable = { _project_building_type = 15 } check_variable = { _project_building_type = 16 } } }
		set_temp_variable = { _infra_bonus = infrastructure_level }
		multiply_temp_variable = { _infra_bonus = 0.1 }
		add_to_temp_variable = { _infra_bonus = 1 }
		multiply_temp_variable = { construction_speed_ = _infra_bonus }
	}

	# Base output per factory per day
	multiply_temp_variable = { construction_speed_ = 2.5 }
}

### Dynamic Queue System - Get building cost
# THIS = state, _project_building_type, _project_province_id, _project_connect_id >>> building_cost_
WA_AI_PC_get_building_cost = {
	set_temp_variable = { building_cost_ = 0 }

	if = { limit = { check_variable = { _project_building_type = 1 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_INFRASTRUCTURE_COST }
		# Level price effect
		set_temp_variable = { _level_cost = infrastructure_level }
		multiply_temp_variable = { _level_cost = 100 }
		add_to_temp_variable = { building_cost_ = _level_cost }
	}
	else_if = { limit = { check_variable = { _project_building_type = 2 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_AIRBASE_COST }
		set_temp_variable = { _level_cost = building_level@air_base }
		# Fix 72: was 250 - air_base.per_level_extra_cost is 300 (00_buildings.txt:86).
		multiply_temp_variable = { _level_cost = 300 }
		add_to_temp_variable = { building_cost_ = _level_cost }
	}
	else_if = { limit = { check_variable = { _project_building_type = 3 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_ANTIAIR_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 4 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_RADAR_COST }
		set_temp_variable = { _level_cost = building_level@radar_station }
		# Fix 72: was 500 - radar_station.per_level_extra_cost is 2500 (00_buildings.txt:117).
		# Largest of the three drifts: a level-5 station was priced 4500 against a real 14500.
		multiply_temp_variable = { _level_cost = 2500 }
		add_to_temp_variable = { building_cost_ = _level_cost }
	}
	else_if = { limit = { check_variable = { _project_building_type = 5 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_MILITARY_FACTORY_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 6 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_CIVILIAN_FACTORY_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 7 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_DOCKYARD_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 8 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_REFINERY_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 9 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_CONVERSION_COST_MIL_TO_CIV }
		set_temp_variable = { _conv_factor = modifier@conversion_cost_mil_to_civ_factor }
		add_to_temp_variable = { _conv_factor = 1.0 }
		multiply_temp_variable = { building_cost_ = _conv_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 10 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_CONVERSION_COST_CIV_TO_MIL }
		set_temp_variable = { _conv_factor = modifier@conversion_cost_civ_to_mil_factor }
		add_to_temp_variable = { _conv_factor = 1.0 }
		multiply_temp_variable = { building_cost_ = _conv_factor }
	}
	else_if = { limit = { check_variable = { _project_building_type = 11 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_HYDROPOWEREDSTEELMILL_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 12 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_HYDROPOWEREDALUMINIUMSMELTER_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 15 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_STEELMILL_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 16 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_ALUMINIUMSMELTER_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 13 } }
		# Fix 73: was a hardcoded `170 + 130 * (1 + existing connection level)`, plus a
		# meta_effect to read that level. Those are VANILLA's rail_way numbers
		# (base_cost = 170, per_level_extra_cost = 130) - the block was a faithful copy of the
		# base game and was never wrong when written. THIS MOD re-priced rail_way to a flat
		# base_cost = 800, per_level_extra_cost = 0 (1000 -> 800 in 42b53eccb, Oct 2023) and
		# the AI's shadow table was never updated, so the AI kept buying rail at the base
		# game's escalating price: 300 for a fresh segment against a real 800.
		#
		# One type-13 project spawns exactly one `build_railway level = 1 path = { from to }`
		# (WA_AI_PC_add_finished_building_by_id), i.e. one level on one province connection,
		# so the charge is one unit of rail_way.base_cost. NO level term: per_level_extra_cost
		# is 0 in this mod. If it is ever made non-zero again, reinstate a level term here -
		# and use `base + per_level * existing` like the air-base and radar branches above, not
		# the `(1 + existing)` form removed here, which over-counted by one level.
		#
		# This is now the same 800 that railway_helpers.txt:25 (@WA_AI_PC_RAILWAY_BASE_COST,
		# "flat per segment") and WA_AI_PC_estimate_railway_cost already use to SCORE routes,
		# so route selection and route payment finally agree.
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_RAILWAY_COST }
	}
	else_if = { limit = { check_variable = { _project_building_type = 14 } }
		set_temp_variable = { building_cost_ = global.WA_AI_PC_BUILDING_NAVAL_BASE_COST }
	}
}

### Dynamic Queue System - Assign factories to projects
# THIS = country
# Allocates available factories to projects from the top of the queue
# Persistent outputs (resynced on every pass): WA_AI_PC_assigned_factories_total,
# WA_AI_PC_active_projects, WA_AI_PC_active_nonrail_projects, and
# WA_AI_PC_air_factories_assigned (Fix 38: civs actually assigned to air-base
# [building_type 2] slots this pass; read CROSS-COUNTRY by the allied air-base
# builder gates in WA_AI_CONSTRUCTION_PRIORITY_strategies.txt).
WA_AI_PC_assign_factories = {
	# Fix 41: resumed-save migration - idempotent clamp of legacy pre-Fix-41 priorities.
	# A save queued under the old 9999/5000 scale carries rails at 9999 (or 10998.9 with the
	# x1.1 high-route multiplier) that would permanently outrank every new 1000-band rail.
	# Anything above @AI_PC_PRIO_LEGACY_MAX (1100, the highest value this build can write)
	# is by definition legacy and is clamped to the rail-war band. One O(queue) loop, no-op
	# on every pass after the first, and on fresh campaigns.
	for_each_loop = { array = WA_AI_PC_queue value = _migr_proj_id
		if = { limit = { check_variable = { WA_AI_PC_priority^_migr_proj_id > @AI_PC_PRIO_LEGACY_MAX } }
			set_variable = { WA_AI_PC_priority^_migr_proj_id = @AI_PC_PRIO_RAIL_WAR }
		}
	}

	# Calculate available factories BEFORE resetting (add back currently assigned to get true available)
	set_temp_variable = { available_factories_ = num_of_civilian_factories_available_for_projects }
	add_to_temp_variable = { available_factories_ = WA_AI_PC_assigned_factories_total }

	# Fix: break the self-referential decay loop. Once the vanilla construction queue absorbs all
	# civs, num_of_civilian_factories_available_for_projects reads 0 and the recurrence becomes
	# T(n+1) = 0.35 * T(n) - pure geometric decay to the clamp floor of 1 (observed: GER 100 -> 9
	# -> 1 over 1943-45, stranding the whole queue). Floor the base at a fraction of TOTAL civilian
	# factories - a number this system's own output cannot influence. Gated so a genuinely
	# collapsing economy degrades to the old behaviour instead of reserving factories it can't spare.
	if = {
		limit = {
			check_variable = { WA_AI_PC_queue^num > 0 }
			num_of_civilian_factories > 19
			has_capitulated = no
			surrender_progress < 0.3
		}
		set_temp_variable = { _pc_stable_base = num_of_civilian_factories }
		multiply_temp_variable = { _pc_stable_base = @AI_PC_STABLE_BASE_FRACTION }
		if = { limit = { check_variable = { available_factories_ < _pc_stable_base } }
			set_temp_variable = { available_factories_ = _pc_stable_base }
		}
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: queue=[?WA_AI_PC_queue^num] raw_avail=[?available_factories_]"
	}

	# Check for override factor (e.g., during high-priority railway construction)
	if = { limit = { has_country_flag = WA_AI_PC_override_max_factories_factor }
		multiply_temp_variable = { available_factories_ = WA_AI_PC_override_max_factories_factor }
	}
	else = {
		multiply_temp_variable = { available_factories_ = @AI_PC_ALLOC_FRACTION }
	}

	# Hard ceiling: the floor above must never drive civilian_factory_use past half the civ pool
	# (civilian_factory_use competes with consumer goods and trade).
	set_temp_variable = { _pc_alloc_cap = num_of_civilian_factories }
	multiply_temp_variable = { _pc_alloc_cap = @AI_PC_ALLOC_HARD_CAP_FRACTION }
	if = { limit = { check_variable = { available_factories_ > _pc_alloc_cap } }
		set_temp_variable = { available_factories_ = _pc_alloc_cap }
	}


	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: after_alloc_fraction=[?available_factories_]"
	}

	# Reset all project factory assignments (use temp for new total)
	for_each_loop = { array = WA_AI_PC_queue value = _project_id
		set_variable = { WA_AI_PC_assigned_factories^_project_id = 0 }
	}
	set_temp_variable = { _new_total = 0 }
	# Fix 38: persistent per-country output - civilian factories assigned to air-base
	# (building_type 2) projects this pass. Read cross-country ("is the host actually
	# funding air bases?") by WA_AI_build_uk_air_hosting_capacity and
	# WA_AI_build_theatre_air_bases, so it must be a real variable, not a temp. An
	# unset read is 0 = "not funding", which fails open toward the rich ally stepping
	# in. Rewritten on every pass for every AI country reaching this effect (weekly
	# on_action, is_ai = yes + has_capitulated = no; also the mid-week completion and
	# railway recursion paths).
	set_variable = { WA_AI_PC_air_factories_assigned = 0 }

	# Sort queue by priority (descending - highest priority first)
	# Step 1: Build temp arrays for sorting
	clear_temp_array = _sort_ids_
	clear_temp_array = _sort_prios_
	for_each_loop = { array = WA_AI_PC_queue value = _project_id
		add_to_temp_array = { _sort_ids_ = _project_id }
		add_to_temp_array = { _sort_prios_ = WA_AI_PC_priority^_project_id }
	}

	# Step 2: Bubble sort descending (higher priority first)
	set_temp_variable = { _sort_n = _sort_ids_^num }
	set_temp_variable = { _sort_i = 0 }
	while_loop_effect = { limit = { check_variable = { _sort_i < _sort_n } }
		set_temp_variable = { _sort_j = _sort_i }
		add_to_temp_variable = { _sort_j = 1 }
		while_loop_effect = { limit = { check_variable = { _sort_j < _sort_n } }
			# Descending: swap if j > i (higher values should come first)
			if = { limit = { check_variable = { _sort_prios_^_sort_j > _sort_prios_^_sort_i } }
				# Swap priorities
				set_temp_variable = { _tmp_prio = _sort_prios_^_sort_i }
				set_temp_variable = { _sort_prios_^_sort_i = _sort_prios_^_sort_j }
				set_temp_variable = { _sort_prios_^_sort_j = _tmp_prio }
				# Swap IDs
				set_temp_variable = { _tmp_id = _sort_ids_^_sort_i }
				set_temp_variable = { _sort_ids_^_sort_i = _sort_ids_^_sort_j }
				set_temp_variable = { _sort_ids_^_sort_j = _tmp_id }
			}
			add_to_temp_variable = { _sort_j = 1 }
		}
		add_to_temp_variable = { _sort_i = 1 }
	}

	# Step 3: Rebuild queue in sorted order
	clear_array = WA_AI_PC_queue
	for_each_loop = { array = _sort_ids_ value = _sorted_id
		add_to_array = { WA_AI_PC_queue = _sorted_id }
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: queue sorted by priority"
		# Log first 3 projects after sort to verify order
		if = { limit = { check_variable = { WA_AI_PC_queue^num > 0 } }
			set_temp_variable = { _dbg_p0 = WA_AI_PC_queue^0 }
			set_temp_variable = { _dbg_prio0 = WA_AI_PC_priority^_dbg_p0 }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: [0] proj=[?_dbg_p0] prio=[?_dbg_prio0]"
		}
		if = { limit = { check_variable = { WA_AI_PC_queue^num > 1 } }
			set_temp_variable = { _dbg_p1 = WA_AI_PC_queue^1 }
			set_temp_variable = { _dbg_prio1 = WA_AI_PC_priority^_dbg_p1 }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: [1] proj=[?_dbg_p1] prio=[?_dbg_prio1]"
		}
		if = { limit = { check_variable = { WA_AI_PC_queue^num > 2 } }
			set_temp_variable = { _dbg_p2 = WA_AI_PC_queue^2 }
			set_temp_variable = { _dbg_prio2 = WA_AI_PC_priority^_dbg_p2 }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: [2] proj=[?_dbg_p2] prio=[?_dbg_prio2]"
		}
	}

	# Fix 41: anti-starvation OVERTAKE LANE - runs BEFORE the winner-takes-most fill.
	# The single most-starved project (highest WA_AI_PC_stall_weeks, and at least
	# @AI_PC_AGING_LANE_WEEKS of them) is revalidated (WA_AI_PC_project_still_relevant,
	# WA_AI_CONSTRUCTION_triggers.txt); if still relevant it is served min(20, pool)
	# factories ahead of the sorted fill. If revalidation FAILS the project is cancelled
	# on the spot (same machinery as the Fix 34 stall sweep, so all queue bookkeeping
	# stays coherent) and the next-most-starved candidate is tried, at most
	# @AI_PC_AGING_LANE_MAX_CANDIDATES per pass.
	#
	# Invariants:
	#   - at most ONE project is overtaking at any time, and it diverts at most 20 factories;
	#   - a served project gets assigned_factories > 0, so the weekly stall maintenance in
	#     WA_AI_PC_update_project_progress resets its stall_weeks to 0 and it LEAVES the lane
	#     (the next-most-starved project becomes eligible on a later week);
	#   - empty pool => the selection block is skipped => empty lane => the Fix 34 stall
	#     sweep keeps its week-25 backstop role unchanged;
	#   - candidates must pass the same target-valid / progress / friendly-controller tests
	#     as the fill: a project frozen on hostile-controlled ground (Fix 34) is NOT lane
	#     material - it is deliberately left to recapture-or-sweep, not cancelled early.
	#
	# Re-entry policy: candidate SELECTION (and its revalidation cost) runs only on the first
	# assign_factories call of the weekly effect chain - the main weekly pass - latched by the
	# chain-persistent temp _pc_lane_selected_ (temp variables persist across nested scripted-
	# effect calls within one execution chain; see WA_AI_PC_get_new_project_ID). The GRANT is
	# re-applied on every re-entry (cleanup / mid-week completion / stall sweep re-call
	# assign_factories, which resets all assignments), otherwise the first re-entry would
	# silently strip the laned project back to 0 factories and void the week.
	if = { limit = { NOT = { check_variable = { _pc_lane_selected_ = 1 } } }
		set_temp_variable = { _pc_lane_selected_ = 1 }
		set_temp_variable = { _pc_lane_proj_ = -1 }
		if = {
			limit = {
				check_variable = { available_factories_ > 0 }
				check_variable = { WA_AI_PC_queue^num > 0 }
			}
			set_temp_variable = { _lane_tries_ = 0 }
			set_temp_variable = { _lane_done_ = 0 }
			while_loop_effect = {
				limit = {
					check_variable = { _lane_done_ = 0 }
					check_variable = { _lane_tries_ < @AI_PC_AGING_LANE_MAX_CANDIDATES }
				}
				add_to_temp_variable = { _lane_tries_ = 1 }

				# most-starved eligible candidate (ties: first found). Cancelled candidates
				# left the queue, so each rescan naturally moves to the next one.
				set_temp_variable = { _lane_best_ = -1 }
				set_temp_variable = { _lane_best_weeks_ = 0 }
				for_each_loop = { array = WA_AI_PC_queue value = _lane_scan_
					if = {
						limit = {
							check_variable = { WA_AI_PC_stall_weeks^_lane_scan_ > _lane_best_weeks_ }
							NOT = { check_variable = { WA_AI_PC_stall_weeks^_lane_scan_ < @AI_PC_AGING_LANE_WEEKS } }
							NOT = { check_variable = { WA_AI_PC_target_state^_lane_scan_ = 0 } }
							check_variable = { WA_AI_PC_progress^_lane_scan_ > 0 }
							# same Fix 34 controller test as the fill below - hostile-frozen
							# projects wait for recapture or the sweep, never the lane
							var:WA_AI_PC_target_state^_lane_scan_ = {
								CONTROLLER = {
									OR = {
										tag = ROOT
										is_in_faction_with = ROOT
										is_subject_of = ROOT
										ROOT = { is_subject_of = PREV }
									}
								}
							}
						}
						set_temp_variable = { _lane_best_ = _lane_scan_ }
						set_temp_variable = { _lane_best_weeks_ = WA_AI_PC_stall_weeks^_lane_scan_ }
					}
				}

				if = { limit = { check_variable = { _lane_best_ < 0 } }
					set_temp_variable = { _lane_done_ = 1 }
				}
				else = {
					set_temp_variable = { _reval_project_id = _lane_best_ }
					if = { limit = { WA_AI_PC_project_still_relevant = yes }
						set_temp_variable = { _pc_lane_proj_ = _lane_best_ }
						set_temp_variable = { _lane_done_ = 1 }
						if = { limit = { has_country_flag = WA_AI_construction_logging }
							log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC LANE: proj=[?_lane_best_] starved [?_lane_best_weeks_] weeks - revalidated, overtaking"
						}
					}
					else = {
						# revalidation failed: cancel NOW instead of letting the week-25 sweep
						# find it - the project is geographically/strategically obsolete
						if = { limit = { has_country_flag = WA_AI_construction_logging }
							log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC LANE: proj=[?_lane_best_] starved [?_lane_best_weeks_] weeks - revalidation FAILED, cancelling"
						}
						clear_temp_array = _cancel_projects_IDS
						add_to_temp_array = { _cancel_projects_IDS = _lane_best_ }
						WA_AI_PC_cancel_projects = yes
						# tlm: pc_aging_reval_cancels - counter, per revalidation-cancel (the
						# cancel above is the verified effect) - consumer: R26
						add_to_variable = { WA_TLM_pc_aging_reval_cancels = 1 }
					}
				}
			}
		}
	}

	# Fix 41: lane GRANT - every pass, so re-entries do not strip the laned project.
	# Re-checks the project is still live (it may have completed mid-week, which is
	# exactly what triggers a re-entry). Control cannot flip inside one effect chain,
	# so the selection-time controller test is not repeated here.
	if = {
		limit = {
			NOT = { check_variable = { _pc_lane_proj_ < 0 } }
			check_variable = { available_factories_ > 0 }
			is_in_array = { WA_AI_PC_queue = _pc_lane_proj_ }
			NOT = { check_variable = { WA_AI_PC_target_state^_pc_lane_proj_ = 0 } }
			check_variable = { WA_AI_PC_progress^_pc_lane_proj_ > 0 }
		}
		set_temp_variable = { assign_amount = available_factories_ }
		clamp_temp_variable = { var = assign_amount min = 1 max = 20 }
		round_temp_variable = assign_amount
		set_variable = { WA_AI_PC_assigned_factories^_pc_lane_proj_ = assign_amount }
		add_to_temp_variable = { _new_total = assign_amount }
		subtract_from_temp_variable = { available_factories_ = assign_amount }
		# Fix 38 accumulator: the laned project counts toward the air share like any other
		if = { limit = { check_variable = { WA_AI_PC_building_type^_pc_lane_proj_ = 2 } }
			add_to_variable = { WA_AI_PC_air_factories_assigned = assign_amount }
		}
		# tlm: pc_aging_grants - counter, ONLY when the lane actually assigned >= 1 factory
		# (the set_variable above is the verified effect); latched so the same weekly grant
		# is not recounted on re-entry re-grants - consumer: R26
		if = { limit = { NOT = { check_variable = { _pc_lane_counted_ = 1 } } }
			set_temp_variable = { _pc_lane_counted_ = 1 }
			add_to_variable = { WA_TLM_pc_aging_grants = 1 }
		}
		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC LANE: proj=[?_pc_lane_proj_] granted [?assign_amount] factories ahead of the fill"
		}
	}

	# Fix 78: AIR-BASING RESERVATION LANE - one slot for the air-basing band, ahead of the
	# rail fill, only while WA_AI_PC_air_basing_reserve is set (30-day flag, refreshed by
	# WA_AI_build_uk_air_hosting_capacity's deficit branch and by nothing else).
	#
	# The starvation this exists for, measured on campaign be18f9c7: ENG's band-300 air-base
	# projects held 0 factories in 20 of 30 sampled months of 1942.1-1944.6 (and its
	# dbg_started counter froze at 17 for nine straight months), because the sorted fill
	# spends the whole pool on band-1000 rails and the Fix 41 lane serves ONE project a week
	# across the entire queue - a starving air project waits ~12+ weeks for its lane turn.
	# The single month ENG's rail queue was short (1943.5, one rail queued), its air projects
	# got 53 factories - the pool was never the constraint, the ordering was. Meanwhile the
	# 3-slot UK budget stayed occupied by the unfunded projects, so no new sites could be
	# queued either: funding starvation and ordering throughput are the same defect.
	#
	# Why not raise the air band above rail instead: the Fix 41 band table deliberately keeps
	# rail-war above air-basing (the livelock that table fixed must stay fixed). A reservation
	# of at most ONE 20-civ slot leaves the band order intact for everything else, mirrors the
	# proven Fix 41 lane machinery (chain-latched selection, per-pass re-grant), and
	# self-releases within 30 days of the deficit closing.
	#
	# Selection: first type-2 project above the strategic band (priority > 250) in the
	# already-sorted queue = the highest-priority air-basing project. Same liveness and
	# Fix 34 controller tests as the fill; skips the Fix 41 lane's pick (already served).
	if = { limit = { NOT = { check_variable = { _pc_air_lane_selected_ = 1 } } }
		set_temp_variable = { _pc_air_lane_selected_ = 1 }
		set_temp_variable = { _pc_air_lane_proj_ = -1 }
		# Fix 87 / P1: second slot, only while the published UK air deficit is deep
		set_temp_variable = { _pc_air_lane2_proj_ = -1 }
		set_temp_variable = { _pc_air_lane2_open_ = 0 }
		if = { limit = { check_variable = { WA_AI_PC_air_deficit_pct > @AI_PC_AIR_LANE_2ND_SLOT_PCT } }
			set_temp_variable = { _pc_air_lane2_open_ = 1 }
		}
		if = {
			limit = {
				has_country_flag = WA_AI_PC_air_basing_reserve
				check_variable = { available_factories_ > 0 }
			}
			for_each_loop = { array = WA_AI_PC_queue value = _air_lane_scan_
				if = {
					limit = {
						OR = {
							check_variable = { _pc_air_lane_proj_ < 0 }
							AND = {
								check_variable = { _pc_air_lane2_open_ = 1 }
								check_variable = { _pc_air_lane2_proj_ < 0 }
							}
						}
						NOT = { check_variable = { _air_lane_scan_ = _pc_lane_proj_ } }
						NOT = { check_variable = { _air_lane_scan_ = _pc_air_lane_proj_ } }
						check_variable = { WA_AI_PC_building_type^_air_lane_scan_ = 2 }
						check_variable = { WA_AI_PC_priority^_air_lane_scan_ > 250 }
						NOT = { check_variable = { WA_AI_PC_target_state^_air_lane_scan_ = 0 } }
						check_variable = { WA_AI_PC_progress^_air_lane_scan_ > 0 }
						var:WA_AI_PC_target_state^_air_lane_scan_ = {
							CONTROLLER = {
								OR = {
									tag = ROOT
									is_in_faction_with = ROOT
									is_subject_of = ROOT
									ROOT = { is_subject_of = PREV }
								}
							}
						}
					}
					if = { limit = { check_variable = { _pc_air_lane_proj_ < 0 } }
						set_temp_variable = { _pc_air_lane_proj_ = _air_lane_scan_ }
					}
					else = {
						set_temp_variable = { _pc_air_lane2_proj_ = _air_lane_scan_ }
					}
				}
			}
		}
	}

	# Fix 78: reservation GRANT - every pass, same re-entry contract as the Fix 41 grant
	# above (a re-entry resets all assignments; without the re-grant the reserved project
	# would be silently stripped back to 0 and the week voided).
	if = {
		limit = {
			NOT = { check_variable = { _pc_air_lane_proj_ < 0 } }
			check_variable = { available_factories_ > 0 }
			is_in_array = { WA_AI_PC_queue = _pc_air_lane_proj_ }
			NOT = { check_variable = { WA_AI_PC_target_state^_pc_air_lane_proj_ = 0 } }
			check_variable = { WA_AI_PC_progress^_pc_air_lane_proj_ > 0 }
		}
		set_temp_variable = { assign_amount = available_factories_ }
		clamp_temp_variable = { var = assign_amount min = 1 max = 20 }
		round_temp_variable = assign_amount
		set_variable = { WA_AI_PC_assigned_factories^_pc_air_lane_proj_ = assign_amount }
		add_to_temp_variable = { _new_total = assign_amount }
		subtract_from_temp_variable = { available_factories_ = assign_amount }
		# Fix 38 accumulator: type 2 by construction of the selection above
		add_to_variable = { WA_AI_PC_air_factories_assigned = assign_amount }
		# tlm: r8_air_lane_grants / _first_t / _last_t - counter + stamp, ONLY when the
		# reservation actually assigned >= 1 factory (the set_variable above is the verified
		# effect); latched so re-entry re-grants of the same weekly grant are not recounted,
		# same shape as pc_aging_grants - consumer: checklist R8 (Fix 78 leg). Retire with it.
		if = { limit = { NOT = { check_variable = { _pc_air_lane_counted_ = 1 } } }
			set_temp_variable = { _pc_air_lane_counted_ = 1 }
			add_to_variable = { WA_TLM_r8_air_lane_grants = 1 }
			# _first_t is stamped once (0 = never granted; the clock is >= 1 from 1936.2, so the
			# init value cannot collide with a real reading). R8 Fix-78 leg asks WHEN the lane
			# started funding against ENG's 1942.1-1944.6 deficit window, not just how often.
			if = { limit = { check_variable = { WA_TLM_r8_air_lane_first_t = 0 } }
				set_variable = { WA_TLM_r8_air_lane_first_t = global.WA_TLM_clock }
			}
			set_variable = { WA_TLM_r8_air_lane_last_t = global.WA_TLM_clock }
		}
		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC AIR LANE: proj=[?_pc_air_lane_proj_] granted [?assign_amount] factories ahead of the fill"
		}
	}

	# Fix 87 / P1: SECOND air-lane grant - same body as the first, for _pc_air_lane2_proj_.
	# Only exists while WA_AI_PC_air_deficit_pct > @AI_PC_AIR_LANE_2ND_SLOT_PCT (selection above).
	if = {
		limit = {
			NOT = { check_variable = { _pc_air_lane2_proj_ < 0 } }
			check_variable = { available_factories_ > 0 }
			is_in_array = { WA_AI_PC_queue = _pc_air_lane2_proj_ }
			NOT = { check_variable = { WA_AI_PC_target_state^_pc_air_lane2_proj_ = 0 } }
			check_variable = { WA_AI_PC_progress^_pc_air_lane2_proj_ > 0 }
		}
		set_temp_variable = { assign_amount = available_factories_ }
		clamp_temp_variable = { var = assign_amount min = 1 max = 20 }
		round_temp_variable = assign_amount
		set_variable = { WA_AI_PC_assigned_factories^_pc_air_lane2_proj_ = assign_amount }
		add_to_temp_variable = { _new_total = assign_amount }
		subtract_from_temp_variable = { available_factories_ = assign_amount }
		add_to_variable = { WA_AI_PC_air_factories_assigned = assign_amount }
		# tlm: r54_air_lane2_grants - weeks the SECOND slot funded >= 1 factory (verified by the
		# set_variable above), latched per pass. Consumer: checklist R54.
		if = { limit = { NOT = { check_variable = { _pc_air_lane2_counted_ = 1 } } }
			set_temp_variable = { _pc_air_lane2_counted_ = 1 }
			add_to_variable = { WA_TLM_r54_air_lane2_grants = 1 }
			if = { limit = { check_variable = { WA_TLM_r54_air_lane2_first_t = 0 } }
				set_variable = { WA_TLM_r54_air_lane2_first_t = global.WA_TLM_clock }
			}
			set_variable = { WA_TLM_r54_air_lane2_last_t = global.WA_TLM_clock }
		}
		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC AIR LANE 2: proj=[?_pc_air_lane2_proj_] granted [?assign_amount] factories ahead of the fill"
		}
	}

	# Assign to projects from top of queue
	# Fix 41: the laned project (_pc_lane_proj_, -1 when no lane) is excluded - it was
	# already served above and must not be double-assigned (the set_variable here would
	# otherwise overwrite its grant and double-count it in _new_total).
	# Fix 78: the air-reservation project is excluded for the same reason.
	for_each_loop = { array = WA_AI_PC_queue value = _project_id break = _assign_break
		if = {
			limit = {
				check_variable = { available_factories_ > 0 }
				NOT = { check_variable = { _project_id = _pc_lane_proj_ } }
				NOT = { check_variable = { _project_id = _pc_air_lane_proj_ } }
				NOT = { check_variable = { _project_id = _pc_air_lane2_proj_ } }	# Fix 87
				# Check state is still valid
				NOT = { check_variable = { WA_AI_PC_target_state^_project_id = 0 } }
				# Check progress remaining
				check_variable = { WA_AI_PC_progress^_project_id > 0 }
				# Fix 34: never feed factories to a project whose target state is hostile-controlled.
				# WA_AI_PC_complete_project_by_id refuses to spawn the building there anyway, so
				# every civ-week spent was pure waste (observed: ROM air base in lost state 143,
				# campaign 2b607968). Progress freezes instead; the stall counter then ages the
				# project into the sweep below, and a recapture within the window resumes the
				# build with progress intact. Same controller test as the completion path
				# (ROOT = builder country in every call chain - weekly on_action and recursion).
				var:WA_AI_PC_target_state^_project_id = {
					CONTROLLER = {
						OR = {
							tag = ROOT
							is_in_faction_with = ROOT
							is_subject_of = ROOT
							ROOT = { is_subject_of = PREV }
						}
					}
				}
			}

			set_temp_variable = { assign_amount = available_factories_ }
			clamp_temp_variable = { var = assign_amount min = 1 max = 20 }
			round_temp_variable = assign_amount

			set_variable = { WA_AI_PC_assigned_factories^_project_id = assign_amount }
			add_to_temp_variable = { _new_total = assign_amount }
			subtract_from_temp_variable = { available_factories_ = assign_amount }
			# Fix 38: accumulate the air-base share of the assignment (see reset above)
			if = { limit = { check_variable = { WA_AI_PC_building_type^_project_id = 2 } }
				add_to_variable = { WA_AI_PC_air_factories_assigned = assign_amount }
			}

			if = { limit = { has_country_flag = WA_AI_construction_logging }
				set_temp_variable = { _dbg_state = WA_AI_PC_target_state^_project_id }
				set_temp_variable = { _dbg_prog = WA_AI_PC_progress^_project_id }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: proj=[?_project_id] assigned=[?assign_amount] state=[?_dbg_state] prog=[?_dbg_prog]"
			}
		}
		else_if = {
			limit = { check_variable = { available_factories_ < 1 } }
			set_temp_variable = { _assign_break = 1 }
		}
		else = {
			# Log why project was skipped
			if = { limit = { has_country_flag = WA_AI_construction_logging }
				set_temp_variable = { _dbg_state = WA_AI_PC_target_state^_project_id }
				set_temp_variable = { _dbg_prog = WA_AI_PC_progress^_project_id }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: SKIPPED proj=[?_project_id] state=[?_dbg_state] prog=[?_dbg_prog] avail=[?available_factories_]"
			}
		}
	}

	# Update total only once at the end, then force single modifier update
	set_variable = { WA_AI_PC_assigned_factories_total = _new_total }

	# Resync the strategy-gate counter with reality: WA_AI_PC_active_projects is maintained by
	# +1/-1 bookkeeping with no reconciliation, and a desync (e.g. the debug queue-clear path)
	# permanently opens or closes every < 5 strategy gate.
	set_variable = { WA_AI_PC_active_projects = WA_AI_PC_queue^num }

	# Fix 29b companion: non-railway count for the strategy gates. Railways queue up to 5
	# segments per route at top priority; counting them against the < 5 gate permanently
	# blocked all radar/refinery/supply-line work for any country with a railway corridor
	# queued (observed on ENG and JAP for most of the campaign).
	# Fix 30: naval bases (type 14) are queued exclusively by the railway system (port
	# upgrades, frontier ports) and are excluded for the same reason - an archipelago power
	# like JAP otherwise holds the < 5 gate permanently shut with port projects.
	# Must stay in sync with the two +1/-1 bookkeeping sites below (start/end project).
	set_temp_variable = { _nonrail_count = 0 }
	for_each_loop = { array = WA_AI_PC_queue value = _count_proj_id
		if = {
			limit = {
				NOT = {
					OR = {
						check_variable = { WA_AI_PC_building_type^_count_proj_id = 13 }
						check_variable = { WA_AI_PC_building_type^_count_proj_id = 14 }
					}
				}
			}
			add_to_temp_variable = { _nonrail_count = 1 }
		}
	}
	set_variable = { WA_AI_PC_active_nonrail_projects = _nonrail_count }

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC_ASSIGN: TOTAL=[?WA_AI_PC_assigned_factories_total]"
	}

	force_update_dynamic_modifier = yes
}

### Dynamic Queue System - Update project progress
# THIS = country
# Called weekly to progress construction and complete finished projects
# Handles mid-week completions by reassigning freed factories immediately
WA_AI_PC_update_project_progress = {
	set_temp_variable = { _remaining_days = 7 }
	set_temp_variable = { _progress_iterations = 0 }

	# Cleanup pass: complete finished projects, and REMOVE broken ones.
	# Broken = target_state 0 (WA_AI_PC_complete_project_by_id no-ops on those, so they could
	# never leave the queue) or cost <= 0 (a failed cost lookup - routing it through completion
	# would spawn a free instant building). Both go straight to WA_AI_PC_end_project_by_id.
	clear_temp_array = _cleanup_project_ids
	clear_temp_array = _orphan_project_ids
	for_each_loop = { array = WA_AI_PC_queue value = _project_id
		if = {
			limit = {
				OR = {
					check_variable = { WA_AI_PC_target_state^_project_id = 0 }
					NOT = { check_variable = { WA_AI_PC_project_cost^_project_id > 0 } }
				}
			}
			add_to_temp_array = { _orphan_project_ids = _project_id }
		}
		else_if = {
			limit = { NOT = { check_variable = { WA_AI_PC_progress^_project_id > 0 } } }
			add_to_temp_array = { _cleanup_project_ids = _project_id }
		}
	}
	if = {
		limit = { check_variable = { _orphan_project_ids^num > 0 } }
		for_each_loop = { array = _orphan_project_ids value = _project_id
			if = { limit = { has_country_flag = WA_AI_construction_logging }
				set_temp_variable = { _dbg_state = WA_AI_PC_target_state^_project_id }
				set_temp_variable = { _dbg_cost = WA_AI_PC_project_cost^_project_id }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC CLEANUP: removing broken proj=[?_project_id] state=[?_dbg_state] cost=[?_dbg_cost]"
			}
			WA_AI_PC_end_project_by_id = yes
			# tlm: pc_orphan_n - counter, one per BROKEN project removed (target_state 0
			# or cost <= 0). Written AFTER end_project_by_id, so it counts removals and not
			# candidates - consumer: standing (the PC termination ledger)
			add_to_variable = { WA_TLM_pc_orphan_n = 1 }
		}
	}
	if = {
		limit = { check_variable = { _cleanup_project_ids^num > 0 } }
		for_each_loop = { array = _cleanup_project_ids value = _project_id
			if = { limit = { has_country_flag = WA_AI_construction_logging }
				set_temp_variable = { _dbg_prog = WA_AI_PC_progress^_project_id }
				set_temp_variable = { _dbg_cost = WA_AI_PC_project_cost^_project_id }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC CLEANUP: proj=[?_project_id] prog=[?_dbg_prog] cost=[?_dbg_cost]"
			}
			WA_AI_PC_complete_project_by_id = yes
		}
	}
	# Reassign factories after any cleanup
	if = {
		limit = {
			OR = {
				check_variable = { _orphan_project_ids^num > 0 }
				check_variable = { _cleanup_project_ids^num > 0 }
			}
		}
		WA_AI_PC_assign_factories = yes
	}

	# Process the week in segments, reassigning factories when projects complete
	while_loop_effect = {
		limit = {
			check_variable = { _remaining_days > 0.1 }
			check_variable = { _progress_iterations < 10 }
		}
		add_to_temp_variable = { _progress_iterations = 1 }

		# Find the earliest completing project
		set_temp_variable = { _earliest_completion_days = 999 }
		set_temp_variable = { _earliest_completion_id = -1 }

		for_each_loop = { array = WA_AI_PC_queue value = _project_id
			if = {
				limit = {
					check_variable = { WA_AI_PC_assigned_factories^_project_id > 0 }
					check_variable = { WA_AI_PC_progress^_project_id > 0 }
				}

				# Get construction speed
				set_temp_variable = { _project_building_type = WA_AI_PC_building_type^_project_id }
				var:WA_AI_PC_target_state^_project_id = {
					WA_AI_PC_get_build_speed = yes
				}

				# Calculate days to completion for this project
				set_temp_variable = { _daily_progress = construction_speed_ }
				multiply_temp_variable = { _daily_progress = WA_AI_PC_assigned_factories^_project_id }

				if = { limit = { check_variable = { _daily_progress > 0 } }
					set_temp_variable = { _days_needed = WA_AI_PC_progress^_project_id }
					divide_temp_variable = { _days_needed = _daily_progress }

					# Check if this completes within remaining days and is earliest
					if = {
						limit = {
							NOT = { check_variable = { _days_needed > _remaining_days } }
							check_variable = { _days_needed < _earliest_completion_days }
						}
						set_temp_variable = { _earliest_completion_days = _days_needed }
						set_temp_variable = { _earliest_completion_id = _project_id }
					}
				}
			}
		}

		# Determine how many days to process this iteration
		if = { limit = { NOT = { check_variable = { _earliest_completion_id < 0 } } }
			# Process up to the completion point
			set_temp_variable = { _days_to_process = _earliest_completion_days }
		}
		else = {
			# No completions - process remaining days
			set_temp_variable = { _days_to_process = _remaining_days }
		}

		# Apply progress for _days_to_process to all active projects
		for_each_loop = { array = WA_AI_PC_queue value = _project_id
			if = {
				limit = {
					check_variable = { WA_AI_PC_assigned_factories^_project_id > 0 }
					check_variable = { WA_AI_PC_progress^_project_id > 0 }
				}

				set_temp_variable = { _project_building_type = WA_AI_PC_building_type^_project_id }
				var:WA_AI_PC_target_state^_project_id = {
					WA_AI_PC_get_build_speed = yes
				}

				set_temp_variable = { _segment_progress = construction_speed_ }
				multiply_temp_variable = { _segment_progress = WA_AI_PC_assigned_factories^_project_id }
				multiply_temp_variable = { _segment_progress = _days_to_process }
				subtract_from_variable = { WA_AI_PC_progress^_project_id = _segment_progress }

				# Update estimated build time
				if = { limit = { check_variable = { construction_speed_ > 0 } }
					set_variable = { WA_AI_PC_build_time^_project_id = WA_AI_PC_progress^_project_id }
					set_temp_variable = { _daily_speed = construction_speed_ }
					multiply_temp_variable = { _daily_speed = WA_AI_PC_assigned_factories^_project_id }
					if = { limit = { check_variable = { _daily_speed > 0 } }
						divide_variable = { WA_AI_PC_build_time^_project_id = _daily_speed }
					}
				}
			}
		}

		subtract_from_temp_variable = { _remaining_days = _days_to_process }

		# If a project completed, handle it and reassign factories
		if = { limit = { NOT = { check_variable = { _earliest_completion_id < 0 } } }
			set_temp_variable = { _project_id = _earliest_completion_id }
			WA_AI_PC_complete_project_by_id = yes
			WA_AI_PC_assign_factories = yes
		}
	}

	# Stall sweep: a project that ends week after week with 0 assigned factories is being starved
	# by higher-priority queue entries. Left alone it blocks the queue forever (the < 5 strategy
	# gate counts it as active). Cancel after @AI_PC_STALL_CANCEL_WEEKS starved weeks.
	# Fix 34: gate on factories actually flowing (assigned total > 0), not on queue length > 5.
	# The old queue gate assumed a short queue drains and the starved project gets its turn -
	# false when the railway system replenishes prio-9999 segments every cycle (observed: ROM,
	# queue of 4, two projects starved 17+ weeks, campaign 2b607968). It also allowed 5 starved
	# nonrail projects to lock every < 5 strategy gate shut with the sweep unable to fire.
	# Total = 0 (economic collapse / full occupation) still cancels nothing: no project is
	# being outranked when nobody gets factories, so starvation proves nothing there.
	clear_temp_array = _cancel_projects_IDS
	for_each_loop = { array = WA_AI_PC_queue value = _project_id
		if = {
			limit = { NOT = { check_variable = { WA_AI_PC_assigned_factories^_project_id > 0 } } }
			add_to_variable = { WA_AI_PC_stall_weeks^_project_id = 1 }
			if = {
				limit = {
					check_variable = { WA_AI_PC_stall_weeks^_project_id > @AI_PC_STALL_CANCEL_WEEKS }
					check_variable = { WA_AI_PC_assigned_factories_total > 0 }
				}
				add_to_temp_array = { _cancel_projects_IDS = _project_id }
				# tlm: pc_sweep_n - counter, one per project the week-30 stall sweep cancels.
				# Everything added to _cancel_projects_IDS here is unconditionally ended by
				# the WA_AI_PC_cancel_projects call a few lines below (its ^num > 0 guard is
				# satisfied by this very add), so this is the cancel, not an attempt at one -
				# consumer: standing (the PC termination ledger)
				add_to_variable = { WA_TLM_pc_sweep_n = 1 }
			}
		}
		else = {
			set_variable = { WA_AI_PC_stall_weeks^_project_id = 0 }
		}
	}
	if = {
		limit = { check_variable = { _cancel_projects_IDS^num > 0 } }
		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC STALL SWEEP: cancelling [?_cancel_projects_IDS^num] starved projects"
		}
		WA_AI_PC_cancel_projects = yes
		WA_AI_PC_assign_factories = yes
	}
}

### Dynamic Queue System - Complete project by ID
# _project_id >>> completes project and spawns building
WA_AI_PC_complete_project_by_id = {
	# Check if state is still controlled by us
	if = {
		limit = {
			NOT = { check_variable = { WA_AI_PC_target_state^_project_id = 0 } }
		}

		# Set project_type for the slot availability trigger
		set_variable = { WA_AI_PC_project_type = _project_id }

		# Check if this building type can be built in allied territory
		# Types: 1=infra, 2=airbase, 3=AA, 4=radar, 13=railway, 14=naval_base
		set_temp_variable = { _can_build_in_allied = 0 }
		if = {
			limit = {
				OR = {
					check_variable = { WA_AI_PC_building_type^_project_id = 1 }
					check_variable = { WA_AI_PC_building_type^_project_id = 2 }
					check_variable = { WA_AI_PC_building_type^_project_id = 3 }
					check_variable = { WA_AI_PC_building_type^_project_id = 4 }
					check_variable = { WA_AI_PC_building_type^_project_id = 13 }
					check_variable = { WA_AI_PC_building_type^_project_id = 14 }
				}
			}
			set_temp_variable = { _can_build_in_allied = 1 }
		}

		var:WA_AI_PC_target_state^_project_id = {
			# Strategic buildings (infra, airbase, AA, radar, railway) can be built in allied territory
			if = {
				limit = {
					check_variable = { _can_build_in_allied = 1 }
					CONTROLLER = {
						OR = {
							tag = ROOT
							is_in_faction_with = ROOT
							is_subject_of = ROOT
							ROOT = { is_subject_of = PREV }
						}
					}
					# Railways and naval_bases don't need slot check, others do
					OR = {
						check_variable = { ROOT.WA_AI_PC_building_type^_project_id = 13 }
						check_variable = { ROOT.WA_AI_PC_building_type^_project_id = 14 }
						WA_AI_PC_building_slot_available = yes
					}
				}

				# Spawn the building
				WA_AI_PC_add_finished_building_by_id = yes
			}
			# Factories and refineries require direct control
			else_if = {
				limit = {
					check_variable = { _can_build_in_allied = 0 }
					CONTROLLER = { tag = ROOT }
					WA_AI_PC_building_slot_available = yes
				}

				# Spawn the building
				WA_AI_PC_add_finished_building_by_id = yes
			}
			# tlm: pc_refused_n / pc_refused_ctrl_n - counters, one per project that reached
			# progress <= 0 (fully PAID FOR) and was then denied its building by the two tests
			# above, after which WA_AI_PC_end_project_by_id discards it anyway. These are the
			# silent failure the whole system had no fingerprint for: from outside, a refused
			# project is indistinguishable from one that completed. The _ctrl_n subset repeats
			# each branch's OWN controller test so the split is exact - a strategic building
			# refused on allied ground is a slot problem, while a factory-type building on the
			# same ground is a control problem - and the remainder (_refused_n - _refused_ctrl_n)
			# is therefore "friendly ground, no free building slot".
			# Runs only on refusals, so the duplicated trigger costs nothing in the common case.
			# consumer: standing (the PC termination ledger)
			else = {
				ROOT = { add_to_variable = { WA_TLM_pc_refused_n = 1 } }
				if = {
					limit = {
						OR = {
							AND = {
								check_variable = { _can_build_in_allied = 1 }
								NOT = {
									CONTROLLER = {
										OR = {
											tag = ROOT
											is_in_faction_with = ROOT
											is_subject_of = ROOT
											ROOT = { is_subject_of = PREV }
										}
									}
								}
							}
							AND = {
								check_variable = { _can_build_in_allied = 0 }
								NOT = { CONTROLLER = { tag = ROOT } }
							}
						}
					}
					ROOT = { add_to_variable = { WA_TLM_pc_refused_ctrl_n = 1 } }
				}
			}
		}

		WA_AI_PC_end_project_by_id = yes
	}
}

### Dynamic Queue System - End project by ID
# _project_id >>> removes project from queue and clears variables
WA_AI_PC_end_project_by_id = {
	# Remove from queue
	remove_from_array = { array = WA_AI_PC_queue value = _project_id }
	subtract_from_variable = { WA_AI_PC_active_projects = 1 }
	# Fix 30: types 13 AND 14 excluded from the nonrail gate counter (must match the resync
	# in WA_AI_PC_assign_factories and the increment in WA_AI_PC_start_project)
	if = {
		limit = {
			NOT = {
				OR = {
					check_variable = { WA_AI_PC_building_type^_project_id = 13 }
					check_variable = { WA_AI_PC_building_type^_project_id = 14 }
				}
			}
		}
		subtract_from_variable = { WA_AI_PC_active_nonrail_projects = 1 }
	}
	# Factory total is recalculated by WA_AI_PC_assign_factories after all completions

	# Remove from state tracking (per-country array stores project IDs)
	# Fix 46: the builder-agnostic counter added by WA_AI_PC_start_project is decremented in
	# the same block, through the same var: accessor, so the pair can never desync. Read the
	# type into a temp FIRST - building_type is cleared a few lines below, and an indexed
	# variable cannot be resolved from inside the state scope once it is gone. The > 0 guard
	# makes this idempotent and safe on saves resumed from a pre-Fix-46 build, where in-flight
	# projects were queued without an increment and would otherwise drive the counter negative
	# (a negative counter reads as "no project here" and would silently disable the guard).
	set_temp_variable = { _pc_end_btype_ = WA_AI_PC_building_type^_project_id }
	if = { limit = { NOT = { check_variable = { WA_AI_PC_target_state^_project_id = 0 } } }
		var:WA_AI_PC_target_state^_project_id = {
			remove_from_array = { array = WA_AI_PC_projects_in_state_of_@ROOT value = _project_id }
			if = {
				limit = { check_variable = { WA_AI_PC_state_type_projects^_pc_end_btype_ > 0 } }
				subtract_from_variable = { WA_AI_PC_state_type_projects^_pc_end_btype_ = 1 }
			}
		}
	}

	# Clear project variables (mark slot as available by setting target_state to 0)
	set_variable = { WA_AI_PC_target_state^_project_id = 0 }
	clear_variable = WA_AI_PC_target_province^_project_id
	clear_variable = WA_AI_PC_connect_province^_project_id
	clear_variable = WA_AI_PC_project_cost^_project_id
	clear_variable = WA_AI_PC_progress^_project_id
	clear_variable = WA_AI_PC_building_type^_project_id
	clear_variable = WA_AI_PC_build_time^_project_id
	clear_variable = WA_AI_PC_priority^_project_id
	clear_variable = WA_AI_PC_assigned_factories^_project_id
	clear_variable = WA_AI_PC_type_id^_project_id
	clear_variable = WA_AI_PC_stall_weeks^_project_id
	# tlm: pc_queued_t - cleared with the slot it belongs to. Slot ids are recycled by
	# WA_AI_PC_get_new_project_ID, so a stale stamp left here would date the NEXT project to
	# occupy this id (and start_project overwrites it anyway - this clear is what keeps a
	# free slot from reading as an ancient queued one).
	clear_variable = WA_TLM_pc_queued_t^_project_id
}

### Cancel multiple projects at once

WA_AI_PC_cancel_projects = {

		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC CANCELLED: project_id=[?_project_id]"
		}
		WA_AI_PC_end_project_by_id = yes

}

### Validate queued provincial projects against current valid paths
# ROOT = BUILDER, _strategy_id (type_id to filter), _valid_provinces array (temp)
# Cancels any projects of the given strategy type whose target province is not in the valid paths
WA_AI_PC_validate_queued_provincial_projects = {
	clear_temp_array = _cancel_projects_IDS

	for_each_loop = { array = WA_AI_PC_queue value = _project_id
		if = {
			limit = {
				# Match the strategy type (railway = type_id 13)
				check_variable = { WA_AI_PC_type_id^_project_id = _strategy_id }
				# Project target province is NOT in the current valid paths
				NOT = { is_in_array = { _valid_provinces = WA_AI_PC_target_province^_project_id } }
			}
			add_to_temp_array = { _cancel_projects_IDS = _project_id }
			# tlm: pc_stale_n - counter, one per queued project cancelled because its target
			# province left the strategy's current valid path set (railway routes, type 13 via
			# Fix 50). Same verified-effect argument as pc_sweep_n: the WA_AI_PC_cancel_projects
			# call below ends every id added here.
			#
			# WHY THIS COUNTER EXISTS, stated without prejudging the answer (corrected
			# 2026-08-15): GER's railway queue fell 80 -> 24 -> 2 across 1944.6-1945.2 on
			# campaign be18f9c7 and the aging lane did not do it (wa_tlm_pc_aging_reval_cancels
			# was FROZEN at 64 across the collapse). Two candidates remain and the shipped
			# comments used to assert BOTH: the 30-week stall sweep (WA_AI_PC_QUEUE_FAIRNESS_
			# DIAGNOSIS.md §2.1 argues the terminal cohort died there, and rebuts the "stall
			# pinned at 26" reading as a monthly-sampling artefact) and this stale-path
			# validator. `pc_sweep_n` and `pc_stale_n` exist precisely to separate them on the
			# NEXT campaign - do not read either comment as the established cause.
			#
			# WATCH ITEM, because Fix 77 widened this path: the skip gate now opens on the
			# LIVE queue count, so this validator runs on far more pulses, against a queue at
			# its admission cap. If a pulse recomputes routes that miss the half-built
			# segments, they are cancelled and their paid-for progress is discarded; and if
			# every WA_AI_PATHFIND_PROV_get_path fails, `_valid_provinces` is EMPTY and this
			# block cancels 100% of type-13 projects in one call. A pc_stale_n spike that
			# tracks a type-13 collapse is that scenario, not evidence about 1944.
			# consumer: standing (the PC termination ledger)
			add_to_variable = { WA_TLM_pc_stale_n = 1 }
		}
	}

	# Cancel invalid projects if any found
	if = { limit = { check_variable = { _cancel_projects_IDS^num > 0 } }
		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC VALIDATION: Cancelling [?_cancel_projects_IDS^num] stale railway projects"
		}
		WA_AI_PC_cancel_projects = yes
	}

	# Clear input arrays
	clear_temp_array = _valid_provinces
}

### Dynamic Queue System - Add finished building by ID
# THIS = state, _project_id >>> spawns the building
WA_AI_PC_add_finished_building_by_id = {
	set_temp_variable = { _build_type = ROOT.WA_AI_PC_building_type^_project_id }

	# tlm: pc_built_n / pc_built_by_type - counters, one per building this system actually
	# spawns. THIS = state, ROOT = builder, so both are written on the builder country.
	# The 1..16 guard is not a formality: it is exactly the range the ladder below covers,
	# so the counter increments if and only if one of its add_building_construction /
	# build_railway branches fires. An out-of-range _build_type falls through the whole
	# ladder silently and must not be counted as a build (honesty rule 1 - verified effect,
	# never code-path entry).
	# pc_built_by_type is INDEXED by the building type code, which makes "did the air bases
	# ever actually finish" one read instead of sixteen metrics. Absence of an index means
	# this country has never completed that type - unambiguous, because the family's
	# presence witness is the zero-initialised pc_built_n beside it.
	# consumer: standing (the PC termination ledger; the success half of it)
	if = {
		limit = {
			check_variable = { _build_type > 0 }
			check_variable = { _build_type < 17 }
		}
		ROOT = {
			add_to_variable = { WA_TLM_pc_built_n = 1 }
			add_to_variable = { WA_TLM_pc_built_by_type^_build_type = 1 }
		}
	}

	if = { limit = { check_variable = { _build_type = 1 } }
		add_building_construction = { type = infrastructure level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 2 } }
		add_building_construction = { type = air_base level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 3 } }
		add_building_construction = { type = anti_air_building level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 4 } }
		add_building_construction = { type = radar_station level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 5 } }
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 6 } }
		add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 7 } }
		add_building_construction = { type = dockyard level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 8 } }
		add_building_construction = { type = synthetic_refinery level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 9 } }
		remove_building = { type = arms_factory level = 1 }
		add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 10 } }
		remove_building = { type = industrial_complex level = 1 }
		add_building_construction = { type = arms_factory level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 11 } }
		add_building_construction = { type = hydro_steel_refinery level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 12 } }
		add_building_construction = { type = hydro_aluminium_refinery level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 15 } }
		add_building_construction = { type = steel_refinery level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 16 } }
		add_building_construction = { type = aluminium_refinery level = 1 instant_build = yes }
	}
	else_if = { limit = { check_variable = { _build_type = 13 } }
		# Railway construction using stored province data
		set_temp_variable = { _prov_from = ROOT.WA_AI_PC_target_province^_project_id }
		set_temp_variable = { _prov_to = ROOT.WA_AI_PC_connect_province^_project_id }

		meta_effect = {
			text = {
				build_railway = {
					level = 1
					path = { [x] [y] }
				}
				set_variable = { global.WA_AI_PC_railway_connections^[x] = 1 }
				add_to_variable = { global.WA_AI_PC_railway_connection_level_[x]^[y] = 1 }
				set_variable = { global.WA_AI_PC_railway_connections^[y] = 1 }
				add_to_variable = { global.WA_AI_PC_railway_connection_level_[y]^[x] = 1 }
			}
			x = "[?_prov_from]"
			y = "[?_prov_to]"
		}

		if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
			ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: Railway built [?_prov_from] -> [?_prov_to]" }
		}
	}
	else_if = { limit = { check_variable = { _build_type = 14 } }
		# Naval base construction using stored province data
		set_temp_variable = { _port_prov = ROOT.WA_AI_PC_target_province^_project_id }

		meta_effect = {
			text = {
				add_building_construction = {
					type = naval_base
					level = 1
					province = { id = [PROV] }
					instant_build = yes
				}
			}
			PROV = "[?_port_prov]"
		}

		if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
			ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: Naval base built at province [?_port_prov]" }
		}
	}
}

### Dynamic Queue System - Start Project
# THIS = state, PREV = builder country
# Input: _project_building_type, _project_province_id, _project_connect_id,
#        _project_priority, _project_queue_num, _project_queue_max, _project_target_level, _project_type_id
# Optional input: _project_queue_max_scoped (Fix 47) - see the latch immediately below
# Creates projects and adds them to the dynamic queue
WA_AI_PC_start_project = {
	# Debug: log entry
	ROOT = { if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC START_PROJECT ENTRY: type=[?_project_building_type] state=[PREV.GetName] prov=[?_project_province_id]->[?_project_connect_id]"
	} }

	# Fix 47: OPT-IN queue_max scoping, latched and immediately cleared.
	#
	# By default _project_queue_max is a country-wide budget over the BUILDING TYPE
	# (queued_amount_ below counts every queued project of the same type, whatever
	# created it). For building type 2 that put two unrelated strategies on one
	# 3-slot budget: WA_AI_build_uk_air_hosting_capacity and WA_AI_build_theatre_air_bases
	# compete first-come-first-served, and campaign 911bed3c caught the consequence -
	# both of ENG's air slots held Pacific theatre projects at 1943.6 (the exact pulse
	# the UK deficit reopened), and all three USA slots were non-UK at 1945.6. Fix 47's
	# priority band cannot arbitrate this: UK and theatre air bases now sit in the SAME
	# band, so only the slot budget decides. Setting _project_queue_max_scoped = 1 makes
	# queue_max count only projects carrying the caller's own _project_type_id, giving
	# each strategy an independent budget.
	#
	# CLEAR-ON-READ on the opt-in switch only. It is a shared temp that persists across the
	# whole execution chain, and this codebase has already been bitten by exactly that:
	# WA_AI_CONSTRUCTION_queue_functions.txt:681-689 records refinery projects inheriting
	# "3 from the two air strategies, 5 from the railway core, 0 from the railway helpers"
	# because they set _project_queue_num but not _project_queue_max. Latching the switch and
	# zeroing the original means the 30 call sites that never set it CANNOT inherit an
	# opted-in state - unscoped stays structurally the default.
	#
	# _project_type_id is deliberately NOT cleared the same way. It is still an input below:
	# the scoped queue budget compares it at :1194 and the new project persists it at :1281.
	# Clearing it here would strip the caller's tag before either use, collapsing the independent
	# type-20/type-21 air budgets back together. Callers that introduce a private tag retire it
	# caller-side. Fix 50 separately pins railway stale-project validation to type 13, so a port
	# helper's type-14 tag can no longer redirect that validator.
	set_temp_variable = { _qmax_scoped_ = _project_queue_max_scoped }
	set_temp_variable = { _project_queue_max_scoped = 0 }

	# Validate location is controlled
	if = {
		limit = {
			OR = {
				is_controlled_by = ROOT
				CONTROLLER = { is_in_faction_with = ROOT }
				CONTROLLER = { is_subject_of = ROOT }
			}
		}

		ROOT = {
			# Default queue_num to 1 if not set
			if = { limit = { check_variable = { _project_queue_num = 0 } }
				set_temp_variable = { _project_queue_num = 1 }
			}

			# Get current queued amount for this building type (for queue_max check)
			# Also check for province-level duplicates (naval bases, supply hubs, etc.)
			# Fix 37: resolve THIS state's id up front so the loop below can also count
			# same-type projects targeting THIS state only (_queued_here_). The
			# target-level test must use the per-state count: the old country-wide
			# queued_amount_ made 3 queued UK air-base projects read as +3 levels on an
			# unrelated Pacific candidate state and reject it (campaign 31eaf7e6).
			# queued_amount_ itself stays country-wide - that is the intended semantic
			# of the queue_max per-country budget. Same state-id resolution idiom as
			# the target-state fallback further down.
			set_temp_variable = { _level_check_state_ = _project_target_state }
			if = { limit = { NOT = { check_variable = { _level_check_state_ > 0 } } }
				PREV = {
					set_temp_variable = { _lvl_first_prov_ = global.WA_AI_MAP_state_province_ids@THIS^0 }
					set_temp_variable = { _level_check_state_ = global.WA_AI_MAP_province_state_id^_lvl_first_prov_ }
				}
			}
			set_temp_variable = { queued_amount_ = 0 }
			set_temp_variable = { _queued_here_ = 0 }
			set_temp_variable = { _province_already_queued = 0 }
			for_each_loop = { array = WA_AI_PC_queue value = _check_proj_id
				if = { limit = { check_variable = { WA_AI_PC_building_type^_check_proj_id = _project_building_type } }
					# Fix 47: queued_amount_ backs the queue_max budget. Unscoped (default) it
					# stays a country-wide count over the building type - unchanged semantics
					# for every existing caller. Scoped, it counts only projects tagged with
					# the caller's own type_id, so two strategies sharing a building type get
					# independent budgets. _queued_here_ and the province-duplicate test below
					# are deliberately NOT scoped: those guard against building twice in one
					# place, which is wrong no matter which strategy queued the other project.
					if = {
						limit = {
							OR = {
								NOT = { check_variable = { _qmax_scoped_ = 1 } }
								check_variable = { WA_AI_PC_type_id^_check_proj_id = _project_type_id }
							}
							# Fix 87b (a): a project frozen on ENEMY-controlled ground (Fix 34) does
							# not hold a budget slot. Same controller test as the fill; state-keyed
							# projects only (target_state 0 = province project, unchanged).
							OR = {
								NOT = { check_variable = { WA_AI_PC_target_state^_check_proj_id > 0 } }
								var:WA_AI_PC_target_state^_check_proj_id = {
									CONTROLLER = {
										OR = {
											tag = ROOT
											is_in_faction_with = ROOT
											is_subject_of = ROOT
											ROOT = { is_subject_of = PREV }
										}
									}
								}
							}
						}
						add_to_temp_variable = { queued_amount_ = 1 }
					}
					# Fix 37: per-state count for the target-level test
					if = { limit = { check_variable = { WA_AI_PC_target_state^_check_proj_id = _level_check_state_ } }
						add_to_temp_variable = { _queued_here_ = 1 }
					}
					# Check for province-level duplicate (same type + same province)
					if = {
						limit = {
							check_variable = { _project_province_id > 0 }
							check_variable = { WA_AI_PC_target_province^_check_proj_id = _project_province_id }
						}
						set_temp_variable = { _province_already_queued = 1 }
					}
				}
			}

			# Skip if this exact province is already queued for this building type
			if = { limit = { check_variable = { _province_already_queued = 1 } }
				if = { limit = { has_country_flag = WA_AI_construction_logging }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC SKIP DUPLICATE: type=[?_project_building_type] prov=[?_project_province_id] already queued"
				}
			}
			else = {

			# Get current building level for target_level check (state-level buildings)
			set_temp_variable = { final_building_level_ = 0 }
			if = { limit = { check_variable = { _project_target_level > 0 } }
				PREV = {
					if = { limit = { check_variable = { _project_building_type = 1 } }
						set_temp_variable = { final_building_level_ = infrastructure_level }
					}
					else_if = { limit = { check_variable = { _project_building_type = 2 } }
						set_temp_variable = { final_building_level_ = building_level@air_base }
					}
					else_if = { limit = { check_variable = { _project_building_type = 4 } }
						set_temp_variable = { final_building_level_ = building_level@radar_station }
					}
				}
				# Fix 37: add only projects already queued FOR THIS STATE to the
				# effective level (was queued_amount_, the country-wide type total)
				add_to_temp_variable = { final_building_level_ = _queued_here_ }
			}

			# Create up to _project_queue_num projects
			while_loop_effect = { break = _start_project_break
				limit = {
					check_variable = { _project_queue_num > 0 }
					# Respect target level limit
					if = { limit = { check_variable = { _project_target_level > 0 } }
						check_variable = { final_building_level_ < _project_target_level }
					}
					# Respect queue max limit
					if = { limit = { check_variable = { _project_queue_max > 0 } }
						check_variable = { queued_amount_ < _project_queue_max }
					}
				}

				# Get new project ID
				WA_AI_PC_get_new_project_ID = yes
				set_temp_variable = { _project_id = new_project_ }

				# Get building cost
				PREV = { WA_AI_PC_get_building_cost = yes }

				# Set project variables
				# Use _project_target_state if provided (for railways), otherwise get from PREV scope
				if = { limit = { check_variable = { _project_target_state > 0 } }
					set_variable = { WA_AI_PC_target_state^_project_id = _project_target_state }
				}
				else = {
					# For state-level buildings called from state scope, get state ID via province mapping
					PREV = {
						set_temp_variable = { _temp_first_prov = global.WA_AI_MAP_state_province_ids@THIS^0 }
						set_temp_variable = { _project_target_state = global.WA_AI_MAP_province_state_id^_temp_first_prov }
					}
					set_variable = { WA_AI_PC_target_state^_project_id = _project_target_state }
				}
				set_variable = { WA_AI_PC_target_province^_project_id = _project_province_id }
				set_variable = { WA_AI_PC_connect_province^_project_id = _project_connect_id }
				set_variable = { WA_AI_PC_project_cost^_project_id = building_cost_ }
				set_variable = { WA_AI_PC_progress^_project_id = building_cost_ }
				set_variable = { WA_AI_PC_building_type^_project_id = _project_building_type }
				set_variable = { WA_AI_PC_priority^_project_id = _project_priority }
				set_variable = { WA_AI_PC_type_id^_project_id = _project_type_id }

				# tlm: pc_queued_t - stamp, INDEXED by project id: the clock at which this
				# slot entered the queue, cleared with the rest of the slot in
				# WA_AI_PC_end_project_by_id. Telemetry rather than system state precisely so
				# that reading global.WA_TLM_clock here stays legal - a wa_ai_* field fed from
				# the clock would be gameplay reading telemetry, which the namespace forbids.
				# It is what makes queue AGE readable at all: nothing else in the queue records
				# when a project arrived. WA_AI_PC_stall_weeks is not a substitute - it resets
				# to 0 every time a project is funded for a week, so it measures the current
				# starvation streak, not time in the queue, and the two diverge exactly on the
				# projects worth asking about.
				# consumer: standing (queue-age questions; `savegame.py pc` prints age_m)
				set_variable = { WA_TLM_pc_queued_t^_project_id = global.WA_TLM_clock }

				# Add to queue
				add_to_array = { WA_AI_PC_queue = _project_id }
				add_to_variable = { WA_AI_PC_active_projects = 1 }
				# Keep the non-railway gate counter live between weekly resyncs (strategies run
				# from a 2-day background event, so waiting for assign_factories would let them
				# over-queue within a week)
				# Fix 30: types 13 AND 14 excluded (must match the resync in
				# WA_AI_PC_assign_factories and the decrement in WA_AI_PC_end_project_by_id)
				if = {
					limit = {
						NOT = {
							OR = {
								check_variable = { WA_AI_PC_building_type^_project_id = 13 }
								check_variable = { WA_AI_PC_building_type^_project_id = 14 }
							}
						}
					}
					add_to_variable = { WA_AI_PC_active_nonrail_projects = 1 }
				}

				# Track in state (per-country array stores project IDs)
				PREV = {
					add_to_array = { WA_AI_PC_projects_in_state_of_@ROOT = _project_id }
				}

				# Fix 46: BUILDER-AGNOSTIC per-state, per-building-type project counter.
				# WA_AI_PC_projects_in_state_of_@ROOT above is keyed by builder, so an allied
				# builder (any building with allied_build = yes - air_base, infrastructure)
				# cannot see the host nation's project in the same state, and vice versa.
				# Campaign 9a4cd657 1944.7: ENG and USA were each funding an air base in
				# state 126 AND state 860 at the same time, one of the two guaranteed to be
				# wasted civ-weeks. WA_AI_PC_has_same_type_project_here reads this counter.
				# Written through var:WA_AI_PC_target_state (NOT PREV) so the increment uses
				# the identical accessor as the decrement in WA_AI_PC_end_project_by_id -
				# PREV is a province scope on the railway path and would desync the pair.
				var:WA_AI_PC_target_state^_project_id = {
					add_to_variable = { WA_AI_PC_state_type_projects^_project_building_type = 1 }
				}

				# Update counters for next iteration
				add_to_temp_variable = { final_building_level_ = 1 }
				add_to_temp_variable = { queued_amount_ = 1 }
				subtract_from_temp_variable = { _project_queue_num = 1 }

				if = { limit = { has_country_flag = WA_AI_construction_logging }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | PC QUEUED: type=[?_project_building_type] cost=[?building_cost_] state=[PREV.GetName] prov=[?_project_province_id]->[?_project_connect_id]"
				}
			}
		} # end else (not duplicate)
	} # end ROOT
	} # end if (controlled)
}
