############################################################################################################
#	World Ablaze - Spanish Civil War opening balance (Nationalist side)
############################################################################################################
#
# WA_SPA_nationalist_opening_reinforcement
#
# Scope   : any country scope. The effect re-scopes to the SPA tag alias itself
#           (original_tag = SPR + SPR_nationalist_spain_flag), so it is safe to call from
#           either branch of lar_spain.2 - the Nationalist branch, where ROOT is the
#           Nationalist, and the Republican branch, where ROOT is the Republican.
# Inputs  : none.
# Outputs : Nationalist army fully equipped, six extra line divisions on the three axes
#           the civil war front actually forms on.
# Timing  : must be called AFTER transfer_units_fraction has split the pre-war army.
#           Calling it earlier would equip the Republican half as well.
#
# Rationale: under the La Resistance split the Republicans open with 10 military factories
# (Madrid 4, Catalonia 3, Granada 2, Valencia 1) against the Nationalists' 6, and the
# Nationalist foreign-aid focus branch (SPA_the_condor_legion / SPA_the_corpo_truppe_volontarie
# / SPA_equipment_shipments) is gated on volunteers actually arriving from GER and ITA.
# This front-loads the materiel edge that foreign support is supposed to represent.

WA_SPA_nationalist_opening_reinforcement = {

	SPA = {

		# Full kit for everything the rising took out of the army split.
		set_equipment_fraction = 1

		# Regulars raised by the garrisons that declared for the rising, placed on the
		# three axes the front forms on: Madrid (via Castille), Catalonia/Valencia
		# (via Aragon) and Andalusia (via Sevilla).
		if = {
			limit = { has_template = "División de Infantería" }

			176 = { # Castille - Madrid approach
				create_unit = {
					division = "division_template = \"División de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
			166 = { # Aragon - eastern front
				create_unit = {
					division = "division_template = \"División de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
			169 = { # Sevilla - southern front
				create_unit = {
					division = "division_template = \"División de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
		}
		# Fallback if the divisional template did not survive the split (e.g. the
		# Republicans took SPR_disband_the_army and the army_ratio transfer left
		# only brigades on the Nationalist side).
		else_if = {
			limit = { has_template = "Brigada de Infantería" }

			176 = {
				create_unit = {
					division = "division_template = \"Brigada de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
			166 = {
				create_unit = {
					division = "division_template = \"Brigada de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
			169 = {
				create_unit = {
					division = "division_template = \"Brigada de Infantería\" start_experience_factor = 0.3 start_equipment_factor = 1"
					owner = SPA
					count = 2
				}
			}
		}
	}
}
