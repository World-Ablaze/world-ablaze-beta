	}

	# permanently holds ~20 railway projects, so the < 5 gate above can never open
	# for the host nation, and the UK capacity deficit grounds the whole allied
	# air war (campaign 5709c8b9: the entire USAAF parked stateside). The strategy
	# bounds itself via _project_queue_max and the per-state level cap.
	WA_AI_build_uk_air_hosting_capacity = yes

WA_AI_build_refinery_resource_shortage_rubber = {

	### track rubber shortage

	if = {
		limit = {
			# there is a rubber deficit
			check_variable = { resource@rubber < 0 }

			# total rubber needed for production
			set_temp_variable = { rubber_needed = resource_consumed@rubber }
			multiply_temp_variable = { rubber_needed = -1 }
			add_to_temp_variable = { rubber_needed = resource@rubber }

			# deficit is >5% of total need
			set_temp_variable = { rubber_deficit = resource@rubber }
			NOT = { check_variable = { rubber_needed = 0 } }
			divide_temp_variable = { rubber_deficit = rubber_needed }
			check_variable = { rubber_deficit > 0.05 }
		}

		add_to_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_rubber_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_projects < 5 }

			NOT = { num_of_civilian_factories_available_for_projects < 20 }

			# resource shortage
			check_variable = { WA_AI_resource_rubber_shortage_months = 3 }
		}

		WA_AI_priority_queue_synthetic_refinery = yes
	}

	###

	else_if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_projects < 5 }

			NOT = { num_of_civilian_factories_available_for_projects < 20 }

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# over 50% of factories are mil
			num_of_factories > 0
			set_temp_variable = { mil_factories = num_of_military_factories }
			divide_temp_variable = { mil_factories = num_of_factories }
			check_variable = { mil_factories > 0.4 }

			# no factory slots left
			NOT = {
				any_of_scopes = { array = controlled_states
					WA_AI_available_SHARED = yes
				}
			}
		}

		WA_AI_priority_convert_MIC_to_CIC = yes
	}
}



WA_AI_build_refinery_resource_shortage_steel = {

	### track steel shortage

	if = {
		limit = {
			# there is a steel deficit / we have enough industry to invest
			OR = {
				check_variable = { resource@steel < 0 }
				AND = {
					check_variable = { resource@iron > -1 }
					check_variable = { resource_imported@steel > 0 }
				}
			}
			OR = {
				num_of_available_civilian_factories > 19
				check_variable = { resource_imported@steel > 80 }
			}
		}

		add_to_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_steel_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_projects < 5 }

			NOT = { num_of_civilian_factories_available_for_projects < 20 }

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }
		}

		# Choose between hydro and non-hydro based on coal availability
		# Prefer non-hydro if coal is abundant (saves IC cost)
		# Prefer hydro if coal is scarce (saves coal consumption)
		if = {
			limit = {
				check_variable = { resource@coal > 50 }
				check_variable = { resource_imported@coal < 20 }
			}
			WA_AI_priority_queue_steel_refinery = yes
		}
		else = {
			WA_AI_priority_queue_hydro_steel_refinery = yes
		}
	}

	###

	else_if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_projects < 5 }

			NOT = { num_of_civilian_factories_available_for_projects < 20 }

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# no factory slots left
			NOT = {
				any_of_scopes = { array = controlled_states
					WA_AI_available_SHARED = yes
				}
			}

	}
}

WA_AI_build_supply_line = {

	set_temp_variable = { pathfind_start = supply_line_start }
	set_temp_variable = { pathfind_target = supply_line_target }
	set_temp_variable = { pathfind_type = 1 }
	WA_AI_PATHFIND_get_path = yes

	if = { limit = { check_variable = { pathfind_success = 1 } }

		# Set up project parameters for infrastructure building
		set_temp_variable = { _project_building_type = 1 }
		set_temp_variable = { _project_province_id = 0 }
		set_temp_variable = { _project_priority = 100 }
		set_temp_variable = { _project_queue_num = 1 }

		### build a route to the front

		set_temp_variable = { target_id = pathfind_path_^0 }
		for_each_loop = { array = pathfind_path_

			var:v = {

				if = { limit = { WA_AI_PC_can_build_project = yes }

					var:target_id = {

						if = { limit = { check_variable = { PREV.building_level@infrastructure < THIS.building_level@infrastructure } }

							set_temp_variable = { target_id = pathfind_path_^i }
						}
					}

					set_temp_variable = { building_path_to_front = 1 }
				}
			}
		}

		if = { limit = { check_variable = { building_path_to_front = 1 } }

			var:target_id = { WA_AI_PC_start_project = yes }
		}

		### build along the front

		if = { limit = { check_variable = { building_path_to_front = 0 } }

			var:supply_line_target = {

				CONTROLLER = {

					set_temp_variable = { supply_line_target_controller = THIS.id }
				}
			}

			set_temp_variable = { front_intersection = pathfind_path_^num }
			subtract_from_temp_variable = { front_intersection = 1 }
			set_temp_variable = { front_intersection = pathfind_path_^front_intersection }

			add_to_temp_array = { front_states = front_intersection }
			add_to_temp_array = { explore_states = front_intersection }

			while_loop_effect = { limit = { check_variable = { explore_states^num > 0 } }

				clear_temp_array = neighbor_states

				for_each_scope_loop = { array = explore_states

					# max infrastructure
					if = { limit = { WA_AI_PC_can_build_project = yes }

						WA_AI_PC_start_project = yes

						set_temp_variable = { break = 1 }
					}

					# check neighbors
					else = {
						every_neighbor_state = {
							limit = {
								is_controlled_by = ROOT

								NOT = { is_in_array = { front_states = THIS.id } }

								any_neighbor_state = {

									CONTROLLER = {

										NOT = { tag = ROOT }

										OR = {
											tag = var:supply_line_target_controller
										}
									}
								}
							}

							add_to_temp_array = { front_states = THIS.id }
							add_to_temp_array = { neighbor_states = THIS.id }
						}
					}
				}

				clear_temp_array = explore_states
				for_each_loop = { array = neighbor_states break = break_2

					add_to_temp_array = { explore_states = v }
				}
			}
		}
	}
}

		every_controlled_state = {
			limit = { state = 177 } # Balearic Islands

			if = { limit = { WA_AI_available_RADAR = yes WA_AI_PC_can_build_project = yes }
				set_temp_variable = { _project_building_type = 4 }
				set_temp_variable = { _project_province_id = 0 }
				set_temp_variable = { _project_priority = 100 }
				set_temp_variable = { _project_queue_num = 1 }
				set_temp_variable = { _project_target_level = 0 }
				WA_AI_PC_start_project = yes
			}

			if = { limit = { WA_AI_available_AA = yes WA_AI_PC_can_build_project = yes }
				set_temp_variable = { _project_building_type = 3 }
				set_temp_variable = { _project_province_id = 0 }
				set_temp_variable = { _project_priority = 100 }
				set_temp_variable = { _project_queue_num = 1 }
				set_temp_variable = { _project_target_level = 0 }
				WA_AI_PC_start_project = yes
			}

			if = { limit = { WA_AI_available_AIR = yes WA_AI_PC_can_build_project = yes }
				set_temp_variable = { _project_building_type = 2 }
				set_temp_variable = { _project_province_id = 0 }
				set_temp_variable = { _project_priority = 100 }
				set_temp_variable = { _project_queue_num = 1 }
				set_temp_variable = { _project_target_level = 0 }
				WA_AI_PC_start_project = yes
			}
		}
	}

# Keep enough air base capacity in central/southern England to host the whole
# faction's aviation (RAF + USAAF + minors). Without this the allied AI parks
# its air force in America: the UK bases in range of the front fill up and the
# engine finds no free capacity worth rebasing to, so the wings never cross the
# Atlantic.
# ENG builds as host nation. USA acts as backup builder (air_base has
# allied_build = yes) whenever ENG cannot act on the deficit itself: ENG is
# player-controlled or lacks civilian factories. This strategy is called outside
# the dispatcher's active_projects/affordability gates (see the caller): ENG's
# queue is permanently railway-saturated, so a busy-queue gate would kill the
# host-nation branch for the whole war.
# ROOT = ENG or USA. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only).
WA_AI_build_uk_air_hosting_capacity = {

	@UK_AIR_CAPACITY_PER_LEVEL = 100	# NDefines.NBuildings.AIRBASE_CAPACITY_MULT (05_defines.lua)
	@UK_AIR_TARGET_LEVEL_CAP = 12		# per-state cap, leaves slots for other buildings
	@UK_AIR_HOSTED_FACTION_SHARE = 0.5	# share of the faction's planes the UK should be able to host

	if = {
		limit = {
			has_war = yes
			any_enemy_country = { capital_scope = { is_on_continent = europe } }
			num_of_civilian_factories_available_for_projects > 15
			OR = {
				original_tag = ENG
				AND = {
					original_tag = USA
					is_in_faction_with = ENG
					# only step in when ENG cannot act on the deficit itself
					ENG = {
						OR = {
							is_ai = no
							num_of_civilian_factories_available_for_projects < 16
						}
					}
				}
			}
		}

		### how many planes the faction may want to base in the UK
		# num_deployed_planes cannot be read as a set_temp_variable rvalue (campaign
		# 5709c8b9: every other gate provably passed and the deficit branch never
		# fired). The per-airframe form num_deployed_planes_with_type@X is script-
		# readable (mission_calculations_effects.txt relies on it), so sum the
		# land-based airframes via WA_AI_PC_add_deployed_land_planes.

		set_temp_variable = { _faction_planes_ = 0 }
		WA_AI_PC_add_deployed_land_planes = yes
		for_each_scope_loop = { array = allies
			WA_AI_PC_add_deployed_land_planes = yes
		}

		set_temp_variable = { _uk_target_capacity_ = _faction_planes_ }
		multiply_temp_variable = { _uk_target_capacity_ = @UK_AIR_HOSTED_FACTION_SHARE }

		### capacity already built or queued in the hosting states

		set_temp_variable = { _uk_hosting_capacity_ = 0 }
		every_state = {
			limit = {
				WA_AI_uk_air_hosting_state = yes
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = { is_in_faction_with = ROOT }
				}
			}
			add_to_temp_variable = { _uk_hosting_capacity_ = building_level@air_base }
		}
		for_each_loop = { array = WA_AI_PC_queue value = _uk_air_proj_id_
			if = { limit = { check_variable = { WA_AI_PC_building_type^_uk_air_proj_id_ = 2 } }
				add_to_temp_variable = { _uk_hosting_capacity_ = 1 }
			}
		}
		multiply_temp_variable = { _uk_hosting_capacity_ = @UK_AIR_CAPACITY_PER_LEVEL }

		if = { limit = { check_variable = { _uk_target_capacity_ > _uk_hosting_capacity_ } }

			### expand the least-developed hosting state first

			set_temp_variable = { _project_building_type = 2 }
			set_temp_variable = { _uk_air_best_level_ = @UK_AIR_TARGET_LEVEL_CAP }
			set_temp_variable = { _uk_air_best_state_ = -1 }
			every_state = {
				limit = {
					WA_AI_uk_air_hosting_state = yes
					check_variable = { building_level@air_base < @UK_AIR_TARGET_LEVEL_CAP }
					OR = {
						AND = {
							is_controlled_by = ROOT
							WA_AI_available_AIR = yes
							WA_AI_PC_can_build_project = yes
						}
						AND = {
							NOT = { is_controlled_by = ROOT }
							WA_AI_available_AIR_allied = yes
							NOT = { WA_AI_PC_has_same_type_project_here = yes }
						}
					}
				}
				if = { limit = { check_variable = { building_level@air_base < _uk_air_best_level_ } }
					set_temp_variable = { _uk_air_best_level_ = building_level@air_base }
					set_temp_variable = { _uk_air_best_state_ = THIS.id }
				}
			}

			if = { limit = { check_variable = { _uk_air_best_state_ > -1 } }
				var:_uk_air_best_state_ = {
					set_temp_variable = { _project_building_type = 2 }
					set_temp_variable = { _project_province_id = 0 }
					set_temp_variable = { _project_target_state = 0 }
					set_temp_variable = { _project_priority = 100 }
					set_temp_variable = { _project_queue_num = 1 }
					set_temp_variable = { _project_queue_max = 3 }
					set_temp_variable = { _project_target_level = @UK_AIR_TARGET_LEVEL_CAP }
					WA_AI_PC_start_project = yes
				}
			}
		}
	}
}
