WA_AI_build_refinery_resource_shortage_rubber = {

	### track rubber shortage

	if = {
		limit = {
			# there is a rubber deficit
			check_variable = { resource@rubber < 0 }

			# total rubber needed for production
			set_temp_variable = { rubber_needed = resource_consumed@rubber }
			multiply_temp_variable = { rubber_needed = -1 }
			add_to_temp_variable = { rubber_needed = resource@rubber }

			# deficit is >5% of total need
			set_temp_variable = { rubber_deficit = resource@rubber }
			NOT = { check_variable = { rubber_needed = 0 } }
			divide_temp_variable = { rubber_deficit = rubber_needed }
			check_variable = { rubber_deficit > 0.05 }
		}

		add_to_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_rubber_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# Fix 40: the dispatcher's WA_AI_PC_can_afford_project used to carry these two
			# for the whole refinery branch; this strategy now runs outside that gate, so
			# it states them itself (same guards the air strategies carry).
			has_capitulated = no
			surrender_progress < 0.3

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_rubber_shortage_months = 3 }

			# Synthetic refineries consume 60 coal/level and, unlike steel/aluminium, have no
			# hydro variant to fall back on: only build when coal is comfortable, otherwise the
			# refinery would feed the coal/steel shortage it competes with. The shortage counter
			# above still ticks either way - it also drives rubber prospecting decisions.
			check_variable = { resource@coal > 50 }
			check_variable = { resource_imported@coal < 20 }
		}

		WA_AI_priority_queue_synthetic_refinery = yes
	}

	# Fix 76 (campaign be18f9c7): a PROACTIVE pre-build else_if was prototyped here as a
	# second host for WA_AI_CONSTRUCTION_should_pre_build_synth_rubber and REMOVED before
	# shipping - it had no measurable beneficiary. Recorded so the same idea is not
	# re-proposed on the same wrong premise.
	#
	# The premise was "ENG's C.10 is silent (checklist R24), so ENG needs a priority-
	# construction host for the pre-build lane". The campaign refutes it. That trigger
	# needs rubber IMPORTS above its floor while the balance is still non-negative, and
	# ENG is never in that state at any sampled date: it owns Malaya, so it imports
	# ZERO rubber across the whole 1936.6-1941.6 positive-balance stretch, and every
	# date on which it does import (1942.6+, 106-210) already carries a negative balance
	# and belongs to the reactive branch above. GER never qualifies either - its rubber
	# imports are 0 from 1938 on, because it is pre-built BY HAND at startup (the
	# original_tag = GER block in events/WA_AI_construction.txt adds 22 refinery levels,
	# park 4 -> 36 by 1940.6). ITA clears the rubber terms three times and is correctly
	# rejected by the coal guard every time - it imports 700-1600 coal. The only
	# qualifying tags in the campaign are USA, JAP and SOV: USA reaches C.10, and JAP
	# and SOV already raise healthy synthetic parks unaided (0 -> 4 and 13 -> 55).
	#
	# The lane therefore lives at ONE site, the WA_AI_C.10 refinery block in
	# events/WA_AI_construction.txt. If a future campaign shows a QUALIFYING country
	# whose C.10 is silent, re-adding an else_if here is the fix - but bring the
	# measurement, and note that this strategy's own `active_nonrail < 5` gate is shut
	# on exactly the big saturated-queue countries such an argument would invoke (the
	# USA's queue held 5 / 8 / 6 non-rail projects, mostly radar, at the three dates
	# sampled).
}



WA_AI_build_refinery_resource_shortage_steel = {

	### track steel shortage

	if = {
		limit = {
			# there is a steel deficit / we have enough industry to invest
			OR = {
				check_variable = { resource@steel < 0 }
				AND = {
					check_variable = { resource@iron > -1 }
					check_variable = { resource_imported@steel > 0 }
				}
			}
			OR = {
				num_of_available_civilian_factories > 19
				check_variable = { resource_imported@steel > 80 }
			}
		}

		add_to_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_steel_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# Fix 40: guards inherited from the dispatcher's WA_AI_PC_can_afford_project,
			# which this strategy no longer runs behind.
			has_capitulated = no
			surrender_progress < 0.3

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# Fix 40 - input self-sufficiency, same form as the synthetic branch's coal
			# test: a steel mill converts IRON into steel, it does not create iron. A
			# country running an iron deficit, or covering its iron from imports the same
			# blockade/convoy war is already strangling, converts nothing and has simply
			# spent 10-12.5k IC to move the shortage one link up the chain - importing the
			# finished steel is strictly cheaper. Domestic iron non-negative and imports
			# below 40 (the aluminium branch below uses the identical shape on bauxite).
			check_variable = { resource@iron > -1 }
			check_variable = { resource_imported@iron < 40 }
		}

		# Choose between hydro and non-hydro based on coal availability
		# Prefer non-hydro if coal is abundant (saves IC cost)
		# Prefer hydro if coal is scarce (saves coal consumption)
		if = {
			limit = {
				check_variable = { resource@coal > 50 }
				check_variable = { resource_imported@coal < 20 }
			}
			WA_AI_priority_queue_steel_refinery = yes
		}
		else = {
			WA_AI_priority_queue_hydro_steel_refinery = yes
		}
	}

	###

	else_if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# Fix 40: the MIC->CIC conversion leg keeps its ORIGINAL reachability. Only the
			# refinery branch above was shown to be wrongly starved by the dispatcher gate;
			# restating can_afford here means moving the strategy out of the dispatcher
			# cannot widen industry conversion as a side effect.
			WA_AI_PC_can_afford_project = yes

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# over 50% of factories are mil
			num_of_factories > 0
			set_temp_variable = { mil_factories = num_of_military_factories }
			divide_temp_variable = { mil_factories = num_of_factories }
			check_variable = { mil_factories > 0.4 }

			# no factory slots left
			NOT = {
				any_of_scopes = { array = controlled_states
					WA_AI_available_SHARED = yes
				}
			}
		}

		WA_AI_priority_convert_MIC_to_CIC = yes
	}
}

WA_AI_build_supply_line_JAP_RAJ = {

	if = {
		limit = {
			tag = JAP
			has_war_with = RAJ
		}

		# Fix 30 (R9): the old hardcoded start 613 (N. Shanghai) could never reach Delhi 439
		# within the A*'s 75-iteration cap. Dynamic anchor instead: BFS outward from the
		# target and take the nearest ROOT-or-subject-controlled state on the SAME landmass
		# (BFS waves are emitted nearest-first; every_neighbor_state can cross straits, the
		# landmass check filters those hits). No anchor found = JAP holds no ground near the
		# RAJ front = no supply line, which is the correct null behaviour.
		set_temp_variable = { _origin_state_id = 439 } # delhi
		set_temp_variable = { _max_distance = 10 }
		WA_AI_PC_railway_get_states_within_distance = yes # >>> states_within_distance_

		set_temp_variable = { _sl_target_state_ = 439 }
		set_temp_variable = { _sl_target_landmass_ = global.WA_AI_MAP_state_landmass^_sl_target_state_ }
		set_temp_variable = { _sl_anchor_found_ = 0 }
		for_each_loop = { array = states_within_distance_ value = _sl_cand_ break = _sl_anchor_found_
			if = {
				limit = { check_variable = { global.WA_AI_MAP_state_landmass^_sl_cand_ = _sl_target_landmass_ } }
				var:_sl_cand_ = {
					if = {
						limit = {
							CONTROLLER = {
								OR = {
									tag = ROOT
									is_subject_of = ROOT
								}
							}
						}
						set_temp_variable = { supply_line_start = THIS.id }
						set_temp_variable = { _sl_anchor_found_ = 1 }
					}
				}
			}
		}

		if = { limit = { check_variable = { _sl_anchor_found_ = 1 } }
			set_temp_variable = { supply_line_target = _sl_target_state_ }
			WA_AI_build_supply_line = yes
		}
	}
}

WA_AI_build_supply_line = {

	# Fix 30 break hygiene: earlier strategies in the same pulse (refinery/AA queue
	# functions, scoring loops) leave the shared `break` temp variable at 1; the
	# default-break loops below (path walk, front exploration) would then abort before
	# their first iteration. The pathfinder resets it too, but reset here so THIS effect
	# is safe regardless of call order.
	set_temp_variable = { break = 0 }
	set_temp_variable = { building_path_to_front = 0 }

	# Fix 30 instrumentation (campaign 66d6b53c: zero supply-line projects all war, and
	# resource-INF shares building type 1, so saves could not tell the two apart).
	# Persistent, save-visible (lowercased in saves); remove once the system is verified.
	#   WA_AI_supply_line_dbg_called      pulses that reached this effect
	#   WA_AI_supply_line_dbg_pf_ok       pulses whose A* returned success
	#   WA_AI_supply_line_dbg_queued      infrastructure projects actually queued here
	#   WA_AI_supply_line_dbg_last_state  state id of the last queued project
	add_to_variable = { WA_AI_supply_line_dbg_called = 1 }

	# Fix: these three inputs were passed WITHOUT the underscore prefix the pathfinder
	# reads (_pathfind_start/_pathfind_target/_pathfind_type in
	# WA_AI_pathfinding_effects.txt), so the A* ran from an unset scope and this whole
	# effect - and all three strategies built on it (North Africa infrastructure,
	# GER->SOV and JAP->RAJ supply lines) - has been silent dead code since authoring:
	# zero type-1 supply-line projects in 8 in-game years of campaign 9be92c89.
	# (Outputs pathfind_success / pathfind_path_ carry no prefix and were correct.)
	set_temp_variable = { _pathfind_start = supply_line_start }
	set_temp_variable = { _pathfind_target = supply_line_target }
	set_temp_variable = { _pathfind_type = 1 }
	WA_AI_PATHFIND_get_path = yes

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | SUPPLY LINE: [?supply_line_start] -> [?supply_line_target] success=[?pathfind_success]"
	}

	if = { limit = { check_variable = { pathfind_success = 1 } }

		add_to_variable = { WA_AI_supply_line_dbg_pf_ok = 1 }

		# Set up project parameters for infrastructure building
		set_temp_variable = { _project_building_type = 1 }
		set_temp_variable = { _project_province_id = 0 }
		set_temp_variable = { _project_priority = 100 }
		set_temp_variable = { _project_queue_num = 1 }

		### build a route to the front

		set_temp_variable = { target_id = pathfind_path_^0 }
		for_each_loop = { array = pathfind_path_

			var:v = {

				if = { limit = { WA_AI_PC_can_build_project = yes }

					var:target_id = {

						if = { limit = { check_variable = { PREV.building_level@infrastructure < THIS.building_level@infrastructure } }

							set_temp_variable = { target_id = pathfind_path_^i }
						}
					}

					set_temp_variable = { building_path_to_front = 1 }
				}
			}
		}

		if = { limit = { check_variable = { building_path_to_front = 1 } }

			set_temp_variable = { _sl_before_ = WA_AI_PC_active_projects }
			var:target_id = { WA_AI_PC_start_project = yes }
			# start_project may no-op (duplicate, level cap) - count only real queue growth
			if = { limit = { check_variable = { WA_AI_PC_active_projects > _sl_before_ } }
				add_to_variable = { WA_AI_supply_line_dbg_queued = 1 }
				set_variable = { WA_AI_supply_line_dbg_last_state = target_id }
			}
		}

		### build along the front

		if = { limit = { check_variable = { building_path_to_front = 0 } }

			var:supply_line_target = {

				CONTROLLER = {

					set_temp_variable = { supply_line_target_controller = THIS.id }
				}
			}

			set_temp_variable = { front_intersection = pathfind_path_^num }
			subtract_from_temp_variable = { front_intersection = 1 }
			set_temp_variable = { front_intersection = pathfind_path_^front_intersection }

			add_to_temp_array = { front_states = front_intersection }
			add_to_temp_array = { explore_states = front_intersection }

			while_loop_effect = { limit = { check_variable = { explore_states^num > 0 } }

				clear_temp_array = neighbor_states

				for_each_scope_loop = { array = explore_states

					# max infrastructure
					if = { limit = { WA_AI_PC_can_build_project = yes }

						set_temp_variable = { _sl_before_ = ROOT.WA_AI_PC_active_projects }
						WA_AI_PC_start_project = yes
						if = { limit = { check_variable = { ROOT.WA_AI_PC_active_projects > _sl_before_ } }
							# Plain state id via the map-data primitive - a scope-hopped PREV.id
							# stores the engine-encoded reference, not the numeric id (R1 lesson).
							# ROOT.-prefixed targets so the RHS temps stay readable in state scope.
							WA_AI_PC_get_state_id = yes # >>> state_id_
							add_to_variable = { ROOT.WA_AI_supply_line_dbg_queued = 1 }
							set_variable = { ROOT.WA_AI_supply_line_dbg_last_state = state_id_ }
						}

						set_temp_variable = { break = 1 }
					}

					# check neighbors
					else = {
						every_neighbor_state = {
							limit = {
								# Fix 30: also walk the front through subject-controlled states
								# (WA_AI_PC_can_build_project already accepts them; a ROOT-only
								# walk filtered the whole UKE/ITL corridor back out)
								OR = {
									is_controlled_by = ROOT
									CONTROLLER = { is_subject_of = ROOT }
								}

								NOT = { is_in_array = { front_states = THIS.id } }

								any_neighbor_state = {

									CONTROLLER = {

										NOT = { tag = ROOT }

										OR = {
											tag = var:supply_line_target_controller
										}
									}
								}
							}

							add_to_temp_array = { front_states = THIS.id }
							add_to_temp_array = { neighbor_states = THIS.id }
						}
					}
				}

				clear_temp_array = explore_states
				for_each_loop = { array = neighbor_states break = break_2

					add_to_temp_array = { explore_states = v }
				}
			}
		}
	}

# Fix 51: queue ONE air-base level in this state. Extracted from the old level ladder
# so the ordered south-first walk and its unordered fallback tail go through a single
# code path and cannot diverge.
#
# THIS = state (already passed WA_AI_uk_air_hosting_state + WA_AI_uk_air_site_buildable
# + the level cap), PREV = ROOT = builder country.
# Inputs, all set by the caller in country scope:
#   _uk_air_prio_        priority band, resolved once per pulse (Fix 47)
#   _uk_air_level_cap_   @UK_AIR_TARGET_LEVEL_CAP, passed in because that constant is
#                        declared inside WA_AI_build_uk_air_hosting_capacity's body
#   _uk_air_rank_        1-11 = south-first rank of this state, 0 = unranked fallback

WA_AI_uk_air_queue_site = {

	# consumed the UK allowance - on 911bed3c both of ENG's slots held Pacific
	# projects on the very pulse the UK deficit reopened.
	set_temp_variable = { _project_type_id = @AI_PC_TYPE_ID_UK_AIR }
	set_temp_variable = { _project_queue_max_scoped = 1 }
	set_temp_variable = { _project_target_level = _uk_air_level_cap_ }

	# retired here instead. Without this, the next type-2 caller that does not
	# set a type_id - WA_AI_fortify_strategic_islands is one - would inherit
	# @AI_PC_TYPE_ID_UK_AIR and be counted against the UK budget.
	set_temp_variable = { _project_type_id = 0 }

	ROOT = {
		# _uk_air_started_ latches on SITE FOUND, not on queue success: if
		# WA_AI_PC_start_project refused (queue max, duplicate), the next candidate
		# would be refused for the same reason, so the walk still stops.
		set_temp_variable = { _uk_air_started_ = 1 }
		set_variable = { WA_AI_uk_air_dbg_best = _uk_air_site_level_ }

		# Fix 46: gate the counters on the queue actually growing. The old
		# unconditional increment counted eligible SITES FOUND, not projects
		# queued, and over-reported by ~4x (campaign 9a4cd657 1944.7: ENG 97 +
		# USA 39 "starts" against 34 air-base levels delivered UK-wide). Same
		# form as WA_AI_thair_dbg_started in WA_AI_build_theatre_air_bases, which
		# has always been gated this way - the two counters are now comparable.

		if = {
			limit = { check_variable = { WA_AI_PC_queue^num > _uk_air_qnum_before_ } }
			add_to_variable = { WA_AI_uk_air_dbg_started = 1 }

			# tlm: WA_TLM_r8_uk_air_q_n / _rank_sum / _rank_max / _fallback_n / _last_t
			# consumer: checklist R8 (concentration leg). Retire with R8.
			add_to_variable = { WA_TLM_r8_uk_air_q_n = 1 }
			set_variable = { WA_TLM_r8_uk_air_last_t = global.WA_TLM_clock }
			if = {
				limit = { check_variable = { _uk_air_rank_ > 0 } }
				add_to_variable = { WA_TLM_r8_uk_air_rank_sum = _uk_air_rank_ }
				if = {
					limit = { check_variable = { _uk_air_rank_ > WA_TLM_r8_uk_air_rank_max } }
					set_variable = { WA_TLM_r8_uk_air_rank_max = _uk_air_rank_ }
				}
			}
			# explicit inverse rather than `else`: the branch above ends in a nested `if`,
			# and an `else` placed after a trailing nested `if` is exactly the binding trap
			# recorded in wa-lessons-learned (the marshal-recruit else).
			if = {
				# rank 0 = queued by the unranked fallback tail, so it is deliberately
				# kept out of _rank_sum / _rank_max (see the registry row for the
				# mean-rank denominator).
				limit = { NOT = { check_variable = { _uk_air_rank_ > 0 } } }
				add_to_variable = { WA_TLM_r8_uk_air_fallback_n = 1 }
			}
		}
	}

# Keep enough air base capacity in central/southern England to host the whole
# faction's aviation (RAF + USAAF + minors). Without this the allied AI parks
# its air force in America: the UK bases in range of the front fill up and the
# engine finds no free capacity worth rebasing to, so the wings never cross the
# Atlantic.
# ENG builds as host nation. USA acts as backup builder (air_base has
# allied_build = yes) whenever ENG cannot act on the deficit itself: ENG is
# player-controlled or lacks civilian factories. This strategy is called outside
# the dispatcher's active_projects/affordability gates (see the caller): ENG's
# queue is permanently railway-saturated, so a busy-queue gate would kill the
# host-nation branch for the whole war.
# ROOT = ENG or USA. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only).
WA_AI_build_uk_air_hosting_capacity = {

	@UK_AIR_CAPACITY_PER_LEVEL = 100	# NDefines.NBuildings.AIRBASE_CAPACITY_MULT (05_defines.lua)
	@UK_AIR_TARGET_LEVEL_CAP = 12		# per-state cap, leaves slots for other buildings
	# Fix 47: 0.5 -> 0.65. Measured on campaign 9a4cd657 (1944.7): 20,503 faction planes
	# against 7,100 capacity - 71 levels held-or-queued across the 11 hosting states, where
	# the old 0.5 asked for 103. The strategy was correctly reporting a 32-level shortfall
	# and still could not host the RAF+USAAF, because half the faction's aviation is the
	# wrong target once the whole Western Front stages out of Britain. 0.65 asks for 133.
	#
	# 0.65 -> 1.0 (2026-08-14, campaign f9321934). 0.65 STILL stood the strategy down
	# 14 months before D-Day. Measured at 1944.6.1: capacity 7,000 (70 levels) against
	# 10,021 faction planes, so the 0.65 target of 6,514 was already satisfied - dbg_best
	# read -2 ("no deficit") continuously from 1942.7, and the last UK air project of the
	# entire campaign started 1943.4. The build stopped at 70 of the 132-level ceiling
	# (53%) with 30% of Allied aviation unhoused. Any share < 1.0 declares victory while
	# part of the faction's air force still has nowhere in Britain to stand; 1.0 is the
	# only value that means what the strategy's own header says it means.
	#
	# STRUCTURAL CEILING: 11 hosting states x @UK_AIR_TARGET_LEVEL_CAP = 132 levels
	# (13,200 capacity), so the ceiling and not this share is what ultimately bounds the
	# build - it still cannot run away. At 1944 plane counts (10,021) the 1.0 target is
	# REACHABLE, which 0.65 was not asking for; beyond ~13,200 faction planes the deficit
	# becomes permanent by construction.
	#
	# ACCEPTED COST, made permanent by this change: air bases hold the 300 band (see the
	# band table in ..._core.txt) for as long as the deficit stands, so from the point the
	# faction passes ~13,200 planes ENG's shortage refineries run on the Fix 41 overtake
	# lane instead of the sorted fill, permanently rather than temporarily. On f9321934
	# that threshold falls in 1945. If the refineries turn out to need their band back more
	# than the RAF needs the last levels, lower this - do NOT raise @UK_AIR_TARGET_LEVEL_CAP
	# or extend the state list, which checklist R8 forbids until R15 justifies more capacity.

	# WHAT THIS CANNOT FIX, stated up front: at 1944.6.1 the island already held 10,100

	@UK_AIR_HOSTED_FACTION_SHARE = 1.0	# share of the faction's planes the UK should be able to host
	# Fix 47: civs/pass a host must put on air bases (2 full 20-civ slots) before an allied

	# DIAGNOSIS INSTRUMENTATION (campaign 0e7e7852: the strategy produced zero

	if = {

	if = {
		limit = {
			has_war = yes
			any_enemy_country = { capital_scope = { is_on_continent = europe } }
			# floored availability: the bare engine variable reads ~0 once the vanilla

			WA_AI_PC_has_project_civs_16 = yes
			OR = {
				original_tag = ENG
				AND = {
					original_tag = USA

					# Fix 47: NO host-condition gate. The USA co-builds until the hosting
					# target is met, whatever ENG's economy is doing.
					#
					# History of this gate, because the shape keeps failing the same way.

					# "ENG assigns < 1 civ to air". Campaign 9a4cd657 then showed that closing
					# too: ENG sat at exactly 1 air civ in 1944.7 with the faction 32 levels
					# short, so the USA - which supplied 585 of the 874 air civ-months measured
					# over 1942-44 - was locked out by a single factory. Promoting air bases to
					# the 300 band would have made that permanent, since ENG now almost always
					# assigns something. A rate test (>= 2 full slots) was the next iteration
					# and is still the wrong shape: any host-spending threshold makes the
					# ALLY's participation a function of the HOST's budget rather than of the
					# shortfall the ally exists to cover.
					#
					# What makes dropping it safe now is Fix 46: WA_AI_PC_has_same_type_project_here
					# became builder-agnostic, so ENG and USA can no longer queue the same state
					# (they demonstrably did, in state 126 and state 860 simultaneously). With
					# collisions impossible, two builders working the same 11-state set is pure
					# added throughput. The deficit branch below (_uk_target_capacity_ >
					# _uk_hosting_capacity_) is the only stop condition either builder needs -
					# it closes for BOTH of them the moment the target is met.
					#
					# WA_AI_build_theatre_air_bases deliberately KEEPS its host-funding gate
					# (@AI_PC_AIR_HOST_FUNDING_FLOOR): it spans every faction member across ten
					# theatres, where an ungated ally would pile into every partner's contested
					# theatre at once. This strategy is a two-builder, eleven-state special case
					# with a known 60-level shortfall, so it does not need that brake.
				}
			}
		}

		### how many planes the faction may want to base in the UK
		# num_deployed_planes cannot be read as a set_temp_variable rvalue (campaign
		# 5709c8b9: every other gate provably passed and the deficit branch never
		# fired). The per-airframe form num_deployed_planes_with_type@X is script-
		# readable (mission_calculations_effects.txt relies on it). The sums are
		# INLINED deliberately - a helper effect is a newly registered name, which a
		# console reload cannot pick up and which added an unprovable failure class
		# to the 0e7e7852 test round; inline statements carry no such risk.

		set_temp_variable = { _faction_planes_ = 0 }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }

		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
		for_each_scope_loop = { array = allies
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }

		set_temp_variable = { _uk_target_capacity_ = _faction_planes_ }

		### capacity already built or queued in the hosting states

		set_temp_variable = { _uk_hosting_capacity_ = 0 }
		every_state = {
			limit = {
				WA_AI_uk_air_hosting_state = yes

					is_controlled_by = ROOT

				limit = {

			# month and never once hit the 0.30 floor. ENG was not civ-poor - it was holding

			#
			# WA_AI_PC_override_max_factories_factor is the existing per-country escape hatch

			# alternate on a last-writer-wins timer.

			# Self-releasing: the 30-day flag is only refreshed while this deficit branch

			if = {
				limit = { original_tag = ENG }
				set_country_flag = { flag = WA_AI_PC_override_max_factories_factor value = 1 days = 30 }
				set_variable = { WA_AI_PC_override_max_factories_factor = 0.6 }
			}

			### Fix 51: fill the SOUTH-EAST first, only move north once it is capped
			#
			# What this replaces and why. The selector used to be a LEVEL ladder: walk
			# _uk_air_try_level_ from 0 upward and take a random hosting state sitting at
			# exactly that level. That is "expand the least-developed state first" - a
			# LEVELLING rule - and it did exactly what it said. Campaign f9321934 (HEAD
			# 289f1318f) at 1946.8: 98 levels spread across 11 states, ~8.9 each against a
			# cap of 12, and the two largest ENG plane concentrations were Gloucestershire
			# (827) and Warwickshire (460) - the two states FURTHEST from France. A wing at
			# Kent covers Northern France; the same wing at Leicestershire does not.
			#
			# This does NOT add capacity. The ceiling is 11 x @UK_AIR_TARGET_LEVEL_CAP = 132
			# levels either way and the deficit is a production race this change does not
			# win (f9321934: 9 800 capacity vs 15 174 faction planes at 1946.8). What it buys
			# is RANGE and mission efficiency for whatever capacity does get built.
			#
			# The level ladder's own reason for existing is preserved. It was written
			# (campaign 9be92c89) to replace a two-pass select that stored THIS.id from STATE
			# scope and compared it to a -1 sentinel in country scope - the stored value is
			# scope-encoded, the guard never passed, and the strategy queued nothing for the
			# entire war. Fifth occurrence of the state-scope temp trap (AIFC, railway Fixes
			# 24/25/27 lineage). Nothing below stores or compares a scope id: the walk is
			# over PLAIN ids from a global array, entered with `var:`, and the project starts
			# directly in state scope per WA_AI_PC_start_project's contract
			# (THIS = state, PREV = builder country) - the same idiom as railway_core's
			# `var:_state_for_project = { WA_AI_PC_start_railway_project = yes }`.
			#
			# Why a top-priority state cannot stall the walk: the moment ANY builder queues
			# an air base in it, WA_AI_PC_has_same_type_project_here (builder-agnostic since
			# Fix 46) makes it fail WA_AI_uk_air_site_buildable, so the next pulse falls
			# through to the next rank. Southern states also run out of BUILDING SLOTS -
			# they hold ENG's arms factories - and the free-slot terms inside that same
			# trigger fall through identically. The walk never stalls on a full state.

			set_temp_variable = { _project_building_type = 2 }
			set_temp_variable = { _uk_air_started_ = 0 }
			# passed to WA_AI_uk_air_queue_site: @ constants are file-scoped AND this one
			# is declared inside this effect's body, so the helper cannot name it directly
			set_temp_variable = { _uk_air_level_cap_ = @UK_AIR_TARGET_LEVEL_CAP }

			# Fix 47: band selection, resolved ONCE in country scope - the trigger is a
			# faction scan and must not run per candidate state inside the walk below.
			set_temp_variable = { _uk_air_prio_ = @AI_PC_PRIO_DEFAULT }
			if = {
				limit = { WA_AI_PC_air_force_needs_basing_priority = yes }
				set_temp_variable = { _uk_air_prio_ = @AI_PC_PRIO_AIR_BASING }
			}

			# Fix 51: build the ordered list once per game. check_variable reads an absent
			# global as 0, so a fresh start and a campaign resumed from a pre-Fix-51 build
			# both take this branch; a later order change takes it again via the version.
			if = {
				limit = { NOT = { check_variable = { global.WA_AI_uk_air_priority_version = @UK_AIR_PRIORITY_VERSION } } }
				WA_AI_uk_air_hosting_priority_init = yes
			}

			set_temp_variable = { _uk_air_rank_ = 0 }
			for_each_loop = {
				array = global.WA_AI_uk_air_priority_states
				value = _uk_air_sid_
				# named break, never the shared default `break` temp - queue functions set
				# plain `break = 1` and never reset it (Fix 30 lesson)
				break = _uk_air_started_

				add_to_temp_variable = { _uk_air_rank_ = 1 }
				# Fix 46: queue depth before the attempt, so dbg_started below can count
				# projects actually QUEUED rather than sites merely found (see the counter).
				set_temp_variable = { _uk_air_qnum_before_ = WA_AI_PC_queue^num }

				var:_uk_air_sid_ = {
					if = {
						limit = {
							# re-tested per candidate: the trigger, not this array, owns
							# membership, so an id dropped from the trigger is filtered out here
							WA_AI_uk_air_hosting_state = yes
							check_variable = { building_level@air_base < @UK_AIR_TARGET_LEVEL_CAP }
							WA_AI_uk_air_site_buildable = yes
						}
						WA_AI_uk_air_queue_site = yes
					}
				}
			}

			# Fix 51: fallback tail - unordered scan over the SAME membership trigger,
			# reached only when the ordered walk queued nothing. This is what makes the
			# duplicated list safe: an id present in WA_AI_uk_air_hosting_state but missing
			# from the array loses its rank and gets built last, instead of never being
			# built at all. In a consistent build it fires only when every ranked state is
			# capped or unbuildable, in which case it also finds nothing - one wasted
			# random_state scan on a pulse that was going to do nothing anyway.
			# WA_TLM_r8_uk_air_fallback_n counts it, so drift is visible in the next campaign.
			if = {
				limit = { check_variable = { _uk_air_started_ < 1 } }
				set_temp_variable = { _uk_air_rank_ = 0 }	# 0 = unranked, see the TLM rows
				set_temp_variable = { _uk_air_qnum_before_ = WA_AI_PC_queue^num }
				random_state = {
					limit = {
						WA_AI_uk_air_hosting_state = yes
						check_variable = { building_level@air_base < @UK_AIR_TARGET_LEVEL_CAP }
						WA_AI_uk_air_site_buildable = yes
					}
					WA_AI_uk_air_queue_site = yes
				}
			}
		}
	}
}

# Build/upgrade air bases inside air theatres that are CONTESTED for ROOT, so the
# strategic_air_importance pulls in common/ai_strategy/WA_AI_MILITARY_DEFAULT_AIR_theatres.txt
# have friendly capacity to rebase into - the engine only rebases wings into in-range friendly
# air-base capacity, so a pull toward a theatre with no basing is inert (that file's KNOWN
# LIMITS names this effect as the follow-up). Campaign 66d6b53c: the USAAF kept 8,595 of
# 13,791 planes in the continental US - no basing existed along the Pacific island chain.
# Theatre membership comes from the WA_AI_MILITARY_AIR_theatre_state_* triggers
# (common/scripted_triggers/WA_AI_MILITARY_triggers.txt), which must-match the contested lists.
#
# NEED GUARDRAILS (all generic, no tags):
# 1. Theatre-scoped sites: projects can only start in theatre member states - this effect can
#    never build on home soil outside a contested theatre (a Canada at peace-depth in its own
#    hemisphere is structurally unreachable).
# 2. Side-wide capacity deficit: a contested theatre only qualifies while the SIDE's existing
#    + ROOT-queued air-base capacity across that theatre's states is below
#    @THEATRE_AIR_HOSTED_SHARE of the side's deployed planes. Capacity counts every state
#    controlled by ROOT, a faction ally or a co-belligerent (same own-side definition as the
#    contested triggers), so once anyone on the side has built enough, every other member
#    stops queuing - a minor never piles onto a theatre its majors already covered, and a
#    co-belligerent outside the faction (e.g. a 1945 SOV vs JAP) sees the US capacity and
#    stays idle.
# 3. Token air forces never enter (<= 99 deployed planes, WA air scale ~1/3 vanilla): a
#    country with no meaningful aviation has no basing need of its own.
# 4. Per-state cap: the level ladder stops selecting a state at @THEATRE_AIR_TARGET_LEVEL,
#    and always raises the least-developed member state first, so capacity spreads 0->4
#    across the chain instead of stacking one level-10 island.
#
# FIX 85 (2026-08-15, campaign af003548) - GROUND-COMMITTED THEATRES FIRST.
#    Measured after D-Day: Western Europe was contested and in deficit for USA and ENG at
#    every save 1944.7-1945.9 (allied-held capacity 21 levels vs a 26-36 level target), the
#    builder ran every pulse (dbg_called +4-5/month, dbg_active 4-6) - and dbg_started FROZE
#    for both majors for 8 months (USA 38, ENG 25). USA never queued a single air base in
#    France (its theatre picks were 163 Balkans, 1002 New Guinea, 1033 Indochina); ENG queued
#    two, states 24 and 785 in 1944.7, both level 0, both starved at 0 factories for five
#    months behind level-8->12 UK bases in the same 300 band, then purged. Net capacity on
#    Allied-controlled French soil: 700 planes (15 at level 4, 1016 at level 3). The USAAF's
#    9 000 land-based planes stayed in the continental US, 92-94% without a mission.
#    Two causes, both in this effect: (a) random_state draws uniformly over EVERY deficit
#    theatre's states from ladder rung 0, so a landing theatre whose only allied states are
#    already at level 3-4 loses the draw to Pacific/Balkan level-0 islands every pulse; (b) a
#    theatre_air project shares the 300 band with uk_air and, sorted behind it, never gets a
#    factory. The remedy keeps every guardrail and adds one discriminator that needs no
#    theatre list and no tag: does ROOT have divisions standing in the theatre?
#      - COMMITTED theatre = a deficit theatre with >= 1 ROOT division on a side-controlled
#        member state (folded into the pass-2 state walk, so no extra scan). While any
#        committed theatre is in deficit, the ladder draws ONLY from committed theatres.
#      - The per-state cap rises to @THEATRE_AIR_TARGET_LEVEL_COMMITTED there: a beachhead
#        of two states must host what ten islands would spread.
#      - Priority @AI_PC_PRIO_AIR_FRONT (350): above uk_air (300) inside the Fix 78 lane and
#        the sorted fill, below rail-prewar (500), so the Fix 41 band order is untouched.
#    Nothing changes for a country with no divisions in any deficit theatre (the Pacific
#    island-chain case this effect was written for keeps its uniform spread).
#    Probe: WA_TLM_r52_thair_front_* (checklist R52).
#
# ROOT = any AI country. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only), outside the dispatcher's
# active_projects/affordability gates for the same reason as the UK hosting strategy above
# (railway-saturated queues would permanently close the < 5 gate for exactly the countries
# this exists for); bounded instead by _project_queue_max, the deficit gate and the
# civilian-factory floor below.
WA_AI_build_theatre_air_bases = {

	@THEATRE_AIR_CAPACITY_PER_LEVEL = 100	# NDefines.NBuildings.AIRBASE_CAPACITY_MULT (05_defines.lua)
	@THEATRE_AIR_TARGET_LEVEL = 4			# per-state ladder cap: the 2-4 level goal
	@THEATRE_AIR_TARGET_LEVEL_COMMITTED = 8	# Fix 85: ladder cap where ROOT's own divisions stand (a two-state beachhead)
	@THEATRE_AIR_QUEUE_MAX = 3				# max concurrent air-base projects (matches UK hosting)
	@THEATRE_AIR_HOSTED_SHARE = 0.25		# share of the side's planes one theatre should be able to host

	if = {
		limit = {
			has_war = yes
			surrender_progress < 0.1
			# guardrail 3: skip token air forces - same floor as the theatre pulls
			has_deployed_air_force_size = { size > 99 }
			# floored availability: the bare engine variable reads ~0 once the vanilla
			# queue saturates (see WA_AI_PC_has_project_civs_16)
			WA_AI_PC_has_project_civs_16 = yes
		}

		# DIAGNOSIS INSTRUMENTATION (savegame-readable; same rationale as the UK hosting
		# dbg family above - temp variables are invisible in saves). Remove once verified.
		#   dbg_called   pulses that passed the outer gate
		#   dbg_target   side capacity target in levels, last gated pulse
		#   dbg_active   contested theatres still in deficit, last gated pulse
		#   dbg_best     ladder level of the last started project
		#   dbg_started  projects actually queued
