		}

		# WA_AI_convert_MIC_to_CIC_industrial_expansion = yes not needed in WA
	}

	# Deliberately outside the active_projects/affordability gates: ENG's queue
	# permanently holds ~20 railway projects, so the < 5 gate above can never open
	# for the host nation, and the UK capacity deficit grounds the whole allied
	# air war (campaign 5709c8b9: the entire USAAF parked stateside). The strategy
	# bounds itself via _project_queue_max and the per-state level cap.

	# (USA, ENG, JAP) keep railway-saturated queues. Self-bounded by a side-wide
	# capacity-deficit gate, _project_queue_max and the per-state level cap.
	WA_AI_build_theatre_air_bases = yes

WA_AI_improve_north_africa_infrastructure = {

	# Fix 30 (R9): both corridor anchors are subject-owned at start (Tripoli 448 = ITL,
	# Cairo 446 = UKE). The old tag gate fired for the subjects too, but a subject never
	# passes the dispatcher's WA_AI_PC_can_afford_project, and the overlord's A* died on the
	# subject-controlled start state. Controller-aware selection instead: whichever country
	# controls its corridor end - directly or through a subject - builds toward the other
	# end, provided the far end is not on its own side. Corridor definition unchanged
	# (448 Tripoli <-> 446 Cairo).

	if = {
		limit = {
			date > 1937.6.1
			448 = { CONTROLLER = { OR = { tag = ROOT is_subject_of = ROOT } } } # tripoli is ours
			446 = { # cairo is not
				CONTROLLER = {
					NOT = { tag = ROOT }
					NOT = { is_subject_of = ROOT }
					NOT = { is_in_faction_with = ROOT }
				}
			}
		}

		set_temp_variable = { supply_line_start = 448.id } # tripoli
		set_temp_variable = { supply_line_target = 446.id } # cairo
		WA_AI_build_supply_line = yes
	}

	if = {
		limit = {
			date > 1937.6.1
			446 = { CONTROLLER = { OR = { tag = ROOT is_subject_of = ROOT } } } # cairo is ours
			448 = { # tripoli is not
				CONTROLLER = {
					NOT = { tag = ROOT }
					NOT = { is_subject_of = ROOT }
					NOT = { is_in_faction_with = ROOT }
				}
			}
		}

		set_temp_variable = { supply_line_start = 446.id }
		set_temp_variable = { supply_line_target = 448.id }
		WA_AI_build_supply_line = yes
	}
}

WA_AI_build_refinery_resource_shortage_rubber = {

	### track rubber shortage

	if = {
		limit = {
			# there is a rubber deficit
			check_variable = { resource@rubber < 0 }

			# total rubber needed for production
			set_temp_variable = { rubber_needed = resource_consumed@rubber }
			multiply_temp_variable = { rubber_needed = -1 }
			add_to_temp_variable = { rubber_needed = resource@rubber }

			# deficit is >5% of total need
			set_temp_variable = { rubber_deficit = resource@rubber }
			NOT = { check_variable = { rubber_needed = 0 } }
			divide_temp_variable = { rubber_deficit = rubber_needed }
			check_variable = { rubber_deficit > 0.05 }
		}

		add_to_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_rubber_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_rubber_shortage_months = 3 }

			# Synthetic refineries consume 60 coal/level and, unlike steel/aluminium, have no
			# hydro variant to fall back on: only build when coal is comfortable, otherwise the
			# refinery would feed the coal/steel shortage it competes with. The shortage counter
			# above still ticks either way - it also drives rubber prospecting decisions.
			check_variable = { resource@coal > 50 }
			check_variable = { resource_imported@coal < 20 }
		}

		WA_AI_priority_queue_synthetic_refinery = yes
	}
}



WA_AI_build_refinery_resource_shortage_steel = {

	### track steel shortage

	if = {
		limit = {
			# there is a steel deficit / we have enough industry to invest
			OR = {
				check_variable = { resource@steel < 0 }
				AND = {
					check_variable = { resource@iron > -1 }
					check_variable = { resource_imported@steel > 0 }
				}
			}
			OR = {
				num_of_available_civilian_factories > 19
				check_variable = { resource_imported@steel > 80 }
			}
		}

		add_to_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_steel_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }
		}

		# Choose between hydro and non-hydro based on coal availability
		# Prefer non-hydro if coal is abundant (saves IC cost)
		# Prefer hydro if coal is scarce (saves coal consumption)
		if = {
			limit = {
				check_variable = { resource@coal > 50 }
				check_variable = { resource_imported@coal < 20 }
			}
			WA_AI_priority_queue_steel_refinery = yes
		}
		else = {
			WA_AI_priority_queue_hydro_steel_refinery = yes
		}
	}

	###

	else_if = {
		limit = {
			check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 }

			# floored availability, was NOT = { ...available_for_projects < 20 }: the
			# bare engine variable reads ~0 once the vanilla queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# over 50% of factories are mil
			num_of_factories > 0
			set_temp_variable = { mil_factories = num_of_military_factories }
			divide_temp_variable = { mil_factories = num_of_factories }
			check_variable = { mil_factories > 0.4 }

			# no factory slots left
			NOT = {
				any_of_scopes = { array = controlled_states
					WA_AI_available_SHARED = yes
				}
			}
		}

		WA_AI_priority_convert_MIC_to_CIC = yes
	}
}

WA_AI_build_supply_line_JAP_RAJ = {

	if = {
		limit = {
			tag = JAP
			has_war_with = RAJ
		}

		# Fix 30 (R9): the old hardcoded start 613 (N. Shanghai) could never reach Delhi 439
		# within the A*'s 75-iteration cap. Dynamic anchor instead: BFS outward from the
		# target and take the nearest ROOT-or-subject-controlled state on the SAME landmass
		# (BFS waves are emitted nearest-first; every_neighbor_state can cross straits, the
		# landmass check filters those hits). No anchor found = JAP holds no ground near the
		# RAJ front = no supply line, which is the correct null behaviour.
		set_temp_variable = { _origin_state_id = 439 } # delhi
		set_temp_variable = { _max_distance = 10 }
		WA_AI_PC_railway_get_states_within_distance = yes # >>> states_within_distance_

		set_temp_variable = { _sl_target_state_ = 439 }
		set_temp_variable = { _sl_target_landmass_ = global.WA_AI_MAP_state_landmass^_sl_target_state_ }
		set_temp_variable = { _sl_anchor_found_ = 0 }
		for_each_loop = { array = states_within_distance_ value = _sl_cand_ break = _sl_anchor_found_
			if = {
				limit = { check_variable = { global.WA_AI_MAP_state_landmass^_sl_cand_ = _sl_target_landmass_ } }
				var:_sl_cand_ = {
					if = {
						limit = {
							CONTROLLER = {
								OR = {
									tag = ROOT
									is_subject_of = ROOT
								}
							}
						}
						set_temp_variable = { supply_line_start = THIS.id }
						set_temp_variable = { _sl_anchor_found_ = 1 }
					}
				}
			}
		}

		if = { limit = { check_variable = { _sl_anchor_found_ = 1 } }
			set_temp_variable = { supply_line_target = _sl_target_state_ }
			WA_AI_build_supply_line = yes
		}
	}
}

WA_AI_build_supply_line = {

	# Fix 30 break hygiene: earlier strategies in the same pulse (refinery/AA queue
	# functions, scoring loops) leave the shared `break` temp variable at 1; the
	# default-break loops below (path walk, front exploration) would then abort before
	# their first iteration. The pathfinder resets it too, but reset here so THIS effect
	# is safe regardless of call order.
	set_temp_variable = { break = 0 }
	set_temp_variable = { building_path_to_front = 0 }

	# Fix 30 instrumentation (campaign 66d6b53c: zero supply-line projects all war, and
	# resource-INF shares building type 1, so saves could not tell the two apart).
	# Persistent, save-visible (lowercased in saves); remove once the system is verified.
	#   WA_AI_supply_line_dbg_called      pulses that reached this effect
	#   WA_AI_supply_line_dbg_pf_ok       pulses whose A* returned success
	#   WA_AI_supply_line_dbg_queued      infrastructure projects actually queued here
	#   WA_AI_supply_line_dbg_last_state  state id of the last queued project
	add_to_variable = { WA_AI_supply_line_dbg_called = 1 }

	# Fix: these three inputs were passed WITHOUT the underscore prefix the pathfinder
	# reads (_pathfind_start/_pathfind_target/_pathfind_type in
	# WA_AI_pathfinding_effects.txt), so the A* ran from an unset scope and this whole
	# effect - and all three strategies built on it (North Africa infrastructure,
	# GER->SOV and JAP->RAJ supply lines) - has been silent dead code since authoring:
	# zero type-1 supply-line projects in 8 in-game years of campaign 9be92c89.
	# (Outputs pathfind_success / pathfind_path_ carry no prefix and were correct.)
	set_temp_variable = { _pathfind_start = supply_line_start }
	set_temp_variable = { _pathfind_target = supply_line_target }
	set_temp_variable = { _pathfind_type = 1 }
	WA_AI_PATHFIND_get_path = yes

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | SUPPLY LINE: [?supply_line_start] -> [?supply_line_target] success=[?pathfind_success]"
	}

	if = { limit = { check_variable = { pathfind_success = 1 } }

		add_to_variable = { WA_AI_supply_line_dbg_pf_ok = 1 }

		# Set up project parameters for infrastructure building
		set_temp_variable = { _project_building_type = 1 }
		set_temp_variable = { _project_province_id = 0 }
		set_temp_variable = { _project_priority = 100 }
		set_temp_variable = { _project_queue_num = 1 }

		### build a route to the front

		set_temp_variable = { target_id = pathfind_path_^0 }
		for_each_loop = { array = pathfind_path_

			var:v = {

				if = { limit = { WA_AI_PC_can_build_project = yes }

					var:target_id = {

						if = { limit = { check_variable = { PREV.building_level@infrastructure < THIS.building_level@infrastructure } }

							set_temp_variable = { target_id = pathfind_path_^i }
						}
					}

					set_temp_variable = { building_path_to_front = 1 }
				}
			}
		}

		if = { limit = { check_variable = { building_path_to_front = 1 } }

			set_temp_variable = { _sl_before_ = WA_AI_PC_active_projects }
			var:target_id = { WA_AI_PC_start_project = yes }
			# start_project may no-op (duplicate, level cap) - count only real queue growth
			if = { limit = { check_variable = { WA_AI_PC_active_projects > _sl_before_ } }
				add_to_variable = { WA_AI_supply_line_dbg_queued = 1 }
				set_variable = { WA_AI_supply_line_dbg_last_state = target_id }
			}
		}

		### build along the front

		if = { limit = { check_variable = { building_path_to_front = 0 } }

			var:supply_line_target = {

				CONTROLLER = {

					set_temp_variable = { supply_line_target_controller = THIS.id }
				}
			}

			set_temp_variable = { front_intersection = pathfind_path_^num }
			subtract_from_temp_variable = { front_intersection = 1 }
			set_temp_variable = { front_intersection = pathfind_path_^front_intersection }

			add_to_temp_array = { front_states = front_intersection }
			add_to_temp_array = { explore_states = front_intersection }

			while_loop_effect = { limit = { check_variable = { explore_states^num > 0 } }

				clear_temp_array = neighbor_states

				for_each_scope_loop = { array = explore_states

					# max infrastructure
					if = { limit = { WA_AI_PC_can_build_project = yes }

						set_temp_variable = { _sl_before_ = ROOT.WA_AI_PC_active_projects }
						WA_AI_PC_start_project = yes
						if = { limit = { check_variable = { ROOT.WA_AI_PC_active_projects > _sl_before_ } }
							# Plain state id via the map-data primitive - a scope-hopped PREV.id
							# stores the engine-encoded reference, not the numeric id (R1 lesson).
							# ROOT.-prefixed targets so the RHS temps stay readable in state scope.
							WA_AI_PC_get_state_id = yes # >>> state_id_
							add_to_variable = { ROOT.WA_AI_supply_line_dbg_queued = 1 }
							set_variable = { ROOT.WA_AI_supply_line_dbg_last_state = state_id_ }
						}

						set_temp_variable = { break = 1 }
					}

					# check neighbors
					else = {
						every_neighbor_state = {
							limit = {
								# Fix 30: also walk the front through subject-controlled states
								# (WA_AI_PC_can_build_project already accepts them; a ROOT-only
								# walk filtered the whole UKE/ITL corridor back out)
								OR = {
									is_controlled_by = ROOT
									CONTROLLER = { is_subject_of = ROOT }
								}

								NOT = { is_in_array = { front_states = THIS.id } }

								any_neighbor_state = {

									CONTROLLER = {

										NOT = { tag = ROOT }

										OR = {
											tag = var:supply_line_target_controller
										}
									}
								}
							}

							add_to_temp_array = { front_states = THIS.id }
							add_to_temp_array = { neighbor_states = THIS.id }
						}
					}
				}

				clear_temp_array = explore_states
				for_each_loop = { array = neighbor_states break = break_2

					add_to_temp_array = { explore_states = v }
				}
			}
		}
	}
}

# Keep enough air base capacity in central/southern England to host the whole
# faction's aviation (RAF + USAAF + minors). Without this the allied AI parks
# its air force in America: the UK bases in range of the front fill up and the
# engine finds no free capacity worth rebasing to, so the wings never cross the
# Atlantic.
# ENG builds as host nation. USA acts as backup builder (air_base has
# allied_build = yes) whenever ENG cannot act on the deficit itself: ENG is
# player-controlled or lacks civilian factories. This strategy is called outside
# the dispatcher's active_projects/affordability gates (see the caller): ENG's
# queue is permanently railway-saturated, so a busy-queue gate would kill the
# host-nation branch for the whole war.
# ROOT = ENG or USA. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only).
WA_AI_build_uk_air_hosting_capacity = {

	@UK_AIR_CAPACITY_PER_LEVEL = 100	# NDefines.NBuildings.AIRBASE_CAPACITY_MULT (05_defines.lua)
	@UK_AIR_TARGET_LEVEL_CAP = 12		# per-state cap, leaves slots for other buildings
	@UK_AIR_HOSTED_FACTION_SHARE = 0.5	# share of the faction's planes the UK should be able to host

	# DIAGNOSIS INSTRUMENTATION (campaign 0e7e7852: the strategy produced zero
	# projects on a build that verifiably contained it, and saves cannot show why -
	# temp variables are not stored). The WA_AI_uk_air_dbg_* persistent variables
	# make each stage readable from the next test campaign's saves:
	#   dbg_called   pulses that reached this effect (ENG/USA only)
	#   dbg_planes   faction plane count of the last gated pulse
	#   dbg_capacity hosting capacity of the last gated pulse
	#   dbg_best     -2 = no deficit, -1 = deficit but no eligible state, else state id
	#   dbg_started  projects actually queued
	# Absence of dbg_called = dispatcher never calls us; dbg_called without the
	# others = the main gate never passed. Remove once the system is verified.
	if = {
		limit = { OR = { original_tag = ENG original_tag = USA } }
		add_to_variable = { WA_AI_uk_air_dbg_called = 1 }
	}

	if = {
		limit = {
			has_war = yes
			any_enemy_country = { capital_scope = { is_on_continent = europe } }
			# floored availability: the bare engine variable reads ~0 once the vanilla
			# queue saturates (see WA_AI_PC_has_project_civs_16)
			WA_AI_PC_has_project_civs_16 = yes
			OR = {
				original_tag = ENG
				AND = {
					original_tag = USA
					is_in_faction_with = ENG
					# only step in when ENG cannot or will not act on the deficit itself:
					# player-controlled, not actually funding air bases, or genuinely poor.
					# Fix 38 (campaign 31eaf7e6): the old capability-only civ test blocked
					# the USA for the whole war while ENG's own air slots ran at 0 assigned
					# factories in 4 of 5 sampled saves - having civs available is not the
					# same as spending them on air bases. WA_AI_PC_air_factories_assigned
					# is written for every AI country by WA_AI_PC_assign_factories (weekly).
					ENG = {
						OR = {
							is_ai = no
							check_variable = { WA_AI_PC_air_factories_assigned < 1 }
							num_of_civilian_factories_available_for_projects < 16
						}
					}
				}
			}
		}

		### how many planes the faction may want to base in the UK
		# num_deployed_planes cannot be read as a set_temp_variable rvalue (campaign
		# 5709c8b9: every other gate provably passed and the deficit branch never
		# fired). The per-airframe form num_deployed_planes_with_type@X is script-
		# readable (mission_calculations_effects.txt relies on it). The sums are
		# INLINED deliberately - a helper effect is a newly registered name, which a
		# console reload cannot pick up and which added an unprovable failure class
		# to the 0e7e7852 test round; inline statements carry no such risk.

		set_temp_variable = { _faction_planes_ = 0 }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
		for_each_scope_loop = { array = allies
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
		}

		set_temp_variable = { _uk_target_capacity_ = _faction_planes_ }
		multiply_temp_variable = { _uk_target_capacity_ = @UK_AIR_HOSTED_FACTION_SHARE }

		### capacity already built or queued in the hosting states

		set_temp_variable = { _uk_hosting_capacity_ = 0 }
		every_state = {
			limit = {
				WA_AI_uk_air_hosting_state = yes
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = { is_in_faction_with = ROOT }
				}
			}
			add_to_temp_variable = { _uk_hosting_capacity_ = building_level@air_base }
		}
		for_each_loop = { array = WA_AI_PC_queue value = _uk_air_proj_id_
			if = {
				limit = {
					check_variable = { WA_AI_PC_building_type^_uk_air_proj_id_ = 2 }
					# Fix 37 companion (campaign 31eaf7e6): only air-base projects actually
					# targeting a UK hosting state count as claimed UK capacity - the USA's
					# Pacific-theatre air projects sitting in the same queue were inflating
					# it by +100 capacity each and masking the real deficit.
					var:WA_AI_PC_target_state^_uk_air_proj_id_ = { WA_AI_uk_air_hosting_state = yes }
				}
				add_to_temp_variable = { _uk_hosting_capacity_ = 1 }
			}
		}
		multiply_temp_variable = { _uk_hosting_capacity_ = @UK_AIR_CAPACITY_PER_LEVEL }

		set_variable = { WA_AI_uk_air_dbg_planes = _faction_planes_ }
		set_variable = { WA_AI_uk_air_dbg_capacity = _uk_hosting_capacity_ }
		set_variable = { WA_AI_uk_air_dbg_best = -2 }

		if = { limit = { check_variable = { _uk_target_capacity_ > _uk_hosting_capacity_ } }

			### expand the least-developed hosting state first
			# Fix: the old two-pass select stored THIS.id from STATE scope and compared it
			# to a -1 sentinel in country scope - the stored value is scope-encoded, the
			# `> -1` guard never passed, and campaign 9be92c89's instrumentation shows
			# dbg_started absent for the entire war (RAF 3.2x overstacked, zero USAAF wings
			# ever based in Britain). Fifth occurrence of the state-scope temp trap (AIFC,
			# railway Fixes 24/25/27 lineage). Level-ladder instead: try level 0 upward,
			# start the project directly in state scope (WA_AI_PC_start_project's documented
			# contract: THIS = state, PREV = builder country), report via a ROOT-hopped
			# plain integer flag.

			set_temp_variable = { _project_building_type = 2 }
			set_temp_variable = { _uk_air_started_ = 0 }
			set_temp_variable = { _uk_air_try_level_ = 0 }
			while_loop_effect = {
				limit = {
					check_variable = { _uk_air_started_ < 1 }
					check_variable = { _uk_air_try_level_ < @UK_AIR_TARGET_LEVEL_CAP }
				}

				random_state = {
					limit = {
						WA_AI_uk_air_hosting_state = yes
						check_variable = { building_level@air_base = _uk_air_try_level_ }
						OR = {
							AND = {
								is_controlled_by = ROOT
								WA_AI_available_AIR = yes
								WA_AI_PC_can_build_project = yes
							}
							AND = {
								NOT = { is_controlled_by = ROOT }
								WA_AI_available_AIR_allied = yes
								NOT = { WA_AI_PC_has_same_type_project_here = yes }
							}
						}
					}

					set_temp_variable = { _project_building_type = 2 }
					set_temp_variable = { _project_province_id = 0 }
					set_temp_variable = { _project_target_state = 0 }
					set_temp_variable = { _project_priority = 100 }
					set_temp_variable = { _project_queue_num = 1 }
					set_temp_variable = { _project_queue_max = 3 }
					set_temp_variable = { _project_target_level = @UK_AIR_TARGET_LEVEL_CAP }
					WA_AI_PC_start_project = yes

					ROOT = {
						set_temp_variable = { _uk_air_started_ = 1 }
						# dbg_best now records the LEVEL a project was started at (-2 = no deficit,
						# unchanged; no start with a deficit leaves it -2 as before)
						set_variable = { WA_AI_uk_air_dbg_best = _uk_air_try_level_ }
						add_to_variable = { WA_AI_uk_air_dbg_started = 1 }
					}
				}

				add_to_temp_variable = { _uk_air_try_level_ = 1 }
			}
		}
	}

# Build/upgrade air bases inside air theatres that are CONTESTED for ROOT, so the
# strategic_air_importance pulls in common/ai_strategy/WA_AI_MILITARY_DEFAULT_AIR_theatres.txt
# have friendly capacity to rebase into - the engine only rebases wings into in-range friendly
# air-base capacity, so a pull toward a theatre with no basing is inert (that file's KNOWN
# LIMITS names this effect as the follow-up). Campaign 66d6b53c: the USAAF kept 8,595 of
# 13,791 planes in the continental US - no basing existed along the Pacific island chain.
# Theatre membership comes from the WA_AI_MILITARY_AIR_theatre_state_* triggers
# (common/scripted_triggers/WA_AI_MILITARY_triggers.txt), which must-match the contested lists.
#
# NEED GUARDRAILS (all generic, no tags):
# 1. Theatre-scoped sites: projects can only start in theatre member states - this effect can
#    never build on home soil outside a contested theatre (a Canada at peace-depth in its own
#    hemisphere is structurally unreachable).
# 2. Side-wide capacity deficit: a contested theatre only qualifies while the SIDE's existing
#    + ROOT-queued air-base capacity across that theatre's states is below
#    @THEATRE_AIR_HOSTED_SHARE of the side's deployed planes. Capacity counts every state
#    controlled by ROOT, a faction ally or a co-belligerent (same own-side definition as the
#    contested triggers), so once anyone on the side has built enough, every other member
#    stops queuing - a minor never piles onto a theatre its majors already covered, and a
#    co-belligerent outside the faction (e.g. a 1945 SOV vs JAP) sees the US capacity and
#    stays idle.
# 3. Token air forces never enter (<= 99 deployed planes, WA air scale ~1/3 vanilla): a
#    country with no meaningful aviation has no basing need of its own.
# 4. Per-state cap: the level ladder stops selecting a state at @THEATRE_AIR_TARGET_LEVEL,
#    and always raises the least-developed member state first, so capacity spreads 0->4
#    across the chain instead of stacking one level-10 island.
#
# ROOT = any AI country. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only), outside the dispatcher's
# active_projects/affordability gates for the same reason as the UK hosting strategy above

		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_burma_india = yes } set_temp_variable = { _thair_bur_ = 1 } }
		set_temp_variable = { _thair_sea_ = 0 }
		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_southeast_asia = yes } set_temp_variable = { _thair_sea_ = 1 } }
		set_temp_variable = { _thair_pac_ = 0 }
		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_pacific_islands = yes } set_temp_variable = { _thair_pac_ = 1 } }

		set_temp_variable = { _thair_any_ = _thair_naf_ }
		add_to_temp_variable = { _thair_any_ = _thair_ita_ }
		add_to_temp_variable = { _thair_any_ = _thair_weu_ }
		add_to_temp_variable = { _thair_any_ = _thair_sca_ }
		add_to_temp_variable = { _thair_any_ = _thair_bal_ }
		add_to_temp_variable = { _thair_any_ = _thair_eeu_ }
		add_to_temp_variable = { _thair_any_ = _thair_mde_ }
		add_to_temp_variable = { _thair_any_ = _thair_bur_ }
		add_to_temp_variable = { _thair_any_ = _thair_sea_ }
		add_to_temp_variable = { _thair_any_ = _thair_pac_ }

		if = { limit = { check_variable = { _thair_any_ > 0 } }

			### pass 2 - guardrail 2: side capacity per contested theatre vs the side's need

			# side's deployed air force (num_deployed_planes is not script-readable; the
			# per-airframe sums are INLINED for the reasons documented in
			# WA_AI_build_uk_air_hosting_capacity above)
			set_temp_variable = { _thair_planes_ = 0 }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
			add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
			for_each_scope_loop = { array = allies
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
				add_to_temp_variable = { _thair_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
			}

			# capacity target per theatre, in air-base LEVELS
			set_temp_variable = { _thair_target_ = _thair_planes_ }
			multiply_temp_variable = { _thair_target_ = @THEATRE_AIR_HOSTED_SHARE }
			divide_temp_variable = { _thair_target_ = @THEATRE_AIR_CAPACITY_PER_LEVEL }

			# capacity the side already holds (or ROOT already queued) per contested theatre.
			# Membership lists are disjoint, so the attribution chain is exact. Allies' PC
			# queues are not visible cross-country - the bounded overshoot (each builder caps
			# itself at @THEATRE_AIR_QUEUE_MAX projects) is accepted.
			set_temp_variable = { _thair_cap_naf_ = 0 }
			set_temp_variable = { _thair_cap_ita_ = 0 }
			set_temp_variable = { _thair_cap_weu_ = 0 }
			set_temp_variable = { _thair_cap_sca_ = 0 }
			set_temp_variable = { _thair_cap_bal_ = 0 }
			set_temp_variable = { _thair_cap_eeu_ = 0 }
			set_temp_variable = { _thair_cap_mde_ = 0 }
			set_temp_variable = { _thair_cap_bur_ = 0 }
			set_temp_variable = { _thair_cap_sea_ = 0 }
			set_temp_variable = { _thair_cap_pac_ = 0 }
			every_state = {
				limit = {
					OR = {
						AND = { check_variable = { _thair_naf_ = 1 } WA_AI_MILITARY_AIR_theatre_state_north_africa = yes }
						AND = { check_variable = { _thair_ita_ = 1 } WA_AI_MILITARY_AIR_theatre_state_italy = yes }
						AND = { check_variable = { _thair_weu_ = 1 } WA_AI_MILITARY_AIR_theatre_state_western_europe = yes }
						AND = { check_variable = { _thair_sca_ = 1 } WA_AI_MILITARY_AIR_theatre_state_scandinavia = yes }
						AND = { check_variable = { _thair_bal_ = 1 } WA_AI_MILITARY_AIR_theatre_state_balkans = yes }
						AND = { check_variable = { _thair_eeu_ = 1 } WA_AI_MILITARY_AIR_theatre_state_eastern_europe = yes }
						AND = { check_variable = { _thair_mde_ = 1 } WA_AI_MILITARY_AIR_theatre_state_middle_east = yes }
						AND = { check_variable = { _thair_bur_ = 1 } WA_AI_MILITARY_AIR_theatre_state_burma_india = yes }
						AND = { check_variable = { _thair_sea_ = 1 } WA_AI_MILITARY_AIR_theatre_state_southeast_asia = yes }
						AND = { check_variable = { _thair_pac_ = 1 } WA_AI_MILITARY_AIR_theatre_state_pacific_islands = yes }
					}
					CONTROLLER = { OR = { tag = ROOT is_in_faction_with = ROOT has_war_together_with = ROOT } }
				}

				set_temp_variable = { _thair_state_levels_ = building_level@air_base }
				if = {
					limit = {
						any_of = { array = WA_AI_PC_projects_in_state_of_@ROOT value = v_thair_proj
							check_variable = { ROOT.WA_AI_PC_building_type^v_thair_proj = 2 }
						}
					}
					add_to_temp_variable = { _thair_state_levels_ = 1 }
				}

				if = { limit = { WA_AI_MILITARY_AIR_theatre_state_north_africa = yes } add_to_temp_variable = { _thair_cap_naf_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_italy = yes } add_to_temp_variable = { _thair_cap_ita_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_western_europe = yes } add_to_temp_variable = { _thair_cap_weu_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_scandinavia = yes } add_to_temp_variable = { _thair_cap_sca_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_balkans = yes } add_to_temp_variable = { _thair_cap_bal_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_eastern_europe = yes } add_to_temp_variable = { _thair_cap_eeu_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_middle_east = yes } add_to_temp_variable = { _thair_cap_mde_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_burma_india = yes } add_to_temp_variable = { _thair_cap_bur_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_southeast_asia = yes } add_to_temp_variable = { _thair_cap_sea_ = _thair_state_levels_ } }
				else_if = { limit = { WA_AI_MILITARY_AIR_theatre_state_pacific_islands = yes } add_to_temp_variable = { _thair_cap_pac_ = _thair_state_levels_ } }
			}

			set_temp_variable = { _thair_contested_ct_ = _thair_any_ }

			# a theatre stays eligible only while the side's capacity is below the target
			if = { limit = { check_variable = { _thair_naf_ = 1 } NOT = { check_variable = { _thair_cap_naf_ < _thair_target_ } } } set_temp_variable = { _thair_naf_ = 0 } }
			if = { limit = { check_variable = { _thair_ita_ = 1 } NOT = { check_variable = { _thair_cap_ita_ < _thair_target_ } } } set_temp_variable = { _thair_ita_ = 0 } }
			if = { limit = { check_variable = { _thair_weu_ = 1 } NOT = { check_variable = { _thair_cap_weu_ < _thair_target_ } } } set_temp_variable = { _thair_weu_ = 0 } }
			if = { limit = { check_variable = { _thair_sca_ = 1 } NOT = { check_variable = { _thair_cap_sca_ < _thair_target_ } } } set_temp_variable = { _thair_sca_ = 0 } }
			if = { limit = { check_variable = { _thair_bal_ = 1 } NOT = { check_variable = { _thair_cap_bal_ < _thair_target_ } } } set_temp_variable = { _thair_bal_ = 0 } }
			if = { limit = { check_variable = { _thair_eeu_ = 1 } NOT = { check_variable = { _thair_cap_eeu_ < _thair_target_ } } } set_temp_variable = { _thair_eeu_ = 0 } }
			if = { limit = { check_variable = { _thair_mde_ = 1 } NOT = { check_variable = { _thair_cap_mde_ < _thair_target_ } } } set_temp_variable = { _thair_mde_ = 0 } }
			if = { limit = { check_variable = { _thair_bur_ = 1 } NOT = { check_variable = { _thair_cap_bur_ < _thair_target_ } } } set_temp_variable = { _thair_bur_ = 0 } }
			if = { limit = { check_variable = { _thair_sea_ = 1 } NOT = { check_variable = { _thair_cap_sea_ < _thair_target_ } } } set_temp_variable = { _thair_sea_ = 0 } }
			if = { limit = { check_variable = { _thair_pac_ = 1 } NOT = { check_variable = { _thair_cap_pac_ < _thair_target_ } } } set_temp_variable = { _thair_pac_ = 0 } }

			set_temp_variable = { _thair_any_ = _thair_naf_ }
			add_to_temp_variable = { _thair_any_ = _thair_ita_ }
			add_to_temp_variable = { _thair_any_ = _thair_weu_ }
			add_to_temp_variable = { _thair_any_ = _thair_sca_ }
			add_to_temp_variable = { _thair_any_ = _thair_bal_ }
			add_to_temp_variable = { _thair_any_ = _thair_eeu_ }
			add_to_temp_variable = { _thair_any_ = _thair_mde_ }
			add_to_temp_variable = { _thair_any_ = _thair_bur_ }
			add_to_temp_variable = { _thair_any_ = _thair_sea_ }
			add_to_temp_variable = { _thair_any_ = _thair_pac_ }

			set_variable = { WA_AI_thair_dbg_target = _thair_target_ }
			set_variable = { WA_AI_thair_dbg_active = _thair_any_ }

			# TEMP R16 live-test logging - remove with the WA_AI_thair_dbg_* family
			if = { limit = { check_variable = { _thair_any_ > 0 } }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: [?_thair_any_|.0] deficit theatre(s) of [?_thair_contested_ct_|.0] contested, side target [?_thair_target_|.0] levels"
			}
			else = {
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: all [?_thair_contested_ct_|.0] contested theatre(s) at capacity (target [?_thair_target_|.0]) - guardrail hold"
			}

			if = { limit = { check_variable = { _thair_any_ > 0 } }

				### pass 3 - guardrail 4: level ladder, raise the least-developed eligible
				### member state first (level-ladder + ROOT-hopped flag per the state-scope
				### temp trap documented in the UK strategy above)

				# _project_building_type must be set BEFORE the ladder: the limit triggers
				# below (WA_AI_PC_can_build_project / WA_AI_PC_has_same_type_project_here)
				# read it
				set_temp_variable = { _project_building_type = 2 }
				set_temp_variable = { _thair_started_ = 0 }
				set_temp_variable = { _thair_try_level_ = 0 }
				while_loop_effect = {
					limit = {
						check_variable = { _thair_started_ < 1 }
						check_variable = { _thair_try_level_ < @THEATRE_AIR_TARGET_LEVEL }
					}

					random_state = {
						limit = {
							OR = {
								AND = { check_variable = { _thair_naf_ = 1 } WA_AI_MILITARY_AIR_theatre_state_north_africa = yes }
								AND = { check_variable = { _thair_ita_ = 1 } WA_AI_MILITARY_AIR_theatre_state_italy = yes }
								AND = { check_variable = { _thair_weu_ = 1 } WA_AI_MILITARY_AIR_theatre_state_western_europe = yes }
								AND = { check_variable = { _thair_sca_ = 1 } WA_AI_MILITARY_AIR_theatre_state_scandinavia = yes }
								AND = { check_variable = { _thair_bal_ = 1 } WA_AI_MILITARY_AIR_theatre_state_balkans = yes }
								AND = { check_variable = { _thair_eeu_ = 1 } WA_AI_MILITARY_AIR_theatre_state_eastern_europe = yes }
								AND = { check_variable = { _thair_mde_ = 1 } WA_AI_MILITARY_AIR_theatre_state_middle_east = yes }
								AND = { check_variable = { _thair_bur_ = 1 } WA_AI_MILITARY_AIR_theatre_state_burma_india = yes }
								AND = { check_variable = { _thair_sea_ = 1 } WA_AI_MILITARY_AIR_theatre_state_southeast_asia = yes }
								AND = { check_variable = { _thair_pac_ = 1 } WA_AI_MILITARY_AIR_theatre_state_pacific_islands = yes }
							}
							check_variable = { building_level@air_base = _thair_try_level_ }
							OR = {
								AND = {	# our own territory
									is_controlled_by = ROOT
									WA_AI_available_AIR = yes
									WA_AI_PC_can_build_project = yes
								}
								AND = {	# subject territory (the engine allows overlord construction in subjects; PC validates the same)
									CONTROLLER = { is_subject_of = ROOT }
									WA_AI_PC_can_build_project = yes
								}
								AND = {	# faction ally's territory (air_base has allied_build = yes),
										# only when the controller cannot or will not act on the
										# deficit itself: player-controlled, not actually funding
										# air bases, or genuinely poor. Fix 38 (campaign 31eaf7e6):
										# capability is not funding - see the UK hosting gate above.
									NOT = { is_controlled_by = ROOT }
									NOT = { CONTROLLER = { is_subject_of = ROOT } }
									WA_AI_available_AIR_allied = yes
									CONTROLLER = {
										OR = {
											is_ai = no
											check_variable = { WA_AI_PC_air_factories_assigned < 1 }
											num_of_civilian_factories_available_for_projects < 16
										}
									}
									NOT = { WA_AI_PC_has_same_type_project_here = yes }
								}
							}
						}

						set_temp_variable = { _project_building_type = 2 }
						set_temp_variable = { _project_province_id = 0 }
						set_temp_variable = { _project_target_state = 0 }
						set_temp_variable = { _project_priority = 100 }
						set_temp_variable = { _project_queue_num = 1 }
						set_temp_variable = { _project_queue_max = @THEATRE_AIR_QUEUE_MAX }
						set_temp_variable = { _project_target_level = @THEATRE_AIR_TARGET_LEVEL }
						set_temp_variable = { _thair_qnum_before_ = ROOT.WA_AI_PC_queue^num }
						WA_AI_PC_start_project = yes

						# Fix: WA_AI_PC_start_project can decline internally (queue_max, same-type
						# dedup, province duplicate) with no return signal - live test 2b607968
						# showed dbg_started over-counting (AST 19 "starts", empty queue) and ENG's
						# theatre starts permanently no-oping against a type-2 queue_max already
						# filled by UK-hosting projects. Only count a start (and stop the ladder)
						# when the queue actually grew; a declined start lets the ladder try the
						# next level instead of silently ending the pulse.
						if = {
							limit = { ROOT = { check_variable = { WA_AI_PC_queue^num > _thair_qnum_before_ } } }

							# TEMP R16 live-test logging - remove with the WA_AI_thair_dbg_* family
							log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: air_base project started at level [?_thair_try_level_|.0] in [This.GetID] [This.GetName]"

							ROOT = {
								set_temp_variable = { _thair_started_ = 1 }
								set_variable = { WA_AI_thair_dbg_best = _thair_try_level_ }
								add_to_variable = { WA_AI_thair_dbg_started = 1 }
							}
						}
					}

					add_to_temp_variable = { _thair_try_level_ = 1 }
				}

				# TEMP R16 live-test logging - remove with the WA_AI_thair_dbg_* family
				if = { limit = { check_variable = { _thair_started_ < 1 } }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: deficit but no eligible member state (ladder exhausted at level [?_thair_try_level_|.0])"
				}
			}
		}
	}
