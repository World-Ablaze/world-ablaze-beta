WA_AI_priority_construction_strategies = {


	# [refineries] the dispatcher is FLAT: no shared admission or affordability gate - every
	# strategy admits on its own terms and its own budget (_project_queue_max, scoped by type_id),
	# and the band table decides who is funded. Assumes the dynamic queue, where admission costs
	# nothing: the sorted fill funds or makes wait, the lanes serve starvation, the 30-week stall
	# sweep clears the unfunded tail. Civ gates are the FLOORED WA_AI_PC_has_project_civs_16/_20/_31
	# (the bare engine availability token reads ~0 on a saturated queue).

	WA_AI_important_radar_constructions = yes

	WA_AI_build_supply_line_GER_SOV = yes

	WA_AI_build_supply_line_JAP_RAJ = yes

	WA_AI_improve_north_africa_infrastructure = yes

	WA_AI_resource_extraction = yes

	WA_AI_fortify_strategic_islands = yes

	# [refineries] the refinery strategies run OUTSIDE any shared affordability gate: an
	# import-strangled major (permanently saturated queue, free civs reading ~0) can never pass
	# one, so the branch was dead for exactly the countries a refinery exists for - and its
	# shortage counters froze with it, starving the prospecting triggers that read them
	# (WA_AI_RESOURCE_NEEDS_triggers.txt). Self-bounding instead: WA_AI_PC_has_project_civs_20 +
	# surrender guards inside each strategy, shortage_months = 3, and _project_queue_max = 1 per
	# type in each queue function - at most ONE refinery project of a given type per country.
	WA_AI_build_refinery_resource_shortage_steel = yes

	WA_AI_build_refinery_resource_shortage_aluminium = yes

	WA_AI_build_refinery_resource_shortage_rubber = yes

	# [uk-air-basing] deliberately admits on its own terms: ENG's queue permanently holds ~20
	# railway projects, so any busy-queue gate would close for the host nation for the whole war,
	# and the UK capacity deficit grounds the entire allied air war. Bounded by _project_queue_max
	# and the per-state level cap.
	WA_AI_build_uk_air_hosting_capacity = yes

	# Same rationale: the countries whose theatre pulls go inert keep railway-saturated queues.
	# Self-bounded by a side-wide capacity-deficit gate, _project_queue_max and the level cap.
	WA_AI_build_theatre_air_bases = yes
}

### qmax.* / air.* in common/script_constants/wa_ai_pc.txt - one declaration shared with core.txt,
### queue_functions.txt and the triggers file (no per-file `@` copies to keep in sync).
### [uk-air-basing] UK air-base throughput is capped by the concurrent-project budget, not by civs:

### 23 islands, 24 supply_line, 25 inf_resource; savegame.py _PC_TYPE_ID names them (registry).

### [refineries] per-strategy budgets sized to reproduce the observed healthy footprint, not widen it.

WA_AI_improve_north_africa_infrastructure = {

	# [railway-landwar] controller-aware corridor selection (448 Tripoli <-> 446 Cairo): whichever
	# country controls one end - directly or through a subject - builds toward the other, provided
	# the far end is not on its own side. Assumes both anchors can be subject-owned (they are at
	# start: Tripoli = ITL, Cairo = UKE), which a tag gate mishandles and which stalls an overlord
	# A* that refuses a subject-controlled start state.

	if = {
		limit = {
			date > 1937.6.1
			448 = { CONTROLLER = { OR = { tag = ROOT is_subject_of = ROOT } } } # tripoli is ours
			446 = { # cairo is not
				CONTROLLER = {
					NOT = { tag = ROOT }
					NOT = { is_subject_of = ROOT }
					NOT = { is_in_faction_with = ROOT }
				}
			}
		}

		set_temp_variable = { supply_line_start = 448.id } # tripoli
		set_temp_variable = { supply_line_target = 446.id } # cairo
		WA_AI_build_supply_line = yes
	}

	if = {
		limit = {
			date > 1937.6.1
			446 = { CONTROLLER = { OR = { tag = ROOT is_subject_of = ROOT } } } # cairo is ours
			448 = { # tripoli is not
				CONTROLLER = {
					NOT = { tag = ROOT }

					NOT = { is_in_faction_with = ROOT }

	}
}

WA_AI_build_refinery_resource_shortage_rubber = {

	### track rubber shortage

	if = {
		limit = {
			# there is a rubber deficit
			check_variable = { resource@rubber < 0 }

			# total rubber needed for production
			set_temp_variable = { rubber_needed = resource_consumed@rubber }
			multiply_temp_variable = { rubber_needed = -1 }
			add_to_temp_variable = { rubber_needed = resource@rubber }

			# deficit is >5% of total need
			set_temp_variable = { rubber_deficit = resource@rubber }
			NOT = { check_variable = { rubber_needed = 0 } }
			divide_temp_variable = { rubber_deficit = rubber_needed }
			check_variable = { rubber_deficit > 0.05 }
		}

		add_to_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_rubber_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_rubber_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			# [refineries] this strategy runs outside any shared affordability gate, so it states
			# its own guards (the same set the air strategies carry).
			WA_AI_PC_economy_alive = yes

			# floored availability: the bare engine variable reads ~0 once the queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_rubber_shortage_months = 3 }

			# Synthetic refineries consume 60 coal/level and, unlike steel/aluminium, have no
			# hydro variant to fall back on: only build when coal is comfortable, otherwise the
			# refinery would feed the coal/steel shortage it competes with. The shortage counter
			# above still ticks either way - it also drives rubber prospecting decisions.
			check_variable = { resource@coal > 50 }
			check_variable = { resource_imported@coal < 20 }
		}

		WA_AI_priority_queue_synthetic_refinery = yes
	}

	# [refineries] REJECTED - do not re-propose a PROACTIVE pre-build else_if here as a second host
	# for WA_AI_CONSTRUCTION_should_pre_build_synth_rubber. It has no beneficiary: that trigger
	# needs rubber IMPORTS above its floor while the balance is still non-negative, and no major is
	# ever in that state - it either owns its rubber, is pre-built by hand at startup
	# (events/WA_AI_construction.txt), is rejected by the coal guard, or already raises a healthy
	# synthetic park unaided. Rubber keeps this REACTIVE strategy only; the whole proactive lane
	# is retired.
}



WA_AI_build_refinery_resource_shortage_steel = {

	### track steel shortage

	if = {
		limit = {
			# [refineries] the counter measures the SHORTAGE only - no civ term. A civ test here is
			# the bare vanilla-free-civs token, which reads ~0 on a saturated queue, so a blockaded
			# major with a real deficit and no imports would never reach 3 and the counter would
			# freeze (which also starves the prospecting triggers that read it). Affordability is
			# the downstream branch's question. The import arm keeps its materiality (> 80: a small
			# import bill is not worth a 10-12.5k IC mill).
			OR = {
				check_variable = { resource@steel < 0 }
				AND = {
					check_variable = { resource@iron > -1 }
					check_variable = { resource_imported@steel > 80 }
				}
			}
		}

		add_to_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}
	else = {
		subtract_from_variable = { WA_AI_resource_steel_shortage_months = 1 }
	}

	clamp_variable = { var = WA_AI_resource_steel_shortage_months min = 0 max = 3 }

	###

	if = {
		limit = {
			# [refineries] this strategy states its own guards - it runs outside any shared
			# affordability gate.
			WA_AI_PC_economy_alive = yes

			# floored availability: the bare engine variable reads ~0 once the queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# [refineries] input self-sufficiency, same form as the synthetic branch's coal test:
			# a steel mill converts IRON into steel, it does not create iron. On an iron deficit,
			# or on iron covered by imports the same blockade is already strangling, the mill
			# converts nothing and has spent 10-12.5k IC to move the shortage one link up the
			# chain - importing finished steel is strictly cheaper. Domestic iron non-negative and
			# imports below 40 (the aluminium branch below uses the identical shape on bauxite).
			check_variable = { resource@iron > -1 }
			check_variable = { resource_imported@iron < 40 }
		}

		# Choose between hydro and non-hydro based on coal availability
		# Prefer non-hydro if coal is abundant (saves IC cost)
		# Prefer hydro if coal is scarce (saves coal consumption)
		if = {
			limit = {
				check_variable = { resource@coal > 50 }
				check_variable = { resource_imported@coal < 20 }
			}
			WA_AI_priority_queue_steel_refinery = yes
		}
		else = {
			WA_AI_priority_queue_hydro_steel_refinery = yes
		}
	}

	###

	else_if = {
		limit = {
			# [refineries] the brake on industry conversion is EXPLICIT, not an affordability gate:
			# WA_AI_priority_convert_MIC_to_CIC carries _project_queue_max =
			# constant:wa_ai_pc.qmax.mic_to_cic (1 conversion project in flight per country, MIC
			# removed only at completion). The guards below are stated as the refinery branch states them.
			WA_AI_PC_economy_alive = yes

			# floored availability: the bare engine variable reads ~0 once the queue saturates
			WA_AI_PC_has_project_civs_20 = yes

			# resource shortage
			check_variable = { WA_AI_resource_steel_shortage_months = 3 }

			# over 50% of factories are mil
			num_of_factories > 0
			set_temp_variable = { mil_factories = num_of_military_factories }
			divide_temp_variable = { mil_factories = num_of_factories }
			check_variable = { mil_factories > 0.4 }

			# no factory slots left
			NOT = {
				any_of_scopes = { array = controlled_states
					WA_AI_available_SHARED = yes
				}
			}
		}

		WA_AI_priority_convert_MIC_to_CIC = yes
	}
}

WA_AI_build_supply_line_JAP_RAJ = {

	if = {
		limit = {
			tag = JAP
			has_war_with = RAJ
		}

		# [railway-landwar] dynamic anchor: BFS outward from the target and take the nearest
		# ROOT-or-subject-controlled state on the SAME landmass (BFS waves are emitted nearest-
		# first; every_neighbor_state can cross straits, so the landmass check filters those hits).
		# A hardcoded far start cannot reach Delhi within the A*'s 75-iteration cap. No anchor
		# found = we hold no ground near the front = no supply line, the correct null behaviour.
		set_temp_variable = { _origin_state_id = 439 } # delhi
		set_temp_variable = { _max_distance = 10 }
		WA_AI_PC_railway_get_states_within_distance = yes # >>> states_within_distance_

		set_temp_variable = { _sl_target_state_ = 439 }
		set_temp_variable = { _sl_target_landmass_ = global.WA_AI_MAP_state_landmass^_sl_target_state_ }
		set_temp_variable = { _sl_anchor_found_ = 0 }
		for_each_loop = { array = states_within_distance_ value = _sl_cand_ break = _sl_anchor_found_
			if = {
				limit = { check_variable = { global.WA_AI_MAP_state_landmass^_sl_cand_ = _sl_target_landmass_ } }
				var:_sl_cand_ = {
					if = {
						limit = {
							CONTROLLER = {
								OR = {
									tag = ROOT
									is_subject_of = ROOT
								}
							}
						}
						set_temp_variable = { supply_line_start = THIS.id }
						set_temp_variable = { _sl_anchor_found_ = 1 }
					}
				}
			}
		}

		if = { limit = { check_variable = { _sl_anchor_found_ = 1 } }
			set_temp_variable = { supply_line_target = _sl_target_state_ }
			WA_AI_build_supply_line = yes
		}
	}
}

WA_AI_build_supply_line = {

	# [railway-landwar] break hygiene: earlier strategies in the same pulse (refinery/AA queue
	# functions, scoring loops) leave the shared `break` temp at 1, and the default-break loops
	# below (path walk, front exploration) would then abort before their first iteration. Reset
	# here so THIS effect is safe regardless of call order.
	set_temp_variable = { break = 0 }
	set_temp_variable = { building_path_to_front = 0 }

	# [railway-landwar] instrumentation, persistent and save-visible: resource-INF also builds
	# building type 1, so a save cannot otherwise tell the two apart. Remove once verified.
	#   WA_AI_supply_line_dbg_called      pulses that reached this effect
	#   WA_AI_supply_line_dbg_pf_ok       pulses whose A* returned success
	#   WA_AI_supply_line_dbg_queued      infrastructure projects actually queued here
	#   WA_AI_supply_line_dbg_last_state  state id of the last queued project
	add_to_variable = { WA_AI_supply_line_dbg_called = 1 }

	# Cross-file contract: the pathfinder reads UNDERSCORE-prefixed inputs
	# (_pathfind_start / _pathfind_target / _pathfind_type in WA_AI_pathfinding_effects.txt) and
	# returns UNprefixed outputs (pathfind_success / pathfind_path_). Drop a prefix on an input and
	# the A* runs from an unset scope - this effect and all three strategies built on it go silently
	# dead with no parse error.
	set_temp_variable = { _pathfind_start = supply_line_start }
	set_temp_variable = { _pathfind_target = supply_line_target }
	set_temp_variable = { _pathfind_type = 1 }
	WA_AI_PATHFIND_get_path = yes

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | SUPPLY LINE: [?supply_line_start] -> [?supply_line_target] success=[?pathfind_success]"
	}

	if = { limit = { check_variable = { pathfind_success = 1 } }

		add_to_variable = { WA_AI_supply_line_dbg_pf_ok = 1 }

		# Set up project parameters for infrastructure building
		set_temp_variable = { _project_building_type = 1 }
		set_temp_variable = { _project_province_id = 0 }
		set_temp_variable = { _project_priority = constant:wa_ai_pc.prio.default }
		set_temp_variable = { _project_queue_num = 1 }
		# [refineries] explicit budget for type-1 supply-line projects (three callers share it),
		# scoped to this strategy's own type_id - WA_AI_priority_queue_INF_resource also creates
		# building type 1, and unscoped the two count each other's projects.
		# _project_queue_max_scoped is clear-on-read in WA_AI_PC_start_project, so it is re-armed
		# before EACH start call below; _project_type_id persists and is cleared at the end here.
		set_temp_variable = { _project_queue_max = constant:wa_ai_pc.qmax.supply_line }
		set_temp_variable = { _project_type_id = constant:wa_ai_pc.type_id.supply_line }

		### build a route to the front

		set_temp_variable = { target_id = pathfind_path_^0 }
		for_each_loop = { array = pathfind_path_

			var:v = {

				if = { limit = { WA_AI_PC_can_build_project = yes }

					var:target_id = {

						if = { limit = { check_variable = { PREV.building_level@infrastructure < THIS.building_level@infrastructure } }

							set_temp_variable = { target_id = pathfind_path_^i }
						}
					}

					set_temp_variable = { building_path_to_front = 1 }
				}
			}
		}

		if = { limit = { check_variable = { building_path_to_front = 1 } }

			set_temp_variable = { _sl_before_ = WA_AI_PC_active_projects }
			set_temp_variable = { _project_queue_max_scoped = 1 }
			var:target_id = { WA_AI_PC_start_project = yes }
			# start_project may no-op (duplicate, level cap) - count only real queue growth
			if = { limit = { check_variable = { WA_AI_PC_active_projects > _sl_before_ } }
				add_to_variable = { WA_AI_supply_line_dbg_queued = 1 }
				set_variable = { WA_AI_supply_line_dbg_last_state = target_id }
			}
		}

		### build along the front

		if = { limit = { check_variable = { building_path_to_front = 0 } }

			var:supply_line_target = {

				CONTROLLER = {

					set_temp_variable = { supply_line_target_controller = THIS.id }
				}
			}

			set_temp_variable = { front_intersection = pathfind_path_^num }
			subtract_from_temp_variable = { front_intersection = 1 }
			set_temp_variable = { front_intersection = pathfind_path_^front_intersection }

			add_to_temp_array = { front_states = front_intersection }
			add_to_temp_array = { explore_states = front_intersection }

			while_loop_effect = { limit = { check_variable = { explore_states^num > 0 } }

				clear_temp_array = neighbor_states

				for_each_scope_loop = { array = explore_states

					# max infrastructure
					if = { limit = { WA_AI_PC_can_build_project = yes }

						set_temp_variable = { _sl_before_ = ROOT.WA_AI_PC_active_projects }
						set_temp_variable = { _project_queue_max_scoped = 1 }
						WA_AI_PC_start_project = yes
						if = { limit = { check_variable = { ROOT.WA_AI_PC_active_projects > _sl_before_ } }
							# Plain state id via the map-data primitive - a scope-hopped PREV.id
							# stores the engine-encoded reference, not the numeric id (R1 lesson).
							# ROOT.-prefixed targets so the RHS temps stay readable in state scope.
							WA_AI_PC_get_state_id = yes # >>> state_id_
							add_to_variable = { ROOT.WA_AI_supply_line_dbg_queued = 1 }
							set_variable = { ROOT.WA_AI_supply_line_dbg_last_state = state_id_ }
						}

						set_temp_variable = { break = 1 }
					}

					# check neighbors
					else = {
						every_neighbor_state = {
							limit = {
								# [railway-landwar] also walk the front through subject-controlled states
								# (WA_AI_PC_can_build_project already accepts them; a ROOT-only
								# walk filtered the whole UKE/ITL corridor back out)
								OR = {
									is_controlled_by = ROOT
									CONTROLLER = { is_subject_of = ROOT }
								}

								NOT = { is_in_array = { front_states = THIS.id } }

								any_neighbor_state = {

									CONTROLLER = {

										NOT = { tag = ROOT }

										OR = {
											tag = var:supply_line_target_controller
										}
									}
								}
							}

							add_to_temp_array = { front_states = THIS.id }
							add_to_temp_array = { neighbor_states = THIS.id }
						}
					}
				}

				clear_temp_array = explore_states
				for_each_loop = { array = neighbor_states break = break_2

					add_to_temp_array = { explore_states = v }
				}
			}
		}

		# retire this strategy's type_id caller-side (see the note at the parameter block)
		set_temp_variable = { _project_type_id = 0 }
	}

#   _uk_air_qnum_before_ WA_AI_PC_queue^num sampled immediately before the call

	# capture the pre-build level HERE, in state scope: WA_AI_uk_air_dbg_best keeps its documented
	# meaning ("the level a project was started at"; -2 = no deficit), which checklist R8's probes

			# explicit inverse rather than `else`: the branch above ends in a nested `if`, and an
			# `else` after a trailing nested `if` is the binding trap recorded in wa-lessons-learned.
			if = {
				# rank 0 = queued by the unranked fallback tail, deliberately kept out of
				# _rank_sum / _rank_max (see the registry row for the mean-rank denominator).
				limit = { NOT = { check_variable = { _uk_air_rank_ > 0 } } }
				add_to_variable = { WA_TLM_r8_uk_air_fallback_n = 1 }
			}
		}
	}
}

# [uk-air-basing] keep enough air base capacity in central/southern England to host the whole
# faction's aviation (RAF + USAAF + minors). Without it the allied AI parks its air force in
# America: the UK bases in range of the front fill up and the engine finds no free capacity worth
# rebasing to, so the wings never cross the Atlantic. ENG builds as host, USA as backup builder
# (air_base has allied_build = yes). Admits on its own terms - ENG's queue is permanently
# railway-saturated, so a busy-queue gate would kill the host branch for the whole war.
# ROOT = ENG or USA. Called from WA_AI_priority_construction_strategies (WA_AI_background.1,
# ~2 day cadence, AI only).
WA_AI_build_uk_air_hosting_capacity = {

	# [equipment-selection] constant:wa_ai_pc.air.uk_hosted_faction_share is 1.0. Any share below 1.0
	# declares victory while part of the faction's air force still has nowhere in Britain to stand,
	# which is not what this strategy's header promises; every lower value tried stood the build
	# down a year or more before D-Day.
	# STRUCTURAL CEILING: 11 hosting states x constant:wa_ai_pc.air.uk_target_level_cap = 132 levels
	# (13,200 capacity), so the ceiling and not this share bounds the build - it cannot run away.
	# Beyond ~13,200 faction planes the deficit becomes permanent by construction.
	# ACCEPTED COST: air bases hold the 300 band (band table in ..._core.txt) while the deficit
	# stands, so past that point ENG's shortage refineries run on the [pc-queue] overtake lane
	# instead of the sorted fill. If the refineries need their band back more than the RAF needs the
	# last levels, lower this share - do NOT raise constant:wa_ai_pc.air.uk_target_level_cap or extend
	# the state list (checklist R8 forbids it until R15 justifies more capacity).
	# WHAT THIS CANNOT FIX: the shortage is positional as well as aggregate - the south-eastern
	# fields saturate while free capacity stands idle further north, and the USAAF's own basing
	# decision (R15) may be the binding constraint. More levels only help where the south-first walk
	# puts them.
	# [uk-air-basing] / P6: the engine books air-base capacity per WING at land_air_wing_size
	# (100 / tac 200 / strat 300 / scout 25), not per plane present, and script can only count
	# planes; dividing the plane count by constant:wa_ai_pc.air.uk_nominal_fill converts the target to
	# wing slots. That fill is a war-year measurement, and the gap grows with attrition, so it
	# understates late-war need.
	# [uk-air-basing] / P1: above constant:wa_ai_pc.alloc.air_lane_2nd_slot_pct the allocator opens a
	# SECOND air-lane slot (WA_AI_PC_assign_factories reads WA_AI_PC_air_deficit_pct, published
	# below). Deliberately a published state variable, not a dbg one - gameplay must not read dbg_*.
	# constant:wa_ai_pc.alloc.air_host_funding_floor (civs/pass a host must put on air bases before an
	# allied builder stands down) is used ONLY by WA_AI_build_theatre_air_bases further down - this
	# UK strategy deliberately has no host-funding gate at all (see its USA branch for why).

	# DIAGNOSIS INSTRUMENTATION - temp variables are not stored in saves, so each stage of this
	# strategy is a persistent WA_AI_uk_air_dbg_* variable. Remove once the system is verified.

	#   dbg_planes   faction plane count of the last gated pulse
	#   dbg_capacity hosting capacity of the last gated pulse
	#   dbg_best     -2 = no deficit; -1 = deficit open but this pulse queued nothing (budget full
	#                or every eligible site blocked); else the AIR-BASE LEVEL of the site queued
	#                this pulse (see queue_site). Single-pulse snapshot - a monthly save reads the
	#                LAST pulse only; trend questions belong to dbg_started, which is cumulative.
	#   dbg_started  projects actually queued

	if = {
		limit = { OR = { original_tag = ENG original_tag = USA } }
		add_to_variable = { WA_AI_uk_air_dbg_called = 1 }
	}

	if = {
		limit = {
			has_war = yes

			# floored availability: the bare engine variable reads ~0 once the queue saturates

			OR = {

				AND = {
					original_tag = USA
					is_in_faction_with = ENG
					# [equipment-selection] NO host-condition gate: the USA co-builds until the
					# hosting target is met, whatever ENG's economy is doing. Every capability
					# or spending test tried here (has civs, assigns < 1 civ, a rate test) is
					# the same wrong shape - it makes the ALLY's participation a function of
					# the HOST's budget rather than of the shortfall the ally exists to cover,
					# and each locked the USA out for whole wars.
					# Safe because [uk-air-basing] made WA_AI_PC_has_same_type_project_here
					# builder-agnostic, so ENG and USA can no longer queue the same state (they
					# demonstrably did). With collisions impossible, two builders on the same
					# 11-state set is pure added throughput, and the deficit branch below
					# (_uk_target_capacity_ > _uk_hosting_capacity_) is the only stop condition
					# either needs - it closes for BOTH the moment the target is met.
					# WA_AI_build_theatre_air_bases deliberately KEEPS its host-funding gate
					# (constant:wa_ai_pc.alloc.air_host_funding_floor): it spans every faction
					# member across ten theatres, where an ungated ally would pile into every
					# partner's contested theatre at once. This is a two-builder, eleven-state

				}

		}

		set_temp_variable = { _faction_planes_ = 0 }

		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
		add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }
		for_each_scope_loop = { array = allies
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_fighter_interceptor_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@small_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_fighter_multirole_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_heavy_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@fast_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@medium_scout_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_bomber_airframe }
			add_to_temp_variable = { _faction_planes_ = num_deployed_planes_with_type@large_maritime_patrol_airframe }

		set_temp_variable = { _uk_target_capacity_ = _faction_planes_ }
		multiply_temp_variable = { _uk_target_capacity_ = constant:wa_ai_pc.air.uk_hosted_faction_share }
		# [uk-air-basing] / P6: planes -> nominal wing slots (see the constant)
		divide_temp_variable = { _uk_target_capacity_ = constant:wa_ai_pc.air.uk_nominal_fill }

		### capacity already built or queued in the hosting states

		set_temp_variable = { _uk_hosting_capacity_ = 0 }
		every_state = {
			limit = {
				WA_AI_uk_air_hosting_state = yes
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = { is_in_faction_with = ROOT }
				}
			}
			add_to_temp_variable = { _uk_hosting_capacity_ = building_level@air_base }

		for_each_loop = { array = WA_AI_PC_queue value = _uk_air_proj_id_
			if = {
				limit = {
					check_variable = { WA_AI_PC_building_type^_uk_air_proj_id_ = 2 }

		# when there is none. Read by the allocator's second air-lane slot

		set_variable = { WA_AI_PC_air_deficit_pct = 0 }

			limit = {
				check_variable = { _uk_target_capacity_ > _uk_hosting_capacity_ }

			subtract_from_variable = { WA_AI_PC_air_deficit_pct = _uk_hosting_capacity_ }
			multiply_variable = { WA_AI_PC_air_deficit_pct = 100 }
			divide_variable = { WA_AI_PC_air_deficit_pct = _uk_hosting_capacity_ }

		if = { limit = { check_variable = { _uk_target_capacity_ > _uk_hosting_capacity_ } }

			# queue-site helper still overwrites this with the site's level when it fires.

			# [pc-queue] reserve ONE allocator slot for the air-basing band while the deficit
			# stands. Read by the reservation lane in WA_AI_PC_assign_factories
			# (WA_AI_CONSTRUCTION_PRIORITY_core.txt) - full rationale there. Set for whichever

			# The host is not civ-poor: it holds idle factories that constant:wa_ai_pc.alloc.fraction
			# forbids PC from touching, so air bases and its shortage refineries trade against each
			# other instead of both being funded.

			# (WA_AI_CONSTRUCTION_PRIORITY_railway_core.txt sets the same pair). The value here is

			# idempotent and a 30-day refresh from either keeps it live. Do not diverge the numbers
			# without giving each system its own flag, or they alternate on a last-writer-wins timer.
			# Self-releasing: the 30-day flag is only refreshed while this deficit branch runs, so
			# allocation falls back to the default within a month of the target being met.
			# Host only - the ally funds its own projects out of its own pool.
			if = {
				limit = { original_tag = ENG }
				set_country_flag = { flag = WA_AI_PC_override_max_factories_factor value = 1 days = 30 }
				set_variable = { WA_AI_PC_override_max_factories_factor = 0.6 }
			}

			### [equipment-selection] fill the SOUTH-EAST first, only move north once it is capped
			# This replaces a LEVEL ladder ("expand the least-developed state first"), which is a
			# LEVELLING rule and put the largest plane concentrations in the states FURTHEST from
			# France. A wing at Kent covers Northern France; the same wing at Leicestershire does
			# not. It does NOT add capacity - the ceiling is
			# 11 x constant:wa_ai_pc.air.uk_target_level_cap either way - it buys RANGE.
			# The ladder's own reason for existing is preserved: it replaced a two-pass select that
			# stored THIS.id from STATE scope and compared it to a sentinel in country scope, where
			# the stored value is scope-encoded, so the guard never passed and the strategy queued
			# nothing all war. Nothing below stores or compares a scope id - the walk is over PLAIN
			# ids from a global array, entered with `var:`, and the project starts directly in state
			# scope per WA_AI_PC_start_project's contract (THIS = state, PREV = builder country).
			# A top-priority state cannot stall the walk: once ANY builder queues an air base there,
			# WA_AI_PC_has_same_type_project_here (builder-agnostic) makes it fail
			# WA_AI_uk_air_site_buildable, and a state out of BUILDING SLOTS falls through the same
			# trigger, so the next pulse takes the next rank.

			set_temp_variable = { _project_building_type = 2 }
			set_temp_variable = { _uk_air_started_ = 0 }
			# passed to WA_AI_uk_air_queue_site as a temp (kept from the file-scoped `@` era; the
			# helper could now read constant:wa_ai_pc.air.uk_target_level_cap itself - left as an
			# explicit input so the helper's inputs stay visible at the call site)
			set_temp_variable = { _uk_air_level_cap_ = constant:wa_ai_pc.air.uk_target_level_cap }

			# [equipment-selection] band selection, resolved ONCE in country scope - the trigger is a
			# faction scan and must not run per candidate state inside the walk below.
			set_temp_variable = { _uk_air_prio_ = constant:wa_ai_pc.prio.default }
			if = {
				limit = { WA_AI_PC_air_force_needs_basing_priority = yes }
				set_temp_variable = { _uk_air_prio_ = constant:wa_ai_pc.prio.air_basing }
			}

			# [equipment-selection] build the ordered list once per game. check_variable reads an
			# absent global as 0, so a fresh start and a campaign resumed from an older build both
			# take this branch; a later order change takes it again via the version stamp.
			if = {
				limit = { NOT = { check_variable = { global.WA_AI_uk_air_priority_version = constant:wa_ai_pc.air.uk_priority_version } } }
				WA_AI_uk_air_hosting_priority_init = yes
			}

			set_temp_variable = { _uk_air_rank_ = 0 }
			for_each_loop = {
				array = global.WA_AI_uk_air_priority_states
				value = _uk_air_sid_
				# named break, never the shared default `break` temp - queue functions set
				# plain `break = 1` and never reset it ([railway-landwar] lesson)
				break = _uk_air_started_

				add_to_temp_variable = { _uk_air_rank_ = 1 }
				# queue depth before the attempt, so dbg_started below counts projects actually
				# QUEUED rather than sites merely found (see the counter).
				set_temp_variable = { _uk_air_qnum_before_ = WA_AI_PC_queue^num }

				var:_uk_air_sid_ = {
					if = {
						limit = {
							# re-tested per candidate: the trigger, not this array, owns
							# membership, so an id dropped from the trigger is filtered out here
							WA_AI_uk_air_hosting_state = yes
							check_variable = { building_level@air_base < constant:wa_ai_pc.air.uk_target_level_cap }
							WA_AI_uk_air_site_buildable = yes
						}
						WA_AI_uk_air_queue_site = yes
					}
				}
			}

			# [equipment-selection] fallback tail - unordered scan over the SAME membership trigger,
			# reached only when the ordered walk queued nothing. This is what makes the duplicated
			# list safe: an id present in WA_AI_uk_air_hosting_state but missing from the array
			# loses its rank and gets built last instead of never. In a consistent build it fires
			# only when every ranked state is capped or unbuildable, and then finds nothing too.
			# WA_TLM_r8_uk_air_fallback_n counts it, so drift is visible in the next campaign.
			if = {
				limit = { check_variable = { _uk_air_started_ < 1 } }
				set_temp_variable = { _uk_air_rank_ = 0 }	# 0 = unranked, see the TLM rows
				set_temp_variable = { _uk_air_qnum_before_ = WA_AI_PC_queue^num }
				random_state = {
					limit = {
						WA_AI_uk_air_hosting_state = yes
						check_variable = { building_level@air_base < constant:wa_ai_pc.air.uk_target_level_cap }
						WA_AI_uk_air_site_buildable = yes
					}
					WA_AI_uk_air_queue_site = yes
				}
			}
		}
	}
}

# Build/upgrade air bases inside air theatres that are CONTESTED for ROOT, so the
# strategic_air_importance pulls in common/ai_strategy/WA_AI_MILITARY_DEFAULT_AIR_theatres.txt
# have friendly capacity to rebase into - the engine only rebases wings into in-range friendly
# air-base capacity, so a pull toward a theatre with no basing is inert (that file's KNOWN
# LIMITS names this effect as the follow-up).
# Theatre membership comes from the WA_AI_MILITARY_AIR_theatre_state_* triggers
# (common/scripted_triggers/WA_AI_MILITARY_triggers.txt), which must-match the contested lists.
#
# NEED GUARDRAILS (all generic, no tags):
# 1. Theatre-scoped sites: projects can only start in theatre member states - this effect can
#    never build on home soil outside a contested theatre (a country at peace-depth in its own
#    hemisphere is structurally unreachable).
# 2. Side-wide capacity deficit: a contested theatre only qualifies while the SIDE's existing
#    + ROOT-queued air-base capacity across that theatre's states is below
#    constant:wa_ai_pc.air.theatre_hosted_share of the side's deployed planes. Capacity counts every state
#    controlled by ROOT, a faction ally or a co-belligerent (same own-side definition as the
#    contested triggers), so once anyone on the side has built enough, every other member
#    stops queuing - a minor never piles onto a theatre its majors already covered, and a
#    co-belligerent outside the faction sees the allied capacity and stays idle.
# 3. Token air forces never enter (<= 99 deployed planes, WA air scale ~1/3 vanilla): a
#    country with no meaningful aviation has no basing need of its own.
# 4. Per-state cap: the level ladder stops selecting a state at constant:wa_ai_pc.air.theatre_target_level,
#    and always raises the least-developed member state first, so capacity spreads 0->4
#    across the chain instead of stacking one level-10 island.
#
# [na-corridor] GROUND-COMMITTED THEATRES FIRST. Without this the ladder draws uniformly over EVERY
#    deficit theatre's states from rung 0, so a landing theatre whose allied states already sit at
#    level 3-4 loses the draw to level-0 islands every pulse, and its projects share the 300 band
#    with uk_air and never get a factory. The discriminator needs no theatre list and no tag: does
#    ROOT have divisions standing in the theatre?
#      - COMMITTED theatre = a deficit theatre with >= 1 ROOT division on a side-controlled
#        member state (folded into the pass-2 state walk, so no extra scan). While any
#        committed theatre is in deficit, the ladder draws ONLY from committed theatres.
#      - The per-state cap rises to constant:wa_ai_pc.air.theatre_target_level_committed there: a beachhead
#        of two states must host what ten islands would spread.
#      - Priority constant:wa_ai_pc.prio.air_front (350): above uk_air (300) inside the [pc-queue] lane and
#        the sorted fill, below rail-prewar (500), so the [pc-queue] band order is untouched.
#    Nothing changes for a country with no divisions in any deficit theatre (the island-chain
#    case this effect was written for keeps its uniform spread).
#    Probe: WA_TLM_r52_thair_front_* (checklist R52).
#
# ROOT = any AI country. Called from WA_AI_priority_construction_strategies
# (WA_AI_background.1, ~2 day cadence, AI only), admitting on its own terms for the same reason as
# the UK hosting strategy above (a railway-saturated queue would permanently close a busy-queue
# gate for exactly the countries this exists for); bounded instead by _project_queue_max, the
# deficit gate and the civilian-factory floor below.
WA_AI_build_theatre_air_bases = {


	if = {
		limit = {
			has_war = yes
			surrender_progress < 0.1
			# guardrail 3: skip token air forces - same floor as the theatre pulls
			has_deployed_air_force_size = { size > 99 }
			# floored availability: the bare engine variable reads ~0 once the queue saturates
			WA_AI_PC_has_project_civs_16 = yes
		}

		# DIAGNOSIS INSTRUMENTATION - same rationale as the UK hosting dbg family above (temp
		# variables are invisible in saves). Remove once verified.
		#   dbg_called   pulses that passed the outer gate
		#   dbg_target   side capacity target in levels, last gated pulse
		#   dbg_active   contested theatres still in deficit, last gated pulse
		#   dbg_best     ladder level of the last started project
		#   dbg_started  projects actually queued
		add_to_variable = { WA_AI_thair_dbg_called = 1 }

		### pass 1 - which theatres are contested for us

		set_temp_variable = { _thair_naf_ = 0 }
		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_north_africa = yes } set_temp_variable = { _thair_naf_ = 1 } }
		set_temp_variable = { _thair_ita_ = 0 }
		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_italy = yes } set_temp_variable = { _thair_ita_ = 1 } }
		set_temp_variable = { _thair_weu_ = 0 }
		if = { limit = { WA_AI_MILITARY_AIR_theatre_contested_western_europe = yes } set_temp_variable = { _thair_weu_ = 1 } }
		set_temp_variable = { _thair_sca_ = 0 }

								AND = { check_variable = { _thair_eeu_ = 1 } WA_AI_MILITARY_AIR_theatre_state_eastern_europe = yes }
								AND = { check_variable = { _thair_mde_ = 1 } WA_AI_MILITARY_AIR_theatre_state_middle_east = yes }
								AND = { check_variable = { _thair_bur_ = 1 } WA_AI_MILITARY_AIR_theatre_state_burma_india = yes }
								AND = { check_variable = { _thair_sea_ = 1 } WA_AI_MILITARY_AIR_theatre_state_southeast_asia = yes }
								AND = { check_variable = { _thair_pac_ = 1 } WA_AI_MILITARY_AIR_theatre_state_pacific_islands = yes }
							}
							check_variable = { building_level@air_base = _thair_try_level_ }
							# [na-corridor] phase 1 restricts the rung to the contested edge; phase 2 lifts it.
							# Placed AFTER the theatre and level tests on purpose - the ring walk is two
							# nested any_neighbor_state and must not run on every state of the map.
							OR = {
								check_variable = { _thair_near_only_ = 0 }
								WA_AI_PC_state_is_near_contested_edge = yes
							}
							OR = {
								AND = {	# our own territory
									is_controlled_by = ROOT
									WA_AI_available_AIR = yes
									WA_AI_PC_can_build_project = yes
								}
								AND = {	# subject territory (the engine allows overlord construction in subjects; PC validates the same)
									CONTROLLER = { is_subject_of = ROOT }
									WA_AI_PC_can_build_project = yes
								}
								AND = {	# faction ally's territory (air_base has allied_build = yes),
										# only when the controller cannot or will not act on the
										# deficit itself: player-controlled, not actually funding
										# air bases, or genuinely poor. [uk-air-basing]: capability
										# is not funding, hence a RATE floor and not a
										# spending-anything test - once air bases hold the 300 band
										# the host almost always assigns SOME civ, and that would
										# lock the ally out permanently.
									NOT = { is_controlled_by = ROOT }
									NOT = { CONTROLLER = { is_subject_of = ROOT } }
									WA_AI_available_AIR_allied = yes
									CONTROLLER = {
										OR = {
											is_ai = no
											check_variable = { WA_AI_PC_air_factories_assigned < constant:wa_ai_pc.alloc.air_host_funding_floor }
											num_of_civilian_factories_available_for_projects < 16
										}
									}
									NOT = { WA_AI_PC_has_same_type_project_here = yes }
									# [uk-air-basing] same free-slot test the own/subject branches get
									# via WA_AI_PC_can_build_project. Without it the allied builder
									# funds a project WA_AI_PC_complete_project_by_id will then
									# refuse to spawn for want of a slot.
									WA_AI_PC_building_slot_available_for_type = yes
								}
							}
						}

						set_temp_variable = { _project_building_type = 2 }
						set_temp_variable = { _project_province_id = 0 }
						set_temp_variable = { _project_target_state = 0 }
						set_temp_variable = { _project_priority = _thair_prio_ }
						set_temp_variable = { _project_queue_num = 1 }
						set_temp_variable = { _project_queue_max = constant:wa_ai_pc.qmax.theatre_air }
						# [equipment-selection] own budget, independent of the UK hosting strategy's - see
						# the matching pair there.
						set_temp_variable = { _project_type_id = constant:wa_ai_pc.type_id.theatre_air }
						set_temp_variable = { _project_queue_max_scoped = 1 }
						set_temp_variable = { _project_target_level = _thair_level_cap_ }
						set_temp_variable = { _thair_qnum_before_ = ROOT.WA_AI_PC_queue^num }
						WA_AI_PC_start_project = yes
						# [equipment-selection] retire the tag - see the matching note in the UK strategy.
						set_temp_variable = { _project_type_id = 0 }

						# WA_AI_PC_start_project can decline internally (queue_max, same-type dedup,
						# province duplicate) with NO return signal. Only count a start (and stop
						# the ladder) when the queue actually grew; a declined start lets the ladder
						# try the next level instead of silently ending the pulse.
						if = {
							limit = { ROOT = { check_variable = { WA_AI_PC_queue^num > _thair_qnum_before_ } } }

							# TEMP R16 live-test logging - remove with the WA_AI_thair_dbg_* family
							log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: air_base project started at level [?_thair_try_level_|.0] in [This.GetID] [This.GetName]"

							ROOT = {
								set_temp_variable = { _thair_started_ = 1 }
								set_variable = { WA_AI_thair_dbg_best = _thair_try_level_ }
								add_to_variable = { WA_AI_thair_dbg_started = 1 }

								# tlm: WA_TLM_r52_thair_front_n / _first_t / _last_t - consumer:
								# checklist R52. VERIFIED effect only (the queue grew, above), and
								# only when the [na-corridor] committed rule was in force for this start.
								if = { limit = { check_variable = { _thair_committed_ = 1 } }
									add_to_variable = { WA_TLM_r52_thair_front_n = 1 }
									if = { limit = { check_variable = { WA_TLM_r52_thair_front_first_t = 0 } }
										set_variable = { WA_TLM_r52_thair_front_first_t = global.WA_TLM_clock }
									}
									set_variable = { WA_TLM_r52_thair_front_last_t = global.WA_TLM_clock }
								}

								# tlm: WA_TLM_r114_thair_near_n / _far_n / _first_t / _last_t - consumer:
								# checklist R78. VERIFIED effect only (the queue grew, above). The PAIR is
								# the metric, never either half alone: near_n counts starts the proximity
								# phase won, far_n starts only the fallback could make, and their RATIO
								# says whether [na-corridor] changed where the planes get based. far_n is
								# not a failure counter - a theatre contested at a distance lands there.
								if = { limit = { check_variable = { _thair_near_only_ = 1 } }
									add_to_variable = { WA_TLM_r114_thair_near_n = 1 }
									if = { limit = { check_variable = { WA_TLM_r114_thair_near_first_t = 0 } }
										set_variable = { WA_TLM_r114_thair_near_first_t = global.WA_TLM_clock }
									}
									set_variable = { WA_TLM_r114_thair_near_last_t = global.WA_TLM_clock }
								}
								else = {
									add_to_variable = { WA_TLM_r114_thair_far_n = 1 }
								}
							}
						}
					}

					add_to_temp_variable = { _thair_try_level_ = 1 }

					# [na-corridor] phase flip. Phase 1 exhausted without a start - drop the proximity filter
					# and walk the rungs again over the whole theatre. Guarded on _thair_near_only_ = 1,
					# so it happens at most once per pulse and the loop is bounded by 2 x _thair_level_cap_.
					# COST: worst case 16 random_state scans per pulse per builder (the committed cap
					# constant:wa_ai_pc.air.theatre_target_level_committed is 8 and both phases can walk
					# all of it). That worst case is ALSO the cheap case for the ring walk - it is only
					# reached when phase 1 found no eligible edge state, i.e. when the two-ring test was
					# answering false quickly.
					if = {
						limit = {
							check_variable = { _thair_near_only_ = 1 }
							check_variable = { _thair_started_ < 1 }
							NOT = { check_variable = { _thair_try_level_ < _thair_level_cap_ } }
						}
						set_temp_variable = { _thair_near_only_ = 0 }
						set_temp_variable = { _thair_try_level_ = 0 }
					}
				}

				# [na-corridor] clear the phase flag on the way out (AGENTS.md editing rule 9): the ladder
				# re-arms it to 1 before every loop, so leaving a value behind is a temp with no owner.
				set_temp_variable = { _thair_near_only_ = 0 }

				# TEMP R16 live-test logging - remove with the WA_AI_thair_dbg_* family
				if = { limit = { check_variable = { _thair_started_ < 1 } }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | THEATRE AIR: deficit but no eligible member state (ladder exhausted at level [?_thair_try_level_|.0])"
				}
			}
		}
	}
