############################################################################################################
#	Fix 115 / 116 - COALITION SEA WEIGHT. ROOT = country. Monthly (WA_AI_misc_on_actions on_monthly).
#	>>> WA_AI_PRODUCTION_coalition_sea_share (0..1), a country variable read by
#	    WA_AI_PRODUCTION_should_build_convoy_relief and, through it, by the convoy arsenal blocks
#	    and the major convoy stop in WA_AI_PRODUCTION_DEFAULT_navy.txt.
#
#	What it protects: the convoy arsenal charges a major 15 dockyards plus 85 unit_ratio to build
#	hulls for its coalition, and it used to open on ONE partner being short. MEASURED on campaign
#	3d68a183: Germany held four convoy lines - three pinned at the 15-dockyard per-line cap, 46 to
#	59 dockyards, 6.0 to 7.6 % of every factory on its military and naval lines - from 1942.6 to
#	1944.6, feeding an Italy that is convoy-short for 56 weeks. The Axis is a land coalition; that
#	industry belonged on the eastern front.
#
#	THE MEASURE: the share of the coalition's INDUSTRY (civ + mil + naval factories) that we cannot
#	reach OVERLAND. Weighted, because an existential test is overturned by any minor - the Axis of
#	that campaign already reaches Africa through ITS and ITL, two colonial subjects with negligible
#	industry (user ruling 2026-08-20).
#
#	Fix 116 replaced the first classifier, capital CONTINENT, the same day and before any campaign
#	ran it. Continents do not measure water, in both directions: London and Berlin are both `europe`
#	with the Channel between them, while Berlin and Cairo are two continents joined by land through
#	Turkey and the Sinai. The classifier is now the state A* the lend-lease relief already uses -
#	WA_AI_LEND_LEASE_relief_land_access, pathfind type 3, capital state to capital state through
#	friendly territory - so "across water" means what it says.
#
#	Assumes: a capital-to-capital land route is a fair stand-in for "our wars are joined by land".
#	It is silent about WHERE a member fights - a coalition sitting on one landmass while campaigning
#	overseas reads 0, which is the Axis-in-Africa case and is the intended answer here, not an error.
#	Fact is gone when: WA_AI_LEND_LEASE_relief_land_access changes its scope contract (it reads
#	ROOT's capital as the start and the scope it is called in as the target - which is why it can be
#	called from inside the every_country walk below with no rebinding).
#
#	Cost: one A* per side member with industry, per candidate, per month; the call site restricts
#	candidates to AI majors at war above the large-dockyard band, and members with no factories are
#	skipped before the pathfind. The relief system already runs this same A* weekly per starving
#	recipient, so this is the cheaper of the two callers.
############################################################################################################

WA_AI_PRODUCTION_update_coalition_sea_weight = {
	set_variable = { WA_AI_PRODUCTION_coalition_sea_share = 0 }

	set_temp_variable = { _csw_side_ic_ = 0 }
	set_temp_variable = { _csw_off_ic_ = 0 }

	# Fix 117 - "our side" HERE IS THE FACTION, not the war, and that is a deliberate deviation from
	# the three-term idiom the rest of the AI uses (self / faction ally / has_war_together_with).
	# MEASURED out of campaign 3d68a183 with the third term IN (scratch replica of this effect,
	# 1943.6): Germany read 0.254 against a 0.25 bar - on the knife edge - because JAPAN, which sits
	# in its own faction and merely shares enemies, put 490 unreachable factories into the Axis's
	# numerator. Restricted to the faction, the same reading is 0.056. That is the user's 'one Iran
	# must not make the Axis maritime' failure wearing a bigger hat.
	# Why it is right and not merely convenient: an arsenal spends dockyards on hulls it will HAND
	# OVER, and what you hand over follows the alliance you supply - not everyone who happens to
	# fight the same enemy on another continent. The transfer path itself stays wider: the convoy
	# pair test (WA_AI_LEND_LEASE_convoy_pair_is_valid) still accepts a co-belligerent.
	# Fix 118 (re-derived 2026-08-20 - the original justification was WITHDRAWN, see below). Four
	# terms, and the fourth one enters OVERLORD only behind an `is_subject = yes` guard:
	#   tag / is_in_faction_with   our faction
	#   is_subject_of = ROOT       our own subjects, which need not be faction members
	#   is_subject + OVERLORD      Fix 27's case - a subject of an ALLY need not be a faction member
	#                              either (ITL is ITA's subject and never joins), and its industry is
	#                              still our side's.
	# WHY THE GUARD RATHER THAN A BARE `OVERLORD = { }`: nobody has measured what a bare scope into a
	# scope that does not exist does to the rest of an OR. The engine's trigger documentation does not
	# say, error.log said nothing, and the reading that was once offered as proof turned out to be an
	# artefact. So this is ASSUMED, and the guard costs one trigger to never have to know.
	# WITHDRAWN CLAIM, recorded so it is not re-derived: this comment used to state that a bare
	# OVERLORD scope "poisons the WHOLE limit", on the strength of `event wa_test.311` returning an
	# empty coalition on 1940.5. That reading was an artefact of the TEST HARNESS's call site, not of
	# this limit - the same harness fired from a different event file read the coalition correctly
	# with the OVERLORD form still in place. See lessons log, "Two call sites, one effect". The
	# OUTCOME here is still right and is independently justified above; the old reason was not.
	# Cost of the fourth term, MEASURED on 3d68a183: nil. Its set is empty there - the Axis roster
	# lists ITS, ITL, RCZ, RDE, RNO, RHO and the Allies UKO, UKT, UKN, RAJ, so subjects ARE faction
	# members in that campaign. It is carried for the world where they are not.
	every_country = {
		limit = {
			exists = yes
			has_capitulated = no
			OR = {
				tag = ROOT
				is_in_faction_with = ROOT
				is_subject_of = ROOT
				AND = {
					is_subject = yes
					OVERLORD = { is_in_faction_with = ROOT }
				}
			}
		}

		# Industry, all three kinds. Dockyards are in on purpose: a coalition whose weight is in
		# shipyards is a maritime coalition by construction.
		set_temp_variable = { _csw_m_ic_ = num_of_civilian_factories }
		add_to_temp_variable = { _csw_m_ic_ = num_of_military_factories }
		add_to_temp_variable = { _csw_m_ic_ = num_of_naval_factories }

		if = {
			limit = { check_variable = { _csw_m_ic_ > 0 } }
			add_to_temp_variable = { _csw_side_ic_ = _csw_m_ic_ }

			# Ourselves: always joined to ourselves by land, and skipping the A* here is what keeps
			# the denominator honest without paying for a pathfind that can only answer one way.
			if = {
				limit = { NOT = { tag = ROOT } }
				# THIS = this member (the A*'s target), ROOT = the candidate (its start). Exactly the
				# scope contract WA_AI_LEND_LEASE_relief_land_access is written against.
				#
				# _llr_donor is an INPUT of pathfind type 3, not of the land-access wrapper: the
				# neighbour filter admits a state controlled by ROOT, by `var:_llr_donor`, by a
				# subject of either, or by a faction ally / co-belligerent of ROOT. The lend-lease
				# caller sets it from its own donor pick; this caller must set it too, or the filter
				# reads an unset variable - which in HOI4 is a scope token, not 0 (lessons log), and
				# the clause silently means something else. Set to the member for the same reason the
				# relief sets it to the donor: the target's own ground is walkable ground.
				set_temp_variable = { _llr_donor = THIS.id }
				WA_AI_LEND_LEASE_relief_land_access = yes
				if = {
					limit = { NOT = { check_variable = { _llr_path_ok = 1 } } }
					add_to_temp_variable = { _csw_off_ic_ = _csw_m_ic_ }
				}
			}
		}
	}

	if = {
		limit = { check_variable = { _csw_side_ic_ > 0 } }
		set_variable = { WA_AI_PRODUCTION_coalition_sea_share = _csw_off_ic_ }
		divide_variable = { WA_AI_PRODUCTION_coalition_sea_share = _csw_side_ic_ }
	}

	if = { limit = { has_country_flag = WA_AI_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONVOY ARSENAL: coalition sea share [?WA_AI_PRODUCTION_coalition_sea_share|.3] (unreachable overland [?_csw_off_ic_|.0] of [?_csw_side_ic_|.0] factories)"
	}
}
