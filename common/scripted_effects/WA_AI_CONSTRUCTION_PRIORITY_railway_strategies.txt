############################################################################################################
#	World Ablaze AI mod - Railway Construction Priority - Strategy Implementations
#	Land War, Overseas War, and Pre-War Preparation strategies
############################################################################################################

### Constants: this file declares none. Priority bands are constant:wa_ai_pc.prio.*, rail levels are
### constant:wa_ai_railway.*; the per-enemy route budget is enforced by
### WA_AI_PC_railway_land_frontline_candidate (WA_AI_CONSTRUCTION_triggers.txt) - this file only
### counts _routes_for_this_enemy up.

############################################################################################################
#	STRATEGY 1: LAND WAR
#	Capital -> frontline states holding a supply hub / naval base, skipping states already at the
#	family cap (native has_railway_level check, so focus- and event-built rail counts).
#	Only frontlines where ROOT or a subject DIRECTLY borders the enemy: a frontline reached through
#	an ally (Italy -> SOV through Germany) is not ours to supply.
############################################################################################################

WA_AI_PC_railway_STRATEGY_land_war = {

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		set_temp_variable = { _log_cap_state = global.WA_AI_MAP_province_state_id^capital_province }
		var:_log_cap_state = { ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: STARTED route_start=[?capital_province] ([Prev.GetName]) landmass=[?capital_landmass]" } }
	}

	# Default route parameters (will be overridden for overseas targets)
	set_temp_variable = { default_route_start = capital_province }
	# [na-corridor] PROVISIONAL floor only: every route is overwritten by WA_AI_PC_rail_size_route in
	# WA_AI_PC_railway_land_consider_frontline. A fixed target here is a level nothing asked for -
	# it lays rail behind an entry port that cannot feed it.
	set_temp_variable = { default_route_level = constant:wa_ai_railway.land_war.rail_level_floor }
	set_temp_variable = { route_priority = _project_priority }
	set_temp_variable = { high_route_priority = route_priority }
	multiply_temp_variable = { high_route_priority = 1.1 }


	### For each RELEVANT enemy we DIRECTLY border (not through allies), find frontline states with supply hubs
	### This prevents building railways to distant frontlines (e.g., Italy -> SOV through Germany)
	### Uses pre-filtered _relevant_enemies_ array to reduce iteration count

	for_each_scope_loop = { array = _relevant_enemies_
		set_temp_variable = { _current_enemy_tag = THIS }

		# Check if ROOT (or ROOT's puppets) directly borders this enemy
		ROOT = {
			set_temp_variable = { _enemy_tag = _current_enemy_tag }
			WA_AI_PC_railway_country_borders_enemy = yes
		}

		# Only process enemies we directly border
		if = {
			limit = { check_variable = { borders_enemy_ = 1 } }

			if = { limit = { has_country_flag = WA_AI_construction_logging }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: processing enemy [?_current_enemy_tag.GetName]"
			}

			### Per-enemy route counter, and the railway level at which a frontline is "done"
			set_temp_variable = { _routes_for_this_enemy = 0 }
			# [na-corridor] must stay equal to the sizer's cap. Set higher, no frontline ever reads
			# "done" and every pass redraws routes that queue no segments.
			set_temp_variable = { _check_railway_level = constant:wa_ai_railway.land_war.rail_level_cap }

			### [na-corridor] three candidate populations, ONE shared body
			### (WA_AI_PC_railway_land_consider_frontline). Order is load-bearing - the per-enemy
			### route budget is consumed top-down, so ROOT's own soil always outranks a subject's
			### and a subject's outranks an ally's.

			### 1. ROOT's own frontline states
			ROOT = {
				every_controlled_state = {
					limit = { WA_AI_PC_railway_land_frontline_candidate = yes }
					WA_AI_PC_railway_land_consider_frontline = yes
				}
			}

			### 2. [railway-pathfinding] ROOT's subject frontline states. Puppet ground is deliberately
			### included - pathfinder type 2 traverses puppet provinces and build_railway works on any
			### province regardless of controller; excluding it was tried and left subject frontlines
			### (UKE Egypt, ITL Libya, RAJ Burma) with no routes at all.
			### The explicit ROOT wrapper is load-bearing: unwrapped, this loop runs in the
			### for_each_scope_loop's scope (THIS = the ENEMY) and walks the ENEMY's puppets, where the
			### acceptance test below can never pass. Tell it is gone: subject frontlines stop scoring.
			ROOT = {
				every_subject_country = {
					every_controlled_state = {
						limit = { WA_AI_PC_railway_land_frontline_candidate = yes }
						WA_AI_PC_railway_land_consider_frontline = yes
					}
				}
			}

			### 3. [na-corridor] faction allies that cannot FUND their own logistics. The PC executor
			### already accepts allied ground (rail and ports are allied_build = yes); only the
			### selectors excluded allies, so a corridor ending on an ally's soil had no builder at
			### all - seen on FRA-held Tunisia and Free-French Normandy.
			### "Cannot fund", not "cannot build": the authority test has no civ-factory term at war,
			### so a broke belligerent reads capable and locks its allies out of ground it cannot
			### develop itself. Anyone above the wartime funding floor stays excluded, which is what
			### keeps two healthy majors out of each other's rail networks.
			### BOUNDED by the per-run route list: routes.max_per_enemy is consumed at population time
			### and this loop runs THIRD, so it only sees budget the ROOT and subject loops left.
			### NOT bounded, and deliberately not claimed to be, by the QUEUE - queue slots are held
			### for months, so an ally's segments can still crowd out ROOT's own.
			ROOT = {
				every_country = {
					limit = {
						exists = yes
						NOT = { tag = ROOT }
						is_in_faction_with = ROOT
						NOT = { is_subject_of = ROOT }
						WA_AI_PC_country_cannot_fund_own_logistics = yes
					}
					every_controlled_state = {
						limit = { WA_AI_PC_railway_land_frontline_candidate = yes }
						WA_AI_PC_railway_land_consider_frontline = yes
					}
				}
			}
		}
	}

	### Sort targets by enemy threat (highest threat enemy's targets first)
	if = { limit = { check_variable = { railway_end_provinces_^num > 1 } }
		WA_AI_PC_railway_score_and_sort_by_enemy_threat = yes
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: COMPLETED - [?railway_end_provinces_^num] targets found"
	}
}

############################################################################################################
#	[na-corridor] shared per-state body for STRATEGY_land_war
#
#	THIS = a frontline state that already passed WA_AI_PC_railway_land_frontline_candidate.
#	ROOT = the builder country (NOT necessarily the state's controller).
#
#	Inputs, all chain-global temps set by the caller: _current_enemy_tag, _routes_for_this_enemy,
#	  _check_railway_level, default_route_start, default_route_level, route_priority, capital_landmass
#	Outputs: appends to railway_start_provinces_ / _end_provinces_ / _target_levels_ / _priorities_ /
#	  _enemy_tags_ / _port_upgrades_, and increments _routes_for_this_enemy.
#
#	One body serves all three populations because the supply-hub acceptance test is the generalised
#	OR below; per-population copies of this body differed only in that term.
############################################################################################################

############################################################################################################
#	[na-corridor] the land-war family's binding of WA_AI_PC_rail_size_route (railway_helpers).
#	THIS = the frontline state (supply_hub_province_ and route_start in scope), ROOT = the builder.
#	Sets `route_level` and nothing else.
#	The CALLER must set the entry mode - the one thing that differs between the three call sites:
#	  _rsize_entry_mode_ = 2  overland from our own capital -> demand decides, no entry cap
#	  _rsize_entry_mode_ = 1  overseas -> _rsize_entry_supply_ = overseas_entry_supply_ (the sea leg's
#	                          bottleneck, published by WA_AI_PC_analyze_overseas_supply_chain)
#	Demand is the ONE frontline hub province, not the whole path: this runs once PER ROUTE (up to
#	routes.max_total per execution) while the sizer's presence walk is every_country x states. The
#	corridor can afford its whole node list only because it sizes once per pass.
#	Floor/cap are constant:wa_ai_railway.land_war.*, NOT the corridor's pair - the two families share
#	a model, not a tuning.
############################################################################################################

WA_AI_PC_railway_land_size_this_route = {
	set_temp_variable = { _rsize_hub_ = supply_hub_province_ }
	ROOT = {
		clear_temp_array = _rsize_prov_
		clear_temp_array = _rsize_ours_
		add_to_temp_array = { _rsize_prov_ = _rsize_hub_ }
		add_to_temp_array = { _rsize_ours_ = 1 }
		set_temp_variable = { _rsize_floor_ = constant:wa_ai_railway.land_war.rail_level_floor }
		set_temp_variable = { _rsize_cap_ = constant:wa_ai_railway.land_war.rail_level_cap }
		WA_AI_PC_rail_size_route = yes
		clear_temp_array = _rsize_prov_
		clear_temp_array = _rsize_ours_
	}
	set_temp_variable = { route_level = rsize_rail_tgt_ }
}

WA_AI_PC_railway_land_consider_frontline = {

	if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
		ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: frontline state [Prev.GetName] passed limit check" }
	}

	# Skip single-node states (only port, no supply hub to connect to)
	WA_AI_PC_coastal_state_is_single_node = yes
	if = {
		limit = { check_variable = { is_single_node_ = 0 } }


		# Find the exact province with the supply hub
		WA_AI_PC_railway_get_supply_hub_province = yes

		if = {
			limit = {
				check_variable = { supply_hub_province_ > 0 }
				NOT = { is_in_array = { railway_end_provinces_ = supply_hub_province_ } }
				# Verify the supply hub PROVINCE is actually held, not just the state - province
				# control diverges from state control on contested fronts (an ITL-controlled
				# Cyrenaica can hold ENG provinces inside it).
				# [na-corridor] ROOT, or this state's controller - which the caller's candidate filter
				# has already vetted as ROOT / a subject of ROOT / a dependent faction ally.
				OR = {
					ROOT = { controls_province = var:supply_hub_province_ }
					CONTROLLER = { controls_province = var:supply_hub_province_ }
				}
			}

			if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
				ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] hub_prov=[?supply_hub_province_] ACCEPTED" }
			}

			WA_AI_PC_get_state_id = yes
			set_temp_variable = { _target_state_id = state_id_ }
			# Get target landmass
			set_temp_variable = { target_landmass = global.WA_AI_MAP_state_landmass^_target_state_id }

			# Determine route start and level based on landmass
			if = {
				limit = { check_variable = { target_landmass = capital_landmass } }
				# Same landmass: use capital as route start.
				# [east-front-rail] this assignment is the ONLY writer of route_start on the overland
				# path; without it the `route_start > 0` gate below fails for every same-landmass
				# frontline and the land-war family emits zero routes.
				# Tell it is gone again: `RAILWAY LAND: COMPLETED - 0 targets found` after ACCEPTED lines.
				set_temp_variable = { route_start = default_route_start }
				# [railway-landwar] the levelling MUST stay after the pathfind - read before it, the
				# path is the PREVIOUS target's or empty, and every land route targets level 1 and
				# queues no segments. Until then this carries the family floor and
				# _route_level_from_path marks the route for the recompute (a sizing, not a ratchet).
				set_temp_variable = { route_level = default_route_level }
				set_temp_variable = { _route_level_from_path = 1 }
			}
			else = {
				# Different landmass: find nearest receiving port via distance-limited search.
				# [railway-pathfinding] one port analysis PER TARGET; caching it per landmass was tried
				# and made every target on a mega-landmass share one port (Singapore for North Africa).
				set_temp_variable = { _target_state_for_port_search = _target_state_id }
				ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
				set_temp_variable = { route_start = overseas_route_start_ }
				# [na-corridor] the port analysis gives ENTRY capacity only, not whether anything on the
				# far end needs it: size against min(demand, entry) in GIVEN mode. An overseas chain is
				# in SERIES, so entry is the BOTTLENECK of home and receiving port, never their sum.
				set_temp_variable = { route_level = overseas_max_railway_level_ }
				set_temp_variable = { _rsize_entry_mode_ = 1 }
				set_temp_variable = { _rsize_entry_supply_ = overseas_entry_supply_ }
				WA_AI_PC_railway_land_size_this_route = yes
				set_temp_variable = { _route_level_from_path = 0 }

				# Queue port upgrade if it's the bottleneck
				if = {
					limit = {
						check_variable = { overseas_max_railway_level_ < constant:wa_ai_railway.land_war.rail_level_cap }
						check_variable = { overseas_receiving_port_state_ > 0 }
						NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
					}
					add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
				}
			}

			# Only proceed if we have a valid route start
			if = {
				limit = { check_variable = { route_start > 0 } }

				### Check for existing project to this destination - skip pathfinding if found
				set_temp_variable = { _skip_pathfinding = 0 }
				for_each_loop = { array = ROOT.WA_AI_PC_queue value = _check_proj break = _found_existing
					if = {
						limit = {
							check_variable = { ROOT.WA_AI_PC_building_type^_check_proj = 13 } # railway
							check_variable = { ROOT.WA_AI_PC_target_province^_check_proj = supply_hub_province_ }
						}
						set_temp_variable = { _skip_pathfinding = 1 }
						set_temp_variable = { _found_existing = 1 }
					}
				}

				if = {
					limit = { check_variable = { _skip_pathfinding = 0 } }

					# Skip if start and end are the same province (already connected)
					if = {
						limit = { NOT = { check_variable = { route_start = supply_hub_province_ } } }

						# Validate land path exists from route start to this frontline
						set_temp_variable = { _pathfind_prov_start = route_start }
						set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
						set_temp_variable = { _pathfind_prov_type = 2 } # Rail networks - only ROOT-controlled provinces
						set_temp_variable = { _pathfind_prov_allow_partial = 1 } # Allow partial paths to frontier
						WA_AI_PATHFIND_PROV_get_path = yes

						# [railway-pathfinding] If same-landmass route got PARTIAL or FAIL, try overseas fallback
						# This handles cases like Italy→Libya where both are on landmass 2 (Afro-Eurasia)
						# but there's no land path across the Mediterranean
						set_temp_variable = { _direct_success = pathfind_prov_success_ }
						if = {
							limit = {
								check_variable = { target_landmass = capital_landmass }
								NOT = { check_variable = { _direct_success = 1 } }
							}
							set_temp_variable = { _target_state_for_port_search = _target_state_id }
							ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
							if = {
								limit = { check_variable = { overseas_route_start_ > 0 } }
								set_temp_variable = { _pathfind_prov_start = overseas_route_start_ }
								set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
								set_temp_variable = { _pathfind_prov_type = 2 }
								set_temp_variable = { _pathfind_prov_allow_partial = 1 }
								WA_AI_PATHFIND_PROV_get_path = yes
								if = {
									limit = {
										check_variable = { pathfind_prov_success_ > 0 }
										OR = {
											check_variable = { _direct_success = 0 }
											check_variable = { pathfind_prov_success_ = 1 }
										}
									}
									if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
										ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] overseas fallback SUCCESS (from port [?overseas_route_start_])" }
									}
									set_temp_variable = { route_start = overseas_route_start_ }
									set_temp_variable = { route_level = overseas_max_railway_level_ }
									set_temp_variable = { _rsize_entry_mode_ = 1 }	# [na-corridor]
									set_temp_variable = { _rsize_entry_supply_ = overseas_entry_supply_ }
									WA_AI_PC_railway_land_size_this_route = yes
									set_temp_variable = { _route_level_from_path = 0 }	# [railway-landwar] level now owned by port analysis
									set_temp_variable = { _direct_success = pathfind_prov_success_ }
									if = {
										limit = {
											check_variable = { overseas_max_railway_level_ < constant:wa_ai_railway.land_war.rail_level_cap }
											check_variable = { overseas_receiving_port_state_ > 0 }
											NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
										}
										add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
									}
								}
								else = {
									# [na-corridor] marking for the recompute is safe even though the fallback
									# pathfind overwrote pathfind_prov_path_: the sizer reads the frontline
									# hub province, not the path, and this branch has not touched it.
									set_temp_variable = { _route_level_from_path = 1 }
									if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
										ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] overseas fallback also PARTIAL — keeping direct route" }
									}
								}
							}
						}

						if = {
							limit = { check_variable = { _direct_success > 0 } }

							# [na-corridor] NO ratchet in the war family - the route is sized by the same
							# helper the corridor uses. "One level above what this route already has,
							# clamped at 5" was tried: it references neither demand nor entry capacity,
							# climbs to the clamp on its own one pass at a time, and overwrites the
							# corridor's lower cap on shared segments because
							# WA_AI_PC_start_railway_project only compares current < target.
							# No `pathfind_prov_path_^num > 1` condition: the sizer reads the frontline
							# hub, not the path, so a 1-node path sizes like any other.
							if = {
								limit = { check_variable = { _route_level_from_path = 1 } }
								set_temp_variable = { _rsize_entry_mode_ = 2 }
								set_temp_variable = { _rsize_entry_supply_ = 0 }
								WA_AI_PC_railway_land_size_this_route = yes
							}

							add_to_temp_array = { railway_start_provinces_ = route_start }
							add_to_temp_array = { railway_end_provinces_ = supply_hub_province_ }
							add_to_temp_array = { railway_target_levels_ = route_level }
							add_to_temp_array = { railway_priorities_ = route_priority }
							add_to_temp_array = { railway_enemy_tags_ = _current_enemy_tag }

							### Increment per-enemy counter
							add_to_temp_variable = { _routes_for_this_enemy = 1 }

						}
					}
				}
			}
		}

		else = {
			# Log why state was skipped
			if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
				if = { limit = { check_variable = { supply_hub_province_ = 0 } }
					ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] SKIPPED - no supply hub province found" }
				}
				else_if = { limit = { is_in_array = { railway_end_provinces_ = supply_hub_province_ } }
					ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] SKIPPED - already in targets" }
				}
				else = {
					ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY LAND: [Prev.GetName] hub_prov=[?supply_hub_province_] SKIPPED - hub province not held by ROOT or the state controller" }
				}
			}
		}
	} # end else (not single-node)
}

############################################################################################################
#	STRATEGY 2: OVERSEAS WAR
#	A) Build max railway + port to best home port (within 5 states of capital)
#	B) Build railways from beachhead port on enemy continent to front lines
############################################################################################################

WA_AI_PC_railway_STRATEGY_overseas_war = {

	if = { limit = { has_country_flag = WA_AI_construction_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: STARTED" }

	set_temp_variable = { route_level = 5 }
	set_temp_variable = { route_priority = _project_priority }

	### PART A: Home port infrastructure
	### Use the new helper to find best port
	WA_AI_PC_railway_find_best_home_port = yes

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: Part A - home port state=[?best_home_port_state_] level=[?best_home_port_level_]"
	}

	### Queue railway from capital to home port (if not capital itself)
	if = {
		limit = {
			check_variable = { best_home_port_state_ > 0 }
			NOT = { check_variable = { best_home_port_state_ = capital_state_id } }
		}
		add_to_temp_array = { railway_start_provinces_ = capital_province }
		add_to_temp_array = { railway_end_provinces_ = best_home_port_province_ }
		add_to_temp_array = { railway_target_levels_ = route_level }
		add_to_temp_array = { railway_priorities_ = route_priority }

		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: Part A - queued capital->home_port route"
		}
	}

	### [railway-pathfinding] Always mark best port for upgrade (even if it's the capital)
	if = {
		limit = { check_variable = { best_home_port_state_ > 0 } }
		add_to_temp_array = { railway_port_upgrades_ = best_home_port_state_ }
	}

	### If best home port IS the capital (and not the same province), queue railway from capital to port's naval base
	if = {
		limit = {
			check_variable = { best_home_port_state_ = capital_state_id }
			check_variable = { best_home_port_province_ > 0 }
			NOT = { check_variable = { capital_province = best_home_port_province_ } }
		}
		add_to_temp_array = { railway_start_provinces_ = capital_province }
		add_to_temp_array = { railway_end_provinces_ = best_home_port_province_ }
		add_to_temp_array = { railway_target_levels_ = route_level }
		add_to_temp_array = { railway_priorities_ = route_priority }

		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: Part A - queued capital->naval_base railway upgrade"
		}
	}

	### PART B: Beachhead expansion
	### For each overseas enemy, find our biggest port on their continent

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: Part B - beachhead expansion starting"
	}

	# Track beachhead states for theatre-aware duplicate detection
	# Replaces continent-based tracking to support multi-theatre scenarios (e.g., Japan vs China + India)
	clear_temp_array = _beachhead_states_

	every_enemy_country = {
		set_temp_variable = { enemy_tag = THIS }
		set_temp_variable = { _check_enemy_tag = THIS }

		# Check if this is an overseas enemy (no land border)
		ROOT = { WA_AI_PC_has_land_border_with_enemy = yes }

		ROOT = { if = { limit = { has_country_flag = WA_AI_construction_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: enemy [?enemy_tag.GetName] has_land_border=[?has_land_border_with_enemy_]" } }

		if = {
			limit = { check_variable = { has_land_border_with_enemy_ = 0 } }

			# Get enemy capital continent using new helper
			capital_scope = {
				WA_AI_PC_railway_get_continent = yes
				set_temp_variable = { enemy_continent = continent_id_ }
				WA_AI_PC_get_state_id = yes
				set_temp_variable = { enemy_capital_state_id = state_id_ }
			}

			# Find our biggest port on that continent (candidate beachhead)
			set_temp_variable = { best_beachhead_state = 0 }
			set_temp_variable = { best_beachhead_port = 0 }
			set_temp_variable = { best_beachhead_province = 0 }

			ROOT = {
				# [na-corridor] the beachhead itself may be an ally/subject port
				every_country = {
					limit = { WA_AI_PC_is_logistics_build_partner = yes }
					every_controlled_state = {
						# Use new continent helper
						WA_AI_PC_railway_get_continent = yes
						WA_AI_PC_get_state_id = yes
						set_temp_variable = { _bh_sid = state_id_ }

						if = {
							limit = {
								check_variable = { continent_id_ = enemy_continent }
								is_coastal = yes
								# [railway-pathfinding] Exclude home territory (same landmass as capital)
								# Beachheads should be on foreign soil, not home island ports
								NOT = { check_variable = { global.WA_AI_MAP_state_landmass^_bh_sid = capital_landmass } }
							}
							# Get naval base province and level using helper (naval_base trigger returns 0 in this scope)
							WA_AI_PC_railway_get_naval_base_province = yes

							if = {
								limit = {
									check_variable = { naval_base_province_ > 0 }
									check_variable = { naval_base_level_ > best_beachhead_port }
								}
								set_temp_variable = { best_beachhead_state = _bh_sid }
								set_temp_variable = { best_beachhead_port = naval_base_level_ }
								set_temp_variable = { best_beachhead_province = naval_base_province_ }
							}
						}
					}
				}
			}

			if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: continent=[?enemy_continent] candidate_beachhead=[?best_beachhead_state] port_level=[?best_beachhead_port]"
			}

			# Theatre separation check: Skip if this beachhead is too close to an existing one
			# This supports multi-theatre scenarios (e.g., Japan vs China + India on same continent)
			if = {
				limit = { check_variable = { best_beachhead_state > 0 } }
				set_temp_variable = { candidate_beachhead_state_ = best_beachhead_state }
				WA_AI_PC_check_theatre_separation = yes

				if = {
					limit = { check_variable = { _is_duplicate_theatre_ = 1 } }
					# Skip - too close to existing beachhead in same theatre
					set_temp_variable = { best_beachhead_state = 0 }

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: beachhead [?candidate_beachhead_state_] SKIPPED - duplicate theatre"
					}
				}
				else = {
					# Register this beachhead for future duplicate checks
					add_to_temp_array = { _beachhead_states_ = best_beachhead_state }

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: beachhead [?best_beachhead_state] ACCEPTED - new theatre"
					}
				}
			}

			### Distance check: Beachhead must be within 15 states of enemy capital
			### This prevents selecting beachheads too far from the theater of war
			if = {
				limit = { check_variable = { best_beachhead_state > 0 } }

				# Get all states within 15 steps of enemy capital
				set_temp_variable = { _origin_state_id = enemy_capital_state_id }
				set_temp_variable = { _max_distance = 15 }
				ROOT = { WA_AI_PC_railway_get_states_within_distance = yes }

				# Check if beachhead is within range (include origin state itself)
				add_to_temp_array = { states_within_distance_ = enemy_capital_state_id }
				if = {
					limit = { NOT = { is_in_array = { states_within_distance_ = best_beachhead_state } } }
					# Beachhead too far from enemy capital - skip it
					set_temp_variable = { best_beachhead_state = 0 }

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: beachhead too far from enemy capital - SKIPPED"
					}
				}
			}

			### Validate beachhead has path to at least one frontline before building railways
			### This prevents useless beachheads like Hong Kong surrounded by enemy territory
			if = {
				limit = { check_variable = { best_beachhead_state > 0 } }

				set_temp_variable = { beachhead_valid_ = 0 }

				# Find any frontline supply hub on this continent and test pathfinding
				ROOT = {
					# [na-corridor] a beachhead may validate against an ally-held frontline
					every_country = {
						limit = { WA_AI_PC_is_logistics_build_partner = yes }
						every_controlled_state = {
							# Use new continent helper
							WA_AI_PC_railway_get_continent = yes

							if = {
								limit = {
									check_variable = { continent_id_ = enemy_continent }
									# Borders enemy (or enemy's puppets)
									any_neighbor_state = {
										OR = {
											is_controlled_by = var:enemy_tag
											controller = { is_subject_of = var:enemy_tag }
										}
									}
									# Has supply hub
									WA_AI_PC_state_has_supply_hub = yes
									# Only test if not already validated (optimization)
									check_variable = { beachhead_valid_ = 0 }
								}

								# Get supply hub province for pathfinding test
								WA_AI_PC_railway_get_supply_hub_province = yes

								if = {
									limit = {
										check_variable = { supply_hub_province_ > 0 }
										# Verify we control the supply hub province (for contested states)
										ROOT = { controls_province = var:supply_hub_province_ }
									}

									# Skip if start and end are the same (beachhead is the supply hub)
									if = {
										limit = { check_variable = { best_beachhead_province = supply_hub_province_ } }
										ROOT = { set_temp_variable = { beachhead_valid_ = 1 } } # Trivially valid
									}
									else = {
										# Test pathfinding from beachhead to this frontline
										set_temp_variable = { _pathfind_prov_start = best_beachhead_province }
										set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
										set_temp_variable = { _pathfind_prov_type = 2 } # [railway-pathfinding] Rail networks - only ROOT-controlled provinces (not allied)
										WA_AI_PATHFIND_PROV_get_path = yes

										if = {
											limit = { check_variable = { pathfind_prov_success_ = 1 } }
											ROOT = { set_temp_variable = { beachhead_valid_ = 1 } }
										}
									}
								}
							}
						}
					}
				}

				# If beachhead invalid (no path to any frontline), skip it
				if = {
					limit = { check_variable = { beachhead_valid_ = 0 } }
					set_temp_variable = { best_beachhead_state = 0 }

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: beachhead has no path to frontlines - INVALID"
					}
				}
				else = {
					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: beachhead VALID - path to frontline exists"
					}
				}
			}

			### Build railways from beachhead to front lines on this continent
			if = {
				limit = { check_variable = { best_beachhead_state > 0 } }

				# Mark beachhead port for upgrade
				if = {
					limit = { NOT = { is_in_array = { railway_port_upgrades_ = best_beachhead_state } } }
					add_to_temp_array = { railway_port_upgrades_ = best_beachhead_state }
				}

				# Find front-line states on this continent with supply hubs
				set_temp_variable = { _check_railway_level = constant:wa_ai_railway.land_war.rail_level_cap }	# [na-corridor] family cap, see the land-war strategy
				ROOT = {
					# [na-corridor] frontline targets on ally/subject soil count too
					every_country = {
						limit = { WA_AI_PC_is_logistics_build_partner = yes }
						every_controlled_state = {
							# Use new continent helper
							WA_AI_PC_railway_get_continent = yes

							if = {
								limit = {
									check_variable = { continent_id_ = enemy_continent }
									# Borders enemy (or enemy's puppets)
									any_neighbor_state = {
										OR = {
											is_controlled_by = var:enemy_tag
											controller = { is_subject_of = var:enemy_tag }
										}
									}
									# Has supply hub (proper check)
									WA_AI_PC_state_has_supply_hub = yes
									# Skip states that already have level 5 railways (native check - includes focus/event built)
									NOT = { WA_AI_PC_state_has_railway_at_level = yes }
								}

								# Find the exact province with the supply hub
								WA_AI_PC_railway_get_supply_hub_province = yes

								if = {
									limit = {
										check_variable = { supply_hub_province_ > 0 }
										NOT = { is_in_array = { railway_end_provinces_ = supply_hub_province_ } }
										# Verify we control the supply hub province (for contested states)
										ROOT = { controls_province = var:supply_hub_province_ }
									}
									# Start from beachhead, not capital.
									# [na-corridor] sized in SUM mode over the two ends - the beachhead
									# port is this chain's entry capacity, the front hub carries the
									# demand. Part A's home-port chain deliberately keeps its flat 5:
									# its target is constant:wa_ai_railway.supply.home_port_target_supply
									# (44 = L5) and it is fed overland from the capital, not by a port.
									set_temp_variable = { _rsize_ob_a_ = best_beachhead_province }
									set_temp_variable = { _rsize_ob_b_ = supply_hub_province_ }
									ROOT = {
										clear_temp_array = _rsize_prov_
										clear_temp_array = _rsize_ours_
										add_to_temp_array = { _rsize_prov_ = _rsize_ob_a_ }
										add_to_temp_array = { _rsize_ours_ = 1 }
										add_to_temp_array = { _rsize_prov_ = _rsize_ob_b_ }
										add_to_temp_array = { _rsize_ours_ = 1 }
										set_temp_variable = { _rsize_entry_mode_ = 0 }
										set_temp_variable = { _rsize_entry_supply_ = 0 }
										set_temp_variable = { _rsize_floor_ = constant:wa_ai_railway.land_war.rail_level_floor }
										set_temp_variable = { _rsize_cap_ = constant:wa_ai_railway.land_war.rail_level_cap }
										WA_AI_PC_rail_size_route = yes
									}
									add_to_temp_array = { railway_start_provinces_ = best_beachhead_province }
									add_to_temp_array = { railway_end_provinces_ = supply_hub_province_ }
									add_to_temp_array = { railway_target_levels_ = rsize_rail_tgt_ }
									add_to_temp_array = { railway_priorities_ = route_priority }

									ROOT = { if = { limit = { has_country_flag = WA_AI_construction_logging }
										log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: ADDED beachhead->[Prev.GetName] route"
									} }
								}
							}
						}
					}
				}
			}
		}
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY OVERSEAS: COMPLETED - [?railway_end_provinces_^num] targets found"
	}
}

############################################################################################################
#	STRATEGY 3: PRE-WAR PREPARATION
#	Build railways toward countries we're justifying against, have wargoals on, or have claims on
############################################################################################################

WA_AI_PC_railway_STRATEGY_prewar_preparation = {

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: STARTED"
	}

	# Default route parameters (may be overridden for overseas targets)
	set_temp_variable = { default_route_start = capital_province }
	# [na-corridor] the prewar family has no sizing of its own: this is the ceiling on the port-derived
	# level for overseas targets and the target for same-landmass ones. Per-route sizing is
	# deliberately NOT used in peace - with no army standing on the target the demand proxy reads a
	# garrison and pins every prewar route at the floor.
	set_temp_variable = { default_route_level = constant:wa_ai_railway.land_war.rail_level_cap }
	set_temp_variable = { route_priority = constant:wa_ai_pc.prio.rail_prewar }

	### Find potential targets: countries we're justifying against, have wargoals on, or have claims on
	### Consolidated from triple loop to single loop with OR conditions
	clear_temp_array = _target_countries_

	### SCRIPTED OVERRIDES: Check for country-specific override targets first (highest priority)
	clear_temp_array = _scripted_override_targets_
	if = {
		limit = { WA_AI_CONFIG_RAILWAY_has_scripted_override = yes }
		WA_AI_PC_railway_get_scripted_override_targets = yes

		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: [?_scripted_override_targets_^num] scripted override targets found"
		}

		# Add override targets to main target list (with safety checks)
		for_each_loop = { array = _scripted_override_targets_ value = _override_target
			var:_override_target = {
				if = {
					limit = {
						exists = yes
						NOT = { has_war_with = ROOT }
						NOT = { is_in_faction_with = ROOT }
						NOT = { is_subject_of = ROOT }
						ROOT = { NOT = { is_subject_of = PREV } }
					}
					add_to_temp_array = { _target_countries_ = THIS }
				}
			}
		}
	}

	### Standard target detection: wargoals and justifications
	every_country = {
		limit = {
			exists = yes
			NOT = { tag = ROOT }
			NOT = { is_in_faction_with = ROOT }
			NOT = { is_subject_of = ROOT }
			# Avoid duplicates from scripted overrides
			NOT = { is_in_array = { _target_countries_ = THIS } }
			OR = {
				ROOT = { is_justifying_wargoal_against = PREV }
				ROOT = { has_wargoal_against = PREV }
			}
		}
		add_to_temp_array = { _target_countries_ = THIS }
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: found [?_target_countries_^num] target countries (including overrides)"
	}

	### For each target, check if land or overseas
	for_each_loop = { array = _target_countries_ value = _target_tag

		# Check for land border
		set_temp_variable = { target_has_land_border = 0 }
		every_controlled_state = {
			limit = {
				any_neighbor_state = {
					is_owned_by = var:_target_tag
				}
			}
			set_temp_variable = { target_has_land_border = 1 }
		}
		# [railway-pathfinding] Also check puppet borders
		every_subject_country = {
			every_controlled_state = {
				limit = {
					any_neighbor_state = {
						is_owned_by = var:_target_tag
					}
				}
				ROOT = { set_temp_variable = { target_has_land_border = 1 } }
			}
		}

		if = { limit = { has_country_flag = WA_AI_construction_logging }
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: target [?_target_tag.GetName] has_land_border=[?target_has_land_border]"
		}

		if = {
			limit = { check_variable = { target_has_land_border = 1 } }

			### LAND TARGET: Build to border states with supply hubs
			set_temp_variable = { _check_railway_level = constant:wa_ai_railway.land_war.rail_level_cap }	# [na-corridor] family cap
			every_controlled_state = {
				limit = {
					any_neighbor_state = {
						is_owned_by = var:_target_tag
					}
					# Has supply hub (proper check)
					WA_AI_PC_state_has_supply_hub = yes
					# Skip states that already have correct level railways (native check - includes focus/event built)
					NOT = { WA_AI_PC_state_has_railway_at_level = yes }
				}

				# Skip single-node states (only port, no supply hub to connect to)
				WA_AI_PC_coastal_state_is_single_node = yes
				if = {
					limit = { check_variable = { is_single_node_ = 0 } }

				# Find the exact province with the supply hub
				WA_AI_PC_railway_get_supply_hub_province = yes

				if = {
					limit = {
						check_variable = { supply_hub_province_ > 0 }
						NOT = { is_in_array = { railway_end_provinces_ = supply_hub_province_ } }
						# Verify we control the supply hub province (for contested states)
						ROOT = { controls_province = var:supply_hub_province_ }
					}

					WA_AI_PC_get_state_id = yes
					set_temp_variable = { _this_state_id = state_id_ }
					# Get target landmass for cross-landmass detection
					set_temp_variable = { target_landmass = global.WA_AI_MAP_state_landmass^_this_state_id }

					# Determine route start and level based on landmass
					if = {
						limit = { check_variable = { target_landmass = capital_landmass } }
						# Same landmass: use capital as route start
						set_temp_variable = { route_start = default_route_start }
						# [railway-landwar] level computed after the pathfind below; floor at default
						set_temp_variable = { route_level = default_route_level }
						set_temp_variable = { _route_level_from_path = 1 }
					}
					else = {
						# Different landmass: find nearest receiving port via distance-limited search
						# [railway-pathfinding] Each target gets its own port analysis (no caching per-landmass)
						set_temp_variable = { _target_state_for_port_search = _this_state_id }
						ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
						set_temp_variable = { route_start = overseas_route_start_ }
						set_temp_variable = { route_level = overseas_max_railway_level_ }
						set_temp_variable = { _route_level_from_path = 0 }
						# Cap at default prewar level
						if = { limit = { check_variable = { route_level > default_route_level } }
							set_temp_variable = { route_level = default_route_level }
						}

						# Queue port upgrade if needed
						if = {
							limit = {
								check_variable = { overseas_receiving_port_state_ > 0 }
								NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
							}
							add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
						}
					}

					# Only proceed if we have a valid route start
					if = {
						limit = { check_variable = { route_start > 0 } }

						# Skip if start and end are the same province (already connected)
						if = {
							limit = { NOT = { check_variable = { route_start = supply_hub_province_ } } }

							# Validate land path exists from route start to this frontline
							set_temp_variable = { _pathfind_prov_start = route_start }
							set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
							set_temp_variable = { _pathfind_prov_type = 2 } # Rail networks - only ROOT-controlled provinces
							set_temp_variable = { _pathfind_prov_allow_partial = 1 } # Allow partial paths to frontier
							WA_AI_PATHFIND_PROV_get_path = yes

							# [railway-pathfinding] Overseas fallback for same-landmass prewar targets
							set_temp_variable = { _direct_success = pathfind_prov_success_ }
							if = {
								limit = {
									check_variable = { target_landmass = capital_landmass }
									NOT = { check_variable = { _direct_success = 1 } }
								}
								set_temp_variable = { _target_state_for_port_search = _this_state_id }
								ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
								if = {
									limit = { check_variable = { overseas_route_start_ > 0 } }
									set_temp_variable = { _pathfind_prov_start = overseas_route_start_ }
									set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
									set_temp_variable = { _pathfind_prov_type = 2 }
									set_temp_variable = { _pathfind_prov_allow_partial = 1 }
									WA_AI_PATHFIND_PROV_get_path = yes
									if = {
										limit = {
											check_variable = { pathfind_prov_success_ > 0 }
											OR = {
												check_variable = { _direct_success = 0 }
												check_variable = { pathfind_prov_success_ = 1 }
											}
										}
										set_temp_variable = { route_start = overseas_route_start_ }
										set_temp_variable = { route_level = overseas_max_railway_level_ }
										set_temp_variable = { _route_level_from_path = 0 }	# [railway-landwar] level now owned by port analysis
										# Cap at prewar level
										if = { limit = { check_variable = { route_level > default_route_level } }
											set_temp_variable = { route_level = default_route_level }
										}
										set_temp_variable = { _direct_success = pathfind_prov_success_ }
										if = {
											limit = {
												check_variable = { overseas_receiving_port_state_ > 0 }
												NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
											}
											add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
										}
									}
									else = {
										# [railway-landwar] fallback pathfind overwrote pathfind_prov_path_ - keep the floor
										set_temp_variable = { _route_level_from_path = 0 }
									}
								}
							}

							if = {
								limit = { check_variable = { _direct_success > 0 } }

								# [railway-landwar] level from the fresh path, capped at the prewar level.
								# [na-corridor] the ratchet is deliberately KEPT in the prewar family, not
								# missed: in peace there is no army on the target, so the demand sizer
								# would pin every prewar route at the floor. "Raise what exists by one,
								# up to the family cap" is the preparation rule; default_route_level is
								# that cap, so the clamp at 5 two lines below can no longer bind.
								if = {
									limit = {
										check_variable = { _route_level_from_path = 1 }
										check_variable = { pathfind_prov_path_^num > 1 }
									}
									WA_AI_PC_get_max_railway_level_on_route = yes
									set_temp_variable = { route_level = _max_route_level_ }
									add_to_temp_variable = { route_level = 1 }
									if = { limit = { check_variable = { route_level > 5 } }
										set_temp_variable = { route_level = 5 }
									}
									if = { limit = { check_variable = { route_level > default_route_level } }
										set_temp_variable = { route_level = default_route_level }
									}
								}

								add_to_temp_array = { railway_start_provinces_ = route_start }
								add_to_temp_array = { railway_end_provinces_ = supply_hub_province_ }
								add_to_temp_array = { railway_target_levels_ = route_level }
								add_to_temp_array = { railway_priorities_ = route_priority }
							}
						}
					}
				}
				} # end if not single-node
			}

			# [railway-pathfinding] Also scan puppet/subject frontline states
			every_subject_country = {
				every_controlled_state = {
					limit = {
						any_neighbor_state = {
							is_owned_by = var:_target_tag
						}
						WA_AI_PC_state_has_supply_hub = yes
						NOT = { WA_AI_PC_state_has_railway_at_level = yes }
					}

					WA_AI_PC_coastal_state_is_single_node = yes
					if = {
						limit = { check_variable = { is_single_node_ = 0 } }

						WA_AI_PC_railway_get_supply_hub_province = yes

						if = {
							limit = {
								check_variable = { supply_hub_province_ > 0 }
								NOT = { is_in_array = { railway_end_provinces_ = supply_hub_province_ } }
								OR = {
									ROOT = { controls_province = var:supply_hub_province_ }
									controller = { is_subject_of = ROOT }
								}
							}

							WA_AI_PC_get_state_id = yes
							set_temp_variable = { _this_state_id = state_id_ }
							set_temp_variable = { target_landmass = global.WA_AI_MAP_state_landmass^_this_state_id }

							if = {
								limit = { check_variable = { target_landmass = capital_landmass } }
								set_temp_variable = { route_start = default_route_start }
								set_temp_variable = { route_level = default_route_level }
							}
							else = {
								set_temp_variable = { _target_state_for_port_search = _this_state_id }
								ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
								set_temp_variable = { route_start = overseas_route_start_ }
								set_temp_variable = { route_level = overseas_max_railway_level_ }
								if = { limit = { check_variable = { route_level > default_route_level } }
									set_temp_variable = { route_level = default_route_level }
								}

								if = {
									limit = {
										check_variable = { overseas_receiving_port_state_ > 0 }
										NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
									}
									add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
								}
							}

							if = {
								limit = { check_variable = { route_start > 0 } }

								if = {
									limit = { NOT = { check_variable = { route_start = supply_hub_province_ } } }

									set_temp_variable = { _pathfind_prov_start = route_start }
									set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
									set_temp_variable = { _pathfind_prov_type = 2 }
									set_temp_variable = { _pathfind_prov_allow_partial = 1 }
									WA_AI_PATHFIND_PROV_get_path = yes

									# [railway-pathfinding] Overseas fallback for same-landmass prewar puppet targets
									set_temp_variable = { _direct_success = pathfind_prov_success_ }
									if = {
										limit = {
											check_variable = { target_landmass = capital_landmass }
											NOT = { check_variable = { _direct_success = 1 } }
										}
										set_temp_variable = { _target_state_for_port_search = _this_state_id }
										ROOT = { WA_AI_PC_analyze_overseas_supply_chain = yes }
										if = {
											limit = { check_variable = { overseas_route_start_ > 0 } }
											set_temp_variable = { _pathfind_prov_start = overseas_route_start_ }
											set_temp_variable = { _pathfind_prov_end = supply_hub_province_ }
											set_temp_variable = { _pathfind_prov_type = 2 }
											set_temp_variable = { _pathfind_prov_allow_partial = 1 }
											WA_AI_PATHFIND_PROV_get_path = yes
											if = {
												limit = {
													check_variable = { pathfind_prov_success_ > 0 }
													OR = {
														check_variable = { _direct_success = 0 }
														check_variable = { pathfind_prov_success_ = 1 }
													}
												}
												set_temp_variable = { route_start = overseas_route_start_ }
												set_temp_variable = { route_level = overseas_max_railway_level_ }
												if = { limit = { check_variable = { route_level > default_route_level } }
													set_temp_variable = { route_level = default_route_level }
												}
												set_temp_variable = { _direct_success = pathfind_prov_success_ }
												if = {
													limit = {
														check_variable = { overseas_receiving_port_state_ > 0 }
														NOT = { is_in_array = { railway_port_upgrades_ = overseas_receiving_port_state_ } }
													}
													add_to_temp_array = { railway_port_upgrades_ = overseas_receiving_port_state_ }
												}
											}
										}
									}

									if = {
										limit = { check_variable = { _direct_success > 0 } }
										add_to_temp_array = { railway_start_provinces_ = route_start }
										add_to_temp_array = { railway_end_provinces_ = supply_hub_province_ }
										add_to_temp_array = { railway_target_levels_ = route_level }
										add_to_temp_array = { railway_priorities_ = route_priority }
									}
								}
							}
						}
					}
				}
			}
		}
		else_if = {
			# [railway-pathfinding] overseas logic only if ROOT has coastal access - without it a
			# landlocked country with a claim on a non-bordering target queues port infrastructure.
			limit = { any_controlled_state = { is_coastal = yes } }

			### OVERSEAS TARGET: Upgrade home port infrastructure
			### Use the new helper
			WA_AI_PC_railway_find_best_home_port = yes

			if = {
				limit = {
					check_variable = { best_home_port_state_ > 0 }
					NOT = { check_variable = { best_home_port_state_ = capital_state_id } }
					NOT = { is_in_array = { railway_end_provinces_ = best_home_port_province_ } }
				}
				add_to_temp_array = { railway_start_provinces_ = capital_province }
				add_to_temp_array = { railway_end_provinces_ = best_home_port_province_ }
				add_to_temp_array = { railway_target_levels_ = route_level }
				add_to_temp_array = { railway_priorities_ = route_priority }

				if = {
					limit = { NOT = { is_in_array = { railway_port_upgrades_ = best_home_port_state_ } } }
					add_to_temp_array = { railway_port_upgrades_ = best_home_port_state_ }
				}

				if = { limit = { has_country_flag = WA_AI_construction_logging }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: ADDED overseas home port route"
				}
			}

			### If best home port IS the capital (and not the same province), queue railway from capital to port's naval base
			if = {
				limit = {
					check_variable = { best_home_port_state_ = capital_state_id }
					check_variable = { best_home_port_province_ > 0 }
					NOT = { check_variable = { capital_province = best_home_port_province_ } }
				}
				add_to_temp_array = { railway_start_provinces_ = capital_province }
				add_to_temp_array = { railway_end_provinces_ = best_home_port_province_ }
				add_to_temp_array = { railway_target_levels_ = route_level }
				add_to_temp_array = { railway_priorities_ = route_priority }

				if = { limit = { has_country_flag = WA_AI_construction_logging }
					log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: Added capital->naval_base railway upgrade"
				}
			}
		}
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY PREWAR: COMPLETED - [?railway_end_provinces_^num] targets found"
	}
}

############################################################################################################
#	STRATEGY 4: THEATRE CORRIDOR ([na-corridor])
#	Selection only. Reads a corridor node list (WA_AI_CONSTRUCTION_PRIORITY_corridor_data.txt) and
#	publishes the routes and hubs ROOT should build THIS pass; WA_AI_PC_railway_corridor_pass
#	(railway_core) does the pathfinding, admission, validation and telemetry.
#
#	ROOT = builder country (country scope).
#	<<< _corridor_id_                          [levant-iraq] which corridor to select, constant:wa_ai_pc.corridor_id.*
#	                                           (north_africa / levant_iraq). Set by the caller before every call.
#	>>> corridor_ran_                          1 = the theatre gate passed and the arrays are meaningful
#	>>> arr:corridor_route_start_ / _end_       province pairs (consecutive nodes both "ours")
#	>>> arr:corridor_route_level_ / _prio_      per route
#	>>> arr:corridor_hub_prov_ / _kind_ / _prio_ hubs to build at a node we hold: kind 17 = supply hub,
#	                                            14 = naval base
#
#	The corridor is SYMMETRIC and tag-free; the priority band is rail_war when any node is enemy-held
#	or when the corridor's define declares _corridor_war_band_ = 1, rail_prewar otherwise (the owner of
#	Libya / Egypt / Tunisia preparing in peace).
#	Sequencing emerges from control: Mechili<->Tobruk is never requested before Mechili is held.
#	Cadence and level are constant:wa_ai_railway.corridor.*.
#
#	[na-corridor] PERMISSION and PAYMENT are two questions, asked separately, both per PROVINCE:
#	  corridor_node_ours_   WA_AI_PC_corridor_node_is_ours       may ROOT build here? ROOT / subject /
#	                                                             ANY faction ally
#	  corridor_node_charge_ WA_AI_PC_corridor_node_is_my_charge  should ROOT pay for it? ROOT / subject /
#	                                                             an ally that cannot fund its own
#	A hop is requested when BOTH ends are ours and AT LEAST ONE is our charge. Requiring charge on both
#	ends was tried and left a hop between a healthy ally's node and our own with no builder anywhere.
#	Hubs keep the charge test ALONE. The pass gate is _corridor_claim_n_ > 0, so permission alone never
#	admits a country with no stake in the theatre.
############################################################################################################

WA_AI_PC_railway_STRATEGY_theatre_corridor = {
	set_temp_variable = { corridor_ran_ = 0 }
	clear_temp_array = corridor_route_start_
	clear_temp_array = corridor_route_end_
	clear_temp_array = corridor_route_level_
	clear_temp_array = corridor_route_prio_
	clear_temp_array = corridor_hub_prov_
	clear_temp_array = corridor_hub_kind_
	clear_temp_array = corridor_hub_prio_
	clear_temp_array = corridor_hub_target_
	clear_temp_array = corridor_node_ours_
	clear_temp_array = corridor_node_charge_
	clear_temp_array = corridor_node_hostile_
	set_temp_variable = { corridor_reach_ = 0 }
	set_temp_variable = { corridor_orphan_ = 0 }

	### Corridor data, selected by _corridor_id_ (set by the caller, WA_AI_PC_railway_corridor_pass).
	### [levant-iraq] A new theatre = a new define here, a new id in the gate dispatch below, a new
	### call in the pass, and a row in the doc's "Theatre corridors" table. Nothing else in this
	### effect is corridor-specific: an unknown id defines no nodes, so the classification loop runs
	### over an empty list, _corridor_claim_n_ stays 0 and the pass is silent - the correct null.
	if = { limit = { check_variable = { _corridor_id_ = constant:wa_ai_pc.corridor_id.north_africa } } WA_AI_PC_CORRIDOR_define_north_africa = yes }
	else_if = { limit = { check_variable = { _corridor_id_ = constant:wa_ai_pc.corridor_id.levant_iraq } } WA_AI_PC_CORRIDOR_define_levant_iraq = yes }
	else = {
		clear_temp_array = corridor_node_prov_
		clear_temp_array = corridor_node_depot_
		clear_temp_array = corridor_node_port_
		clear_temp_array = corridor_node_fwdport_
		set_temp_variable = { _corridor_war_band_ = 0 }
	}

	### Classify every node once: ours (ROOT may build) / our charge (ROOT should pay) / enemy-held
	set_temp_variable = { _corridor_hostile_n_ = 0 }
	set_temp_variable = { _corridor_ours_n_ = 0 }
	set_temp_variable = { _corridor_claim_n_ = 0 }
	for_each_loop = { array = corridor_node_prov_ value = _corridor_prov break = _cn_b_
		if = {
			limit = { WA_AI_PC_corridor_node_is_ours = yes }
			add_to_temp_array = { corridor_node_ours_ = 1 }
			add_to_temp_variable = { _corridor_ours_n_ = 1 }
		}
		else = {
			add_to_temp_array = { corridor_node_ours_ = 0 }
		}
		if = {
			limit = { WA_AI_PC_corridor_node_is_my_charge = yes }
			add_to_temp_array = { corridor_node_charge_ = 1 }
			add_to_temp_variable = { _corridor_claim_n_ = 1 }
		}
		else = {
			add_to_temp_array = { corridor_node_charge_ = 0 }
		}
		if = {
			limit = { WA_AI_PC_corridor_node_is_hostile = yes }
			add_to_temp_array = { corridor_node_hostile_ = 1 }
			add_to_temp_variable = { _corridor_hostile_n_ = 1 }
		}
		else = {
			add_to_temp_array = { corridor_node_hostile_ = 0 }
		}
	}

	# [na-corridor] gate on the CHARGE count, not the permission count. Gating on permission lets any
	# faction member with no stake in the theatre run the whole pass - JAP passes the North Africa
	# theatre gate the moment ENG holds Egypt - and it would then classify, pathfind and run
	# stale-path validation over an empty valid set.
	if = {
		limit = {
			check_variable = { _corridor_claim_n_ > 0 }
			# [levant-iraq] one theatre gate per corridor id. AND-pairs, never an `if` with a limit:
			# an `if` whose limit is false is VACUOUSLY TRUE inside a trigger, which would arm every
			# corridor for every id (wa-pdxscript, and the defect measured in the rail-corridor cheat
			# on 2026-09-01).
			OR = {
				AND = {
					check_variable = { _corridor_id_ = constant:wa_ai_pc.corridor_id.north_africa }
					WA_AI_PC_corridor_theatre_live_north_africa = yes
				}
				AND = {
					check_variable = { _corridor_id_ = constant:wa_ai_pc.corridor_id.levant_iraq }
					WA_AI_PC_corridor_theatre_live_levant_iraq = yes
				}
			}
		}
		set_temp_variable = { corridor_ran_ = 1 }

		### [na-corridor] size the corridor before emitting anything. Publishes corridor_rail_tgt_ (the
		### route level below) and corridor_port_tgt_ (the level a port node should reach, 0 = "place
		### if missing"). Must run here: it reads corridor_node_ours_, just filled by the loop above.
		WA_AI_PC_corridor_compute_sizing = yes

		### Band: war when an enemy holds a node of this corridor, preparation otherwise.
		### [levant-iraq] second term: a corridor whose own gate IS a war gate declares
		### _corridor_war_band_ = 1 in its define, because its node list can be entirely behind our
		### own lines and would then price a wartime measure as peacetime preparation. North Africa
		### declares 0, so its band is decided exactly as before.
		if = {
			limit = {
				OR = {
					check_variable = { _corridor_hostile_n_ > 0 }
					check_variable = { _corridor_war_band_ = 1 }
				}
			}
			set_temp_variable = { _corridor_prio_ = constant:wa_ai_pc.prio.rail_war }
		}
		else = {
			set_temp_variable = { _corridor_prio_ = constant:wa_ai_pc.prio.rail_prewar }
		}

		### Routes: every consecutive pair of nodes both ours
		set_temp_variable = { _corridor_last_i_ = corridor_node_prov_^num }
		subtract_from_temp_variable = { _corridor_last_i_ = 1 }
		for_each_loop = { array = corridor_node_prov_ index = _ci_ value = _ci_prov_ break = _ci_b_
			if = {
				limit = { check_variable = { _ci_ < _corridor_last_i_ } }
				set_temp_variable = { _ci_next_ = _ci_ }
				add_to_temp_variable = { _ci_next_ = 1 }
				# [na-corridor] permission on BOTH ends, payment on AT LEAST ONE.
				if = {
					limit = {
						check_variable = { corridor_node_ours_^_ci_ = 1 }
						check_variable = { corridor_node_ours_^_ci_next_ = 1 }
					}
					set_temp_variable = { _hop_charge_n_ = corridor_node_charge_^_ci_ }
					add_to_temp_variable = { _hop_charge_n_ = corridor_node_charge_^_ci_next_ }
					if = {
						limit = { check_variable = { _hop_charge_n_ > 0 } }
						add_to_temp_array = { corridor_route_start_ = _ci_prov_ }
						add_to_temp_array = { corridor_route_end_ = corridor_node_prov_^_ci_next_ }
						add_to_temp_array = { corridor_route_level_ = corridor_rail_tgt_ }
						add_to_temp_array = { corridor_route_prio_ = _corridor_prio_ }
						# tlm gauge [na-corridor]: hops requestable only because one end is a healthy
						# ally's - a both-ends-charge rule refuses them.
						if = {
							limit = { check_variable = { _hop_charge_n_ = 1 } }
							add_to_temp_variable = { corridor_reach_ = 1 }
						}
					}
					else = {
						# tlm gauge [na-corridor]: both ends ours, NEITHER our charge - both ends are
						# held by partners that can fund, so the hop belongs to someone else and a
						# transient reading is normal. Sustained high WHILE the hop stays unbuilt on
						# the map is the signal: the owner is not building it either.
						add_to_temp_variable = { corridor_orphan_ = 1 }
					}
				}
			}
			### Hubs: a depot / port wanted at a node we hold.
			# [na-corridor] CHARGE, not permission - a hub is a whole building on one node, so unlike a
			# hop there is no "we have a foot on it" half to reason about.
			if = {
				limit = { check_variable = { corridor_node_charge_^_ci_ = 1 } }
				if = {
					limit = { check_variable = { corridor_node_depot_^_ci_ = 1 } }
					add_to_temp_array = { corridor_hub_prov_ = _ci_prov_ }
					add_to_temp_array = { corridor_hub_kind_ = 17 }
					add_to_temp_array = { corridor_hub_prio_ = _corridor_prio_ }
					# [na-corridor] a supply_node has no level (LOGISTICS_MODEL section 3) - always 1.
					add_to_temp_array = { corridor_hub_target_ = 1 }
				}
				if = {
					limit = { check_variable = { corridor_node_port_^_ci_ = 1 } }
					add_to_temp_array = { corridor_hub_prov_ = _ci_prov_ }
					add_to_temp_array = { corridor_hub_kind_ = 14 }
					add_to_temp_array = { corridor_hub_prio_ = _corridor_prio_ }
					# [na-corridor] the level this port should REACH. corridor_port_tgt_ is 0 unless
					# sizing found ENTRY CAPACITY to be what binds; the hub loop reads 0 as "place a
					# missing port at level 1". Carried as a PARALLEL ARRAY, not a chain temp, on
					# purpose: the hub loop in railway_core re-sets _hub_prov_ / _kind_ / _prio_ per
					# iteration, so a fourth chain temp would leak from one port to the next.
					add_to_temp_array = { corridor_hub_target_ = corridor_port_tgt_ }
				}
				### FORWARD port - the attacker's supply port at a spearhead node: only the side
				### attacking along the corridor builds it, never the defender on its own ground.
				### Attacker = the node's province is OWNED by an enemy (WA_AI_PC_corridor_node_is_
				### conquered - we stand on conquered soil); spearhead = one neighbouring node is
				### enemy-held (the front is right there) and the other is ours (the rear already
				### fell). A corridor end node has one neighbour only, which must then be the hostile
				### one.
				### [na-corridor] the two _fwd_rear_ tests deliberately read corridor_node_ours_
				### (PERMISSION), not charge: "has the rear fallen to our side" is a question about
				### SIDES, and an ally standing in the rear secures it whether or not it could fund a
				### railway. Deliberate widening; the port stays bounded by three unchanged terms -
				### outer gate is CHARGE (we need a stake in the node itself), one neighbour must be
				### enemy-held, and the conquered-soil test excludes the defender.
				if = {
					limit = { check_variable = { corridor_node_fwdport_^_ci_ = 1 } }
					set_temp_variable = { _fwd_front_ = 0 }
					set_temp_variable = { _fwd_rear_ = 0 }
					if = {
						limit = { check_variable = { _ci_ > 0 } }
						set_temp_variable = { _ci_prev_ = _ci_ }
						subtract_from_temp_variable = { _ci_prev_ = 1 }
						if = { limit = { check_variable = { corridor_node_hostile_^_ci_prev_ = 1 } } set_temp_variable = { _fwd_front_ = 1 } }
						if = { limit = { check_variable = { corridor_node_ours_^_ci_prev_ = 1 } } set_temp_variable = { _fwd_rear_ = 1 } }
					}
					if = {
						limit = { check_variable = { _ci_ < _corridor_last_i_ } }
						set_temp_variable = { _ci_next_ = _ci_ }
						add_to_temp_variable = { _ci_next_ = 1 }
						if = { limit = { check_variable = { corridor_node_hostile_^_ci_next_ = 1 } } set_temp_variable = { _fwd_front_ = 1 } }
						if = { limit = { check_variable = { corridor_node_ours_^_ci_next_ = 1 } } set_temp_variable = { _fwd_rear_ = 1 } }
					}
					set_temp_variable = { _corridor_prov = _ci_prov_ }
					if = {
						limit = {
							check_variable = { _fwd_front_ = 1 }
							check_variable = { _fwd_rear_ = 1 }
							WA_AI_PC_corridor_node_is_conquered = yes
						}
						add_to_temp_array = { corridor_hub_prov_ = _ci_prov_ }
						add_to_temp_array = { corridor_hub_kind_ = 14 }
						add_to_temp_array = { corridor_hub_prio_ = _corridor_prio_ }
						# [na-corridor] a FORWARD port is placed, never raised: it injects supply at a
						# spearhead on conquered soil, where the RANGE model fails and not the flow
						# model (LOGISTICS_MODEL section 6), and the spearhead moves, so a multi-level
						# climb would pay for a port the front has already left. Level 1, deliberately.
						add_to_temp_array = { corridor_hub_target_ = 1 }
					}
				}
			}
		}

		if = { limit = { has_country_flag = WA_AI_construction_logging }
