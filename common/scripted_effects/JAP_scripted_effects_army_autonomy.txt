# Below Kantokuen (JAP_defensive_posture_against_the_soviets): staging toward the Soviet border.
# Level 3 air bases in the four border states (the same level Northern Indochina gives Tonkin).
# Each railway that crosses into the Soviet Far East gets one more level up to the border, and a supply hub
# where it meets the border. Hokiang already has a hub at 9798 from Kantokuen, Heilungkiang at 7832 next to 14348.
#   Amur (Heilungkiang 714):       1588 ... 14348 -> Birobidzhan, level 1 -> 2
#   Ussuri (Sungkiang 717):        9766 783 -> Vladivostok, level 1 -> 2, hub at 783

# Southern Resource Area (JAP_seize_the_southern_resource_area): staging for the southern operation, the pair of
# JAP_soviet_border_staging. The Malaya force gathered on Hainan and in French Indochina and its air units flew from
# around Saigon; the Philippine force came from Taiwan. Air bases +2 in Cochinchina, Hainan and Taiwan, and ports +1
# at Vung Tau (instead of Saigon) and Haikou. Taiwan's port at Kaohsiung (12068) is already level 10.
# Other countries' states are only built up when they are not held by an enemy (Hainan: only when held by our side).
JAP_southern_staging = {
	524 = {	# Taiwan
		add_building_construction = { type = air_base level = 2 instant_build = yes }
	}
	286 = {	# Cochinchina (Saigon)
		if = {
			limit = { CONTROLLER = { NOT = { has_war_with = ROOT } } }
			add_building_construction = { type = air_base level = 2 instant_build = yes }
		}
	}
	1064 = {	# Vung Tau
		if = {
			limit = { CONTROLLER = { NOT = { has_war_with = ROOT } } }
			add_building_construction = { type = naval_base level = 1 instant_build = yes province = 14168 }
		}
	}
	591 = {	# Hainan
		if = {
			limit = {
				CONTROLLER = {
					OR = {
						tag = ROOT
						is_subject_of = ROOT
						is_in_faction_with = ROOT
					}
				}
			}
			add_building_construction = { type = air_base level = 2 instant_build = yes }
			add_building_construction = { type = naval_base level = 1 instant_build = yes province = 1070 }
		}
	}
}

# The army faction in power loses one step of influence: the Toseiha once the Kodoha has been purged, otherwise the
# Kodoha. Same steps as JAP_fifty_division_plan_timeout_effect (JAP_scripted_effects.txt), without the navy's gain.
JAP_army_influence_down_effect = {
	if = {
		limit = { JAP_kodoha_faction_alive_trigger = no }
		custom_effect_tooltip = JAP_toseiha_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_toseiha_influence > 0 } }
				add_to_variable = { JAP_toseiha_influence = -1 }
				add_to_variable = { JAP_toseiha_production_speed_arms_factory_factor = -0.04 }
			}
		}
	}
	else = {
		custom_effect_tooltip = JAP_kodoha_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kodoha_influence > 0 } }
				add_to_variable = { JAP_kodoha_influence = -1 }
				add_to_variable = { JAP_kodoha_production_speed_arms_factory_factor = -0.04 }
			}
		}
	}
}

# The army faction in power gains one step of influence (cap 7): the mirror of JAP_army_influence_down_effect.

	if = {
		limit = { JAP_kodoha_faction_alive_trigger = no }
		custom_effect_tooltip = JAP_toseiha_influence_up_tt

				add_to_variable = { JAP_toseiha_influence = 1 }
				add_to_variable = { JAP_toseiha_production_speed_arms_factory_factor = 0.04 }
			}
		}
	}
	else = {
		custom_effect_tooltip = JAP_kodoha_influence_up_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kodoha_influence < 7 } }
				add_to_variable = { JAP_kodoha_influence = 1 }
				add_to_variable = { JAP_kodoha_production_speed_arms_factory_factor = 0.04 }
			}
		}
	}
}
