# Japan's newest infantry kit into the stockpile of the country in scope (Japan or Manchukuo).
# Amounts come from the temp variables JAP_kit_inf, JAP_kit_hv, JAP_kit_sup and JAP_kit_art.
# The model is picked from Japan's own research.
JAP_add_latest_infantry_kit = {
	if = {
		limit = { JAP = { has_tech = jap_infantry_weapons_6 } }
		add_equipment_to_stockpile = { type = jap_inf_6 amount = JAP_kit_inf producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_infantry_weapons_5 } }
		add_equipment_to_stockpile = { type = jap_inf_5 amount = JAP_kit_inf producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_infantry_weapons_3 } }
		add_equipment_to_stockpile = { type = jap_inf_3 amount = JAP_kit_inf producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_infantry_weapons_2 } }
		add_equipment_to_stockpile = { type = jap_inf_2 amount = JAP_kit_inf producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_infantry_weapons_4 } }
		add_equipment_to_stockpile = { type = jap_inf_4 amount = JAP_kit_inf producer = JAP }
	}
	else = {
		add_equipment_to_stockpile = { type = jap_inf_1 amount = JAP_kit_inf producer = JAP }
	}
	if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_7 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_7 amount = JAP_kit_hv producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_6 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_6 amount = JAP_kit_hv producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_4 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_4 amount = JAP_kit_hv producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_5 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_5 amount = JAP_kit_hv producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_3 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_3 amount = JAP_kit_hv producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_heavy_infantry_weapons_2 } }
		add_equipment_to_stockpile = { type = jap_hv_inf_2 amount = JAP_kit_hv producer = JAP }
	}
	else = {
		add_equipment_to_stockpile = { type = jap_hv_inf_1 amount = JAP_kit_hv producer = JAP }
	}
	add_equipment_to_stockpile = { type = support_equipment_1 amount = JAP_kit_sup producer = JAP }
	if = {
		limit = { JAP = { has_tech = jap_artillery_4 } }
		add_equipment_to_stockpile = { type = jap_art_4 amount = JAP_kit_art producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_artillery_3 } }
		add_equipment_to_stockpile = { type = jap_art_3 amount = JAP_kit_art producer = JAP }
	}
	else_if = {
		limit = { JAP = { has_tech = jap_artillery_2 } }
		add_equipment_to_stockpile = { type = jap_art_2 amount = JAP_kit_art producer = JAP }
	}
	else = {
		add_equipment_to_stockpile = { type = jap_art_1 amount = JAP_kit_art producer = JAP }
	}
}
