# Southern Resource Area (JAP_seize_the_southern_resource_area): staging for the southern operation, the pair of

# around Saigon; the Philippine force came from Taiwan. Air bases +2 in Cochinchina, Hainan and Taiwan, and ports +1
# at Vung Tau (instead of Saigon) and Haikou. Taiwan's port at Kaohsiung (12068) is already level 10.
# Other countries' states are only built up when they are not held by an enemy (Hainan: only when held by our side).

	524 = {	# Taiwan
		add_building_construction = { type = air_base level = 2 instant_build = yes }
	}

			limit = { CONTROLLER = { NOT = { has_war_with = ROOT } } }

		}
	}
	1064 = {	# Vung Tau
		if = {
			limit = { CONTROLLER = { NOT = { has_war_with = ROOT } } }

		}
	}

# The army faction in power loses one step of influence: the Toseiha once the Kodoha has been purged, otherwise the
# Kodoha. Same steps as JAP_fifty_division_plan_timeout_effect (JAP_scripted_effects.txt), without the navy's gain.
JAP_army_influence_down_effect = {
	if = {
		limit = { JAP_kodoha_faction_alive_trigger = no }
		custom_effect_tooltip = JAP_toseiha_influence_down_tt_ij
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_toseiha_influence > 0 } }
				add_to_variable = { JAP_toseiha_influence = -1 }
				add_to_variable = { JAP_toseiha_production_speed_arms_factory_factor = -0.04 }
			}
		}
	}
	else = {
		custom_effect_tooltip = JAP_kodoha_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kodoha_influence > 0 } }
				add_to_variable = { JAP_kodoha_influence = -1 }
				add_to_variable = { JAP_kodoha_production_speed_arms_factory_factor = -0.04 }
			}
		}
	}
}
