############################################################################################################
#	World Ablaze AI mod - AI Force Concentration (AIFC) - Helpers Module
#	Candidate collection, scoring, sector construction, and armour steering (front_armor_score emission)
############################################################################################################
#	System overview and value bands: common/ai_strategy/WA_AI_MILITARY_DEFAULT_FRONT_aifc.txt
#	Lifecycle and cadence:           common/scripted_effects/WA_AI_AIFC_core.txt
#	Behavioural switches:            common/scripted_triggers/WA_AI_AIFC_triggers.txt
#
#	SELECTION MODEL
#	---------------
#	One sector per country. Chosen once a week, at most, and only when the previous one went stale.
#
#		1. Collect every enemy-held state that touches territory we control. These are the only places a
#		   ground offensive can actually begin, which keeps the candidate set naturally small.
#		1c. Measure each candidate's launching pad - our own divisions in the friendly states next to it -
#		   and let the strongest pad pick the MAIN ENEMY. Campaign 5709c8b9 showed why: scoring that only
#		   looked at the defender sent GER's schwerpunkt to Kurdufan and SOV's to Finland while both
#		   armies massed at Rzhev. A schwerpunkt follows our mass; a thin seam on a front we are not
#		   actually on is not a schwerpunkt (same front-local lesson as the posture calculus).
#		2. Score each candidate ON THAT FRONT on how good a breakthrough point it is - terrain, how
#		   thinly it is held, what is behind it worth taking - plus its own launching pad.
#		3. The winner becomes the anchor. The corridor is the anchor plus its enemy-held neighbours,
#		   capped at @WA_AI_AIFC_SECTOR_MAX_STATES.
#		4. Objectives are the anchor plus the best-scoring state in the corridor - the depth target the
#		   push aims for rather than merely the point it enters at.
#
#	Deliberately NOT modelled in this version, to keep the weekly cost bounded:
#		- distance weighting (every candidate is adjacent to us by construction, so it adds little)
#		- multi-hop pathing into the enemy rear (WA_AI_PATHFIND_get_path exists if this is ever wanted)
#		- coordinating sectors between faction members
#
#	DEPENDENCY
#	----------
#	Reuses WA_AI_PC_get_state_id from WA_AI_CONSTRUCTION_PRIORITY_railway_primitives.txt. It is a generic
#	state-scope -> numeric-ID primitive despite the railway prefix; do not duplicate it here.
############################################################################################################

### AIFC sector constants (must match common/scripted_effects/WA_AI_AIFC_core.txt)
@WA_AI_AIFC_SECTOR_MAX_STATES = 8	# Corridor cap. Every entry costs on each Layer 4 is_in_array lookup.


############################################################################################################
#	Select one sector for ROOT
#	ROOT = country scope
#	Writes: WA_AI_AIFC_sector_states, WA_AI_AIFC_sector_objectives,
#	        WA_AI_AIFC_sector_anchor, WA_AI_AIFC_sector_age
############################################################################################################

WA_AI_AIFC_select_sector = {

	WA_AI_AIFC_clear_sector = yes

	clear_temp_array = _aifc_cand

	############################################################
	# 1. Candidate collection - enemy-held states in contact
	############################################################

	every_controlled_state = {
		every_neighbor_state = {
			limit = {
				CONTROLLER = { has_war_with = ROOT }
			}

			WA_AI_PC_get_state_id = yes
			set_temp_variable = { _aifc_sid = state_id_ }

			if = {
				limit = { NOT = { is_in_array = { _aifc_cand = _aifc_sid } } }
				add_to_temp_array = { _aifc_cand = _aifc_sid }
			}
		}
	}

	############################################################
	# 1b. Expeditionary fallback - no native land contact
	############################################################
	# Fix 28: every_controlled_state cannot see allied-controlled soil, so an expeditionary power
	# fighting entirely from a faction member's territory (USA in liberated France) never found an
	# anchor and never got a sector. Same scope trap as railway Fixes 24/25/27 (puppets/controllers).
	# Fallback: enemy states in contact with faction-member or subject-relation territory where ROOT
	# itself has a meaningful ground presence. Gated on the native scan coming up empty so the extra
	# state walk is only ever paid by countries that need it.

	# Gate relaxed from "no native candidate at all" to "fewer than 3": campaign 5709c8b9 showed a
	# single token native contact (one captured island) suppressing the fallback entirely, leaving an
	# expeditionary power's real army - fighting from allied soil on another continent - sector-less
	# or anchored on the irrelevant native candidate.

	# FIX 65 (2026-08-14, campaign f9321934): the ally filter used to carry `has_capitulated = no`,
	# which is the one country in the faction whose soil an expeditionary army is MOST likely to be
	# standing on. `has_capitulated` answers "did this government fall", NOT "does this country still
	# exist" and NOT "is this territory usable": a capitulated tag keeps its faction membership, keeps
	# owning and controlling territory, and REGAINS CONTROL OF LIBERATED HOME SOIL while still
	# reporting capitulated = yes. So the liberator fights its whole campaign on ground the filter
	# throws away.
	#
	# Measured: ENG held NO sector from 1944.5 to 1946.7 - ~118 consecutive weeks, 116-143 divisions,
	# standing execute orders vs GER - because it controlled zero continental states of its own (its
	# only land contact, Italian North Africa, had been ANNEXED, and Italy had changed sides) while
	# every large concentration it had sat on FRA-controlled soil: 735 French Alpes, then 15 Lower
	# Normandy (15-17 divisions) and 1 Corsica. FRA read capitulated = yes in EVERY save of that
	# campaign while owning 60 states and controlling 50. Empty candidate array -> _aifc_best_id = 0 ->
	# clear_sector, every week, for the entire Western campaign.
	#
	# The term was never load-bearing. The real guard is the divisions_in_state test below: ROOT must
	# have >2 of its OWN divisions standing on that ground, which excludes a capitulated ally's
	# territory we are not actually on. `exists = yes` replaces it to keep dead tags out of the walk -
	# that is the check the old term was reaching for. The launching-pad measurement in section 1c
	# below was written in the SAME commit as this block (4ffb8e442, Fix 28) and already omits the
	# capitulation test, which is the tell that it was an incidental paste of the "usable ally" idiom.
	#
	# Keep in lockstep with the Fix 28 validity mirror in WA_AI_AIFC_core.txt - if selection may
	# anchor on a capitulated ally's front, validity must accept the same one, or the sector is
	# invalidated the week after it is chosen and the country churns forever.
	#
	# NOT a fix for any front. AIFC issues no attack orders and Layers 1-2 are not sector-gated, so
	# this restores a scripted AXIS, not concentration itself. Do not expect front movement from it.
	if = {
		limit = { check_variable = { _aifc_cand^num < 3 } }

		every_country = {
			limit = {
				NOT = { tag = ROOT }
				exists = yes
				OR = {
					is_in_faction_with = ROOT
					is_subject_of = ROOT
					ROOT = { is_subject_of = PREV }
				}
			}

			every_controlled_state = {
				limit = {
					# ROOT has boots on this allied ground - at least 3 of its own divisions.
					ROOT = { divisions_in_state = { state = PREV size > 2 } }
				}

				every_neighbor_state = {
					limit = {
						CONTROLLER = { has_war_with = ROOT }
					}

					WA_AI_PC_get_state_id = yes
					set_temp_variable = { _aifc_sid = state_id_ }

					if = {
						limit = { NOT = { is_in_array = { _aifc_cand = _aifc_sid } } }
						add_to_temp_array = { _aifc_cand = _aifc_sid }
					}
				}
			}
		}
	}

	############################################################
	# 1c. Launching pads and the main enemy
	############################################################
	# Per candidate: band-count ROOT's own divisions in the adjacent friendly states (own or, for
	# expeditionary powers, faction/subject-relation controlled - mirrors the Fix 28 scopes). The
	# candidate with the strongest pad names the MAIN ENEMY; only that enemy's candidates compete for
	# the anchor. This is what keeps the schwerpunkt on the front where the army actually stands
	# instead of the thinnest seam anywhere in the world (campaign 5709c8b9: GER->Kurdufan,
	# SOV->Finland, armour at Rzhev).
	# _aifc_pad_arr and _aifc_ctrl_arr are index-aligned with _aifc_cand.

	clear_temp_array = _aifc_pad_arr
	clear_temp_array = _aifc_ctrl_arr

	set_temp_variable = { _aifc_main_enemy = 0 }
	set_temp_variable = { _aifc_max_pad = 0 }

	for_each_loop = { array = _aifc_cand value = _aifc_c

		set_temp_variable = { _aifc_pad = 0 }

		var:_aifc_c = {
			every_neighbor_state = {
				limit = {
					OR = {
						is_controlled_by = ROOT
						CONTROLLER = {
							OR = {
								is_in_faction_with = ROOT
								is_subject_of = ROOT
								ROOT = { is_subject_of = PREV }
							}
						}
					}
				}

				if = {
					limit = { ROOT = { divisions_in_state = { state = PREV size > 9 } } }
					add_to_temp_variable = { _aifc_pad = 3 }
				}
				else_if = {
					limit = { ROOT = { divisions_in_state = { state = PREV size > 4 } } }
					add_to_temp_variable = { _aifc_pad = 2 }
				}
				else_if = {
					limit = { ROOT = { divisions_in_state = { state = PREV size > 0 } } }
					add_to_temp_variable = { _aifc_pad = 1 }
				}
			}

			CONTROLLER = {
				ROOT = { set_temp_variable = { _aifc_ctrl = PREV.id } }
			}
		}

		add_to_temp_array = { _aifc_pad_arr = _aifc_pad }
		add_to_temp_array = { _aifc_ctrl_arr = _aifc_ctrl }

		if = {
			limit = { check_variable = { _aifc_pad > _aifc_max_pad } }
			set_temp_variable = { _aifc_max_pad = _aifc_pad }
			set_temp_variable = { _aifc_main_enemy = _aifc_ctrl }
		}
	}

	############################################################
	# 2. Score the main enemy's candidates, keep the single best
	############################################################
	# _aifc_main_enemy = 0 means no candidate has any launching pad (all fronts remote); fall back to
	# scoring everything rather than selecting nothing.

	set_temp_variable = { _aifc_best_id = 0 }
	set_temp_variable = { _aifc_best_score = -9999 }

	for_each_loop = { array = _aifc_cand index = _aifc_i value = _aifc_c

		if = {
			limit = {
				OR = {
					check_variable = { _aifc_main_enemy = 0 }
					check_variable = { _aifc_ctrl_arr^_aifc_i = _aifc_main_enemy }
				}
			}

			set_temp_variable = { _aifc_target = _aifc_c }
			WA_AI_AIFC_score_state = yes		# >>> _aifc_score

			# Launching-pad bonus: among this front's entry points, prefer the ones our divisions can
			# actually pour through. 5 per band step, capped at 60.
			set_temp_variable = { _aifc_pad_bonus = _aifc_pad_arr^_aifc_i }
			multiply_temp_variable = { _aifc_pad_bonus = 5 }
			if = {
				limit = { check_variable = { _aifc_pad_bonus > 60 } }
				set_temp_variable = { _aifc_pad_bonus = 60 }
			}
			add_to_temp_variable = { _aifc_score = _aifc_pad_bonus }

			if = {
				limit = { check_variable = { _aifc_score > _aifc_best_score } }
				set_temp_variable = { _aifc_best_score = _aifc_score }
				set_temp_variable = { _aifc_best_id = _aifc_c }
			}
		}
	}

	############################################################
	# 3 + 4. Build the corridor and pick the depth objective
	############################################################

	if = {
		limit = { check_variable = { _aifc_best_id > 0 } }

		set_variable = { WA_AI_AIFC_sector_anchor = _aifc_best_id }
		# Age starts at 1, not 0: a variable stored at 0 fails has_variable/check_variable > 0 guards,
		# which made the weekly ageing in WA_AI_AIFC_core.txt treat a fresh sector as "no sector" and
		# re-select every week. 1..MAX_AGE = live sector; 0/absent = no sector.
		set_variable = { WA_AI_AIFC_sector_age = 1 }

		# Publish who the sector is against: the anchor's controller, stored as a country id on ROOT
		# and marked with a per-owner flag on the enemy. The flag is what ai_strategy country_trigger
		# blocks can read cheaply (armor-follows-schwerpunkt in WA_AI_MILITARY_DEFAULT_FRONT_aifc.txt);
		# clearing happens in WA_AI_AIFC_clear_sector, which runs at the head of every re-selection.
		var:_aifc_best_id = {
			CONTROLLER = {
				ROOT = { set_variable = { WA_AI_AIFC_sector_enemy = PREV.id } }
			}
		}
		var:WA_AI_AIFC_sector_enemy = {
			set_country_flag = WA_AI_AIFC_sector_enemy_of_@PREV
		}

		add_to_array = { WA_AI_AIFC_sector_states = _aifc_best_id }
		add_to_array = { WA_AI_AIFC_sector_objectives = _aifc_best_id }
		# Fix 31: engine-encoded twins for the Layer 4 state_triggers. Their is_in_array
		# compares against THIS.id of a state, which is the engine-encoded scope reference,
		# not the plain map-data id (live-confirmed: anchor=228 vs scoped=-10737.41596) - so
		# Layer 4 needs arrays stored in the SAME encoding. PREV.id through a ROOT hop stores
		# exactly that encoding. Kept in lockstep with the plain arrays; cleared together in
		# WA_AI_AIFC_clear_sector.
		var:_aifc_best_id = {
			ROOT = {
				add_to_array = { WA_AI_AIFC_sector_states_ref = PREV.id }
				add_to_array = { WA_AI_AIFC_sector_objectives_ref = PREV.id }
			}
		}

		# Corridor: enemy-held neighbours of the anchor.
		set_temp_variable = { _aifc_depth_id = 0 }
		set_temp_variable = { _aifc_depth_score = -9999 }

		var:_aifc_best_id = {
			every_neighbor_state = {
				limit = {
					CONTROLLER = { has_war_with = ROOT }
				}

				WA_AI_PC_get_state_id = yes
				set_temp_variable = { _aifc_nid = state_id_ }

				if = {
					limit = {
						check_variable = { ROOT.WA_AI_AIFC_sector_states^num < @WA_AI_AIFC_SECTOR_MAX_STATES }
						NOT = { is_in_array = { ROOT.WA_AI_AIFC_sector_states = _aifc_nid } }
					}

					ROOT = {
						add_to_array = { WA_AI_AIFC_sector_states = _aifc_nid }
						# Fix 31: encoded twin (PREV = the neighbour state) - see anchor site
						add_to_array = { WA_AI_AIFC_sector_states_ref = PREV.id }
					}

					# Track the most valuable corridor state as the depth objective.
					set_temp_variable = { _aifc_target = _aifc_nid }
					ROOT = { WA_AI_AIFC_score_state = yes }

					if = {
						limit = { check_variable = { _aifc_score > _aifc_depth_score } }
						set_temp_variable = { _aifc_depth_score = _aifc_score }
						set_temp_variable = { _aifc_depth_id = _aifc_nid }
					}
				}
			}
		}

		if = {
			limit = { check_variable = { _aifc_depth_id > 0 } }
			add_to_array = { WA_AI_AIFC_sector_objectives = _aifc_depth_id }
			# Fix 31: encoded twin - see anchor site
			var:_aifc_depth_id = {
				ROOT = { add_to_array = { WA_AI_AIFC_sector_objectives_ref = PREV.id } }
			}
		}
	}
	else = {
		# No land contact with any enemy - amphibious or air war only. Leave the sector empty so Layer 4
		# stays inert rather than suppressing targets the engine could otherwise use.
		WA_AI_AIFC_clear_sector = yes
	}

	clear_temp_array = _aifc_cand
	clear_temp_array = _aifc_pad_arr
	clear_temp_array = _aifc_ctrl_arr
}


############################################################################################################
#	Score one enemy state as a breakthrough point
#	ROOT = country scope, _aifc_target = state ID  >>> _aifc_score
#
#	Baseline 100, adjusted by three questions:
#		Can armour move through it?   terrain, from precomputed province data
#		Is it thinly held?            division density - the point of a schwerpunkt is to hit a weak seam
#		Is it worth taking?           industry, and whether it is occupied territory rather than core
#
#	All terrain data is read from globals in country scope, so only the density and industry checks pay
#	for a scope change.
############################################################################################################

WA_AI_AIFC_score_state = {

	set_temp_variable = { _aifc_score = 100 }

	############################################################
	# Terrain of the state's main settlement - can we exploit here?
	############################################################

	set_temp_variable = { _aifc_vp_prov = global.WA_AI_MAP_state_vp_province^_aifc_target }
	set_temp_variable = { _aifc_terrain = global.WA_AI_MAP_province_terrain_type^_aifc_vp_prov }

	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_plains } }   add_to_temp_variable = { _aifc_score = 30 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_desert } }   add_to_temp_variable = { _aifc_score = 20 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_hills } }    subtract_from_temp_variable = { _aifc_score = 10 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_forest } }   subtract_from_temp_variable = { _aifc_score = 15 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_urban } }    subtract_from_temp_variable = { _aifc_score = 20 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_jungle } }   subtract_from_temp_variable = { _aifc_score = 35 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_marsh } }    subtract_from_temp_variable = { _aifc_score = 40 } }
	if = { limit = { check_variable = { _aifc_terrain = global.WA_AI_MAP_PROV_TYPE_mountain } } subtract_from_temp_variable = { _aifc_score = 50 } }

	############################################################
	# Garrison density and strategic value
	############################################################

	var:_aifc_target = {

		# A thin seam is the whole point. A heavily stacked state is where a concentrated push dies.
		if = {
			limit = { CONTROLLER = { divisions_in_state = { state = PREV size < 3 } } }
			ROOT = { add_to_temp_variable = { _aifc_score = 40 } }
		}
		else_if = {
			limit = { CONTROLLER = { divisions_in_state = { state = PREV size > 7 } } }
			ROOT = { subtract_from_temp_variable = { _aifc_score = 40 } }
		}

		# Industry behind the line is worth reaching.
		if = {
			limit = { arms_factory > 0 }
			ROOT = { add_to_temp_variable = { _aifc_score = 15 } }
		}
		if = {
			limit = { industrial_complex > 1 }
			ROOT = { add_to_temp_variable = { _aifc_score = 15 } }
		}

		# Occupied territory resists less than the defender's own core.
		if = {
			limit = { NOT = { is_core_of = CONTROLLER } }
			ROOT = { add_to_temp_variable = { _aifc_score = 20 } }
		}
	}
}


############################################################################################################
#	ARMOR-FOLLOWS-SCHWERPUNKT - scripted front_armor_score steering
############################################################################################################
#	Pulls ROOT's armour toward the sector enemy (+400) and eases it off every other war enemy's front
#	(-150), tag-free, for any country in any world state. Switch: WA_AI_AIFC_armor_steering_enabled in
#	WA_AI_AIFC_triggers.txt.
#
#	WHY SCRIPT AND NOT AN ai_strategy FILE
#	--------------------------------------
#	front_armor_score is id-keyed only - it accepts id = "TAG" and none of the generic targeting fields
#	(the country_trigger version in WA_AI_MILITARY_DEFAULT_FRONT_aifc.txt never parsed:
#	"Unexpected token: country_trigger"). A dynamic target therefore needs add_ai_strategy with a tag
#	that is only known at runtime, which is what the meta_effect emitters below do: [?var.GetTag]
#	renders the country id in WA_AI_AIFC_sector_enemy as its literal tag inside the compiled effect
#	(same idiom as ASG_scripted_effects.txt tag curves and the railway [?state_id_] array access).
#
#	RECONCILIATION MODEL
#	--------------------
#	There is no remove_ai_strategy, and front_armor_score entries are additive per id - so an entry is
#	"retired" by adding its exact negation, and the books must track what is currently applied:
#
#		WA_AI_AIFC_armor_boost        country id currently holding the +400 entry (absent = none)
#		WA_AI_AIFC_armor_suppressed   array of country ids currently holding a -150 entry
#
#	WA_AI_AIFC_armor_reconcile runs every weekly tick (end of WA_AI_AIFC_update_sector) and diffs the
#	books against the desired state. Unchanged entries are NOT touched - a stable sector against a
#	stable enemy set reconciles to zero work. Entries churn only when the sector enemy changes, a new
#	enemy joins the war, an enemy leaves it, or the steering switch flips.
#
#	ENTRY ACCUMULATION (accepted, bounded)
#	--------------------------------------
#	Every churn leaves a +X/-X pair of strategy instances on the country, serialized in the save.
#	Sectors re-select at most every @WA_AI_AIFC_SECTOR_MAX_AGE weeks and the enemy only changes on a
#	subset of re-selections, so a full campaign accumulates at most a few hundred inert paired entries
#	per major - accepted as the cost of there being no remove effect.
#
#	KNOWN GAP: a tracked country that ceases to exist (annexed) cannot receive the negation entry -
#	the emitter is skipped and its live entry goes stale. If that tag later revives into a war with
#	ROOT it carries a stale +-400/150 bias until the next enemy change reconciles it. Rare and
#	self-correcting; noted here so nobody hunts it as a mystery.
#
#	VALUES are literals inside the meta_effect text blocks (runtime-compiled text does not see
#	file-scoped @constants). BOOST = 400, SUPPRESS = -150 - the same band the removed ai_strategy
#	version used: below the explicit faction/country steering entries (+500 GER etc.), which add on
#	top where both apply, so the schwerpunkt bias never overrides an explicit historical rule.
############################################################################################################

############################################################
# Entry emitters
# ROOT = country scope, _aifc_armor_id = target country id
# Each adds ONE front_armor_score instance on ROOT. Callers
# own the bookkeeping; these just emit.
############################################################

WA_AI_AIFC_armor_add_boost = {
	meta_effect = {
		text = {
			add_ai_strategy = { type = front_armor_score id = [ETAG] value = 400 }
		}
		ETAG = "[?_aifc_armor_id.GetTag]"
	}
}

WA_AI_AIFC_armor_cancel_boost = {
	meta_effect = {
		text = {
			add_ai_strategy = { type = front_armor_score id = [ETAG] value = -400 }
		}
		ETAG = "[?_aifc_armor_id.GetTag]"
	}
}

WA_AI_AIFC_armor_add_suppress = {
	meta_effect = {
		text = {
			add_ai_strategy = { type = front_armor_score id = [ETAG] value = -150 }
		}
		ETAG = "[?_aifc_armor_id.GetTag]"
	}
}

WA_AI_AIFC_armor_cancel_suppress = {
	meta_effect = {
		text = {
			add_ai_strategy = { type = front_armor_score id = [ETAG] value = 150 }
		}
		ETAG = "[?_aifc_armor_id.GetTag]"
	}
}


############################################################################################################
#	Reconcile applied armour steering against the current sector
#	ROOT = country scope
#	Reads:  WA_AI_AIFC_sector_enemy (absent = no sector), WA_AI_AIFC_armor_steering_enabled
#	Writes: WA_AI_AIFC_armor_boost, WA_AI_AIFC_armor_suppressed, and front_armor_score entries on ROOT
############################################################################################################

WA_AI_AIFC_armor_reconcile = {

	# Desired boost target: the sector enemy, if steering applies at all. 0 = steer nothing, which
	# retires everything below - covering sector loss, war end, armour loss, and the master switch.
	set_temp_variable = { _aifc_want = 0 }
	if = {
		limit = {
			WA_AI_AIFC_armor_steering_enabled = yes
			has_variable = WA_AI_AIFC_sector_enemy
		}
		set_temp_variable = { _aifc_want = WA_AI_AIFC_sector_enemy }
	}

	set_temp_variable = { _aifc_cur = 0 }
	if = {
		limit = { has_variable = WA_AI_AIFC_armor_boost }
		set_temp_variable = { _aifc_cur = WA_AI_AIFC_armor_boost }
	}

	############################################################
	# 1. Boost target changed - retire the old, install the new
	############################################################

	if = {
		limit = { NOT = { check_variable = { _aifc_want = _aifc_cur } } }
		if = {
			limit = { check_variable = { _aifc_cur > 0 } }
			set_temp_variable = { _aifc_armor_id = _aifc_cur }
			if = {
				limit = { var:_aifc_armor_id = { exists = yes } }
				WA_AI_AIFC_armor_cancel_boost = yes
			}
			clear_variable = WA_AI_AIFC_armor_boost
		}

		if = {
			limit = { check_variable = { _aifc_want > 0 } }
			set_temp_variable = { _aifc_armor_id = _aifc_want }
			WA_AI_AIFC_armor_add_boost = yes
			set_variable = { WA_AI_AIFC_armor_boost = _aifc_want }
			# If the new target was under suppression (yesterday's secondary front is today's
			# schwerpunkt), the rebuild loop below retires that -150 - the "id == want" arm - so
			# the target's net lands at exactly +400.
		}
	}

	############################################################
	# 2. Retire suppressions that no longer apply
	#    (enemy left the war, became the boost target, or
	#     steering itself turned off)
	############################################################
	# Rebuild rather than remove-in-place: copy the books to a temp array, clear them, and re-add
	# only the entries that survive. Avoids remove_from_array's value form, which has no usage
	# precedent in this mod - if it ever failed silently, a stale id would be re-cancelled +150 on
	# every weekly tick, an unbounded positive drift on that enemy's armour score.

	clear_temp_array = _aifc_supp_old
	for_each_loop = { array = WA_AI_AIFC_armor_suppressed value = _aifc_sid_v
		add_to_temp_array = { _aifc_supp_old = _aifc_sid_v }
	}
	clear_array = WA_AI_AIFC_armor_suppressed

	for_each_loop = { array = _aifc_supp_old value = _aifc_sid_v
		if = {
			limit = {
				OR = {
					check_variable = { _aifc_want < 1 }
					check_variable = { _aifc_sid_v = _aifc_want }
					NOT = {
						var:_aifc_sid_v = {
							exists = yes
							has_war_with = ROOT
						}
					}
				}
			}
			# Retire: emit the negation (skipped for an annexed tag - see KNOWN GAP above) and
			# do not re-add to the books.
			set_temp_variable = { _aifc_armor_id = _aifc_sid_v }
			if = {
				limit = { var:_aifc_sid_v = { exists = yes } }
				WA_AI_AIFC_armor_cancel_suppress = yes
			}
		}
		else = {
			# Keep: entry still valid, back into the books untouched.
			add_to_array = { WA_AI_AIFC_armor_suppressed = _aifc_sid_v }
		}
	}

	############################################################
	# 3. Suppress war enemies not yet tracked
	#    Only while a boost is installed - suppression without a
	#    boost is exactly the unpaired case the header forbids.
	############################################################

	if = {
		limit = { check_variable = { _aifc_want > 0 } }
		every_enemy_country = {
			set_temp_variable = { _aifc_eid = THIS.id }
			if = {
				limit = {
					NOT = { check_variable = { _aifc_eid = _aifc_want } }
					NOT = { is_in_array = { ROOT.WA_AI_AIFC_armor_suppressed = _aifc_eid } }
				}
				set_temp_variable = { _aifc_armor_id = _aifc_eid }
				ROOT = {
					WA_AI_AIFC_armor_add_suppress = yes
					add_to_array = { WA_AI_AIFC_armor_suppressed = _aifc_eid }
				}
			}
		}
	}

	clear_temp_array = _aifc_supp_old
}
