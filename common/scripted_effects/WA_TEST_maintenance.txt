	#    trigger. When B and C disagree the fault is in one of the two, not hidden between them.
	############################################################

	set_temp_variable = { _mnt_c_short = 0 }
	if = {
		limit = { check_variable = { _mnt_at_t > _mnt_tm } check_variable = { _mnt_at_r < _mnt_en } }
		set_temp_variable = { _mnt_c_short = 1 }
	}
	set_temp_variable = { _mnt_b_short = 0 }
	if = { limit = { WA_AI_PRODUCTION_is_anti_tank_short = yes } set_temp_variable = { _mnt_b_short = 1 } }
	set_temp_variable = { _mnt_ok_agree = 0 }
	if = { limit = { check_variable = { _mnt_b_short = _mnt_c_short } } set_temp_variable = { _mnt_ok_agree = 1 } }
	log = "  C walk: anti_tank short - shipped=[?_mnt_b_short|.0]  re-derived=[?_mnt_c_short|.0]"

	############################################################
	# D. Verdict
	#    agree      B and C ask the same question of anti-tank and get the same answer
	#    schmitt    no key is both short and supplied - the two bars have not crossed
	#    exclusive  no ARMOUR key is maintained while the armour category push is armed
	#               (a need-blind floor under a raised demand double-counts)
	#    useful     no key is maintained while its own build decision still holds
	#    deep       the deep half never arms without the base half
	############################################################

	log = "  VERDICT: shipped-vs-walk=[?_mnt_ok_agree|.0]  schmitt-disjoint=[?_mnt_ok_schmitt|.0]  exclusive-with-category-push=[?_mnt_ok_excl|.0]  floor-is-useful=[?_mnt_ok_use|.0]  deep-implies-base=[?_mnt_ok_deep|.0]   (all five must read 1)"
	log = "=== END MAINTENANCE TEST ==="
}
