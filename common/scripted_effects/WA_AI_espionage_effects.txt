# [intel-adversary-targeting] score factor a current intel target keeps over challengers, so rank
# 2/3 swaps between 2-day pulses do not pull operatives off a half-built network.
# Single reader in this file: WA_AI_rank_intel_candidate.
@INTEL_INCUMBENT_FACTOR = 1.2

WA_AI_set_boost_resistance_target = {

	set_temp_variable = { owned_states_controlled_by_enemies = 0 }

	if = { limit = { has_war = yes }

		for_each_scope_loop = { array = owned_states

			if = { limit = { CONTROLLER = { has_war_with = ROOT } }

				add_to_temp_variable = { owned_states_controlled_by_enemies = 1 }
			}
		}
	}

	if = {
		limit = {
			check_variable = { owned_states_controlled_by_enemies > 10 }
			check_variable = { operatives > 1 }
		}

		set_temp_variable = { best_boost_resistance_target = 0 }
		set_temp_variable = { best_boost_resistance_states_num = 0 }

		for_each_loop = { array = enemies

			if = { limit = { any_home_area_neighbor_country = { tag = var:v } }

				set_temp_variable = { num_owned_states_controlled_by_enemy = 0 }

				for_each_scope_loop = { array = ROOT.owned_states

					if = { limit = { is_controlled_by = var:v }

						add_to_temp_variable = { num_owned_states_controlled_by_enemy = 1 }
					}
				}

				if = { limit = { check_variable = { num_owned_states_controlled_by_enemy > best_boost_resistance_states_num } }

					set_temp_variable = { best_boost_resistance_target = var:v }
					set_temp_variable = { best_boost_resistance_states_num = num_owned_states_controlled_by_enemy }
				}
			}
		}

		if = { limit = { NOT = { check_variable = { best_boost_resistance_target = 0 } } }

			set_variable = { WA_AI_boost_resistance_operation_target = best_boost_resistance_target }
		}
	}
	else = {
		clear_variable = WA_AI_boost_resistance_operation_target
	}
}

WA_AI_set_rescue_operative_target = {

	if = {
		limit = {
			NOT = {
				AND = {
					has_variable = WA_AI_rescue_operative_target
					var:WA_AI_rescue_operative_target = { has_captured_operative = ROOT }
				}
			}
		}

		if = { limit = { any_operative_leader = { is_operative_captured = yes } }

			random_operative = { limit = { is_operative_captured = yes }

				set_variable = { ROOT.WA_AI_rescue_operative_target = operative_captor }
			}
		}
		else = {
			clear_variable = WA_AI_rescue_operative_target
		}
	}
}

# [intel-adversary-targeting] COUNTRY scope (ROOT = the spying AI), every background pulse. Names
# the two strongest adversaries as the intel-network targets WA_AI_espionage_strategies.txt reads:
# at war its living enemies, at peace the majors WA_AI_THIS_is_threat_to_PREV names; power is
# WA_AI_MATH_country_power_score. How to tell it is wrong: the AI builds its networks on a minor
# while at war with a major.
WA_AI_set_intel_network_targets = {
	set_temp_variable = { _intel_prev_1 = 0 }
	set_temp_variable = { _intel_prev_2 = 0 }
	if = { limit = { has_variable = WA_AI_intel_target_1 } set_temp_variable = { _intel_prev_1 = WA_AI_intel_target_1 } }
	if = { limit = { has_variable = WA_AI_intel_target_2 } set_temp_variable = { _intel_prev_2 = WA_AI_intel_target_2 } }
	clear_variable = WA_AI_intel_target_1
	clear_variable = WA_AI_intel_target_2

	if = { limit = { check_variable = { operatives > 0 } }

		set_temp_variable = { _intel_score_1 = -1 }
		set_temp_variable = { _intel_score_2 = -1 }
		set_temp_variable = { _intel_tag_1 = 0 }
		set_temp_variable = { _intel_tag_2 = 0 }

		if = { limit = { has_war = yes }
			every_enemy_country = {
				limit = {
					exists = yes
					has_capitulated = no
				}
				WA_AI_rank_intel_candidate = yes
			}
		}
		else = {
			every_other_country = {
				limit = {
					is_major = yes
					WA_AI_THIS_is_threat_to_PREV = yes
				}
				WA_AI_rank_intel_candidate = yes
			}
		}

		if = { limit = { NOT = { check_variable = { _intel_tag_1 = 0 } } }
			set_variable = { WA_AI_intel_target_1 = _intel_tag_1 }
		}
		if = { limit = { NOT = { check_variable = { _intel_tag_2 = 0 } } }
			set_variable = { WA_AI_intel_target_2 = _intel_tag_2 }
		}
	}
}

# [intel-adversary-targeting] COUNTRY scope, THIS = one candidate, called only by
# WA_AI_set_intel_network_targets. Keeps the two best scores in its _intel_* temps; a current
# target scores x @INTEL_INCUMBENT_FACTOR.
WA_AI_rank_intel_candidate = {
	WA_AI_MATH_country_power_score = yes
	set_temp_variable = { _intel_score = _power_score_ }
	if = {
		limit = {
			OR = {
				check_variable = { _intel_prev_1 = THIS.id }
				check_variable = { _intel_prev_2 = THIS.id }
			}
		}
		multiply_temp_variable = { _intel_score = @INTEL_INCUMBENT_FACTOR }
	}

	if = { limit = { check_variable = { _intel_score > _intel_score_1 } }
		set_temp_variable = { _intel_score_2 = _intel_score_1 }
		set_temp_variable = { _intel_tag_2 = _intel_tag_1 }
		set_temp_variable = { _intel_score_1 = _intel_score }
		set_temp_variable = { _intel_tag_1 = THIS.id }
	}
	else_if = { limit = { check_variable = { _intel_score > _intel_score_2 } }
		set_temp_variable = { _intel_score_2 = _intel_score }
		set_temp_variable = { _intel_tag_2 = THIS.id }
	}
}
