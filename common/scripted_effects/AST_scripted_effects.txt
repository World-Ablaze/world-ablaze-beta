# Creates (or refreshes) the "Australian Infantry Division" template.
#
# Composition mirrors WA_AI_TEMPLATES_GENERIC_HEAVY_INFANTRY_30_MIX_MOT
# (common/ai_templates/WA_AI_TEMPLATES_infantry.txt, flag value 1006) so the
# player and the AI field the same division from this focus tree.
#
# The template is built in two steps: a base every country can equip, then the
# parts that depend on what has actually been researched.
#
# - The maintenance company is dropped when tech_maintenance_company is missing.
#   That technology must never become a focus prerequisite: its AI research
#   weight is zeroed unless WA_AI_CONFIG_DIVISIONS_use_armored_divisions or
#   _use_mechanized_divisions is true for the country, so a focus gated on it
#   could become permanently unreachable for the AI.
# - The anti-tank battalions/companies and the anti-air company are dropped when
#   the country has no variant of anti_tank_equipment / anti_air_equipment.
#   WA_AI_RESEARCH_needs_anti_tank and _needs_anti_air are false for a non-major
#   before 1940, so AI Australia spawns these divisions with no AT or AA design
#   unlocked; create_unit then asks the equipment pool to fill an archetype that
#   has no variant ("equipmentpool.cpp: Trying to fill variant where none exist
#   from type: anti_tank_equipment belonging to Australia", once per division).
#
# Degrading the template here keeps the focus reachable whatever the AI has
# researched. Every other sub-unit below is either active by default or enabled
# by a technology whose WA_AI_RESEARCH_needs_* trigger is always = yes.
#
# Must be called immediately before any create_unit that spawns it: the AI
# renames and reshapes its own templates at will, so no pre-existing template
# name can be relied on when the country is AI-controlled. Re-running the effect
# is safe - the division_template block below redefines the template from
# scratch before the conditional parts are re-added.
AST_create_australian_infantry_division_template = {
	division_template = {
		name = "Australian Infantry Division"
		division_names_group = AST_INF_01
		regiments = {
			infantry_heavy_horse_battalion_line = { x = 0 y = 0 }
			infantry_heavy_horse_battalion_line = { x = 0 y = 1 }
			infantry_heavy_horse_battalion_line = { x = 0 y = 2 }
			infantry_heavy_horse_battalion_line = { x = 0 y = 3 }

			infantry_heavy_horse_battalion_line = { x = 1 y = 0 }
			infantry_heavy_horse_battalion_line = { x = 1 y = 1 }
			infantry_heavy_horse_battalion_line = { x = 1 y = 2 }

			infantry_heavy_horse_battalion_line = { x = 2 y = 0 }
			infantry_heavy_horse_battalion_line = { x = 2 y = 1 }
			infantry_heavy_horse_battalion_line = { x = 2 y = 2 }

			artillery_horse_battalion_line = { x = 3 y = 0 }
			artillery_horse_battalion_line = { x = 3 y = 1 }
			artillery_horse_battalion_line = { x = 3 y = 2 }
		}
		regimental_support = {
			pack_artillery_mot_company_regimental = { x = 0 y = 0 }
			pack_artillery_mot_company_regimental = { x = 1 y = 0 }
			pack_artillery_mot_company_regimental = { x = 2 y = 0 }
		}
		support = {
			engineer_mot_battalion_divisional = { x = 0 y = 0 }
			recon_mot_company_divisional = { x = 0 y = 1 }
			field_hospital_mot_company_divisional = { x = 0 y = 2 }
			signal_mot_company_divisional = { x = 0 y = 3 }
			logistics_mot_company_divisional = { x = 0 y = 4 }

			heavy_artillery_mot_company_divisional = { x = 1 y = 0 }
			military_police_mot_company_divisional = { x = 1 y = 1 }
		}
	}
	if = {
		limit = {
			has_design_based_on = anti_tank_equipment
		}
		add_units_to_division_template = {
			template_name = "Australian Infantry Division"
			regiments = {
				anti_tank_horse_battalion_line = 4
				anti_tank_horse_battalion_line = 4
			}
			regimental_support = {
				anti_tank_mot_company_regimental = 0
				anti_tank_mot_company_regimental = 1
				anti_tank_mot_company_regimental = 2
			}
		}
	}
	if = {
		limit = {
			has_design_based_on = anti_air_equipment
		}
		add_units_to_division_template = {
			template_name = "Australian Infantry Division"
			support = {
				anti_air_mot_company_divisional = 1
			}
		}
	}
	if = {
		limit = {
			has_tech = tech_maintenance_company
		}
		add_units_to_division_template = {
			template_name = "Australian Infantry Division"
			support = {
				maintenance_mot_company_divisional = 1
			}
		}
	}
}
