# Rewards and penalties of the naval armaments plan missions (common/decisions/JAP_factions.txt). The merged
# mission gives both rewards, so taking the Fourth Circle Plan early does not cost the Third Circle Plan's.

# Third Naval Armaments Replenishment Plan
JAP_third_naval_plan_reward_effect = {
	add_political_power = 25
	navy_experience = 25
	WA_add_mastery_capital_ship_100 = yes
	WA_add_mastery_carrier_100 = yes
	if = {
		limit = {
			has_dlc = "Arms Against Tyranny"
		}
		custom_effect_tooltip = JAP_Navy_MIO_Funds_tt
		effect_tooltip = {
			mio:JAP_yokosuka_naval_arsenal_organization = {
				add_mio_funds = 600
			}
			mio:JAP_kure_naval_arsenal_organization = {
				add_mio_funds = 600
			}
			mio:JAP_sasebo_naval_arsenal_organization = {
				add_mio_funds = 600
			}
			mio:JAP_maizuru_naval_arsenal_organization = {
				add_mio_funds = 600
			}
			mio:JAP_kawasaki_shipbuilding_organization = {
				add_mio_funds = 600
			}
			mio:JAP_mitsubishi_organization = {
				add_mio_funds = 600
			}
		}
		hidden_effect = {
			if = {
				limit = {
					mio:JAP_yokosuka_naval_arsenal_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_yokosuka_naval_arsenal_organization = {
					add_mio_funds = 600
				}
			}
			if = {
				limit = {
					mio:JAP_kure_naval_arsenal_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_kure_naval_arsenal_organization = {
					add_mio_funds = 600
				}
			}
			if = {
				limit = {
					mio:JAP_sasebo_naval_arsenal_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_sasebo_naval_arsenal_organization = {
					add_mio_funds = 600
				}
			}
			if = {
				limit = {
					mio:JAP_maizuru_naval_arsenal_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_maizuru_naval_arsenal_organization = {
					add_mio_funds = 600
				}
			}
			if = {
				limit = {
					mio:JAP_kawasaki_shipbuilding_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_kawasaki_shipbuilding_organization = {
					add_mio_funds = 600
				}
			}
			if = {
				limit = {
					mio:JAP_mitsubishi_organization = {
						has_mio_size > 1
					}
				}
				mio:JAP_mitsubishi_organization = {
					add_mio_funds = 600
				}
			}
		}
	}

	if = {
		limit = {
			JAP_kodoha_faction_alive_trigger = no
		}
		custom_effect_tooltip = JAP_kantaiha_mission_influence_up_tt
		custom_effect_tooltip = JAP_toseiha_mission_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kantaiha_influence < 7 } }
				add_to_variable = { JAP_kantaiha_influence = 1 }
			}
			add_to_variable = { JAP_kantaiha_industrial_capacity_dockyard = 0.1 }
			if = {
				limit = { check_variable = { JAP_toseiha_influence > 0 } }
				add_to_variable = { JAP_toseiha_influence = -1 }
			}
			add_to_variable = { JAP_toseiha_industrial_capacity_factory = -0.1 }
		}
	}
	if = {
		limit = {
			JAP_kodoha_faction_alive_trigger = no
		}
		custom_effect_tooltip = JAP_kantaiha_mission_influence_up_tt
		custom_effect_tooltip = JAP_toseiha_mission_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kantaiha_influence < 7 } }
				add_to_variable = { JAP_kantaiha_influence = 1 }
			}
			add_to_variable = { JAP_kantaiha_industrial_capacity_dockyard = 0.1 }
			if = {
				limit = { check_variable = { JAP_toseiha_influence > 0 } }
				add_to_variable = { JAP_toseiha_influence = -1 }
			}
			add_to_variable = { JAP_toseiha_industrial_capacity_factory = -0.1 }
		}
	}

	else = {
		custom_effect_tooltip = JAP_kantaiha_mission_influence_up_tt
		custom_effect_tooltip = JAP_kodoha_mission_influence_down_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kantaiha_influence < 7 } }
				add_to_variable = { JAP_kantaiha_influence = 1 }
			}
			add_to_variable = { JAP_kantaiha_industrial_capacity_dockyard = 0.1 }
			if = {
				limit = { check_variable = { JAP_kodoha_influence > 0 } }
				add_to_variable = { JAP_kodoha_influence = -1 }
			}
			add_to_variable = { JAP_kodoha_industrial_capacity_factory = -0.1 }
		}
	}
}
