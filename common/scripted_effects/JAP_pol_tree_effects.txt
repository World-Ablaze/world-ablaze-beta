# Effects of the new political tree that are written by hand (the generator of the tree does not touch this file).

# Goodwill towards the West is paid for at home: 10 political power for every step of the Subjects' influence
# (JAP_shinmin_influence, 0 to 7). Used by the focuses that raise the West's opinion of Japan.
JAP_pol_us_goodwill_cost_effect = {
	custom_effect_tooltip = JAP_pol_us_goodwill_cost_tt
	if = {
		limit = { check_variable = { JAP_shinmin_influence > 6 } }
		add_political_power = -70
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 5 } }
		add_political_power = -60
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 4 } }
		add_political_power = -50
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 3 } }
		add_political_power = -40
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 2 } }
		add_political_power = -30
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 1 } }
		add_political_power = -20
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 0 } }
		add_political_power = -10
	}
}
