############################################################################################################
#	World Ablaze AI mod - AI Force Concentration (AIFC) - Core Module
#	Entry point, sector lifecycle, staleness detection
############################################################################################################
#	System overview and value bands: common/ai_strategy/WA_AI_MILITARY_DEFAULT_FRONT_aifc.txt
#	Behavioural switches:            common/scripted_triggers/WA_AI_AIFC_triggers.txt
#	Candidate scoring:               common/scripted_effects/WA_AI_AIFC_helpers.txt
#
#	WHAT THIS PRODUCES
#	------------------
#	Two persistent country arrays, consumed by the Layer 4 ai_strategy blocks through state_trigger:
#
#		WA_AI_AIFC_sector_states       the corridor - enemy-held states forming the chosen offensive axis
#		WA_AI_AIFC_sector_objectives   the deep targets inside that corridor
#
#	Plus two bookkeeping variables:
#
#		WA_AI_AIFC_sector_anchor       state ID the sector was built around
#		WA_AI_AIFC_sector_age          weeks since selection, drives forced re-evaluation
#
#	And, downstream of the sector, dynamic front_armor_score strategy entries steering armour toward
#	the sector enemy (WA_AI_AIFC_armor_reconcile in the helpers, called at the end of every weekly
#	update - the type is id-keyed so this cannot be a static ai_strategy block).
#
#	WHY ARRAYS AND NOT HARDCODED STATE LISTS
#	----------------------------------------
#	The engine re-evaluates AIFC every AIFC_UPDATE_FREQUENCY_DAYS (5) and calls the ai_strategy
#	state_trigger per candidate state, per front. Anything expensive in that trigger is paid hundreds of
#	times. So all the thinking happens here, once a week, and the strategy layer only does a single
#	is_in_array lookup. This is the same split Expert AI uses and it is the reason its triggers are
#	one-liners.
#
#	CADENCE
#	-------
#	Called from common/on_actions/WA_AI_misc_on_actions.txt on_weekly, behind
#	WA_AI_AIFC_should_select_sector.
############################################################################################################

### AIFC eligibility constants (must match common/scripted_triggers/WA_AI_AIFC_triggers.txt)
### The trigger file is authoritative - it is where the eligibility test that reads them lives.
@WA_AI_AIFC_MIN_STATES = 2			# Skip micro-nations; nothing to concentrate with
@WA_AI_AIFC_MIN_DIV = 12			# ...and armies too small to have a surplus (Fix 56; rationale is
									# in the trigger file, which is where the test lives)
@WA_AI_AIFC_MAX_SURRENDER = 0.35	# Skip countries already collapsing...
@WA_AI_AIFC_RECOVERY_MIN_STATES = 20	# ...unless the army is intact: states above this AND
@WA_AI_AIFC_RECOVERY_MIN_DIV = 40		# divisions above this (Fix 43 escape hatch)

### AIFC sector constants (declared only here; helpers has no copy)
@WA_AI_AIFC_SECTOR_MAX_AGE = 4		# Weeks before a sector is re-evaluated regardless of validity
									# (age is 1-based: selected at 1, re-selected when age exceeds this)


############################################################################################################
#	ENTRY POINT - weekly sector maintenance
#	ROOT = country scope
#
#	Three outcomes:
#		not eligible          -> clear any sector, so Layer 4 goes inert and the engine falls back to
#		                         its own target scoring rather than pushing at a stale axis
#		sector still good     -> age it and do nothing
#		sector stale/invalid  -> re-select
############################################################################################################

WA_AI_AIFC_update_sector = {

	if = {
		limit = { NOT = { WA_AI_AIFC_should_select_sector = yes } }

		WA_AI_AIFC_clear_sector = yes
	}
	else = {
		# Age the existing sector. A country with no sector is treated as maximally stale so that the
		# first eligible week always selects one. The age is 1-based (set to 1 on selection in
		# WA_AI_AIFC_helpers.txt): 0/absent both mean "no sector" here, so a zero-valued age can never
		# be mistaken for a live one - has_variable is unreliable for a counter whose range includes 0.
		if = {
			limit = { check_variable = { WA_AI_AIFC_sector_age > 0 } }
			add_to_variable = { WA_AI_AIFC_sector_age = 1 }
		}
		else = {
			set_variable = { WA_AI_AIFC_sector_age = 99 }
		}

		WA_AI_AIFC_check_sector_valid = yes		# >>> _aifc_sector_valid

		# Fix 31 verification telemetry - remove when checklist item R1 retires.
		# Healthy: age climbs 2..5 with valid=1, re-selection only at age > 4.
		if = {
			limit = { is_major = yes }
			log = "AIFC-DIAG | [GetDateText] | [Root.GetName] | entry: age=[?WA_AI_AIFC_sector_age] corridor=[?WA_AI_AIFC_sector_states^num] anchor=[?WA_AI_AIFC_sector_anchor] valid=[?_aifc_sector_valid]"
		}

		if = {
			limit = {
				OR = {
					# Nothing selected yet
					NOT = { check_variable = { WA_AI_AIFC_sector_states^num > 0 } }
					# Timed refresh - keeps the AI from committing forever to a dead axis
					check_variable = { WA_AI_AIFC_sector_age > @WA_AI_AIFC_SECTOR_MAX_AGE }
					# The front moved, or we took the corridor
					check_variable = { _aifc_sector_valid < 1 }
				}
			}

			WA_AI_AIFC_select_sector = yes

			if = {
				limit = { has_country_flag = WA_AI_AIFC_logging }
				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | AIFC: sector re-selected, anchor=[?WA_AI_AIFC_sector_anchor] corridor=[?WA_AI_AIFC_sector_states^num]"
			}
		}
	}

	# Armour steering follows whatever the maintenance above decided. Runs every weekly tick, not just
	# on re-selection: it diffs applied front_armor_score entries against the current sector enemy and
	# war set (WA_AI_AIFC_armor_reconcile, helpers), so a stable sector reconciles to zero work while
	# enemies joining/leaving the war are picked up within a week. Also handles full retirement when
	# the branch above cleared the sector or steering became ineligible.
	WA_AI_AIFC_armor_reconcile = yes
}


############################################################################################################
#	Wipe the sector and its bookkeeping
#	ROOT = country scope
#
#	Called when the country stops being eligible, and at the head of every re-selection. Leaving a stale
#	corridor in place is worse than having none: the Layer 4 pairing suppresses everything OUTSIDE the
#	corridor, so a corridor we already captured would suppress all remaining real targets.
############################################################################################################

WA_AI_AIFC_clear_sector = {

	# Fix 31 verification telemetry - remove when checklist item R1 retires.
	if = {
		limit = { is_major = yes }
		log = "AIFC-DIAG | [GetDateText] | [Root.GetName] | clear_sector"
	}

	clear_array = WA_AI_AIFC_sector_states
	clear_array = WA_AI_AIFC_sector_objectives
	# Fix 31: engine-encoded twins consumed by the Layer 4 state_triggers
	# (WA_AI_MILITARY_DEFAULT_FRONT_aifc.txt) - kept in lockstep with the plain arrays.
	clear_array = WA_AI_AIFC_sector_states_ref
	clear_array = WA_AI_AIFC_sector_objectives_ref
	clear_variable = WA_AI_AIFC_sector_anchor
	clear_variable = WA_AI_AIFC_sector_age

	# Retract the sector-enemy marker flag (set in WA_AI_AIFC_select_sector). The armour steering
	# that reads the sector enemy (WA_AI_AIFC_armor_reconcile, helpers) works off the
	# WA_AI_AIFC_sector_enemy variable cleared just below; the flag is a cheap public marker kept
	# in sync for any future ai_strategy country_trigger consumer on a type that supports one.
	if = {
		limit = { has_variable = WA_AI_AIFC_sector_enemy }
		var:WA_AI_AIFC_sector_enemy = {
			clr_country_flag = WA_AI_AIFC_sector_enemy_of_@PREV
		}
		clear_variable = WA_AI_AIFC_sector_enemy
	}
}


############################################################################################################
#	Is the current sector still a live front?
#	ROOT = country scope >>> _aifc_sector_valid (0 or 1)
#
#	Valid means: at least one state we control still borders at least one enemy-held state that is in the
#	corridor. That single check catches both failure modes - the corridor was captured (no longer enemy
#	held) and the front moved away (no longer adjacent to us).
#
#	Follows the WA_AI_PC_has_land_border_with_enemy pattern from the railway primitives.
############################################################################################################

WA_AI_AIFC_check_sector_valid = {

	set_temp_variable = { _aifc_sector_valid = 0 }

	if = {
		limit = { check_variable = { WA_AI_AIFC_sector_states^num > 0 } }

		# Fix 31 (R1): the old passes tested corridor membership with
		#   is_in_array = { ROOT.WA_AI_AIFC_sector_states = THIS.id }
		# but the corridor arrays hold PLAIN map-data ids while THIS.id yields the engine-encoded
		# scope reference - live-confirmed 1943.3 (anchor=228 vs scoped=-10737.41596): equality
		# never held, validity was permanently 0 and the sector re-selected weekly for the
		# system's whole life (sector_age pinned at 1 in every campaign since creation; fixes
		# 4bfea363d and 128cc7995 patched real-but-downstream defects). Never compare a plain
		# stored id against .id of a scope - walk the corridor array and ask each state directly.
		# The array is capped at @WA_AI_AIFC_SECTOR_MAX_STATES, so this is at most 8 state hops.
		for_each_loop = {
			array = WA_AI_AIFC_sector_states
			value = _aifc_v_sid

			if = {
				limit = {
					# stop scanning once proven valid
					check_variable = { _aifc_sector_valid < 1 }
					var:_aifc_v_sid = {
						# still enemy-held...
						CONTROLLER = { has_war_with = ROOT }
						OR = {
							# ...and our frontline still touches it (native case)
							any_neighbor_state = { is_controlled_by = ROOT }
							# Fix 28 mirror: expeditionary sectors - the corridor also stays
							# valid while it borders faction/subject territory holding ROOT
							# divisions (anchored on allied soil the native case cannot see).
							any_neighbor_state = {
								CONTROLLER = {
									NOT = { tag = ROOT }
									has_capitulated = no
									OR = {
										is_in_faction_with = ROOT
										is_subject_of = ROOT
										ROOT = { is_subject_of = PREV }
									}
								}
								ROOT = { divisions_in_state = { state = PREV size > 2 } }
							}
						}
					}
				}
				set_temp_variable = { _aifc_sector_valid = 1 }
			}
		}
	}
}
