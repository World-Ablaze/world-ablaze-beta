############################################################################################################
#	World Ablaze - AIFC sector-selection console harness. harness-contract: v1 (wa-testing SKILL).
############################################################################################################
#	[aifc-traction] L2 instrument: prints every stage of the weekly sector selection for ROOT so a
#	human can read WHICH stage zeroes out (eligibility / candidate collection / expeditionary pass /
#	feasibility floor / main-enemy election by per-enemy pad-band sums). Written for the "USA holds
#	0 sectors in 47 war months" diagnosis; also the owner-run test for the anchor feasibility floor
#	(constant:wa_ai_aifc.selection.min_pad) and the sum election. Section E dumps the armour
#	steering books, including the [aifc-revived-tag-residue] pending-cancel debt ledgers.
#
#	Recipe: events/wa_test_aifc.txt header. Fire `event wa_aifc.1 USA` from ANY tag - do NOT tag
#	into the country under test (tag-switching makes it human). Output: logs/game.log, "AIFC SECTOR TEST".
#
#	The walks below DUPLICATE the shipped selection's limits (WA_AI_AIFC_helpers.txt sections 1/1b/1c
#	and the section-2 floor) on purpose - contract rule 4: when instrument and system disagree, the
#	fault is localised. Only script constants are shared (they ARE the bars being measured).
############################################################################################################

############################################################
#	Per-candidate probe: dedup, pad, floor, logging
#	THIS = enemy-held candidate state, called from the walks below
#	Reads/writes the _aifct_* temps of WA_TEST_AIFC_report
############################################################

WA_TEST_AIFC_probe_candidate = {

	# Dedup on the engine-encoded ref (a candidate borders several of our states) - encoding is
	# consistent within one run, which is all is_in_array needs (the Layer 4 _ref idiom).
	ROOT = { set_temp_variable = { _aifct_ref = PREV.id } }

	if = {
		limit = { NOT = { is_in_array = { _aifct_seen = _aifct_ref } } }

		add_to_temp_array = { _aifct_seen = _aifct_ref }
		add_to_temp_variable = { _aifct_n = 1 }

		# Launching pad - the shipped 1c banding duplicated: per adjacent friendly (own or
		# faction/subject-relation) state, ROOT divisions >9 = 3, >4 = 2, >0 = 1, summed.
		set_temp_variable = { _aifct_pad = 0 }
		every_neighbor_state = {
			limit = {
				OR = {
					is_controlled_by = ROOT
					CONTROLLER = {
						OR = {
							is_in_faction_with = ROOT
							is_subject_of = ROOT
							ROOT = { is_subject_of = PREV }
						}
					}
				}
			}
			if = {
				limit = { ROOT = { divisions_in_state = { state = PREV size > 9 } } }
				add_to_temp_variable = { _aifct_pad = 3 }
			}
			else_if = {
				limit = { ROOT = { divisions_in_state = { state = PREV size > 4 } } }
				add_to_temp_variable = { _aifct_pad = 2 }
			}
			else_if = {
				limit = { ROOT = { divisions_in_state = { state = PREV size > 0 } } }
				add_to_temp_variable = { _aifct_pad = 1 }
			}
		}

		# Anchor feasibility floor ([aifc-traction])
		set_temp_variable = { _aifct_okflag = 0 }
		if = {
			limit = { check_variable = { _aifct_pad > constant:wa_ai_aifc.selection.min_pad } }
			set_temp_variable = { _aifct_okflag = 1 }
			add_to_temp_variable = { _aifct_ok = 1 }
		}

		if = {
			limit = { check_variable = { _aifct_pad > _aifct_max_pad } }
			set_temp_variable = { _aifct_max_pad = _aifct_pad }
		}

		# Main-enemy election books ([aifc-traction]) - independent copy of the shipped 1d
		# election (contract rule 4): floor-eligible candidates add their pad to their
		# controller's sum; the largest sum is elected in the report. min_pad is shared
		# (it IS the bar); the find-or-append structure is re-typed.
		if = {
			limit = { check_variable = { _aifct_okflag > 0 } }
			CONTROLLER = {
				ROOT = { set_temp_variable = { _aifct_ctrl = PREV.id } }
			}
			set_temp_variable = { _aifct_efound = 0 }
			set_temp_variable = { _aifct_eb_ = 0 }
			for_each_loop = { array = _aifct_eids index = _aifct_j value = _aifct_eid break = _aifct_eb_
				if = {
					limit = { check_variable = { _aifct_eid = _aifct_ctrl } }
					set_temp_variable = { _aifct_newsum = _aifct_esums^_aifct_j }
					add_to_temp_variable = { _aifct_newsum = _aifct_pad }
					set_temp_variable = { _aifct_esums^_aifct_j = _aifct_newsum }
					set_temp_variable = { _aifct_efound = 1 }
					set_temp_variable = { _aifct_eb_ = 1 }
				}
			}
			if = {
				limit = { check_variable = { _aifct_efound < 1 } }
				add_to_temp_array = { _aifct_eids = _aifct_ctrl }
				add_to_temp_array = { _aifct_esums = _aifct_pad }
			}
		}

		if = {
			limit = { check_variable = { _aifct_n < 16 } }
			log = "  cand  : [This.GetName]  pad=[?_aifct_pad|.0]  anchor-eligible=[?_aifct_okflag|.0]"
		}
	}
}


############################################################
#	The report
#	ROOT = country under test (fire the event WITH a target)
############################################################

WA_TEST_AIFC_report = {

	log = "=== AIFC SECTOR TEST === [Root.GetName] [GetDateText]"

	### CONTEXT HEADER - printed FIRST, on every run. harness-contract: v1 (wa-testing SKILL).
	log = "  who   : THIS = [This.GetName]   ROOT = [Root.GetName]   FROM = [From.GetName]"
	set_temp_variable = { _aifct_e1_ = 0 }
	set_temp_variable = { _aifct_e2_ = 0 }
	set_temp_variable = { _aifct_e3_ = 0 }
	set_temp_variable = { _aifct_e4_ = 0 }
	set_temp_variable = { _aifct_e5_ = 0 }
	if = { limit = { always = yes } set_temp_variable = { _aifct_e1_ = 1 } }
	if = { limit = { tag = ROOT } set_temp_variable = { _aifct_e2_ = 1 } }
	if = { limit = { tag = THIS } set_temp_variable = { _aifct_e3_ = 1 } }
	if = { limit = { ROOT = { always = yes } } set_temp_variable = { _aifct_e4_ = 1 } }
	if = { limit = { NOT = { always = yes } } set_temp_variable = { _aifct_e5_ = 1 } }
	log = "  scope : always=[?_aifct_e1_|.0]  I-am-ROOT=[?_aifct_e2_|.0]  I-am-THIS=[?_aifct_e3_|.0]  ROOT-scope-usable=[?_aifct_e4_|.0]  control-false=[?_aifct_e5_|.0]"
	log = "            (anything but 1 1 1 1 0 here: STOP. Nothing below is a measurement. Re-fire the same report from another event file - tag GER then event wa_iso.3 - before editing a line of this effect.)"

	############################################################
	# A. Eligibility - each term of WA_AI_AIFC_should_select_sector, independently
	############################################################

	set_temp_variable = { _aifct_ai = 0 }
	if = { limit = { is_ai = yes } set_temp_variable = { _aifct_ai = 1 } }
	set_temp_variable = { _aifct_war = 0 }
	if = { limit = { has_war = yes } set_temp_variable = { _aifct_war = 1 } }
	set_temp_variable = { _aifct_cap = 0 }
	if = { limit = { has_capitulated = yes } set_temp_variable = { _aifct_cap = 1 } }
	set_temp_variable = { _aifct_enemy = 0 }
	if = { limit = { any_enemy_country = { exists = yes } } set_temp_variable = { _aifct_enemy = 1 } }
	set_temp_variable = { _aifct_states = num_of_controlled_states }
	set_temp_variable = { _aifct_div = num_divisions }
	set_temp_variable = { _aifct_bar_states = constant:wa_ai_aifc.eligibility.min_states }
	set_temp_variable = { _aifct_bar_div = constant:wa_ai_aifc.eligibility.min_div }
	set_temp_variable = { _aifct_armysz = 0 }
	if = {
		limit = { has_army_size = { size > constant:wa_ai_aifc.eligibility.min_div } }
		set_temp_variable = { _aifct_armysz = 1 }
	}
	set_temp_variable = { _aifct_sur = 0 }
	if = {
		limit = { surrender_progress < constant:wa_ai_aifc.eligibility.max_surrender }
		set_temp_variable = { _aifct_sur = 1 }
	}
	set_temp_variable = { _aifct_rec = 0 }
	if = {
		limit = {
			num_of_controlled_states > constant:wa_ai_aifc.eligibility.recovery_min_states
			has_army_size = { size > constant:wa_ai_aifc.eligibility.recovery_min_div }
		}
		set_temp_variable = { _aifct_rec = 1 }
	}
	# The shipped aggregate, for localisation when it disagrees with the terms above.
	set_temp_variable = { _aifct_sel = 0 }
	if = { limit = { WA_AI_AIFC_should_select_sector = yes } set_temp_variable = { _aifct_sel = 1 } }

	log = "  elig  : is_ai=[?_aifct_ai|.0]  war=[?_aifct_war|.0]  capitulated=[?_aifct_cap|.0]  enemy-exists=[?_aifct_enemy|.0]  states=[?_aifct_states|.0](bar>[?_aifct_bar_states|.0])  div=[?_aifct_div|.0]  army>[?_aifct_bar_div|.0]=[?_aifct_armysz|.0]  surrender-ok=[?_aifct_sur|.0]  recovery-hatch=[?_aifct_rec|.0]"
	log = "  elig  : shipped WA_AI_AIFC_should_select_sector reads [?_aifct_sel|.0] (0 with all terms above green = trigger and terms disagree - localise there)"

	############################################################
	# B. Native candidate collection (helpers section 1 duplicated)
	############################################################

	clear_temp_array = _aifct_seen
	clear_temp_array = _aifct_eids
	clear_temp_array = _aifct_esums
	set_temp_variable = { _aifct_n = 0 }
	set_temp_variable = { _aifct_ok = 0 }
	set_temp_variable = { _aifct_max_pad = 0 }

	every_controlled_state = {
		every_neighbor_state = {
			limit = { CONTROLLER = { has_war_with = ROOT } }
			WA_TEST_AIFC_probe_candidate = yes
		}
	}
	set_temp_variable = { _aifct_n_native = _aifct_n }
	set_temp_variable = { _aifct_native_maxpad = _aifct_max_pad }

	############################################################
	# C. Expeditionary pass (helpers section 1b) - run UNCONDITIONALLY here, with the
	#    shipped gate (capability-based since [aifc-traction]: walk runs unless a native
	#    pad clears selection.exp_gate_pad) reported beside it, so the ally-soil
	#    presence is visible even when the shipped code would have skipped the walk.
	############################################################

	set_temp_variable = { _aifct_gate_bar = constant:wa_ai_aifc.selection.exp_gate_pad }
	set_temp_variable = { _aifct_exp_gate = 0 }
	if = {
		limit = { NOT = { check_variable = { _aifct_native_maxpad > constant:wa_ai_aifc.selection.exp_gate_pad } } }
		set_temp_variable = { _aifct_exp_gate = 1 }
	}
	set_temp_variable = { _aifct_pad_src = 0 }

	every_country = {
		limit = {
			NOT = { tag = ROOT }
			exists = yes
			OR = {
				is_in_faction_with = ROOT
				is_subject_of = ROOT
				ROOT = { is_subject_of = PREV }
			}
		}
		every_controlled_state = {
			limit = {
				# independent copy of the shipped merge bar (contract rule 4): an ally
				# state counts only as a band-2 pad on its own, 5+ ROOT divisions.
				ROOT = { divisions_in_state = { state = PREV size > 4 } }
			}
			add_to_temp_variable = { _aifct_pad_src = 1 }
			every_neighbor_state = {
				limit = { CONTROLLER = { has_war_with = ROOT } }
				WA_TEST_AIFC_probe_candidate = yes
			}
		}
	}

	############################################################
	# D. Stored sector + summary
	############################################################

	if = {
		limit = { check_variable = { WA_AI_AIFC_sector_states^num > 0 } }
		set_temp_variable = { _aifct_corr = WA_AI_AIFC_sector_states^num }
		log = "  stored: sector LIVE  anchor=[?WA_AI_AIFC_sector_anchor|.0]  corridor=[?_aifct_corr|.0]  age=[?WA_AI_AIFC_sector_age|.0]wk"
		if = {
			limit = { has_variable = WA_AI_AIFC_sector_enemy }
			var:WA_AI_AIFC_sector_enemy = {
				log = "  stored: sector enemy = [This.GetName]"
			}
		}
	}
	else = {
		log = "  stored: NO sector on [Root.GetName] (weekly tick last wrote nothing, or cleared)"
	}

	log = "  sum   : cand_native=[?_aifct_n_native|.0]  native-maxpad=[?_aifct_native_maxpad|.0]  cand_total=[?_aifct_n|.0]  exp-gate(native-maxpad<=[?_aifct_gate_bar|.0])=[?_aifct_exp_gate|.0]  ally-states-with-5+div=[?_aifct_pad_src|.0]  anchor-eligible(pad>floor)=[?_aifct_ok|.0]  max_pad=[?_aifct_max_pad|.0]"

	# Election readout ([aifc-traction]): one line per enemy with floor-eligible candidates,
	# then the elected main enemy (largest sum). Computed over ALL candidates (walk is
	# unconditional here) - when the shipped exp gate skips the ally walk, the shipped
	# election sees native-only books; that display divergence is the instrument's job.
	set_temp_variable = { _aifct_bestsum = 0 }
	set_temp_variable = { _aifct_elect = 0 }
	for_each_loop = { array = _aifct_eids index = _aifct_j value = _aifct_eid
		set_temp_variable = { _aifct_esum_p = _aifct_esums^_aifct_j }
		var:_aifct_eid = {
			log = "  enemy : [This.GetName]  eligible-pad-sum=[?_aifct_esum_p|.0]"
		}
		if = {
			limit = { check_variable = { _aifct_esum_p > _aifct_bestsum } }
			set_temp_variable = { _aifct_bestsum = _aifct_esum_p }
			set_temp_variable = { _aifct_elect = _aifct_eid }
		}
	}
	if = {
		limit = { check_variable = { _aifct_elect > 0 } }
		var:_aifct_elect = {
			log = "  sum   : main enemy (largest eligible pad SUM=[?_aifct_bestsum|.0]) = [This.GetName] - shipped selection only anchors on THIS enemy's candidates"
		}
	}
	else = {
		log = "  sum   : no enemy elected (no floor-eligible candidate) - shipped selection writes NO sector"
	}
	############################################################
	# E. Armour steering books ([aifc-revived-tag-residue])
	############################################################
	# Read-only state dump: the reconcile's live books and the pending-cancel debt books (cancels
	# owed to tags that were dead at retirement). target-exists=1 on a pending line is the DEFECT:
	# the weekly sweep must emit the owed cancel on the first tick where the tag exists again.

	if = {
		limit = { has_variable = WA_AI_AIFC_armor_boost }
		var:WA_AI_AIFC_armor_boost = {
			log = "  armor : boost target = [This.GetName]"
		}
	}
	else = {
		log = "  armor : boost target = none"
	}
	set_temp_variable = { _aifct_supn = 0 }
	set_temp_variable = { _aifct_pbn = 0 }
	set_temp_variable = { _aifct_psn = 0 }
	if = {
		limit = { check_variable = { WA_AI_AIFC_armor_suppressed^num > 0 } }
		set_temp_variable = { _aifct_supn = WA_AI_AIFC_armor_suppressed^num }
	}
	if = {
		limit = { check_variable = { WA_AI_AIFC_armor_pending_boost^num > 0 } }
		set_temp_variable = { _aifct_pbn = WA_AI_AIFC_armor_pending_boost^num }
	}
	if = {
		limit = { check_variable = { WA_AI_AIFC_armor_pending_supp^num > 0 } }
		set_temp_variable = { _aifct_psn = WA_AI_AIFC_armor_pending_supp^num }
	}
	log = "  armor : suppressed=[?_aifct_supn|.0]  pending-boost-cancels=[?_aifct_pbn|.0]  pending-suppress-cancels=[?_aifct_psn|.0]"
	for_each_loop = { array = WA_AI_AIFC_armor_pending_boost value = _aifct_pid
		set_temp_variable = { _aifct_pex = 0 }
		if = { limit = { var:_aifct_pid = { exists = yes } } set_temp_variable = { _aifct_pex = 1 } }
		log = "  armor : pending boost-cancel  id=[?_aifct_pid]  target-exists=[?_aifct_pex|.0]  (1 = DEFECT, sweep failed)"
	}
	for_each_loop = { array = WA_AI_AIFC_armor_pending_supp value = _aifct_pid
		set_temp_variable = { _aifct_pex = 0 }
		if = { limit = { var:_aifct_pid = { exists = yes } } set_temp_variable = { _aifct_pex = 1 } }
		log = "  armor : pending suppress-cancel  id=[?_aifct_pid]  target-exists=[?_aifct_pex|.0]  (1 = DEFECT, sweep failed)"
	}

	log = "  read  : elig 0 -> gate; cand_total 0 -> no land contact; anchor-eligible 0 with cand_total > 0 -> the feasibility floor (or all pads 0) - and for an expeditionary power check ally-states-with-5+div first"
	log = "=== AIFC SECTOR TEST END === [Root.GetName]"

	clear_temp_array = _aifct_seen
	clear_temp_array = _aifct_eids
	clear_temp_array = _aifct_esums
}
