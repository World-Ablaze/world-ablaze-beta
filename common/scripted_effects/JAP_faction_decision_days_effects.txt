# Keeps the length of Japan's faction decisions current. The decision view only shows the "N days:" heading over a
# decision's modifiers when the variable behind days_remove is set, and each decision used to set it only when taken.
# Same steps as the complete_effect of each decision in common/decisions/JAP_factions.txt. Called at game start
# (history/countries/JAP - Japan.txt), weekly (common/on_actions/JAP_faction_decision_days_on_actions.txt) and when
# the Subjects' attitude or influence moves (JAP_shinmin_effects.txt).
JAP_faction_decision_days_update = {
	if = {
		limit = { check_variable = { JAP_toseiha_influence > 4 } }
		set_variable = { TOSEIHA_INFLUENCE_LEVEL = 35 }
	}
	else_if = {
		limit = { check_variable = { JAP_toseiha_influence > 3 } }
		set_variable = { TOSEIHA_INFLUENCE_LEVEL = 28 }
	}
	else_if = {
		limit = { check_variable = { JAP_toseiha_influence > 2 } }
		set_variable = { TOSEIHA_INFLUENCE_LEVEL = 21 }
	}
	else_if = {
		limit = { check_variable = { JAP_toseiha_influence > 1 } }
		set_variable = { TOSEIHA_INFLUENCE_LEVEL = 14 }
	}
	else = {
		set_variable = { TOSEIHA_INFLUENCE_LEVEL = 7 }
	}
	if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 6 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 140 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 5 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 120 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 4 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 100 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 3 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 80 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 2 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 60 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 1 } }
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 40 }
	}
	else = {
		set_variable = { ZAIBATSUS_DECISION_CONSTRUCTION_DAYS = 20 }
	}
	if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 6 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 140 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 5 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 120 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 4 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 100 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 3 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 80 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 2 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 60 }
	}
	else_if = {
		limit = { check_variable = { JAP_zaibatsus_influence > 1 } }
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 40 }
	}
	else = {
		set_variable = { ZAIBATSUS_DECISION_OUTPUT_DAYS = 20 }
	}
	if = {
		limit = { check_variable = { JAP_shinmin_attitude > 1 } }
		set_variable = { SHINMIN_ATTITUDE_DAYS = 80 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_attitude > 0 } }
		set_variable = { SHINMIN_ATTITUDE_DAYS = 60 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_attitude > -1 } }
		set_variable = { SHINMIN_ATTITUDE_DAYS = 40 }
	}
	else = {
		set_variable = { SHINMIN_ATTITUDE_DAYS = 20 }
	}
	if = {
		limit = { check_variable = { JAP_shinmin_influence > 6 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 14 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 5 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 28 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 4 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 42 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 3 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 56 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 2 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 70 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 1 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 84 }
	}
	else_if = {
		limit = { check_variable = { JAP_shinmin_influence > 0 } }
		set_variable = { SHINMIN_INFLUENCE_DAYS = 98 }
	}
	else = {
		set_variable = { SHINMIN_INFLUENCE_DAYS = 112 }
	}
}
