# [light-medium-conversion] Trade ONE light-majority tank division for a fresh medium one.
# CREATE FIRST, destroy only once the division count proves the spawn landed: a name miss must
# cost nothing, and destroy_unit is the only division-scope removal the engine has - it refunds
# neither men nor equipment. Rate, cost table and the rejected replace_with route: WORK.md, slug.
# Country scope, AI only. Gate: WA_AI_TEMPLATES_should_convert_light_armor_division.
WA_AI_TEMPLATES_convert_one_light_armor_division = {
	set_temp_variable = { _wa_lmc_before = num_divisions }
	# The division carries NO `name =`: an unnamed create_unit takes the engine's automatic name
	# from the country's division-names group (common/units/names_divisions/<TAG>_*), the way an
	# AI-deployed division does. Naming it after the template printed "Medium Tank template C".
	meta_effect = {
		text = {
			capital_scope = {
				create_unit = {
					division = "division_template = \"[MEDIUM_TEMPLATE]\" start_experience_factor = 0.5 start_equipment_factor = 0.1 start_manpower_factor = 1"
					owner = PREV
				}
			}
		}
		MEDIUM_TEMPLATE = "[WA_AI_TEMPLATES_GetMediumTankTemplateName]"
	}
	set_temp_variable = { _wa_lmc_after = num_divisions }
	if = {
		limit = { check_variable = { _wa_lmc_after > _wa_lmc_before } }
		random_country_division = {
			limit = {
				division_has_majority_template = light_armor_battalion_line
			}
			destroy_unit = yes
		}
	}
}

# [war-start-suppression-template] The engine can leave its lettered cavalry templates competing
# with the named suppression template. Remove that A..Z ladder before creating or retaining the
# named template. Country scope: THIS/ROOT = the country entering the war.
WA_AI_TEMPLATES_purge_light_cavalry_templates = {
	if = {
		limit = { is_ai = yes }
		WA_AI_TEMPLATES_delete_light_cavalry_lettered_templates = yes
	}
}

# [war-cavalry-disband] Disband the fielded cavalry of an AI entering its first war, templates kept.
# By name: the starting OOB templates whose regiments are mostly cavalry/camelry (history/units;
# police and garrison ones included: only the fielded units go) and the engine's
# lettered copies, named after the majority unit's localisation. The Light Cavalry A..Z templates
# themselves are still removed by the suppression purge that runs next. Country scope: THIS = ROOT.
WA_AI_TEMPLATES_disband_cavalry_divisions_on_war = {
	if = {
		limit = { WA_AI_TEMPLATES_should_disband_cavalry_on_war = yes }
		set_country_flag = WA_AI_TEMPLATES_war_cavalry_disbanded
		WA_AI_TEMPLATES_delete_starting_cavalry_units = yes
		WA_AI_TEMPLATES_delete_lettered_cavalry_units = yes
	}
}

# One-way latch behind WA_AI_CONFIG_DIVISIONS_can_motorize_support: a coalition member that has
# once reached 20 military factories keeps motorised support for the rest of the campaign.
# Country scope, called monthly for EVERY country (players included - the reserve system and the
# division creator read the trigger too).
#
# Why a latch and not a live threshold: the trigger selects WHICH ai_template target is enabled,
# and every time the enabled target changes the engine gives the role to a fresh copy and
# decommissions the template that held it. A live `num_of_military_factories > 20` would do that on
# construction, bomb damage, state loss, faction change and exile - and Norway and Belgium sit at
# exactly 20 in 1940. Never add a clearing branch here without reading the decommission entry in
# wa-lessons-learned.
WA_AI_TEMPLATES_update_motorized_support_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_motorized_support_earned }
			OR = {
				is_in_faction_with = USA
				is_in_faction_with = ENG
			}
			num_of_military_factories > 20
		}
		set_country_flag = WA_AI_TEMPLATES_motorized_support_earned
	}
}

WA_AI_TEMPLATES_calculate_templates = {
	if = {
		limit = {
			WA_AI_CONFIG_DIVISIONS_use_default_templates = yes
		}
		# Block template creation if T1 doctrine is picked but required spirits are not yet selected
		if = {
			limit = {
				WA_has_one_t1_land_doctrine = yes
				NOT = { WA_AI_TEMPLATES_has_required_spirits = yes }
			}
			# Ensure correct spirits are assigned
			WA_AI_TEMPLATES_ensure_correct_spirits = yes
			# Still block templates this tick (spirits will be ready next check)
			WA_AI_TEMPLATES_block_all_templates = yes
			# [historical-tank-park] Preserve SOV's exact starting corps target during the
			# one-tick doctrine-spirit setup block; it must be trainable from game start.
			if = {
				limit = {
					WA_AI_CONFIG_pursues_historical_tank_park = yes
					WA_AI_PRODUCTION_should_continue_historical_tank_park = yes
				}
				WA_AI_TEMPLATES_calculate_light_support_armor_template = yes
			}
		}
		else = {
			WA_AI_TEMPLATES_calculate_all_templates = yes
		}
	}
}

# Template type codes:
# 1 = INFANTRY, 2 = MOTORIZED, 3 = MECHANIZED, 4 = LIGHT_ARMOR, 5 = MEDIUM_ARMOR, 6 = HEAVY_ARMOR
# 7 = free (was MODERN_ARMOR), 8 = free (was GARRISON), 9 = MARINES, 10 = AIRBORNE, 11 = MOUNTAINEER, 12 = RANGERS

WA_AI_TEMPLATES_update_target_template = {
	#Inputs :
	#_template_type_code : integer (1-12, see codes above)
	#_template_value : integer (must be set before calling this effect)
	#Note: HOI4 variables cannot store strings, so we use numeric codes and map them
	meta_effect = {
		text = {
			if = {
				limit = {
					has_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE }
				}
				clr_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE }
			}
			if = {
				limit = {
					NOT = { check_variable = { [TEMPLATE_VALUE] = 0 } }
				}
				set_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE value = [TEMPLATE_VALUE] }
			}
		}
		TEMPLATE_TYPE = "[WA_AI_TEMPLATE_TYPE]"
		TEMPLATE_VALUE = "[?_template_value]"
	}
}

WA_AI_TEMPLATES_ensure_correct_army_spirit = {
	# Democratic -> relief_of_command_spirit
	if = {
		limit = {
			has_government = democratic
			NOT = { has_idea = relief_of_command_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = relief_of_command_spirit
	}
	# Fascist/Neutral -> professional_officer_corps_spirit
	# Exception: Italy uses state_serves_the_military_spirit until adequate_army reforms are done
	else_if = {
		limit = {
			OR = {
				has_government = fascism
				has_government = neutrality
			}
			NOT = { has_idea = professional_officer_corps_spirit }
			# Exclude Italy without adequate_army (they can't pick professional_officer_corps yet)
			NOT = {
				AND = {
					tag = ITA
					NOT = { has_country_flag = adequate_army }
				}
			}
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = professional_officer_corps_spirit
	}
	# Italy fallback: state_serves_the_military_spirit until adequate_army reforms complete
	else_if = {
		limit = {
			tag = ITA
			has_government = fascism
			NOT = { has_country_flag = adequate_army }
			NOT = { has_idea = state_serves_the_military_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = state_serves_the_military_spirit
	}
	# Communist -> ideological_loyalty_spirit
	else_if = {
		limit = {
			has_government = communism
			NOT = { has_idea = ideological_loyalty_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = ideological_loyalty_spirit
	}
}

WA_AI_TEMPLATES_ensure_correct_academy_spirit = {

			limit = { has_country_flag = WA_AI_TEMPLATES_light_support_mixed }
			set_temp_variable = { _template_value = 15004 }
		}
		else_if = {
			limit = { WA_AI_TEMPLATES_use_mechanized_templates = yes }
			set_temp_variable = { _template_value = 15003 }
		}
	}
	WA_AI_TEMPLATES_update_target_template = yes
}
