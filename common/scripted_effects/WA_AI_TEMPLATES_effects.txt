# One-way latch behind WA_AI_CONFIG_DIVISIONS_can_motorize_support: a coalition member that has
# once reached 20 military factories keeps motorised support for the rest of the campaign.
# Country scope, called monthly for EVERY country (players included - the reserve system and the
# division creator read the trigger too).
#
# Why a latch and not a live threshold: the trigger selects WHICH ai_template target is enabled,
# and every time the enabled target changes the engine gives the role to a fresh copy and
# decommissions the template that held it. A live `num_of_military_factories > 20` would do that on
# construction, bomb damage, state loss, faction change and exile - and Norway and Belgium sit at
# exactly 20 in 1940. Never add a clearing branch here without reading the decommission entry in
# wa-lessons-learned.
WA_AI_TEMPLATES_update_motorized_support_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_motorized_support_earned }
			OR = {
				is_in_faction_with = USA
				is_in_faction_with = ENG
			}
			num_of_military_factories > 20
		}
		set_country_flag = WA_AI_TEMPLATES_motorized_support_earned
	}
}

# [templates-admission] one-way deadline latch: a country whose free-design focus never comes
# (unreachable on this path, or war-gated like RAJ's while at peace) still enters the template
# system past these dates, paying full XP for the design. One-way flag for the same reason as
# the motorized-support latch above: a gate that flickers re-runs the engine's template
# decommission pass. Read by WA_AI_TEMPLATES_has_infantry/tank_focus_completed.
WA_AI_TEMPLATES_update_design_deadline_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_design_deadline_passed }
			OR = {
				date > 1941.1.1
				AND = {
					has_war = yes
					date > 1939.6.1
				}
			}
		}
		set_country_flag = WA_AI_TEMPLATES_design_deadline_passed
	}
}

# [modern-chassis-tier] One-way latch: once the medium role fields the modern chassis it never
# goes back. Same reason as the two latches above - the flag decides WHICH ai_template is enabled,
# and a gate that flickers re-runs the engine's template decommission pass. Two of the three terms
# are NOT monotone: use_mechanized_templates falls back under 100 military factories, and
# use_medium_armor_templates falls under the early-expansion override.
# The mechanized term is deliberate: the modern-chassis mirror is mechanized-only, so a
# focus-medium country with no mechanized never switches. Owner rule 2026-08-28.
WA_AI_TEMPLATES_update_modern_chassis_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_modern_chassis_earned }
			WA_AI_TEMPLATES_use_medium_armor_templates = yes
			WA_AI_TEMPLATES_has_modern_armor_unlocked = yes
			WA_AI_TEMPLATES_use_mechanized_templates = yes
		}
		set_country_flag = WA_AI_TEMPLATES_modern_chassis_earned
	}
}

# [mot-field-hospital] One-way latch: 300 military factories buy the MOTORISED field hospital
# inside an otherwise horse-drawn line. motorised_infantry is required because that tech is what
# unlocks motorized_equipment (common/technologies/armor.txt) - without it the mirror target is
# unfillable. One-way for the same reason as the three latches above: the flag decides WHICH
# ai_template is enabled, and a gate that flickers re-runs the template decommission pass.
WA_AI_TEMPLATES_update_motorized_hospital_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_motorized_hospital_earned }
			num_of_military_factories > 300
			WA_AI_TEMPLATES_has_motorized_unlocked = yes
		}
		set_country_flag = WA_AI_TEMPLATES_motorized_hospital_earned
	}
}

# [mot-field-hospital] +100 turns a 100%-horse target into the same line with a motorised field
# hospital. Call it ONLY on a value that HAS a mirror block - 1003/1004 (infantry), 2000/2002
# (mountaineers), 10000/10002 (marines). On any other value it lands on a target that does not
# exist and the role is left with no enabled template at all.
WA_AI_TEMPLATES_apply_motorized_hospital_mirror = {
	if = {
		limit = { WA_AI_TEMPLATES_can_motorize_field_hospital = yes }
		add_to_temp_variable = { _template_value = 100 }
	}
}

# [mech-window] One-way latch: once the mechanization window has opened it never closes.
# Same reason as the four latches above - the flag decides WHICH ai_template is enabled, and a
# gate that flickers re-runs the engine's template decommission pass. The window's GER branch
# reads deployed manpower, which falls in war, so without the latch a bloodied GER would flip
# its whole armour force back to motorised and then forward again.
# Window definition and its cost: WA_AI_CONFIG_DIVISIONS_mechanization_window_open.
WA_AI_TEMPLATES_update_mechanization_latch = {
	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_TEMPLATES_mechanization_earned }
			WA_AI_CONFIG_DIVISIONS_mechanization_window_open = yes
		}
		set_country_flag = WA_AI_TEMPLATES_mechanization_earned
	}
}

WA_AI_TEMPLATES_calculate_templates = {
	if = {
		limit = {
			WA_AI_CONFIG_DIVISIONS_use_default_templates = yes
		}
		# Block template creation if T1 doctrine is picked but required spirits are not yet selected
		if = {
			limit = {
				WA_has_one_t1_land_doctrine = yes
				NOT = { WA_AI_TEMPLATES_has_required_spirits = yes }
			}
			# Ensure correct spirits are assigned
			WA_AI_TEMPLATES_ensure_correct_spirits = yes
			# Still block templates this tick (spirits will be ready next check)
			WA_AI_TEMPLATES_block_all_templates = yes
		}
		else = {
			WA_AI_TEMPLATES_calculate_all_templates = yes
		}
	}
}

# Template type codes:
# 1 = INFANTRY, 2 = MOTORIZED, 3 = MECHANIZED, 4 = LIGHT_ARMOR, 5 = MEDIUM_ARMOR, 6 = HEAVY_ARMOR
# 7 = free (was MODERN_ARMOR), 8 = free (was GARRISON), 9 = MARINES, 10 = AIRBORNE, 11 = MOUNTAINEER, 12 = RANGERS

WA_AI_TEMPLATES_update_target_template = {
	#Inputs :
	#_template_type_code : integer (1-12, see codes above)
	#_template_value : integer (must be set before calling this effect)
	#Note: HOI4 variables cannot store strings, so we use numeric codes and map them
	meta_effect = {
		text = {
			if = {
				limit = {
					has_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE }
				}
				clr_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE }
			}
			if = {
				limit = {
					NOT = { check_variable = { [TEMPLATE_VALUE] = 0 } }
				}
				set_country_flag = { flag = WA_[TEMPLATE_TYPE]_TEMPLATE value = [TEMPLATE_VALUE] }
			}
		}
		TEMPLATE_TYPE = "[WA_AI_TEMPLATE_TYPE]"
		TEMPLATE_VALUE = "[?_template_value]"
	}
}

WA_AI_TEMPLATES_ensure_correct_spirits = {
	# Replace incorrect army_spirit if needed (MUST be ideology-specific)
	WA_AI_TEMPLATES_ensure_correct_army_spirit = yes
	# Add academy_spirit ONLY if empty (any existing is valid)
	WA_AI_TEMPLATES_ensure_correct_academy_spirit = yes
}

WA_AI_TEMPLATES_ensure_correct_army_spirit = {
	# Democratic -> relief_of_command_spirit
	if = {
		limit = {
			has_government = democratic
			NOT = { has_idea = relief_of_command_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = relief_of_command_spirit
	}
	# Fascist/Neutral -> professional_officer_corps_spirit
	# Exception: Italy uses state_serves_the_military_spirit until adequate_army reforms are done
	else_if = {
		limit = {
			OR = {
				has_government = fascism
				has_government = neutrality
			}
			NOT = { has_idea = professional_officer_corps_spirit }
			# Exclude Italy without adequate_army (they can't pick professional_officer_corps yet)
			NOT = {
				AND = {
					tag = ITA
					NOT = { has_country_flag = adequate_army }
				}
			}
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = professional_officer_corps_spirit
	}
	# Italy fallback: state_serves_the_military_spirit until adequate_army reforms complete
	else_if = {
		limit = {
			tag = ITA
			has_government = fascism
			NOT = { has_country_flag = adequate_army }
			NOT = { has_idea = state_serves_the_military_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = state_serves_the_military_spirit
	}
	# Communist -> ideological_loyalty_spirit
	else_if = {
		limit = {
			has_government = communism
			NOT = { has_idea = ideological_loyalty_spirit }
		}
		WA_AI_TEMPLATES_remove_wrong_army_spirits = yes
		add_ideas = ideological_loyalty_spirit
	}
}

WA_AI_TEMPLATES_ensure_correct_academy_spirit = {
	# ONLY add if no academy spirit exists (any existing is valid)
	if = {
		limit = {
			NOT = { WA_AI_TEMPLATES_has_any_academy_spirit = yes }
		}
		# Democratic -> best_of_the_best_spirit
		if = {
			limit = { has_government = democratic }
			add_ideas = best_of_the_best_spirit
		}
		# Non-Democratic -> political_loyalty_spirit
		else = {
			add_ideas = political_loyalty_spirit
		}
	}
}

WA_AI_TEMPLATES_calculate_all_templates = {
	if = {
		limit = {
			# Ensure certain focuses have been completed for free template design
			WA_AI_TEMPLATES_has_infantry_focus_completed = yes
		}
		WA_AI_TEMPLATES_calculate_infantry_template = yes
		WA_AI_TEMPLATES_calculate_marines_template = yes
		WA_AI_TEMPLATES_calculate_suppression_template = yes
	}
	if = {
		limit = {
			# Ensure certain focuses have been completed for free template design
			WA_AI_TEMPLATES_has_tank_focus_completed = yes
		}
		WA_AI_TEMPLATES_calculate_motorized_template = yes
		WA_AI_TEMPLATES_calculate_mountaineer_template = yes
		WA_AI_TEMPLATES_calculate_mechanized_template = yes
		WA_AI_TEMPLATES_calculate_light_support_armor_template = yes
		WA_AI_TEMPLATES_calculate_light_armor_template = yes
		WA_AI_TEMPLATES_calculate_medium_armor_template = yes
		WA_AI_TEMPLATES_calculate_heavy_armor_template = yes
	}
}

WA_AI_TEMPLATES_calculate_infantry_template = {
	set_temp_variable = { _template_type_code = 1 }
	set_temp_variable = { _template_value = 0 }
	set_temp_variable = { _template_claimed = 0 }
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			NOT = { WA_AI_TEMPLATES_use_heavy_infantry = yes }
		}
		set_temp_variable = { _template_value = 1000 }
		set_temp_variable = { _template_claimed = 1 }
	}
	# [templates-motorised-20w] the motorised twin of 1001, the shape 1005/1006 already
	# have at 30 width. Without it a rich army in marsh or mountain terrain fields the
	# horse-drawn 20-width line and its motorised regimental companies go unused.
	# 1001 takes no hospital mirror: no 1101 exists.
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			WA_AI_CONFIG_DIVISIONS_is_expected_marshes_or_mountains = yes
			WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
		}
		set_temp_variable = { _template_value = 1002 }
		set_temp_variable = { _template_claimed = 1 }
	}
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			WA_AI_CONFIG_DIVISIONS_is_expected_marshes_or_mountains = yes
		}
		set_temp_variable = { _template_value = 1001 }
		set_temp_variable = { _template_claimed = 1 }
	}
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			WA_AI_CONFIG_DIVISIONS_use_line_support = yes
			WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
		}
		set_temp_variable = { _template_value = 1006 }
		set_temp_variable = { _template_claimed = 1 }
	}
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			WA_AI_CONFIG_DIVISIONS_use_line_support = yes
		}
		set_temp_variable = { _template_value = 1004 }
		WA_AI_TEMPLATES_apply_motorized_hospital_mirror = yes
		set_temp_variable = { _template_claimed = 1 }
	}
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
			WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
		}
		set_temp_variable = { _template_value = 1005 }
		set_temp_variable = { _template_claimed = 1 }
	}
	if = {
		limit = {
			check_variable = { _template_claimed = 0 }
		}
		set_temp_variable = { _template_value = 1003 }
		WA_AI_TEMPLATES_apply_motorized_hospital_mirror = yes
		set_temp_variable = { _template_claimed = 1 }
	}
	WA_AI_TEMPLATES_update_target_template = yes
}
