WA_AI_recruit_general = {

	### ratio of divisions:generals

	set_temp_variable = { WA_AI_general_count = 0 }
	set_temp_variable = { WA_AI_divisions_to_generals_ratio = num_divisions }

	every_army_leader = {
		limit = { NOT = { is_field_marshal = yes } }

		add_to_temp_variable = { WA_AI_general_count = 1 } # count the number of generals
	}

	divide_temp_variable = { WA_AI_divisions_to_generals_ratio = WA_AI_general_count }

	# [recruit-loop] registration watchdog: a creation this counter never sees would re-fire the
	# recruit forever (the count gate cannot close), burning the CP income and flooding the save
	# with orphan characters. Oracle = the count the gate itself reads (character EXISTENCE is the
	# wrong oracle - orphans exist in the DB); a net leader loss inside one 2-day verify window can
	# fake a failure, which is why one strike refunds and only two consecutive strikes latch.
	# The pending flag guarantees WA_AI_expected_generals was set in the same pass.
	if = {
		limit = { has_country_flag = WA_AI_recruit_general_pending }
		if = {
			limit = { check_variable = { WA_AI_general_count < WA_AI_expected_generals } }
			add_to_variable = { WA_AI_recruit_general_fail_n = 1 }
			add_command_power = WA_AI_last_general_cost
			if = {
				limit = { check_variable = { WA_AI_recruit_general_fail_n > 1 } }
				set_country_flag = WA_AI_recruit_general_broken
				set_variable = { WA_AI_broken_at_generals = WA_AI_general_count }
				if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: general creation does not register - latched off" }
			}
		}
		else = {
			set_variable = { WA_AI_recruit_general_fail_n = 0 }
		}
		clr_country_flag = WA_AI_recruit_general_pending
	}
	# [recruit-loop] latch exit: the roster growing ABOVE its count at latch time (focus/event
	# generals) proves registration can move again - give the script another chance; a still-broken
	# registration re-latches after two refunded strikes. Growth-since-latch, not a threshold: a
	# threshold exit would re-arm every pulse for a ratio-branch looper whose count never moves.
	if = {
		limit = {
			has_country_flag = WA_AI_recruit_general_broken
			check_variable = { WA_AI_general_count > WA_AI_broken_at_generals }
		}
		clr_country_flag = WA_AI_recruit_general_broken
		set_variable = { WA_AI_recruit_general_fail_n = 0 }
	}

	### cost of recruiting a new general

	set_temp_variable = { WA_AI_recruit_cost = 0 }

	every_army_leader = {
		add_to_temp_variable = { WA_AI_recruit_cost = 1 }
	}

	# recruitment is paid in command power, mirroring NDefines.NPolitics.ARMY_LEADER_COST
	# (2 x existing leaders). This previously charged political power at 5 x roster size,
	# which silently consumed a major's entire wartime PP income (ENG: 225+ PP per general).
	# The ratio gate below already prevents recruit spam, so the cost is clamped for majors.
	multiply_temp_variable = { WA_AI_recruit_cost = 2 }
	clamp_temp_variable = { var = WA_AI_recruit_cost min = 10 max = 50 }

	### recruit a general

	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_recruit_general_broken }
			NOT = { has_country_flag = WA_AI_recruit_general_pending }
			OR = {
				check_variable = { WA_AI_divisions_to_generals_ratio > 18 }
				check_variable = { WA_AI_general_count < 3 } # get at least 3
			}
			NOT = { check_variable = { command_power < WA_AI_recruit_cost } }
		}

		# remembered so a failed registration refunds exactly what was debited
		set_variable = { WA_AI_last_general_cost = WA_AI_recruit_cost }
		multiply_temp_variable = { WA_AI_recruit_cost = -1 }
		add_command_power = WA_AI_recruit_cost

		# [recruit-loop] generate_character = documented "create + recruit" (random name from the
		# country's own lists when omitted); the bare create_corps_commander it replaces left every
		# creation as an unregistered orphan invisible to every_army_leader. Token must be unique
		# per creation, hence the per-country sequence in the meta.
		add_to_variable = { WA_AI_gen_seq = 1 }
		meta_effect = {
			text = {
				generate_character = {
					token_base = WA_AI_gen_[TAG]_[SEQ]
					corps_commander = {
						skill = 1
					}
				}
			}
			TAG = "[ROOT.GetTag]"
			SEQ = "[?WA_AI_gen_seq|.0]"
		}

		set_variable = { WA_AI_expected_generals = WA_AI_general_count }
		add_to_variable = { WA_AI_expected_generals = 1 }
		set_country_flag = WA_AI_recruit_general_pending

		if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: general" }
	}
}

WA_AI_recruit_marshal = {
	### ratio of generals:marshals

	set_temp_variable = { WA_AI_generals_needed = num_divisions }
	divide_temp_variable = { WA_AI_generals_needed = 18 }  # divisions per general
	set_temp_variable = { WA_AI_marshal_count = 0 }

	every_army_leader = {
		limit = { is_field_marshal = yes }

		add_to_temp_variable = { WA_AI_marshal_count = 1 }
	}

	divide_temp_variable = { WA_AI_generals_needed = WA_AI_marshal_count }

	# [recruit-loop] same registration watchdog as WA_AI_recruit_general - see the comments there
	if = {
		limit = { has_country_flag = WA_AI_recruit_marshal_pending }
		if = {
			limit = { check_variable = { WA_AI_marshal_count < WA_AI_expected_marshals } }
			add_to_variable = { WA_AI_recruit_marshal_fail_n = 1 }
			add_command_power = WA_AI_last_marshal_cost
			if = {
				limit = { check_variable = { WA_AI_recruit_marshal_fail_n > 1 } }
				set_country_flag = WA_AI_recruit_marshal_broken
				set_variable = { WA_AI_broken_at_marshals = WA_AI_marshal_count }
				if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: marshal creation does not register - latched off" }
			}
		}
		else = {
			set_variable = { WA_AI_recruit_marshal_fail_n = 0 }
		}
		clr_country_flag = WA_AI_recruit_marshal_pending
	}
	# [recruit-loop] latch exit (growth-since-latch) - see WA_AI_recruit_general
	if = {
		limit = {
			has_country_flag = WA_AI_recruit_marshal_broken
			check_variable = { WA_AI_marshal_count > WA_AI_broken_at_marshals }
		}
		clr_country_flag = WA_AI_recruit_marshal_broken
		set_variable = { WA_AI_recruit_marshal_fail_n = 0 }
	}

	### Check if a new marshal is needed

	if = {
		limit = {
			OR = {
				check_variable = { WA_AI_marshal_count < 2 } # get at least 2 marshals
				check_variable = { WA_AI_generals_needed > 3.5 } # there are currently over this many generals ideally per marshal
			}
		}

		### Check if it possible to recruit a new field marshal, else check if a general should be promoted

		set_temp_variable = { WA_AI_recruit_cost = 0 }

		every_army_leader = {
			add_to_temp_variable = { WA_AI_recruit_cost = 1 }
		}

		# see WA_AI_recruit_general - recruitment is paid in command power, not political power
		multiply_temp_variable = { WA_AI_recruit_cost = 2 }
		clamp_temp_variable = { var = WA_AI_recruit_cost min = 10 max = 50 }

		# total = recruitment cost + 30 command power promotion cost
		set_temp_variable = { WA_AI_marshal_total_cost = WA_AI_recruit_cost }
		add_to_temp_variable = { WA_AI_marshal_total_cost = 30 }

		if = { # recruit a new marshal
			limit = {
				NOT = { has_country_flag = WA_AI_recruit_marshal_broken }
				NOT = { has_country_flag = WA_AI_recruit_marshal_pending }
				NOT = { check_variable = { command_power < WA_AI_marshal_total_cost } }
			}

			# remembered so a failed registration refunds exactly what was debited
			set_variable = { WA_AI_last_marshal_cost = WA_AI_marshal_total_cost }
			multiply_temp_variable = { WA_AI_marshal_total_cost = -1 }
			add_command_power = WA_AI_marshal_total_cost

			# [recruit-loop] generate_character - see WA_AI_recruit_general
			add_to_variable = { WA_AI_gen_seq = 1 }
			meta_effect = {
				text = {
					generate_character = {
						token_base = WA_AI_gen_[TAG]_[SEQ]
						field_marshal = {
							skill = 1
						}
					}
				}
				TAG = "[ROOT.GetTag]"
				SEQ = "[?WA_AI_gen_seq|.0]"
			}

			set_variable = { WA_AI_expected_marshals = WA_AI_marshal_count }
			add_to_variable = { WA_AI_expected_marshals = 1 }
			set_country_flag = WA_AI_recruit_marshal_pending

			if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: marshal" }
		}

		else = { # cannot afford a new marshal - promote an existing general instead
			# (this block was previously nested as the else of the logging if above, so every
			# scripted marshal recruit with logging off ALSO promoted a general to marshal,
			# creating a second marshal and burning an extra 15-30 command power)

				set_temp_variable = { WA_AI_general_count = 0 }
				set_temp_variable = { WA_AI_divs_per_general = num_divisions }
				every_army_leader = { limit = { NOT = { is_field_marshal = yes } }

					add_to_temp_variable = { WA_AI_general_count = 1 } # count the number of generals
				}

				divide_temp_variable = { WA_AI_divs_per_general = WA_AI_general_count }

				if = {
					limit = {
						check_variable = { WA_AI_general_count > 5 }
						check_variable = { WA_AI_divs_per_general < 18 }
						NOT = { command_power < 30 } # promotion cost
					}

					# Find the lowest level general to promote
					set_temp_variable = { level = 1 }
					while_loop_effect = { limit = { check_variable = { level < 9 } }

						add_to_temp_variable = { level = 1 }

						if = {
							limit = {
								meta_trigger = {
									text = { any_army_leader = { NOT = { is_field_marshal = yes } skill < [x] } }
									x = "[?level|.0]"
								}
							}

							meta_effect = {
								text = {
									random_army_leader = { limit = { NOT = { is_field_marshal = yes } skill < [x] }
										promote_leader = THIS

										if = { limit = { OR = { has_trait = politically_connected has_trait = war_hero } } ROOT = { add_command_power = -15 } }
										else_if = { limit = { has_trait = career_officer } ROOT = { add_command_power = -22 } }
										else = { ROOT = { add_command_power = -30 } }
									}
								}
								x = "[?level|.0]"
							}

							if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | promoted a general" }

							set_temp_variable = { break = 1 }
						}
					}
				}
			}
		}
}

WA_AI_recruit_navy = {

	### ratio of ships:admirals

	set_temp_variable = { WA_AI_navy_leader_count = 0 }
	set_temp_variable = { WA_AI_ships_to_navy_leaders_ratio = num_ships }

	every_navy_leader = {
		add_to_temp_variable = { WA_AI_navy_leader_count = 1 } # count the number of admirals
	}

	divide_temp_variable = { WA_AI_ships_to_navy_leaders_ratio = WA_AI_navy_leader_count }

	# [recruit-loop] same registration watchdog as WA_AI_recruit_general - see the comments there
	if = {
		limit = { has_country_flag = WA_AI_recruit_navy_pending }
		if = {
			limit = { check_variable = { WA_AI_navy_leader_count < WA_AI_expected_navy_leaders } }
			add_to_variable = { WA_AI_recruit_navy_fail_n = 1 }
			add_command_power = WA_AI_last_navy_cost
			if = {
				limit = { check_variable = { WA_AI_recruit_navy_fail_n > 1 } }
				set_country_flag = WA_AI_recruit_navy_broken
				set_variable = { WA_AI_broken_at_navy_leaders = WA_AI_navy_leader_count }
				if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: navy leader creation does not register - latched off" }
			}
		}
		else = {
			set_variable = { WA_AI_recruit_navy_fail_n = 0 }
		}
		clr_country_flag = WA_AI_recruit_navy_pending
	}
	# [recruit-loop] latch exit (growth-since-latch) - see WA_AI_recruit_general
	if = {
		limit = {
			has_country_flag = WA_AI_recruit_navy_broken
			check_variable = { WA_AI_navy_leader_count > WA_AI_broken_at_navy_leaders }
		}
		clr_country_flag = WA_AI_recruit_navy_broken
		set_variable = { WA_AI_recruit_navy_fail_n = 0 }
	}

	### cost of recruiting a new admiral

	set_temp_variable = { WA_AI_recruit_cost = 0 }

	every_navy_leader = {
		add_to_temp_variable = { WA_AI_recruit_cost = 1 }
	}

	# recruitment is paid in command power, mirroring NDefines.NPolitics.NAVY_LEADER_COST
	# (2 x existing navy leaders). Same clamp as army recruitment.
	multiply_temp_variable = { WA_AI_recruit_cost = 2 }
	clamp_temp_variable = { var = WA_AI_recruit_cost min = 10 max = 50 }

	### recruit an admiral

	if = {
		limit = {
			NOT = { has_country_flag = WA_AI_recruit_navy_broken }
			NOT = { has_country_flag = WA_AI_recruit_navy_pending }
			OR = {
				AND = {
					check_variable = { WA_AI_ships_to_navy_leaders_ratio > 10 }
					check_variable = { num_ships > 10 } # enough ships to justify a second admiral
				}
				AND = {
					check_variable = { WA_AI_navy_leader_count < 1 } # get at least 1
					check_variable = { num_ships > 4 } # but only if there is a navy to lead
				}
			}
			NOT = { check_variable = { command_power < WA_AI_recruit_cost } }
		}

		# remembered so a failed registration refunds exactly what was debited
		set_variable = { WA_AI_last_navy_cost = WA_AI_recruit_cost }
		multiply_temp_variable = { WA_AI_recruit_cost = -1 }
		add_command_power = WA_AI_recruit_cost

		# [recruit-loop] generate_character = documented "create + recruit" (random name from the
		# country's own lists when omitted); navy_leader is a valid character role block.
		# Token must be unique per creation, hence the per-country sequence in the meta.
		add_to_variable = { WA_AI_gen_seq = 1 }
		meta_effect = {
			text = {
				generate_character = {
					token_base = WA_AI_gen_[TAG]_[SEQ]
					navy_leader = {
						skill = 1
					}
				}
			}
			TAG = "[ROOT.GetTag]"
			SEQ = "[?WA_AI_gen_seq|.0]"
		}

		set_variable = { WA_AI_expected_navy_leaders = WA_AI_navy_leader_count }
		add_to_variable = { WA_AI_expected_navy_leaders = 1 }
		set_country_flag = WA_AI_recruit_navy_pending

		if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RECRUIT: navy leader" }
	}
}
