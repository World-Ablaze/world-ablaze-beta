### functions for AI construction

# Fix 88: breadth-first placement in the shared-slot queues.
# The engine merges same-(state, building) adds into ONE construction line (amount = N); a line
# takes at most MAX_CIV_FACTORIES_PER_LINE (WA: 20) civilian factories and builds its levels
# serially. Every WA_AI_queue_<X> used to break on the first eligible state of the score-sorted
# array, so a burst of N calls became one slow line instead of N parallel ones (reporter: "US has
# ~20 parallel lines, I queued 20 alu mills and it occupies 2 or 3").
# Each queue function now makes ONE walk over WA_AI_shared_slot_scores and classifies every
# eligible state (WA_AI_bf_consider) into four tiers, best first:
#   A  rank < K, under the per-line depth cap, not yet picked this pulse   -> the normal spread
#   B  rank < K, under depth, already picked this pulse                    -> every top-K state has
#      one; wrap around (picked set is cleared, round-robin restarts)
#   C  rank < K, over depth, not yet picked                                -> the depth cap is a
#      PREFERENCE, not a wall: a burst bigger than K x depth keeps spreading
#   D  first eligible state in score order, whatever its depth/picked state -> exactly what the
#      pre-Fix-88 loop picked, so no country can ever build LESS than before (picked set cleared)
# The first tier-A hit ends the walk (the array is score-sorted, so it is the best). K =
# ceil((civs - civs already assigned to priority construction) / @WA_AI_BF_CIVS_PER_LINE), clamped
# to [1, @WA_AI_BF_MAX_K] = the lines the country can actually feed. Score order still decides
# WHICH K states are candidates; breadth only decides how a burst spreads among them. All eight
# shared-slot types (CIC/MIC/NIC + the five refineries) carry the Fix 81 sum WA_AI_committed_<X>
# and a depth trigger WA_AI_queue_depth_ok_<X> (WA_AI_CONSTRUCTION_triggers.txt). Not applied to
# the non-shared queues (RADAR/AIR/AA/SILO/INF): different arrays, small caps - out of scope.
# WA_AI_bf_k, WA_AI_bf_picked_<X> and every _wa_bf_* are TEMP (execution-scoped = one event fire =
# one burst). The selection loop breaks on its own _wa_bf_break_; the plain `break = 1` is still set
# on every add so the downstream contract (queue_INF fallback, MIC-replacement legs, R48 save/
# restore) is unchanged whatever the engine does with `break` on loop entry.
# Cadence: C.10 is gated on idle civs > 1 AND re-arms for 75-1000 days after a firing, so absorbing
# more idle civs per firing does not lower the firing rate. Timeline, 300-civ major (K = 15), one
# C.10 firing with 20 CIC calls: t0 old = 1 line x 20 civs (20 serial builds), new = 15 lines x 20
# civs then wrap; t1 (next day) old: idle civs > 1 but the timer blocks C.10 anyway; t2 = first
# completion: old 1 CIC, new 15. Same civs, same firings, ~K x the throughput. A 25-civ minor has
# K = 2 and behaves almost as before. Bulk fire_only_once bursts are the largest change: WA_AI_C.72
# (USA 1940.7, 354 MIC calls) goes from ~8 lines to up to 20 - the build order's intent (its author
# is the reporter); priority construction keeps its floored share (Fix 40 stable base, 30% of civs)
# and K is computed net of the civs PC already holds: USA 1940.7 ~350 civs, ~35% on PC -> K =
# ceil(227 / 20) = 12 lines x 20 = 240 civs for the standard queue, never the whole pool. The old
# "~8 lines" for C.72 is an estimate (354 / cap 45), not a measured baseline - R53 measures both.
# Loop-entry `break` semantics: the R48 comment says a stale `break = 1` aborts a following queue
# call; the cut-3 save BHU_1941_06_14_15 shows C.63's five consecutive WA_AI_queue_AR calls landing
# 4+1 levels, i.e. the loop DID run five times. v2 does not depend on either reading: the selection
# walk breaks on its own _wa_bf_break_ and never reads the plain `break` on entry.
# num_of_civilian_factories (TOTAL) minus WA_AI_PC_assigned_factories_total on purpose - the idle
# reading *_available_for_projects sits at ~0 once the vanilla queue saturates (see lessons-log).
# Telemetry: WA_TLM_sq_* (standing, family "sq" = standard queue) - see WA_TLM_core.txt / doc 6d.
# MUST MATCH NDefines.NProduction.MAX_CIV_FACTORIES_PER_LINE in common/defines/05_defines.lua (WA = 20,
# vanilla 15) - it is the divisor that turns "civs" into "lines the country can feed".
@WA_AI_BF_CIVS_PER_LINE = 20
# Hard ceiling on breadth, independent of civ count. Rank K+1.. of the score order is never used by
# tiers A-C even for a 600-civ USA; a burst larger than K wraps around inside the top K. It bounds
# two costs of breadth: how far down the score order (infra, supply, exposure) a burst may reach,
# and how many civs one burst may take from the engine's own queue at once.
@WA_AI_BF_MAX_K = 20

# ROOT = THIS = country. Computes WA_AI_bf_k once per pulse (same "compute if 0" idiom as
# WA_AI_has_shared_slot_scores).
WA_AI_queue_breadth_prepare = {
	if = { limit = { check_variable = { WA_AI_bf_k = 0 } }
		set_temp_variable = { WA_AI_bf_k = num_of_civilian_factories }
		subtract_from_temp_variable = { WA_AI_bf_k = WA_AI_PC_assigned_factories_total }
		divide_temp_variable = { WA_AI_bf_k = @WA_AI_BF_CIVS_PER_LINE }
		clamp_temp_variable = { var = WA_AI_bf_k min = 1 max = @WA_AI_BF_MAX_K }
	}
}

# Resets the per-selection temporaries. Country scope, called right before the walk.
WA_AI_bf_select_begin = {
	set_temp_variable = { _wa_bf_break_ = 0 }
	set_temp_variable = { _wa_bf_rank_ = 0 }
	set_temp_variable = { _wa_bf_a_ = 0 }
	set_temp_variable = { _wa_bf_b_ = 0 }
	set_temp_variable = { _wa_bf_c_ = 0 }
	set_temp_variable = { _wa_bf_d_ = 0 }
	# found flags: THIS.id is a scope-encoded reference (can be negative - Fix 51 lesson), so the id
	# temporaries above are only ever used with var: / is_in_array, never compared numerically.
	set_temp_variable = { _wa_bf_a_found_ = 0 }
	set_temp_variable = { _wa_bf_b_found_ = 0 }
	set_temp_variable = { _wa_bf_c_found_ = 0 }
	set_temp_variable = { _wa_bf_d_found_ = 0 }
}

# THIS = an ELIGIBLE state (the caller has already applied the type's availability limit).
# Inputs (temp): _wa_bf_depth_ok_ (0/1), _wa_bf_is_picked_ (0/1), _wa_bf_rank_ (eligible states seen
# so far), WA_AI_bf_k. Records the first state of each tier (A/B/C/D above) by id; a tier-A hit
# also raises _wa_bf_break_ so the walk stops. Increments the rank.
WA_AI_bf_consider = {
	if = { limit = { check_variable = { _wa_bf_rank_ < WA_AI_bf_k } }
		if = { limit = { check_variable = { _wa_bf_depth_ok_ = 1 } }
			if = { limit = { check_variable = { _wa_bf_is_picked_ = 0 } check_variable = { _wa_bf_a_found_ = 0 } }
				set_temp_variable = { _wa_bf_a_ = THIS.id }
				set_temp_variable = { _wa_bf_a_found_ = 1 }
				set_temp_variable = { _wa_bf_break_ = 1 }
			}
			if = { limit = { check_variable = { _wa_bf_b_found_ = 0 } } set_temp_variable = { _wa_bf_b_ = THIS.id } set_temp_variable = { _wa_bf_b_found_ = 1 } }
		}
		if = { limit = { check_variable = { _wa_bf_is_picked_ = 0 } check_variable = { _wa_bf_c_found_ = 0 } } set_temp_variable = { _wa_bf_c_ = THIS.id } set_temp_variable = { _wa_bf_c_found_ = 1 } }
	}
	if = { limit = { check_variable = { _wa_bf_d_found_ = 0 } } set_temp_variable = { _wa_bf_d_ = THIS.id } set_temp_variable = { _wa_bf_d_found_ = 1 } }
	add_to_temp_variable = { _wa_bf_rank_ = 1 }
}

# Country scope, after the walk. Picks the best tier into _wa_bf_target_ (scope-encoded state id, only
# ever used with var:) and _wa_bf_target_found_ (0 = nothing eligible), sets _wa_bf_clear_ = 1 when the picked set must restart (tiers B and D), and
# _wa_bf_tier_ (1-4) for telemetry.
WA_AI_bf_choose = {
	set_temp_variable = { _wa_bf_target_ = 0 }
	set_temp_variable = { _wa_bf_target_found_ = 0 }
	set_temp_variable = { _wa_bf_clear_ = 0 }
	set_temp_variable = { _wa_bf_tier_ = 0 }
	if = { limit = { check_variable = { _wa_bf_a_found_ = 1 } }
		set_temp_variable = { _wa_bf_target_ = _wa_bf_a_ }
		set_temp_variable = { _wa_bf_target_found_ = 1 }
		set_temp_variable = { _wa_bf_tier_ = 1 }
	}
	else_if = { limit = { check_variable = { _wa_bf_b_found_ = 1 } }
		set_temp_variable = { _wa_bf_target_ = _wa_bf_b_ }
		set_temp_variable = { _wa_bf_target_found_ = 1 }
		set_temp_variable = { _wa_bf_clear_ = 1 }
		set_temp_variable = { _wa_bf_tier_ = 2 }
	}
	else_if = { limit = { check_variable = { _wa_bf_c_found_ = 1 } }
		set_temp_variable = { _wa_bf_target_ = _wa_bf_c_ }
		set_temp_variable = { _wa_bf_target_found_ = 1 }
		set_temp_variable = { _wa_bf_tier_ = 3 }
	}
	else_if = { limit = { check_variable = { _wa_bf_d_found_ = 1 } }
		set_temp_variable = { _wa_bf_target_ = _wa_bf_d_ }
		set_temp_variable = { _wa_bf_target_found_ = 1 }
		set_temp_variable = { _wa_bf_clear_ = 1 }
		set_temp_variable = { _wa_bf_tier_ = 4 }
	}
}

# Country scope, after a successful selection add. Telemetry only.
# tlm: WA_TLM_sq_tier_a_n / _b_n / _c_n / _d_n (standing, family sq) - which tier placed the add;
# d_n > 0 means breadth was exhausted (every top-K candidate over depth AND already picked, or
# nothing eligible inside the top K). Consumers: checklist R53, R49.
WA_AI_bf_record = {
	if = { limit = { check_variable = { _wa_bf_tier_ = 1 } } add_to_variable = { WA_TLM_sq_tier_a_n = 1 } }
	else_if = { limit = { check_variable = { _wa_bf_tier_ = 2 } } add_to_variable = { WA_TLM_sq_tier_b_n = 1 } }
	else_if = { limit = { check_variable = { _wa_bf_tier_ = 3 } } add_to_variable = { WA_TLM_sq_tier_c_n = 1 } }
	else_if = { limit = { check_variable = { _wa_bf_tier_ = 4 } } add_to_variable = { WA_TLM_sq_tier_d_n = 1 } }
	if = { limit = { check_variable = { WA_AI_bf_k > WA_TLM_sq_k_max } } set_variable = { WA_TLM_sq_k_max = WA_AI_bf_k } }
}

# civilian factories
WA_AI_queue_CIC = {

	if = { limit = { check_variable = { WA_AI_has_shared_slot_scores = 0 } }

		WA_AI_get_shared_slot_scores = yes
	}

	# Fix 88: breadth-first placement - one selection walk, four tiers (see WA_AI_bf_consider).
	WA_AI_queue_breadth_prepare = yes
	WA_AI_bf_select_begin = yes
	for_each_scope_loop = { array = WA_AI_shared_slot_scores break = _wa_bf_break_
		if = {
			limit = {
				WA_AI_available_SHARED = yes
				infrastructure > 3
				WA_AI_available_CIC = yes # Fix 88: state cap incl. queued (Fix 81 sum)
			}
			set_temp_variable = { _wa_bf_depth_ok_ = 0 }
			if = { limit = { WA_AI_queue_depth_ok_CIC = yes } set_temp_variable = { _wa_bf_depth_ok_ = 1 } }
			set_temp_variable = { _wa_bf_is_picked_ = 0 }
			if = { limit = { is_in_array = { array = WA_AI_bf_picked_CIC value = THIS.id } } set_temp_variable = { _wa_bf_is_picked_ = 1 } }
			WA_AI_bf_consider = yes
		}
	}
	WA_AI_bf_choose = yes
	if = { limit = { check_variable = { _wa_bf_clear_ = 1 } } clear_temp_array = WA_AI_bf_picked_CIC }
	if = { limit = { check_variable = { _wa_bf_target_found_ = 1 } }
		var:_wa_bf_target_ = { WA_AI_add_CIC = yes }
		add_to_temp_array = { WA_AI_bf_picked_CIC = _wa_bf_target_ }
		WA_AI_bf_record = yes
		set_temp_variable = { break = 1 } # downstream contract (queue_INF fallback, MIC-replacement legs, R48)
	}

	if = { limit = { check_variable = { break = 0 } }

		WA_AI_queue_INF = yes
	}
}

# Fix 39: substitution cascade for WA_AI_queue_MIC under industry overextension.
# THIS = ROOT = country (same scope contract as WA_AI_queue_MIC).
# Refinery leg gates: synth tech researched; input self-sufficiency on coal (same
# form as WA_AI_build_refinery_resource_shortage_rubber - refineries eat 60 coal/
# level, only build on domestic coal); economy law at war economy or above
# (WA economy ladder in common/ideas/_economic.txt: undisturbed_isolation <
# isolation < civilian_economy < low_economic_mobilisation <
# partial_economic_mobilisation < war_economy < tot_economic_mobilisation <
# over_mobilisation). Dedup: the refinery leg is skipped while a type-8 (synthetic
# refinery) project already sits in WA_AI_PC_queue, so the cascade cannot pile on
# top of WA_AI_build_refinery_resource_shortage_rubber's own type-8 queueing; the
# per-state same-type check inside WA_AI_PC_can_build_project still applies below.
# If the refinery leg does not start a project (queue unchanged - no eligible
# state), falls through to the CIC leg: a substituted call never builds nothing
# beyond what WA_AI_queue_CIC itself guarantees (it has its own INF fallback).
# Debug counters (persistent country variables, savegame-visible):
#   WA_AI_overext_dbg_mic_redirects_refinery, WA_AI_overext_dbg_mic_redirects_cic
WA_AI_overext_substitute_MIC = {

	set_temp_variable = { _overext_queued_ = 0 }

	if = {
		limit = {
			has_tech = synth_oil_experiments

			# input self-sufficiency (same test as WA_AI_build_refinery_resource_shortage_rubber)
			check_variable = { resource@coal > 50 }
			check_variable = { resource_imported@coal < 20 }

			# economy law at or above war economy
			OR = {
				has_idea = war_economy
				has_idea = tot_economic_mobilisation
				has_idea = over_mobilisation
			}

			# dedup: no synthetic refinery (type 8) already in the PC queue
			NOT = {
				any_of = { array = WA_AI_PC_queue value = v_overext_
					check_variable = { WA_AI_PC_building_type^v_overext_ = 8 }
				}
			}
		}

		set_temp_variable = { _overext_q_before_ = WA_AI_PC_queue^num }
		WA_AI_priority_queue_synthetic_refinery = yes

		if = { limit = { check_variable = { WA_AI_PC_queue^num > _overext_q_before_ } }

			set_temp_variable = { _overext_queued_ = 1 }
			add_to_variable = { WA_AI_overext_dbg_mic_redirects_refinery = 1 }
			if = { limit = { has_country_flag = WA_AI_construction_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: overextended - MIC redirected to synthetic refinery" }
		}
	}

	if = { limit = { check_variable = { _overext_queued_ = 0 } }

		WA_AI_queue_CIC = yes
		add_to_variable = { WA_AI_overext_dbg_mic_redirects_cic = 1 }
		if = { limit = { has_country_flag = WA_AI_construction_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: overextended - MIC redirected to CIC" }
	}
}

# Fix 40 (campaign 31eaf7e6) - three changes shared by all five refinery queue
# functions below (types 8 synthetic / 11 hydro steel / 12 hydro aluminium /
# 15 steel / 16 aluminium; mapping per WA_AI_PC_get_build_speed and
# WA_AI_PC_get_building_cost in WA_AI_CONSTRUCTION_PRIORITY_core.txt):
#
# 1. FILTERED DRAW. The old form was a BARE random_scope_in_array over
#    controlled_states with the eligibility test in an inner `if`: the engine drew
#    ONE state per pulse and then usually discarded it, so the hit rate was
#    (eligible states / controlled states) - under 9% for a 1943 ENG holding tens
#    of controlled states with a handful of buildable ones. The tests now live in
#    `limit`, the engine's own filtered form (same shape as the radar strategy in
#    WA_AI_CONSTRUCTION_PRIORITY_strategies.txt), so the draw can only land on an
#    eligible state.
#
# 2. EXPLICIT _project_queue_max = 1. These functions set _project_queue_num but
#    never _project_queue_max, so WA_AI_PC_start_project inherited whatever an
#    earlier caller left in that shared temp variable (3 from the two air
#    strategies, 5 from the railway core, 0 = unlimited from the railway helpers).
#    One refinery project of a given type in flight per country is the intended
#    budget, and it is now also what bounds these strategies, which since Fix 40
#    run outside the dispatcher's affordability gate. _project_target_state and
#    _project_target_level are zeroed in the same spirit: a stale non-zero
#    target_state would redirect the project to an entirely unrelated state.
#
# 3. TWO-PASS SITE SELECTION. Pass 1 keeps the historical home-area criterion
#    (WA_AI_refinery_region_priority - a thin wrapper over WA_AI_region_priority,
#    so SOV/CHI keep their central-Russia / western-China rule). If pass 1 queued
#    nothing - detected by queue growth, the same idiom Fix 39 uses in
#    WA_AI_overext_substitute_MIC above - pass 2 retries over the widened set
#    (WA_AI_refinery_region_priority_wide: ROOT-owned and not on a hostile
#    border). A saturated home area was the second reason ENG never built a
#    refinery: every home-area state already held one or had no free shared slot.
#
# Contract preserved for Fix 39: a successful queue is still visible as growth of
# WA_AI_PC_queue within the same call, so WA_AI_overext_substitute_MIC's
# before/after delta around WA_AI_priority_queue_synthetic_refinery still detects
# success (both passes end in WA_AI_PC_start_project, which is what grows it).
#
# THIS = ROOT = country for all five (called from
# WA_AI_build_refinery_resource_shortage_* and, for the synthetic one, from
# WA_AI_overext_substitute_MIC).

WA_AI_priority_queue_synthetic_refinery = {

	if = { limit = { check_variable = { ROOT.WA_AI_PC_active_nonrail_projects < 5 } }

		set_temp_variable = { _project_building_type = 8 }
		set_temp_variable = { _project_province_id = 0 }
		# Fix 41: strategic band (250) - a shortage-driven refinery outranks default-100
		# projects but never a war railway; band table in WA_AI_CONSTRUCTION_PRIORITY_core.txt
		set_temp_variable = { _project_priority = 250 }
		set_temp_variable = { _project_queue_num = 1 }
		# Fix 40: explicit bounds / leak hygiene - see the header note above
		set_temp_variable = { _project_queue_max = 1 }
		set_temp_variable = { _project_target_state = 0 }
		set_temp_variable = { _project_target_level = 0 }

		set_temp_variable = { _refinery_q_before_ = WA_AI_PC_queue^num }

		# Fix 40 pass 1 - preferred sites (filtered draw)
		random_scope_in_array = { array = controlled_states
			limit = { WA_AI_PC_can_build_project = yes WA_AI_refinery_region_priority = yes }

			WA_AI_PC_start_project = yes
			set_temp_variable = { break = 1 }
		}

		# Fix 40 pass 2 - widened sites, only when pass 1 queued nothing
		if = { limit = { NOT = { check_variable = { WA_AI_PC_queue^num > _refinery_q_before_ } } }

			random_scope_in_array = { array = controlled_states
				limit = { WA_AI_PC_can_build_project = yes WA_AI_refinery_region_priority_wide = yes }

				WA_AI_PC_start_project = yes
				set_temp_variable = { break = 1 }
			}
		}
	}
}

		set_temp_variable = { _project_building_type = 11 }
		set_temp_variable = { _project_province_id = 0 }
		# Fix 41: strategic band (250) - see the synthetic-refinery note above
		set_temp_variable = { _project_priority = 250 }
		set_temp_variable = { _project_queue_num = 1 }
		# Fix 40: explicit bounds / leak hygiene - see the header note above
		set_temp_variable = { _project_queue_max = 1 }
		set_temp_variable = { _project_target_state = 0 }
		set_temp_variable = { _project_target_level = 0 }

		set_temp_variable = { _refinery_q_before_ = WA_AI_PC_queue^num }

		# Fix 40 pass 1 - preferred sites (filtered draw)
		random_scope_in_array = { array = controlled_states
			limit = { WA_AI_PC_can_build_project = yes WA_AI_refinery_region_priority = yes }

			WA_AI_PC_start_project = yes
			set_temp_variable = { break = 1 }
		}

		# Fix 40 pass 2 - widened sites, only when pass 1 queued nothing
		if = { limit = { NOT = { check_variable = { WA_AI_PC_queue^num > _refinery_q_before_ } } }

			random_scope_in_array = { array = controlled_states
				limit = { WA_AI_PC_can_build_project = yes WA_AI_refinery_region_priority_wide = yes }

				WA_AI_PC_start_project = yes
				set_temp_variable = { break = 1 }
			}
		}
	}
}

# for building INF to increase supply
WA_AI_queue_INF_supply = {
	if = {
		limit = {
			any_of_scopes = { array = controlled_states
				WA_AI_available_INF = yes
				free_building_slots = { building = infrastructure size > 5 include_locked = no }
				ROOT = { divisions_in_state = { state = PREV size > 5 } }
				any_neighbor_state = { CONTROLLER = { has_war_with = ROOT strength_ratio = { tag = ROOT ratio > 0.5 } } }
			}
		}

		random_scope_in_array = { array = controlled_states
			limit = {
				WA_AI_available_INF = yes
				free_building_slots = { building = infrastructure size > 5 include_locked = no }
				ROOT = { divisions_in_state = { state = PREV size > 5 } }
				any_neighbor_state = { CONTROLLER = { has_war_with = ROOT strength_ratio = { tag = ROOT ratio > 0.5 } } }
			}
			WA_AI_add_INF = yes
		}
	}
	else_if = {
		limit = {
			any_of_scopes = { array = controlled_states
				WA_AI_available_INF = yes
				free_building_slots = { building = infrastructure size > 5 include_locked = no }
				ROOT = { divisions_in_state = { state = PREV size > 5 } }
				any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
			}
		}
		random_scope_in_array = { array = controlled_states
			limit = {
				WA_AI_available_INF = yes
				free_building_slots = { building = infrastructure size > 5 include_locked = no }
				ROOT = { divisions_in_state = { state = PREV size > 5 } }
				any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
			}
			WA_AI_add_INF = yes
		}
	}
}

# build forts on victory points in a state
WA_AI_queue_FORT_CITIES = {

	random_scope_in_array = { array = controlled_states
		limit = {
			is_in_home_area = yes
			WA_AI_has_city = yes
			check_variable = { WA_AI_forts_constructed_on_cities_@ROOT < 3 }
		}

		WA_AI_add_FORT_CITY = yes
	}
}
