			limit = {
				check_variable = { _use_stockpile = 1 }
			}

		}

			}

			}

				limit = {

						text = {

							}
						}

					}
				}
				if = {
					limit = {
						check_variable = { _use_stockpile = 1 }
					}
					WA_AI_DIVISION_remove_stockpiles = yes
				}
			}
			else = {
				if = {
					limit = { is_debug = yes }
					log = "WA_AI_DIVISION_spawn_divisions create_unit meta effect failed. Country: [ROOT.GetName]. 	}

		limit = {
			check_variable = { success = 0 }
		}

			limit = { is_debug = yes }
			log = "WA_AI_DIVISION_spawn_divisions bad input. Country: [ROOT.GetName]. # Spawns the required amount of divisions for the country to reach the total number provided in the country's capital state with configurable options via input parameters for AI only
# All required input parameters described below must be set via set_temp_variable before using this effect
# Scope: Country
# [INPUT] [REQUIRED] _num_divisions_total - The number of total divisions the country needs. The number of divisions spawned will be this number minus the existing number of divisions. Must be greater than 1. If less than the country's current division count, does nothing.
# [INPUT] [OPTIONAL] _state - Integer. The state in which to spawn the divisions. Defaults to the country's capital.
# [INPUT] [OPTIONAL] _use_stockpile - Integer Boolean. Whether to use the country's stockpiled equipment to supply the divisions. Must be 0 or 1 (false or true). Defaults to 1.
# [INPUT] [OPTIONAL] _division_template - Integer. The template of divisions to spawn. 1 = Infantry, 2 = Motorized, 3 = Mechanized, 4 = Armored, 5 = Militia, 6 = Paratrooper, 7 = 20Width Infantry, 8 = Marines, 9 = Amtrac + Tanks, 10 = snlf, 11 = Armoured snlf. Defaults to 1
# [INPUT] [OPTIONAL] _start_experience - Float. The starting experience of the spawned divisions. Must be between 0.1 and 1. Defaults to 0.2 (Trained 1)
# [INPUT] [OPTIONAL] _start_equipment - Float. The starting equipment of the spawned divisions AND will be taken from the stockpile of the country. Must be between 0.1 and 1. Defaults to 0.1
# [INPUT] [OPTIONAL] _start_manpower - Float. The starting manpower of the spawned divisions. Must be between 0.1 and 1. Defaults to 0.1
WA_AI_DIVISION_spawn_divisions_up_to = {
	# Validate required input parameters
	set_temp_variable = { success = 1 }
	if = {
		limit = {
			is_ai = no
			is_debug = no
		}
		set_temp_variable = { success = 0 }
	}
	else_if = {
		limit = {
			check_variable = { _num_divisions_total < 1 }
		}
		set_temp_variable = { success = 0 }
	}
	else_if = {
		limit = {
			NOT = { check_variable = { _num_divisions_total > num_divisions } } # Input must be greater than the country's current division count
		}
		set_temp_variable = { success = 2 } # Skip, but not an error
	}

	# Execute the effect to spawn divisions if parameters are valid
	if = {
		limit = {
			check_variable = { success = 1 }
		}
		set_temp_variable = { _num_divisions_spawn = _num_divisions_total }
		subtract_from_temp_variable = { _num_divisions_spawn = num_divisions }
		clamp_temp_variable = {
			var = _num_divisions_spawn
			min = 1
			max = 100
		}
		WA_AI_DIVISION_spawn_divisions = yes
	}
	else_if = {
		limit = {
			check_variable = { success = 0 }
		}
		if = {
			limit = { is_debug = yes }
			log = "WA_AI_DIVISION_spawn_divisions_up_to bad input. Country: [ROOT.GetName]. Parameters: _num_divisions_total = [?_num_divisions_total]. If all parameters valid, potentially not AI"
		}
	}
}

# Scope: Country
WA_AI_DIVISION_delete_spawned_divisions = {
	delete_unit = {
		id = 999
	}
}

					check_variable = { h_art_eq > 0 }

				divide_temp_variable = { num_divs_h_art_eq = h_art_eq }
				add_to_temp_array = { eq_requirements = num_divs_h_art_eq }
			}
			if = {
				limit = {
					check_variable = { h_aa_eq > 0 }
				}
				set_temp_variable = { num_divs_h_aa_eq = num_equipment@heavy_anti_air_equipment }
				divide_temp_variable = { num_divs_h_aa_eq = h_aa_eq }
				add_to_temp_array = { eq_requirements = num_divs_h_aa_eq }
			}
			if = {
				limit = {
					check_variable = { h_at_eq > 0 }
				}
				set_temp_variable = { num_divs_h_at_eq = num_equipment@heavy_anti_tank_equipment }
				divide_temp_variable = { num_divs_h_at_eq = h_at_eq }
				add_to_temp_array = { eq_requirements = num_divs_h_at_eq }
			}

			# Find the lowest number of divisions we can make with equipment limitations
			find_lowest_in_array = {
				array = eq_requirements
				value = _num_division_spawn_calculated_eq
			}

			# The below round needs to be floor(var), so we calculate it manually since modulo function doesn't work with decimals
			set_temp_variable = { _num_division_spawn_calculated_eq_rounded = _num_division_spawn_calculated_eq }
			round_temp_variable = _num_division_spawn_calculated_eq_rounded
			if = {
				limit = {
					check_variable = { _num_division_spawn_calculated_eq_rounded > _num_division_spawn_calculated_eq }
				}
				subtract_from_temp_variable = { _num_division_spawn_calculated_eq_rounded = 1 }
			}
			set_temp_variable = { _num_division_spawn_calculated_eq = _num_division_spawn_calculated_eq_rounded}
		}

		######################################################### Handle Manpower #########################################################
		set_temp_variable = { tot_mp = 0 }

		if = {
			limit = {
				check_variable = { _division_template = 1 }
			}
			set_temp_variable = { tot_mp = 15900 }
			if = {

				limit = {
					WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}

				check_variable = { _division_template = 3 }

						WA_AI_CONFIG_DIVISIONS_use_tank_destroyers = yes
						WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes

		else_if = {
			limit = {
				check_variable = { _division_template = 4 }
			}
			set_temp_variable = { tot_mp = 11200 }
			if = {

					OR = {
						WA_AI_CONFIG_DIVISIONS_use_tank_destroyers = yes
						WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
					}

				add_to_temp_variable = { tot_mp = 300 }
			}
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 5 }
			}
			set_temp_variable = { tot_mp = 10500 }
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 6 }
			}
			set_temp_variable = { tot_mp = 15400 }
		}
		else_if = {

				check_variable = { _division_template = 7 }

			set_temp_variable = { tot_mp = 11900 }
			if = {

			if = {
				limit = {
					WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 8 }
			}
			set_temp_variable = { tot_mp = 15900 }
			if = {
				limit = {
					WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
			if = {
				limit = {
					WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 9 }
			}
			set_temp_variable = { tot_mp = 16200 }
			if = {
				limit = {
					OR = {
						WA_AI_CONFIG_DIVISIONS_use_tank_destroyers = yes
						WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
					}
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 10 }
			}
			set_temp_variable = { tot_mp = 17100 }
			if = {
				limit = {
					WA_AI_CONFIG_DIVISIONS_can_motorize_support = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
			if = {
				limit = {
					WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
		}
		else_if = {
			limit = {
				check_variable = { _division_template = 11 }
			}

				limit = {
					OR = {
						WA_AI_CONFIG_DIVISIONS_use_tank_destroyers = yes
						WA_AI_CONFIG_DIVISIONS_use_heavy_at = yes
					}
				}
				add_to_temp_variable = { tot_mp = 300 }
			}
		}

		# Calculate total manpower actually spawned based on % equipped
		multiply_temp_variable = { tot_mp = _start_manpower }

		# Only calculate with respect to available manpower if force use is disabled
		if = {
			limit = {
				check_variable = { _force_use_equipment_manpower = 0 }
			}
			# Divide the country's available manpower by the calculated manpower requirement per division
			set_temp_variable = { _num_division_spawn_calculated_mp = manpower_k }
			divide_temp_variable = { _num_division_spawn_calculated_mp = tot_mp }
			multiply_temp_variable = { _num_division_spawn_calculated_mp = 1000 } # Since the country manpower var is 1/1000
			# The below round needs to be floor(var), so we calculate it manually since modulo function doesn't work with decimals
			set_temp_variable = { _num_division_spawn_calculated_mp_rounded = _num_division_spawn_calculated_mp }
			round_temp_variable = _num_division_spawn_calculated_mp_rounded
			if = {
				limit = {
					check_variable = { _num_division_spawn_calculated_mp_rounded > _num_division_spawn_calculated_mp }
				}
				subtract_from_temp_variable = { _num_division_spawn_calculated_mp_rounded = 1 }
			}
			set_temp_variable = { _num_division_spawn_calculated_mp = _num_division_spawn_calculated_mp_rounded}

			######################################################### Final Check #########################################################
			# Do the final check to determine which limiting factor, equipment or manpower, will determine how many possible divisions can be spawned
			if = {
				limit = {
					check_variable = { _num_division_spawn_calculated_eq < _num_division_spawn_calculated_mp }
				}
				set_temp_variable = { _num_division_spawn_calculated = _num_division_spawn_calculated_eq }
			}
			else = {
				set_temp_variable = { _num_division_spawn_calculated = _num_division_spawn_calculated_mp }
			}
		}
		if = {
			limit = { is_debug = yes }
			if = {
				limit = { check_variable = { _force_use_equipment_manpower = 0 }}
				log = "WA_AI_DIVISION_calculate_equipment_usage Country: [ROOT.GetName]. Divisions spawnable based on equipment: [?_num_division_spawn_calculated_eq]. Divisions spawnable based on manpower: [?_num_division_spawn_calculated_mp]. Template: [WA_AI_DIVISION_GetSpawnDivisionTemplate]"
			}
			else = {
				log = "WA_AI_DIVISION_calculate_equipment_usage Country: [ROOT.GetName]. Force use equipment and manpower: [?_force_use_equipment_manpower]. Template: [WA_AI_DIVISION_GetSpawnDivisionTemplate]"
			}
		}
	}
	else = {
		if = {
			limit = { is_debug = yes }
			log = "WA_AI_DIVISION_calculate_equipment_usage bad input. Country: [ROOT.GetName]. Parameters: _division_template = [?_division_template], _start_equipment = [?_start_equipment], _start_manpower = [?_start_manpower], _force_use_equipment_manpower = [?_force_use_equipment_manpower]. If all parameters valid, potentially not AI"
		}
	}
}

# Calls the necessary effects to create units on specified enemy states.
# [INPUT] [REQUIRED] _invasion_state - Integer. The State ID of a coastal state where the invasion divisions are spawned.
# [INPUT] [REQUIRED] _invasion_state_provinces - Array of integer Province IDs. The Province IDs in _invasion_state where the divisions will be spawned.
# [INPUT] [REQUIRED] _target_country - Country ID. The country the invasion is targeting.
# [INPUT] [OPTIONAL] _divisions_per_province - Integer. The number of divisions to spawn in each state. Must be between 1 and 10 inclusive. Defaults to 1.
# [INPUT] [OPTIONAL] _division_template - Integer. The template of divisions to spawn. 1 = Infantry, 2 = Motorized, 3 = Mechanized, 4 = Armored, 5 = Militia, 6 = Paratrooper, 7 = 20Width Infantry, 8 = Marines, 9 = Amtrac + Tanks, 10 = snlf, 11 = Armoured snlf. Defaults to 1
# [INPUT] [OPTIONAL] _start_experience - Float. The starting experience of the spawned divisions. Must be between 0.1 and 1. Defaults to 0.45 (Regular 1).
# [INPUT] [OPTIONAL] _start_equipment - Float. The starting equipment of the spawned divisions AND will be taken from the stockpile of the country. Must be between 0.1 and 1. Defaults to 1.
# [INPUT] [OPTIONAL] _start_manpower - Float. The starting manpower of the spawned divisions. Must be between 0.1 and 1. Defaults to 1.
WA_AI_DIVISION_spawn_invasion = {
	# Validate required input parameters
	set_temp_variable = { success = 1 }
	if = {
		limit = {
			is_ai = no
			is_debug = no
		}
		set_temp_variable = { success = 0 }
	}
	else_if = {
		limit = {
			NOT = {
				check_variable = { _invasion_state > 0 }
				check_variable = { _invasion_state < 1500 } # We don't have this many states
			}
		}
		set_temp_variable = { success = 0 }
	}

	# Optional input parameters get a default value set if not provided
	if = {
		limit = {
			OR = {
				check_variable = { _divisions_per_province < 1 }
				check_variable = { _divisions_per_province > 10 }
			}
		}
		set_temp_variable = { _divisions_per_province = 1 }
	}
	if = {
		limit = {
			OR = {
				check_variable = { _division_template < 1 }
				check_variable = { _division_template > 11 }
			}
		}
		set_temp_variable = { _division_template = 1 }
	}
	if = {
		limit = {
			OR = {
				check_variable = { _start_experience > 1 }
				check_variable = { _start_experience < 0.1 }
			}
		}
		set_temp_variable = { _start_experience = 0.45 } # Regular I
	}
	if = {
		limit = {
			OR = {
				check_variable = { _start_equipment > 1 }
				check_variable = { _start_equipment < 0.1 }
			}
		}
		set_temp_variable = { _start_equipment = 1 }
	}
	if = {
		limit = {
			OR = {
				check_variable = { _start_manpower > 1 }
				check_variable = { _start_manpower < 0.1 }
			}
		}
		set_temp_variable = { _start_manpower = 1 }
	}

	if = {
		limit = {
			check_variable = { success = 1 }
		}
		# Setup for the divisions to be created
		WA_AI_DIVISION_setup = yes
		set_temp_variable = { _force_use_equipment_manpower = 1 }
		set_temp_variable = { _num_divisions_spawn = _divisions_per_province }

		# Manually create the divisions in each of the provided states
		for_each_loop = {
			array = _invasion_state_provinces
			value = _province_id

			WA_AI_DIVISION_calculate_equipment_usage = yes
			WA_AI_DIVISION_remove_stockpiles = yes

			meta_effect = {
				text = {
					[STATE_ID] = {
						CONTROLLER = {
							ROOT = {
								set_province_controller = [PROVINCE_ID]
								[STATE_ID] = {
									create_unit = {
										division = "division_template=\"[TEMPLATE_NAME]\" start_experience_factor=[START_XP] start_equipment_factor=[START_EQ] start_manpower_factor=[START_MAN]"
										owner = ROOT
										count = [DIV_COUNT]
										prioritize_location = [PROVINCE_ID]
										allow_spawning_on_enemy_provs = yes
									}
								}
							}
							set_province_controller = [PROVINCE_ID]
						}
					}
				}
				STATE_ID = "[?_invasion_state]"
				PROVINCE_ID = "[?_province_id]"
				TEMPLATE_NAME = "[WA_AI_DIVISION_GetSpawnDivisionTemplate]"
				START_XP = "[?_start_experience]"
				START_EQ = "[?_start_equipment]"
				START_MAN = "[?_start_manpower]"
				DIV_COUNT = "[?_divisions_per_province]"
				# debug = yes
			}
		}
		# Manually apply an attack penalty to emulate the in-battle naval invasion stat penalty
		meta_effect = {
			text = {
				add_relation_modifier = {
					target = [COUNTRY_TAG]
					modifier = WA_AI_invasions_modifier
				}
			}
			COUNTRY_TAG = "[?_target_country.GetTag]"
			# debug = yes
		}
		set_variable = { most_recent_invasion_target = _target_country }
		country_event = { id = WA_AI_invasions.1 days = 14 }
	}
	else_if = {
		limit = {
			check_variable = { success = 0 }
		}
		if = {
			limit = { is_debug = yes }
			log = "WA_AI_DIVISION_spawn_invasion bad input. Country: [ROOT.GetName]. Parameters: _invasion_state = [?_invasion_state], _invasion_state_provinces first element = [?_invasion_state_provinces^0], _divisions_per_province = [?_divisions_per_province], _num_divisions_spawn = [?_num_divisions_spawn], _division_template = [?_division_template], experience = [?_start_experience|.2], equipment = [?_start_equipment|.2], manpower = [?_start_manpower|.2]. If all parameters valid, potentially not AI"
		}
	}
}

#
# [INPUT] [REQUIRED] _divisions_per_province - Integer. The base number of divisions to spawn per invasion province on normal difficulty. Must be between 1 and 10.
WA_AI_DIVISION_adjust_invasion_for_difficulty = {
	# Validate required input parameters
	set_temp_variable = { success = 1 }
	if = {
		limit = {
			is_ai = no
			is_debug = no
		}
		set_temp_variable = { success = 0 }
	}
	else_if = {
		limit = {
			OR = {
				check_variable = { _divisions_per_province < 1 }
				check_variable = { _divisions_per_province > 10 }
			}
		}
		set_temp_variable = { success = 0 }
	}

	if = {
		limit = {
			check_variable = { success = 1 }
		}
		if = {
			limit = { WA_AI_DIFFICULTY_is_historical_easy = yes }
			multiply_temp_variable = { _divisions_per_province = 0.5 }
			round_temp_variable = _divisions_per_province
		}
		else_if = {
			limit = {
				OR = {
					WA_AI_DIFFICULTY_is_historical_hard = yes
					WA_AI_DIFFICULTY_is_competitive_hard = yes
					log = "yes"
				}
			}
			multiply_temp_variable = { _divisions_per_province = 2 }
		}
	}
	else = {
		if = {
			limit = { is_debug = yes }
			log = "WA_AI_DIVISION_adjust_invasion_for_difficulty bad input. Country: [ROOT.GetName]. Parameters: _divisions_per_province = [?_divisions_per_province]. If all parameters valid, potentially not AI"
		}
	}
}
