### Constants: this file declares none - it reads constant:wa_ai_railway.* (queue size, supply model,
### scoring costs; common/script_constants/wa_ai_railway.txt) and the PC-wide
### constant:wa_ai_pc.type_id.rail / prio.rail_war / prio.rail_prewar (wa_ai_pc.txt).
# [pc-queue] admission cap on queued type-13 segments, country-wide =
# constant:wa_ai_railway.routes.queue_full, the SAME key railway_core uses as its
# recalculation-skip threshold - they must stay one key. Without the cap one recalculation queues
# every segment of up to 8 routes at once, which parks the queue above the threshold for months and
# stops route recalculation AND stale-path validation exactly when the front is moving. Capping at
# the threshold lets the queue drain below it within an interval, so every interval recomputes,
# validates and tops back up. Throughput is unchanged - the winner-takes-most allocator never funds
# more than ~pool/20 segments anyway, and segments not admitted this pulse are re-derived from the
# CURRENT front next pulse instead of rotting in the queue.

### Supply-flow model, home-port search, building costs (from 00_buildings.txt, registered
### mirrors) and cost estimation: constant:wa_ai_railway.supply.* / .cost.* - see wa_ai_railway.txt.

#	1=europe, 2=north_america, 3=south_america, 4=asia, 5=africa, 6=middle_east, 7=australia

WA_AI_PC_railway_get_continent = {
	set_temp_variable = { continent_id_ = 0 }

	if = { limit = { is_on_continent = europe } set_temp_variable = { continent_id_ = 1 } }
	else_if = { limit = { is_on_continent = north_america } set_temp_variable = { continent_id_ = 2 } }
	else_if = { limit = { is_on_continent = south_america } set_temp_variable = { continent_id_ = 3 } }
	else_if = { limit = { is_on_continent = asia } set_temp_variable = { continent_id_ = 4 } }
	else_if = { limit = { is_on_continent = africa } set_temp_variable = { continent_id_ = 5 } }
	else_if = { limit = { is_on_continent = middle_east } set_temp_variable = { continent_id_ = 6 } }
	else_if = { limit = { is_on_continent = australia } set_temp_variable = { continent_id_ = 7 } }

}

############################################################################################################
#	NEW HELPER: Find Best Home Port (with cost-based scoring)
#	Uses scoring: maximize supply first, minimize cost as tiebreaker
#	capital_state_id >>> best_home_port_state_, best_home_port_level_, best_home_port_province_, best_home_port_score_
############################################################################################################

WA_AI_PC_railway_find_best_home_port = {
	set_temp_variable = { best_home_port_state_ = 0 }
	set_temp_variable = { best_home_port_level_ = 0 }
	set_temp_variable = { best_home_port_province_ = 0 }
	set_temp_variable = { best_home_port_score_ = -999999999 }

	# Use configurable search distance
	set_temp_variable = { _origin_state_id = capital_state_id }
	set_temp_variable = { _max_distance = constant:wa_ai_railway.supply.home_port_search_distance }
	WA_AI_PC_railway_get_states_within_distance = yes

	# Also check capital state itself (distance 0)
	add_to_temp_array = { states_within_distance_ = capital_state_id }

	# Build distance lookup: states are added in BFS order, so index approximates distance
	# Capital is always last (we just added it), so its effective distance is 0
	set_temp_variable = { _capital_idx = states_within_distance_^num }
	subtract_from_temp_variable = { _capital_idx = 1 }

	# Evaluate each controlled coastal state within range
	every_controlled_state = {
		WA_AI_PC_get_state_id = yes
		set_temp_variable = { _this_state_id = state_id_ }

		if = {
			limit = {
				is_in_array = { states_within_distance_ = _this_state_id }
				is_coastal = yes
			}
			# Get naval base province and level
			WA_AI_PC_railway_get_naval_base_province = yes

			if = {
				limit = { check_variable = { naval_base_province_ > 0 } }

				# Calculate BFS distance: find index in array, use as distance proxy
				# Capital (last added) has distance 0, others have distance based on index
				set_temp_variable = { _bfs_distance = 0 }
				for_each_loop = { array = states_within_distance_ index = _dist_idx value = _check_state break = _found_dist
					if = { limit = { check_variable = { _check_state = _this_state_id } }
						# If this is capital, distance is 0; otherwise use index as distance
						if = { limit = { check_variable = { _this_state_id = capital_state_id } }
							set_temp_variable = { _bfs_distance = 0 }
						}
						else = {
							# Estimate distance from index (earlier = closer)
							# Scale index to max_distance range
							set_temp_variable = { _bfs_distance = _dist_idx }
							add_to_temp_variable = { _bfs_distance = 1 }
							# Cap at max distance
							if = { limit = { check_variable = { _bfs_distance > constant:wa_ai_railway.supply.home_port_search_distance } }
								set_temp_variable = { _bfs_distance = constant:wa_ai_railway.supply.home_port_search_distance }
							}
						}
						set_temp_variable = { _found_dist = 1 }
					}
				}

				# Score this port candidate
				set_temp_variable = { _candidate_port_level = naval_base_level_ }
				WA_AI_PC_score_port_candidate = yes

				# Check if this is the best so far
				if = {
					limit = { check_variable = { port_score_ > best_home_port_score_ } }
					set_temp_variable = { best_home_port_state_ = _this_state_id }
					set_temp_variable = { best_home_port_level_ = naval_base_level_ }
					set_temp_variable = { best_home_port_province_ = naval_base_province_ }
					set_temp_variable = { best_home_port_score_ = port_score_ }
				}
			}
		}
	}

	if = { limit = { has_country_flag = WA_AI_construction_logging }
		if = { limit = { check_variable = { best_home_port_state_ > 0 } }
			var:best_home_port_state_ = {
				ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY HELPER: find_best_home_port FOUND [Prev.GetName] level=[?best_home_port_level_] score=[?best_home_port_score_]" }
			}
		}
		else = {
			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY HELPER: find_best_home_port NONE FOUND"
		}
	}
}

############################################################################################################
#	HELPER: Check if a candidate beachhead is in a separate theatre from existing beachheads
#	Uses landmass for fast-path (different landmass = separate theatres)
#	Uses BFS distance for same-landmass check (>10 states apart = separate theatres)
#	candidate_beachhead_state_, _beachhead_states_ array >>> _is_duplicate_theatre_
############################################################################################################

WA_AI_PC_check_theatre_separation = {
	set_temp_variable = { _is_duplicate_theatre_ = 0 }
	set_temp_variable = { _found_duplicate = 0 }

	# Get candidate's landmass
	set_temp_variable = { _cand_landmass = global.WA_AI_MAP_state_landmass^candidate_beachhead_state_ }

	for_each_loop = {
		array = _beachhead_states_
		value = _existing_bh_state
		break = _found_duplicate

		# Get existing beachhead's landmass
		set_temp_variable = { _exist_landmass = global.WA_AI_MAP_state_landmass^_existing_bh_state }

		# Same landmass = need distance check
		if = {
			limit = { check_variable = { _cand_landmass = _exist_landmass } }

			# Check if candidate is within BFS distance of existing beachhead
			set_temp_variable = { _origin_state_id = _existing_bh_state }
			set_temp_variable = { _max_distance = constant:wa_ai_railway.routes.theatre_separation_distance }
			WA_AI_PC_railway_get_states_within_distance = yes

			# Include origin state in check
			add_to_temp_array = { states_within_distance_ = _existing_bh_state }

			# If candidate is within range of existing beachhead, it's a duplicate theatre
			if = {
				limit = { is_in_array = { states_within_distance_ = candidate_beachhead_state_ } }
				set_temp_variable = { _is_duplicate_theatre_ = 1 }
				set_temp_variable = { _found_duplicate = 1 }
			}
		}
		# Different landmass = definitely separate theatres (fast path, no action needed)
	}
}

############################################################################################################
#	HELPER: Pre-filter enemies to reduce iteration count
#	Only includes: majors, enemies we directly border, or enemies with significant industry
############################################################################################################

WA_AI_PC_railway_get_relevant_enemies = { # >>> arr:_relevant_enemies_

	clear_temp_array = _relevant_enemies_

	every_enemy_country = {
		set_temp_variable = { _include_enemy = 0 }

		### Always include majors
		if = {
			limit = { is_major = yes }
			set_temp_variable = { _include_enemy = 1 }
		}

		### Include enemies with 50+ factories (significant industrial threat)
		if = {
			limit = {
				check_variable = { _include_enemy = 0 }
				num_of_factories > 50
			}
			set_temp_variable = { _include_enemy = 1 }
		}

		### Include enemies we directly border
		if = {
			limit = { check_variable = { _include_enemy = 0 } }
			set_temp_variable = { _check_enemy_for_border = THIS }
			set_temp_variable = { _found_border = 0 }
			ROOT = {
				every_controlled_state = {
					limit = {
						check_variable = { _found_border = 0 }
						any_neighbor_state = {
							is_controlled_by = var:_check_enemy_for_border
						}
					}
					set_temp_variable = { _found_border = 1 }
				}
			}
			if = {
				limit = { check_variable = { _found_border = 1 } }
				set_temp_variable = { _include_enemy = 1 }
			}
		}

		if = {
			limit = { check_variable = { _include_enemy = 1 } }
			add_to_temp_array = { _relevant_enemies_ = THIS }
		}
	}

}

############################################################################################################
#	HELPER: Get states within N adjacency steps from a state (BFS)
############################################################################################################

WA_AI_PC_railway_get_states_within_distance = { # _origin_state_id, _max_distance >>> states_within_distance_

	clear_temp_array = states_within_distance_
	clear_temp_array = _states_current_wave_
	clear_temp_array = _states_next_wave_
	clear_temp_array = _states_visited_

	add_to_temp_array = { _states_current_wave_ = _origin_state_id }
	add_to_temp_array = { _states_visited_ = _origin_state_id }

	set_temp_variable = { _current_distance = 0 }

	while_loop_effect = {
		limit = {
			check_variable = { _current_distance < _max_distance }
			check_variable = { _states_current_wave_^num > 0 }
		}

		# Process current wave
		for_each_loop = { array = _states_current_wave_ value = _wave_state
			var:_wave_state = {
				every_neighbor_state = {
					WA_AI_PC_get_state_id = yes
					set_temp_variable = { _neighbor_id = state_id_ }
					if = {
						limit = {
							NOT = { is_in_array = { _states_visited_ = _neighbor_id } }
						}
						add_to_temp_array = { _states_next_wave_ = _neighbor_id }
						add_to_temp_array = { _states_visited_ = _neighbor_id }
						add_to_temp_array = { states_within_distance_ = _neighbor_id }
					}
				}
			}
		}

		# Move to next wave
		clear_temp_array = _states_current_wave_
		for_each_loop = { array = _states_next_wave_ value = _next_state
			add_to_temp_array = { _states_current_wave_ = _next_state }
		}
		clear_temp_array = _states_next_wave_

		add_to_temp_variable = { _current_distance = 1 }
	}
}

############################################################################################################
#	HELPER: Check if ROOT (or ROOT's puppets) directly borders an enemy country
#	Used to prevent building railways to frontlines the country doesn't directly touch
############################################################################################################

WA_AI_PC_railway_country_borders_enemy = { # _enemy_tag >>> borders_enemy_
	set_temp_variable = { borders_enemy_ = 0 }

	# [railway-pathfinding] CONTROLLED states, not owned: frontline detection uses controlled states
	# too, and a state ROOT owns but the enemy occupies is not a border of ours.
	# Puppet borders are deliberately INCLUDED (below): pathfinder type 2 traverses puppet provinces
	# and build_railway works on any province regardless of controller. Excluding them was tried and
	# left subject frontlines (Libya, Egypt, Burma) reading as "we do not border this enemy".
	every_controlled_state = {
		limit = {
			any_neighbor_state = {
				OR = {
					is_controlled_by = var:_enemy_tag
					controller = { is_subject_of = var:_enemy_tag }
				}
			}
		}
		ROOT = { set_temp_variable = { borders_enemy_ = 1 } }
	}

	# [railway-pathfinding] ROOT's subjects/puppets border the enemy on our behalf
	every_subject_country = {
		every_controlled_state = {
			limit = {
				any_neighbor_state = {
					OR = {
						is_controlled_by = var:_enemy_tag
						controller = { is_subject_of = var:_enemy_tag }
					}
				}
			}
			ROOT = { set_temp_variable = { borders_enemy_ = 1 } }
		}
	}

}

############################################################################################################
#	HELPER: Score and sort targets by enemy threat (multi-front war support)
#	Prioritizes targets facing the most dangerous enemy first
#	Threat score = num_of_factories + (num_divisions * 5)
############################################################################################################

WA_AI_PC_railway_score_and_sort_by_enemy_threat = {

	### Step 1: Build unique enemy list and calculate threat scores
	clear_temp_array = _unique_enemies_
	clear_temp_array = _enemy_scores_

	for_each_loop = { array = railway_enemy_tags_ value = _check_enemy
		# Check if this enemy is already in our unique list
		set_temp_variable = { _enemy_already_listed = 0 }
		for_each_loop = { array = _unique_enemies_ value = _listed_enemy
			if = { limit = { check_variable = { _check_enemy = _listed_enemy } }
				set_temp_variable = { _enemy_already_listed = 1 }
			}
		}

		# If not listed, add enemy and calculate threat score
		if = { limit = { check_variable = { _enemy_already_listed = 0 } }
			add_to_temp_array = { _unique_enemies_ = _check_enemy }

			# Calculate threat score: factories + (divisions * 5)
			set_temp_variable = { _threat_score = 0 }
			var:_check_enemy = {
				add_to_temp_variable = { _threat_score = num_of_factories }
				set_temp_variable = { _div_score = num_divisions }
				multiply_temp_variable = { _div_score = 5 }
				add_to_temp_variable = { _threat_score = _div_score }
			}
			add_to_temp_array = { _enemy_scores_ = _threat_score }
		}
	}

	### Step 2: Sort enemies by threat score (descending) using bubble sort
	set_temp_variable = { _num_enemies = _unique_enemies_^num }
	set_temp_variable = { _outer_limit = _num_enemies }
	subtract_from_temp_variable = { _outer_limit = 1 }

	while_loop_effect = {
		limit = { check_variable = { _outer_limit > 0 } }

		set_temp_variable = { _inner_idx = 0 }
		while_loop_effect = {
			limit = { check_variable = { _inner_idx < _outer_limit } }

			set_temp_variable = { _next_idx = _inner_idx }
			add_to_temp_variable = { _next_idx = 1 }

			# If current score < next score, swap (we want descending order)
			if = { limit = { check_variable = { _enemy_scores_^_inner_idx < _enemy_scores_^_next_idx } }
				# Swap scores
				set_temp_variable = { _tmp_score = _enemy_scores_^_inner_idx }
				set_temp_variable = { _enemy_scores_^_inner_idx = _enemy_scores_^_next_idx }
				set_temp_variable = { _enemy_scores_^_next_idx = _tmp_score }

				# Swap enemy tags
				set_temp_variable = { _tmp_enemy = _unique_enemies_^_inner_idx }
				set_temp_variable = { _unique_enemies_^_inner_idx = _unique_enemies_^_next_idx }
				set_temp_variable = { _unique_enemies_^_next_idx = _tmp_enemy }
			}

			add_to_temp_variable = { _inner_idx = 1 }
		}

		subtract_from_temp_variable = { _outer_limit = 1 }
	}

	### Step 3: Rebuild target arrays in enemy-priority order
	# Copy current arrays to temporary storage
	clear_temp_array = _tmp_start_provinces_
	clear_temp_array = _tmp_end_provinces_
	clear_temp_array = _tmp_target_levels_
	clear_temp_array = _tmp_priorities_
	clear_temp_array = _tmp_enemy_tags_

	for_each_loop = { array = railway_start_provinces_ index = _copy_i
		add_to_temp_array = { _tmp_start_provinces_ = railway_start_provinces_^_copy_i }
		add_to_temp_array = { _tmp_end_provinces_ = railway_end_provinces_^_copy_i }
		add_to_temp_array = { _tmp_target_levels_ = railway_target_levels_^_copy_i }
		add_to_temp_array = { _tmp_priorities_ = railway_priorities_^_copy_i }
		add_to_temp_array = { _tmp_enemy_tags_ = railway_enemy_tags_^_copy_i }
	}

	# Clear original arrays
	clear_temp_array = railway_start_provinces_
	clear_temp_array = railway_end_provinces_
	clear_temp_array = railway_target_levels_
	clear_temp_array = railway_priorities_
	clear_temp_array = railway_enemy_tags_

	# Rebuild in sorted order: for each enemy (highest threat first), add all their targets
	for_each_loop = { array = _unique_enemies_ value = _sorted_enemy
		for_each_loop = { array = _tmp_enemy_tags_ index = _target_i value = _target_enemy
			if = { limit = { check_variable = { _target_enemy = _sorted_enemy } }
				add_to_temp_array = { railway_start_provinces_ = _tmp_start_provinces_^_target_i }
				add_to_temp_array = { railway_end_provinces_ = _tmp_end_provinces_^_target_i }
				add_to_temp_array = { railway_target_levels_ = _tmp_target_levels_^_target_i }
				add_to_temp_array = { railway_priorities_ = _tmp_priorities_^_target_i }
				add_to_temp_array = { railway_enemy_tags_ = _target_enemy }
			}
		}
	}

	# Boost priority of first target (highest threat enemy's first target)
	if = { limit = { check_variable = { railway_priorities_^num > 0 } }
		set_temp_variable = { _boosted_priority = railway_priorities_^0 }
		multiply_temp_variable = { _boosted_priority = 1.1 }
		set_temp_variable = { railway_priorities_^0 = _boosted_priority }
	}

}

############################################################################################################
#	LANDMASS-AWARE SUPPLY CHAIN HELPERS (Single Responsibility)
#	Used to handle island nations like Japan that need overseas supply routes
############################################################################################################

### Find highest level naval base on a specific landmass
WA_AI_PC_get_best_port_on_landmass = { # _search_landmass >>> best_port_level_, best_port_province_, best_port_state_, best_port_supply_

	set_temp_variable = { best_port_level_ = 0 }
	set_temp_variable = { best_port_province_ = 0 }
	set_temp_variable = { best_port_state_ = 0 }
	set_temp_variable = { best_port_supply_ = 0 }

	# [na-corridor] ONE walk over ROOT, its subjects AND faction allies that cannot build their own
	# logistics. Without the ally leg a route TO ally soil still starts from ROOT's own nearest port
	# and has to pathfind across the theatre, while the ally's own big port sits at the target.
	every_country = {
		limit = { WA_AI_PC_is_logistics_build_partner = yes }
		every_controlled_state = {
			WA_AI_PC_get_state_id = yes
			set_temp_variable = { _this_state_id = state_id_ }

			if = {
				limit = {
					check_variable = { global.WA_AI_MAP_state_landmass^_this_state_id = _search_landmass }
					is_coastal = yes
				}

				# Get the naval base province and level
				WA_AI_PC_railway_get_naval_base_province = yes

				if = {
					limit = {
						check_variable = { naval_base_province_ > 0 }
						check_variable = { naval_base_level_ > best_port_level_ }
					}
					set_temp_variable = { best_port_level_ = naval_base_level_ }
					set_temp_variable = { best_port_state_ = _this_state_id }
					set_temp_variable = { best_port_province_ = naval_base_province_ }
				}
			}
		}
	}

	# Calculate supply capacity using configurable formula
	set_temp_variable = { _port_level = best_port_level_ }
	WA_AI_PC_calculate_port_supply = yes
	set_temp_variable = { best_port_supply_ = port_supply_ }
}

############################################################################################################
#	HELPER: Find Best Port Near a Target State (distance-limited)
#	BFS from target state up to N states, then finds the highest-level controlled port
#	on the same landmass within that radius. Uses every_controlled_state to include puppets.
#	[railway-pathfinding] the radius is what keeps the port near the target: a landmass-only search
#	picks ports thousands of km away (Singapore for a North African target on Afro-Eurasia).
############################################################################################################

WA_AI_PC_get_best_port_near_state = { # _target_state_for_port, _port_search_radius >>> best_port_level_, best_port_province_, best_port_state_, best_port_supply_

	# Step 1: BFS from target state to get all states within radius
	set_temp_variable = { _origin_state_id = _target_state_for_port }
	set_temp_variable = { _max_distance = _port_search_radius }
	WA_AI_PC_railway_get_states_within_distance = yes
	# Include origin state itself (BFS helper excludes it)
	add_to_temp_array = { states_within_distance_ = _target_state_for_port }

	# Step 2: Get target landmass for filtering
	set_temp_variable = { _ps_target_lm = global.WA_AI_MAP_state_landmass^_target_state_for_port }

	# Step 3: Among ROOT's controlled states (and puppet states), find best port within BFS range on same landmass
	set_temp_variable = { best_port_level_ = 0 }
	set_temp_variable = { best_port_province_ = 0 }
	set_temp_variable = { best_port_state_ = 0 }
	set_temp_variable = { best_port_supply_ = 0 }

	# [na-corridor] ONE walk over ROOT, its subjects and dependent faction allies, so a receiving port
	# can be a subject's (Tripoli for Derna) or an ally's sitting at the target, instead of ROOT's
	# nearest own port a theatre away.
	every_country = {
		limit = { WA_AI_PC_is_logistics_build_partner = yes }
		every_controlled_state = {
			WA_AI_PC_get_state_id = yes
			set_temp_variable = { _ps_state_id = state_id_ }
			if = {
				limit = {
					# Must be on same landmass as target
					check_variable = { global.WA_AI_MAP_state_landmass^_ps_state_id = _ps_target_lm }
					# Must be within BFS search radius
					is_in_array = { states_within_distance_ = _ps_state_id }
					is_coastal = yes
				}
				WA_AI_PC_railway_get_naval_base_province = yes
				if = {
					limit = {
						check_variable = { naval_base_province_ > 0 }
						check_variable = { naval_base_level_ > best_port_level_ }
					}
					set_temp_variable = { best_port_level_ = naval_base_level_ }
					set_temp_variable = { best_port_state_ = _ps_state_id }
					set_temp_variable = { best_port_province_ = naval_base_province_ }
				}
			}
		}
	}

	# Calculate supply capacity
	set_temp_variable = { _port_level = best_port_level_ }
	WA_AI_PC_calculate_port_supply = yes
	set_temp_variable = { best_port_supply_ = port_supply_ }
}

### Calculate supply bottleneck (minimum of two supply values)
WA_AI_PC_calculate_supply_bottleneck = { # _supply_a, _supply_b >>> bottleneck_supply_

	if = { limit = { check_variable = { _supply_a < _supply_b } }
		set_temp_variable = { bottleneck_supply_ = _supply_a }
	}
	else = {
		set_temp_variable = { bottleneck_supply_ = _supply_b }
	}
}

### Convert supply capacity to maximum useful railway level
### Uses configurable constants: BASE + PER_LEVEL * level
WA_AI_PC_supply_to_railway_level = { # _supply_capacity >>> max_railway_level_
	# Formula: level = (supply - BASE) / PER_LEVEL, clamped to 0-5
	set_temp_variable = { max_railway_level_ = _supply_capacity }
	subtract_from_temp_variable = { max_railway_level_ = constant:wa_ai_railway.supply.railway_base }
	divide_temp_variable = { max_railway_level_ = constant:wa_ai_railway.supply.railway_per_level }

	# Clamp to valid range
	if = { limit = { check_variable = { max_railway_level_ < 0 } }
		set_temp_variable = { max_railway_level_ = 0 }
	}
	if = { limit = { check_variable = { max_railway_level_ > 5 } }
		set_temp_variable = { max_railway_level_ = 5 }
	}
}

### Calculate railway supply capacity for a given level
### Formula: BASE + PER_LEVEL * level (e.g., 4 + 8*level)
WA_AI_PC_calculate_railway_supply = { # _railway_level >>> railway_supply_
	set_temp_variable = { railway_supply_ = _railway_level }
	multiply_temp_variable = { railway_supply_ = constant:wa_ai_railway.supply.railway_per_level }
	add_to_temp_variable = { railway_supply_ = constant:wa_ai_railway.supply.railway_base }
}

### Calculate port supply capacity for a given level
### Formula: level * PER_LEVEL (e.g., level * 5)
WA_AI_PC_calculate_port_supply = { # _port_level >>> port_supply_
	set_temp_variable = { port_supply_ = _port_level }
	multiply_temp_variable = { port_supply_ = constant:wa_ai_railway.supply.port_per_level }
}

### Calculate minimum railway level needed for target supply
### Inverse of: supply = BASE + PER_LEVEL * level
WA_AI_PC_supply_to_target_railway = { # _target_supply >>> target_railway_level_
	set_temp_variable = { target_railway_level_ = _target_supply }
	subtract_from_temp_variable = { target_railway_level_ = constant:wa_ai_railway.supply.railway_base }
	divide_temp_variable = { target_railway_level_ = constant:wa_ai_railway.supply.railway_per_level }
	# Round up by adding 0.99 before implicit floor
	add_to_temp_variable = { target_railway_level_ = 0.99 }
	round_temp_variable = target_railway_level_

	# Clamp to 1-5 range
	if = { limit = { check_variable = { target_railway_level_ < 1 } }
		set_temp_variable = { target_railway_level_ = 1 }
	}
	if = { limit = { check_variable = { target_railway_level_ > 5 } }
		set_temp_variable = { target_railway_level_ = 5 }
	}
}

### Calculate minimum port level needed for target supply
### Inverse of: supply = level * PER_LEVEL
WA_AI_PC_supply_to_target_port = { # _target_supply >>> target_port_level_
	set_temp_variable = { target_port_level_ = _target_supply }
	divide_temp_variable = { target_port_level_ = constant:wa_ai_railway.supply.port_per_level }
	# Round up by adding 0.99 before implicit floor
	add_to_temp_variable = { target_port_level_ = 0.99 }
	round_temp_variable = target_port_level_

	# Clamp to 1-10 range (naval base max)
	if = { limit = { check_variable = { target_port_level_ < 1 } }
		set_temp_variable = { target_port_level_ = 1 }
	}
	if = { limit = { check_variable = { target_port_level_ > 10 } }
		set_temp_variable = { target_port_level_ = 10 }
	}
}

############################################################################################################

### Estimate railway upgrade cost based on BFS distance
### Uses estimated segments per state and base cost per segment
WA_AI_PC_estimate_railway_cost = { # _bfs_distance, _target_level >>> estimated_railway_cost_
	# Estimate total segments: distance * segments_per_state
	set_temp_variable = { estimated_segments_ = _bfs_distance }
	multiply_temp_variable = { estimated_segments_ = constant:wa_ai_railway.cost.railway_segments_per_state }

	# Total cost: segments * base_cost * target_level (higher levels need more upgrades)
	set_temp_variable = { estimated_railway_cost_ = estimated_segments_ }
	multiply_temp_variable = { estimated_railway_cost_ = constant:wa_ai_pc.cost.railway }
	multiply_temp_variable = { estimated_railway_cost_ = _target_level }
}

### Calculate port upgrade cost from current to target level
### Naval base cost: base_cost + per_level_cost * current_level (per_level is negative, so decreasing)
### Total cost = sum from current to target-1 of (base + per_level * i)
WA_AI_PC_calculate_port_upgrade_cost = { # _current_port_level, _target_port_level >>> port_upgrade_cost_
	set_temp_variable = { port_upgrade_cost_ = 0 }

	# If already at or above target, no cost
	if = {
		limit = { NOT = { check_variable = { _current_port_level < _target_port_level } } }
		# Cost stays 0
	}
	else = {
		# Calculate sum of costs for each level upgrade
		# Cost per level = base + per_level * current_level
		# For naval base: 10000 + (-556) * level
		set_temp_variable = { _upgrade_level = _current_port_level }

		while_loop_effect = {
			limit = { check_variable = { _upgrade_level < _target_port_level } }

			# Cost for this level: base + per_level * current_level
			set_temp_variable = { _level_cost = _upgrade_level }
			multiply_temp_variable = { _level_cost = constant:wa_ai_railway.cost.naval_base_per_level }
			add_to_temp_variable = { _level_cost = constant:wa_ai_pc.cost.naval_base }

			add_to_temp_variable = { port_upgrade_cost_ = _level_cost }
			add_to_temp_variable = { _upgrade_level = 1 }
		}
	}
}

### Score a candidate port for home port selection
### Strategy: Maximize supply first, then minimize cost as tiebreaker
### Score = achievable_supply * 100000 - total_cost
WA_AI_PC_score_port_candidate = { # _candidate_port_level, _bfs_distance >>> port_score_
	# Calculate target levels from configured target supply
	set_temp_variable = { _target_supply = constant:wa_ai_railway.supply.home_port_target_supply }
	WA_AI_PC_supply_to_target_railway = yes
	WA_AI_PC_supply_to_target_port = yes

	# Cap target port level to 10 (max naval base)
	if = { limit = { check_variable = { target_port_level_ > 10 } }
		set_temp_variable = { target_port_level_ = 10 }
	}

	# Calculate achievable supply (bottleneck of port and railway at target levels)
	set_temp_variable = { _railway_level = target_railway_level_ }
	WA_AI_PC_calculate_railway_supply = yes
	set_temp_variable = { _railway_achievable = railway_supply_ }

	set_temp_variable = { _port_level = target_port_level_ }
	WA_AI_PC_calculate_port_supply = yes
	set_temp_variable = { _port_achievable = port_supply_ }

	# Bottleneck is the minimum
	if = { limit = { check_variable = { _port_achievable < _railway_achievable } }
		set_temp_variable = { achievable_supply_ = _port_achievable }
	}
	else = {
		set_temp_variable = { achievable_supply_ = _railway_achievable }
	}

	# Estimate railway cost
	set_temp_variable = { _target_level = target_railway_level_ }
	WA_AI_PC_estimate_railway_cost = yes

	# Calculate port upgrade cost
	set_temp_variable = { _current_port_level = _candidate_port_level }
	WA_AI_PC_calculate_port_upgrade_cost = yes

	# Total cost
	set_temp_variable = { total_cost_ = estimated_railway_cost_ }
	add_to_temp_variable = { total_cost_ = port_upgrade_cost_ }

	# Score: supply * 100000 - cost
	# This ensures higher supply always wins, cost breaks ties
	set_temp_variable = { port_score_ = achievable_supply_ }
	multiply_temp_variable = { port_score_ = 100000 }
	subtract_from_temp_variable = { port_score_ = total_cost_ }
}

### Check if state has only a single logistics node (port acting as hub, no actual supply_node)
### States like Sakhalin have only a port - building railways is useless
WA_AI_PC_coastal_state_is_single_node = { # THIS = state >>> is_single_node_ (1 = only port, no supply hub)

	set_temp_variable = { is_single_node_ = 0 }

	# If state has a naval base but no supply_node, it's a single-node state
	if = {
		limit = {
			is_coastal = yes
			NOT = { supply_node > 0 }
		}
		# Check for naval base using helper (trigger returns 0 in some scopes)
		WA_AI_PC_railway_get_naval_base_province = yes
		if = {
			limit = { check_variable = { naval_base_province_ > 0 } }
			set_temp_variable = { is_single_node_ = 1 }
		}
	}
}

### Analyze overseas supply chain for a target on a different landmass

### (WA_AI_PC_get_best_port_near_state); a whole-landmass search picks ports a theatre away.
WA_AI_PC_analyze_overseas_supply_chain = { # capital_landmass, target_landmass, _target_state_for_port_search >>> overseas_route_start_, overseas_max_railway_level_, overseas_receiving_port_state_, overseas_home_port_state_, overseas_entry_supply_ ([na-corridor])

	set_temp_variable = { overseas_route_start_ = 0 }
	set_temp_variable = { overseas_max_railway_level_ = 0 }

	# Find home port (on capital landmass) - unchanged

	# [railway-pathfinding] receiving port near the target state (BFS radius below, same landmass)
	set_temp_variable = { _target_state_for_port = _target_state_for_port_search }

	WA_AI_PC_get_best_port_near_state = yes

	set_temp_variable = { overseas_entry_supply_ = bottleneck_supply_ }
}

WA_AI_PC_start_railway_project = { # THIS = state, _project_province_id, _project_connect_id, _project_target_level, _project_priority
	# Optional input: _railway_family_ (temp, [na-corridor]) - 1 = the segment belongs to a THEATRE
	# CORRIDOR: tagged with [levant-iraq]'s _corridor_type_id_ (that corridor's own
	# constant:wa_ai_pc.type_id.*, derived once at the entry of WA_AI_PC_railway_corridor_run_one,
	# the only caller that sets _railway_family_), budgeted by
	# constant:wa_ai_railway.corridor.queue_max under scoped queue_max. Absent / 0 = the rail family
	# (type_id rail, budget routes.queue_full).
	# LATCHED AND ZEROED HERE, at the top, before any early return: it is a chain temp, the corridor
	# sets it before EVERY segment call, and a leaked 1 tags the main pass's first land-war segment
	# as corridor.
	set_temp_variable = { _rail_family_latched_ = _railway_family_ }
	set_temp_variable = { _railway_family_ = 0 }

	# Check if railway already at target level
	meta_effect = {
		text = {
			set_temp_variable = { current_level = global.WA_AI_PC_railway_connection_level_[x]^[y] }
		}
		x = "[?_project_province_id]"
		y = "[?_project_connect_id]"
	}

	# Count already queued projects for this segment
	set_temp_variable = { _queued_for_segment = 0 }
	ROOT = {
		for_each_loop = { array = WA_AI_PC_queue value = _check_proj
			if = {
				limit = {
					check_variable = { WA_AI_PC_building_type^_check_proj = 13 }
					check_variable = { WA_AI_PC_target_province^_check_proj = _project_province_id }
					check_variable = { WA_AI_PC_connect_province^_check_proj = _project_connect_id }
				}
				add_to_temp_variable = { _queued_for_segment = 1 }
			}
		}
	}

	# [na-corridor] CONNECT BEFORE CONSOLIDATE. Read the RAW map level, before the queued upgrades are
	# folded in below: the question is "does a railway exist on this connection", not "has someone
	# asked for one". A corridor segment at 0 is a BREAK in the chain and must outrank every
	# level-raising segment, because the corridor pass emits ONE priority for all its hops and
	# WA_AI_PC_assign_factories sorts by priority alone - at a tie the bubble sort serves insertion
	# order, so the rear hops take the factories and the break stays a break.
	# Corridor family only (_rail_family_latched_ = 1) by decision: land_war derives its routes
	# dynamically, so the same rule there is a global re-ordering for every AI country.
	# This does NOT create construction capacity - it re-targets capacity already being spent.
	set_temp_variable = { _rail_map_level_ = current_level }

	# Effective level = current level + queued upgrades
	add_to_temp_variable = { current_level = _queued_for_segment }

	if = {
		limit = {
			check_variable = { _rail_family_latched_ = 1 }
			check_variable = { _rail_map_level_ < 1 }
		}
		set_temp_variable = { _project_priority = constant:wa_ai_pc.prio.rail_connect }
	}

	ROOT = { if = { limit = { has_country_flag = WA_AI_construction_logging }
		log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY DEBUG: prov=[?_project_province_id]->[?_project_connect_id] current=[?current_level] target=[?_project_target_level] queued=[?_queued_for_segment]"
	} }

	if = {
		limit = {
			check_variable = { current_level < _project_target_level }
		}
		# tlm r103_corridor_blocked_n - counter, VERIFIED EFFECT: this corridor segment wanted a
		# level (the check above passed, so it is neither built nor already queued) and the
		# state-scoped ADMISSION gate refuses it. That refusal is silent in WA_AI_PC_start_project,
		# which is how a hop on an ally-held province inside an enemy-controlled state can be
		# requested and pathfound for years and never queued. Deliberately placed AFTER the level
		# check - before it, every already-finished hop on a hostile-state end would book a refusal
		# that never happened. THIS = state, ROOT = builder (both call sites scope into the state),
		# the scope WA_AI_PC_state_controller_allows_admission requires - the same trigger the
		# executor calls, so probe and gate cannot drift. Corridor family only.
		# [levant-iraq] corridor 1 only, like every other corridor WA_TLM name: the metric is read as
		# a North Africa refusal count and a second corridor summing into it would retarget it, not
		# extend it. Corridor 2 gets no blocked_n by design - its three nodes and every province
		# between them are in ONE state, so the province/state scope mismatch this counter exists to
		# expose cannot arise there.
		if = {
			limit = {
				check_variable = { _rail_family_latched_ = 1 }
				check_variable = { _corridor_id_ = constant:wa_ai_pc.corridor_id.north_africa }
				NOT = { WA_AI_PC_state_controller_allows_admission = yes }
			}
			ROOT = { add_to_variable = { WA_TLM_r103_corridor_blocked_n = 1 } }
		}
		# Use new PC system - call WA_AI_PC_start_project
		set_temp_variable = { _project_building_type = 13 }
		set_temp_variable = { _project_queue_num = 1 }
		if = {
			limit = { check_variable = { _rail_family_latched_ = 1 } }
			# [na-corridor] theatre-corridor segment - own tag, own scoped budget, so the corridor
			# neither eats the land-war family's budget nor is refused because that family is full.
			# [levant-iraq] the tag is PER CORRIDOR (two corridors under one tag cancel each other's
			# queue - see WA_AI_PC_railway_corridor_run_one). CONTRACT: _corridor_type_id_ is derived
			# at the entry of run_one, which is the only caller that sets _railway_family_ = 1, so it
			# is always written before this line. Deliberately NO `> 0` fallback: an UNSET temp reads
			# as a scope token, not as 0, so such a guard passes and tags the project with garbage.
			set_temp_variable = { _project_type_id = _corridor_type_id_ }
			set_temp_variable = { _project_queue_max = constant:wa_ai_railway.corridor.queue_max }
			set_temp_variable = { _project_queue_max_scoped = 1 }
		}
		else = {
			set_temp_variable = { _project_type_id = constant:wa_ai_pc.type_id.rail }
			# [pc-queue] this budget must never be 0: it is the only cap on the railway family, and
			# WA_AI_PC_start_project counts queued type-13 country-wide, so it bounds land war,
			# overseas and prewar together. See the constant's header for the threshold relationship.
			# [na-corridor] "country-wide" means every type-13 EXCEPT the corridor's - the skip gate
			# in railway_core and this budget both key on WA_AI_PC_type_id = rail.
			set_temp_variable = { _project_queue_max = constant:wa_ai_railway.routes.queue_full }
			set_temp_variable = { _project_queue_max_scoped = 1 }
		}
		set_temp_variable = { _project_target_level = 0 }  # Already checked above
		# Set state ID using province mapping (states don't have .id property)
		set_temp_variable = { _project_target_state = global.WA_AI_MAP_province_state_id^_project_province_id }

		WA_AI_PC_start_project = yes
	}
}

#	Upgrades ports marked in railway_port_upgrades_ array
############################################################################################################

WA_AI_PC_process_port_upgrades = {
	# Process all port upgrades for overseas operations via PC system
	# Note: Uses temp array which is cleared each railway pass, so we process all at once
	for_each_loop = { array = railway_port_upgrades_ value = _port_state

		var:_port_state = {
			if = {
				limit = {
					# [na-corridor] NOT is_controlled_by = ROOT: naval_base is allied_build = yes and
					# WA_AI_PC_start_project already accepts allied ground, so the receiving port of
					# an ally-soil corridor is upgradable - it only lacked a selector.
					# This site and the frontier port below set _project_queue_max = 0, i.e. NO type
					# budget, so the exposure is stated rather than called bounded: the sites are fed
					# by the route pass (capped at routes.max_total pathfinds per run), the
					# province-duplicate guard in WA_AI_PC_start_project refuses a second project of
					# the same building type on the same province while the first is queued, and the
					# eligible set only shrinks (this site stops at naval_base > 8, the frontier site
					# only fires where there is no port at all). The unbounded quantity is the SET of
					# eligible states, which geography caps - not a per-pass rate. Queue-slot
					# competition against ROOT's own projects is unproven, not safe.
					WA_AI_PC_can_build_logistics_here = yes
					is_coastal = yes
					# Only upgrade if below max useful level (L9 already exceeds L5 railway capacity)
					NOT = { naval_base > 8 }
				}

				# Get the naval base province (naval bases are built on provinces, not states)
				WA_AI_PC_railway_get_naval_base_province = yes

				if = {
					limit = { check_variable = { naval_base_province_ > 0 } }

					# Queue port construction via PC system
					set_temp_variable = { _project_building_type = 14 }  # naval_base
					set_temp_variable = { _project_type_id = constant:wa_ai_pc.type_id.port }
					set_temp_variable = { _project_province_id = naval_base_province_ }
					set_temp_variable = { _project_connect_id = 0 }  # Not used for ports
					set_temp_variable = { _project_priority = constant:wa_ai_pc.prio.rail_prewar }  # Lower priority for upgrades
					set_temp_variable = { _project_queue_num = 1 }
					set_temp_variable = { _project_queue_max = 0 }
					set_temp_variable = { _project_target_level = constant:wa_ai_railway.supply.port_max_useful_level }
					set_temp_variable = { _project_target_state = _port_state }

					WA_AI_PC_start_project = yes

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY: Queued port upgrade at province [?naval_base_province_] in state [?_port_state]" }
					}
				}
			}
		}
	}
}

WA_AI_PC_create_frontier_port = { # _frontier_province_id >>> frontier_port_created_

	set_temp_variable = { _frontier_state_id = global.WA_AI_MAP_province_state_id^_frontier_province_id }

	if = {
		limit = { check_variable = { _frontier_state_id > 0 } }

		var:_frontier_state_id = {
			if = {
				limit = {
					# [na-corridor] NOT is_controlled_by = ROOT - a partial route that dead-ends on

					WA_AI_PC_province_is_coastal = yes

					set_temp_variable = { _project_target_level = constant:wa_ai_railway.supply.port_max_useful_level }
					set_temp_variable = { _project_target_state = _frontier_state_id }

					WA_AI_PC_start_project = yes

					set_temp_variable = { frontier_port_created_ = 1 }

					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY: Queued frontier port at province [?_build_port_province] in state [?_frontier_state_id]" }
					}
				}
				else = {
					if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
						ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY: Frontier state [?_frontier_state_id] already has naval base at [?naval_base_province_]" }
					}

############################################################################################################
#	HELPER: Get Maximum Railway Level on Route
#	Input: pathfind_prov_path_ (province path from the most recent WA_AI_PATHFIND_PROV_get_path call -
#	       call this AFTER the pathfind for the route you are levelling, never before)
#	Output: _max_route_level_ (highest level found on any segment in the route)
#	Purpose: Find the max level segment in a route to sync all segments to max+1
#	[railway-landwar] it must read pathfind_prov_path_ DIRECTLY - set_temp_variable does not copy
#	arrays, so a scalar "copy" of the path returns 0 and every route then targets level 1, which
#	existing rail already satisfies, so nothing is ever queued.
############################################################################################################

WA_AI_PC_get_max_railway_level_on_route = {
	# Initialize max to 0
	set_temp_variable = { _max_route_level_ = 0 }

	# Iterate through the fresh pathfind result to find all segments
	for_each_loop = { array = pathfind_prov_path_ index = seg_i value = seg_prov
		# Calculate next index (segment endpoint)
		set_temp_variable = { next_seg_i = seg_i }
		add_to_temp_variable = { next_seg_i = 1 }

		# Only process if there's a next segment (prevents index out of bounds)
		if = { limit = { check_variable = { next_seg_i < pathfind_prov_path_^num } }
			# Get endpoint of this segment
			set_temp_variable = { seg_start = pathfind_prov_path_^seg_i }
			set_temp_variable = { seg_end = pathfind_prov_path_^next_seg_i }

			# Read the current level of this segment from global array
			meta_effect = {
				text = {
					set_temp_variable = { seg_level = global.WA_AI_PC_railway_connection_level_[x]^[y] }
				}
				x = "[?seg_start]"
				y = "[?seg_end]"
			}

			# Update max if this segment is higher than current max
			if = { limit = { check_variable = { seg_level > _max_route_level_ } }
				set_temp_variable = { _max_route_level_ = seg_level }
			}
		}
	}
}

############################################################################################################
#	HELPER [na-corridor]: SIZE A RAIL ROUTE - what does the ground on this route need, what can
#	our entry capacity inject, and what railway / port level does that imply.
#	ROOT = THIS = builder country (COUNTRY scope: the presence walk is `every_country` and the
#	capital test is `capital_scope`). ONE middle layer, TWO callers:
#	  - WA_AI_PC_corridor_compute_sizing (just below), for a theatre corridor;
#	  - WA_AI_PC_railway_land_consider_frontline (railway_strategies), for a land-war / overseas route.
#	Implements documentation/WA_AI_LOGISTICS_MODEL.md 11.4-11.6 for both.
#
#	Inputs, all temps set by the caller:
#	  _rsize_prov_         ARRAY of province ids whose STATES carry the demand. The corridor passes its
#	                       whole node list; the land-war caller passes the ONE frontline hub province,
#	                       because the presence walk below is every_country x states and that caller
#	                       runs it once PER ROUTE, up to routes.max_total times per execution.
#	  _rsize_ours_         ARRAY parallel to _rsize_prov_, 1 = this node's port counts as ours.
#	                       Read in SUM mode only.
#	  _rsize_entry_mode_   0 = SUM      - add up the naval bases we hold on the route. A corridor is fed
#	                                      by every port along it and injections ADD (LOGISTICS_MODEL 7).
#	                       1 = GIVEN    - the caller already knows the entry capacity and passes it in
#	                                      _rsize_entry_supply_. An overseas chain is in SERIES, so its
#	                                      entry is the BOTTLENECK of home port and receiving port.
#	                       2 = OVERLAND - there is no port entry to cap against: the chain runs from our
#	                                      own capital across our own network, fed by CAPITAL_SUPPLY_BASE.
#	                                      Demand alone decides. This is NOT the same as passing a set
#	                                      that happens to contain no port: a frontline state that DOES
#	                                      have a small harbour would otherwise cap an overland chain at
#	                                      that harbour's capacity, which is the wrong physics.
#	                       THE SPLIT IS DELIBERATE AND MUST NOT BE UNIFIED. Each rule is right in its own
#	                       topology and any single one would be wrong in the other two. This is the whole
#	                       reason the helper takes a mode instead of computing injection for everybody.
#	  _rsize_entry_supply_ entry capacity in supply units. GIVEN mode only; ignored in SUM mode.
#	  _rsize_floor_        never size below this.
#	  _rsize_cap_          never size above this. Callers pass their own family's constants
#	                       (corridor.rail_level_* / land_war.rail_level_*), which is what lets the two
#	                       families be retuned apart while sharing one model.
#	>>> rsize_presence_ (demand proxy), rsize_demand_, rsize_inject_ (-1 = cap not applied),
#	    rsize_rail_tgt_ (floor..cap), rsize_port_tgt_ (0 = no port case open)
#
#	Protects against: a flat hardcoded rail target. Both families shipped one in turn - a ceiling by
#	construction on one side, a demand-blind ratchet on the other - which is why they now share a
#	model instead of a convention.
#	Assumes: the two engine flow ladders, rail 4 + 8xlevel and port 5xlevel
#	(constant:wa_ai_railway.supply.*, registered mirrors of 05_defines.lua). ACTUAL supply is not
#	readable from script, so this MODELS flow from levels and can never observe whether it was right;
#	WA_TLM r107 against a campaign map reading is the only validation path.
#	Fact is gone when: either ladder is retuned, or a script trigger for real supply appears.
#
#	rsize_presence_ is NOT a division count and must not be read as one. `divisions_in_state` is a
#	comparison trigger, so no exact per-state count exists in script; the ladder below is the
#	WA_AI_MILITARY_posture_count_state_divs idiom (posture_effects.txt) with bucket midpoints instead
#	of its power weights, and WITHOUT its armour doubling (a combat-power proxy, not a supply one).
#	COPIED rather than called on purpose: that effect writes _post_accum, a military-system temp.
#	Known bias, ASSUMED: it over-reads a DISPERSED army, because each state pays its own bucket floor.
#	The bias is upward, i.e. toward overbuilding;
#	constant:wa_ai_railway.corridor.supply_per_division is the knob that compensates for it.
############################################################################################################

WA_AI_PC_rail_size_route = {
	set_temp_variable = { rsize_presence_ = 0 }
	set_temp_variable = { rsize_demand_ = 0 }
	set_temp_variable = { rsize_inject_ = 0 }
	set_temp_variable = { rsize_rail_tgt_ = _rsize_floor_ }
	set_temp_variable = { rsize_port_tgt_ = 0 }
	clear_temp_array = _rz_states_

	### Distinct states of the route's nodes. Deduplicated because several corridor nodes share one
	### state - counting a state twice inflates the demand proxy by its own garrison.
	for_each_loop = { array = _rsize_prov_ value = _rz_prov_
		set_temp_variable = { _rz_state_ = global.WA_AI_MAP_province_state_id^_rz_prov_ }
		if = {
			limit = {
				check_variable = { _rz_state_ > 0 }
				NOT = { is_in_array = { _rz_states_ = _rz_state_ } }
			}
			add_to_temp_array = { _rz_states_ = _rz_state_ }
		}
	}

	### Presence of OUR SIDE on those states. Every side holder counts - a coalition ally's divisions
	### eat the same route's supply as ours. WA_AI_PC_is_corridor_side_holder is the same three-term
	### permission test the corridor selector uses, reused here so a widening of "our side" moves both
	### together. (The name still says corridor because that is where it is declared and where its
	### header lives; it has no corridor-specific term in it.)
	every_country = {
		limit = { WA_AI_PC_is_corridor_side_holder = yes }
		for_each_loop = { array = _rz_states_ value = _rz_cnt_state
			var:_rz_cnt_state = {
				if = {
					limit = { PREV = { divisions_in_state = { state = PREV size > 0 } } }
					if      = { limit = { PREV = { divisions_in_state = { state = PREV size > 30 } } } ROOT = { add_to_temp_variable = { rsize_presence_ = 35 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 20 } } } ROOT = { add_to_temp_variable = { rsize_presence_ = 25 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 15 } } } ROOT = { add_to_temp_variable = { rsize_presence_ = 17 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 10 } } } ROOT = { add_to_temp_variable = { rsize_presence_ = 12 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 7 } } }  ROOT = { add_to_temp_variable = { rsize_presence_ = 8 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 5 } } }  ROOT = { add_to_temp_variable = { rsize_presence_ = 6 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 3 } } }  ROOT = { add_to_temp_variable = { rsize_presence_ = 4 } } }
					else_if = { limit = { PREV = { divisions_in_state = { state = PREV size > 1 } } }  ROOT = { add_to_temp_variable = { rsize_presence_ = 2 } } }
					else    = { ROOT = { add_to_temp_variable = { rsize_presence_ = 1 } } }
				}
			}
		}
	}

	set_temp_variable = { rsize_demand_ = rsize_presence_ }
	multiply_temp_variable = { rsize_demand_ = constant:wa_ai_railway.corridor.supply_per_division }
	multiply_temp_variable = { rsize_demand_ = constant:wa_ai_railway.corridor.target_ratio }

	### Injection - see _rsize_entry_mode_ in the header for why this is the caller's decision.
	if = {
		limit = { check_variable = { _rsize_entry_mode_ = 2 } }
		### OVERLAND: nothing to cap against. Left at 0, which the cap test below reads as
		### "we hold no entry capacity" and turns into the -1 "cap not applied" state.
		set_temp_variable = { rsize_inject_ = 0 }
	}
	else_if = {
		limit = { check_variable = { _rsize_entry_mode_ = 1 } }
		### GIVEN: an overseas chain in SERIES. The caller passes the bottleneck it already computed
		### (WA_AI_PC_analyze_overseas_supply_chain -> overseas_entry_supply_).
		set_temp_variable = { rsize_inject_ = _rsize_entry_supply_ }
	}
	else = {
		### SUM: what OUR ports on this route can put into the network. Per PROVINCE, never per state
		### (WA_AI_PC_prov_get_naval_base_level's header records why). Injections ADD across nodes
		### (LOGISTICS_MODEL section 7), so this is a sum, not a max.
		###
		### BUILT levels only - a port project in flight is invisible here, deliberately. It under-states
		### future capacity, so the rail target comes out LOWER and rises when the port completes: the
		### safe error direction, because railways cannot be downgraded (LOGISTICS_MODEL section 10
		### rule 3), and it sequences port first, rail catching up. It cannot loop - the
		### province-duplicate guard in WA_AI_PC_start_project admits at most one naval-base project
		### per province at a time. The RAIL side has the opposite convention on purpose
		### (WA_AI_PC_start_railway_project folds _queued_for_segment in before comparing): there the
		### risk is queueing the same segment twice, not sizing it low.
		for_each_loop = { array = _rsize_prov_ index = _rz_inj_i_ value = _rz_inj_prov_
			if = {
				limit = { check_variable = { _rsize_ours_^_rz_inj_i_ = 1 } }
				set_temp_variable = { _rz_inj_state_ = global.WA_AI_MAP_province_state_id^_rz_inj_prov_ }
				if = {
					limit = { check_variable = { _rz_inj_state_ > 0 } }
					var:_rz_inj_state_ = {
						set_temp_variable = { _province_hub_id = _rz_inj_prov_ }
						WA_AI_PC_prov_get_naval_base_level = yes
						set_temp_variable = { _port_level = prov_naval_base_level_ }
						WA_AI_PC_calculate_port_supply = yes
						ROOT = { add_to_temp_variable = { rsize_inject_ = port_supply_ } }
					}
				}
			}
		}
	}

	### When the injection cap must NOT be applied (LOGISTICS_MODEL section 11.5): the network is also
	### fed by land and rsize_inject_ under-states it. Two cases, both written as -1 so the probe can
	### tell "cap not applied" from "we hold no port" - a 0 would conflate them.
	###   (a) we hold no entry capacity at all: min(demand, 0) would pin every target at the floor,
	###       which is a regression, not a measurement.
	###   (b) ROOT's capital sits ON the route: a native power is supplied overland from its capital
	###       (CAPITAL_SUPPLY_BASE), which no port on this list carries. This is also what makes the
	###       helper correct for a land-war route that starts at the capital - the SUM there is zero
	###       or near it, and it is the capital, not a port, that feeds the chain.
	set_temp_variable = { _rz_capped_ = 1 }
	if = { limit = { NOT = { check_variable = { rsize_inject_ > 0 } } }
		set_temp_variable = { _rz_capped_ = 0 }
	}
	# Capital state resolved OUTSIDE the test: WA_AI_PC_get_state_id is an EFFECT and cannot run
	# inside a limit block. state_id_ is pre-zeroed so a tag with no capital cannot match state 0.
	set_temp_variable = { state_id_ = 0 }
	capital_scope = { WA_AI_PC_get_state_id = yes }
	set_temp_variable = { _rz_cap_state_ = state_id_ }
	if = {
		limit = {
			check_variable = { _rz_cap_state_ > 0 }
			is_in_array = { _rz_states_ = _rz_cap_state_ }
		}
		set_temp_variable = { _rz_capped_ = 0 }
	}
	if = { limit = { check_variable = { _rz_capped_ = 0 } }
		set_temp_variable = { rsize_inject_ = -1 }
	}

	### Rail target. The chain is in SERIES (LOGISTICS_MODEL section 10 rule 3), so the useful flow is
	### the minimum of the entry injection and every hop - a level above the weakest link buys nothing.
	set_temp_variable = { _target_supply = rsize_demand_ }
	if = {
		limit = {
			check_variable = { _rz_capped_ = 1 }
			check_variable = { rsize_inject_ < rsize_demand_ }
		}
		set_temp_variable = { _target_supply = rsize_inject_ }
	}
	WA_AI_PC_supply_to_target_railway = yes
	set_temp_variable = { rsize_rail_tgt_ = target_railway_level_ }
	if = { limit = { check_variable = { rsize_rail_tgt_ < _rsize_floor_ } }
		set_temp_variable = { rsize_rail_tgt_ = _rsize_floor_ }
	}
	if = { limit = { check_variable = { rsize_rail_tgt_ > _rsize_cap_ } }
		set_temp_variable = { rsize_rail_tgt_ = _rsize_cap_ }
	}

	### Port target - only when the entry capacity is what binds (LOGISTICS_MODEL section 11.7). Left
	### at 0 otherwise, which the corridor hub loop reads as "place a missing port at level 1".
	### WHETHER to act on this target is decided by the chain-length arbitration in railway_core, not
	### here: this effect runs before the routes are pathfound, so it cannot know N.
	if = {
		limit = {
			check_variable = { _rz_capped_ = 1 }
			check_variable = { rsize_inject_ < rsize_demand_ }
		}
		set_temp_variable = { _target_supply = rsize_demand_ }
		WA_AI_PC_supply_to_target_port = yes
		set_temp_variable = { rsize_port_tgt_ = target_port_level_ }
		if = { limit = { check_variable = { rsize_port_tgt_ > constant:wa_ai_railway.supply.port_max_useful_level } }
			set_temp_variable = { rsize_port_tgt_ = constant:wa_ai_railway.supply.port_max_useful_level }
		}
	}

	### Editing rule 9: this effect's working set does not travel with the rest of the weekly chain.
	clear_temp_array = _rz_states_
}

############################################################################################################
#	HELPER [na-corridor]: size the theatre corridor. The MODEL lives in WA_AI_PC_rail_size_route
#	above, shared with the land-war family; this is only the corridor's binding of it - which array
#	is the route, which nodes' ports count, which floor/cap.
#	ROOT = THIS = builder country. Runs once per corridor pass, AFTER the node classification loop
#	(it reads corridor_node_ours_).
#	Inputs (temp arrays, filled by the caller): corridor_node_prov_, corridor_node_ours_.
#	>>> corridor_presence_, corridor_demand_, corridor_inject_ (-1 = cap not applied),
#	    corridor_rail_tgt_ (floor..cap), corridor_port_tgt_ (0 = no port case open)
############################################################################################################

WA_AI_PC_corridor_compute_sizing = {

		add_to_temp_array = { _rsize_prov_ = _cz_prov_ }
		add_to_temp_array = { _rsize_ours_ = corridor_node_ours_^_cz_i_ }
	}

	# SUM: a corridor is fed by every port along it (see the mode note in the helper's header).
	set_temp_variable = { _rsize_entry_mode_ = 0 }
	set_temp_variable = { _rsize_entry_supply_ = 0 }
	set_temp_variable = { _rsize_floor_ = constant:wa_ai_railway.corridor.rail_level_floor }
	set_temp_variable = { _rsize_cap_ = constant:wa_ai_railway.corridor.rail_level_cap }
	WA_AI_PC_rail_size_route = yes

	set_temp_variable = { corridor_presence_ = rsize_presence_ }
	set_temp_variable = { corridor_demand_ = rsize_demand_ }
	set_temp_variable = { corridor_inject_ = rsize_inject_ }
	set_temp_variable = { corridor_rail_tgt_ = rsize_rail_tgt_ }
	set_temp_variable = { corridor_port_tgt_ = rsize_port_tgt_ }

	clear_temp_array = _rsize_prov_
	clear_temp_array = _rsize_ours_
}

############################################################################################################
#	HELPER ([na-corridor]): queue a theatre-corridor HUB - a supply hub (17) or a naval base (14) - at a
#	named province, tagged type_id corridor under the corridor's scoped budget.
#	ROOT = builder country (country scope). Inputs (temp): _hub_prov_ (province id),
#	_hub_kind_ (17 = supply_node, 14 = naval_base), _hub_prio_ (priority band),
#	_hub_target_level_ (ports only: the level to reach; 1 = "place if missing". ALWAYS set by the
#	caller from the corridor_hub_target_ array - never a chain temp left over from the previous hub).
#	>>> corridor_hub_admitted_ (1 iff WA_AI_PC_start_project grew the queue - the verified effect
#	    the telemetry counts; "we asked" is not "it was admitted": queue_max, the province-duplicate
#	    guard and the controller test can all refuse)
#	Skips silently when the building already exists at that province (a corridor node keeps its
#	attribute forever; the existence test is what makes the request idempotent), when the state
#	mapping is missing, or - for a port - when the state is not coastal.
############################################################################################################

WA_AI_PC_corridor_start_hub = {
	set_temp_variable = { corridor_hub_admitted_ = 0 }
	set_temp_variable = { _hub_state_ = global.WA_AI_MAP_province_state_id^_hub_prov_ }
	if = {
		limit = { check_variable = { _hub_state_ > 0 } }
		var:_hub_state_ = {
			set_temp_variable = { _province_hub_id = _hub_prov_ }
			set_temp_variable = { _hub_wanted_ = 0 }
			if = {
				limit = {
					check_variable = { _hub_kind_ = 17 }
					NOT = { WA_AI_PC_prov_has_supply_node = yes }
				}
				set_temp_variable = { _hub_wanted_ = 1 }
			}
			else_if = {
				limit = {
					check_variable = { _hub_kind_ = 14 }
					is_coastal = yes
				}
				# [na-corridor] a port is RAISED, not merely placed. A bare existence test
				# (`NOT = { WA_AI_PC_prov_has_naval_base = yes }`) means a port someone dropped at
				# level 1 can never be improved by anyone, ever. _hub_target_level_ = 1 reproduces
				# that behaviour exactly (0 < 1 only when absent).
				# THIS level test is the only gate on repetition that works: WA_AI_PC_start_project's
				# _project_target_level check is INERT for building type 14 - it reads
				# final_building_level_ only for types 1 / 2 / 4 (PRIORITY_core.txt), so a naval base
				# always reads 0 there and always passes. The second guard is the province-duplicate
				# test in the same effect (at most one naval-base project per province at a time);
				# since one completed project spawns exactly +1 level, the pair is what makes a
				# multi-level climb one level per pass instead of a loop.
				WA_AI_PC_prov_get_naval_base_level = yes
				if = {
					limit = { check_variable = { prov_naval_base_level_ < _hub_target_level_ } }
					set_temp_variable = { _hub_wanted_ = 1 }
				}
			}
			if = {
				limit = { check_variable = { _hub_wanted_ = 1 } }
				set_temp_variable = { _project_building_type = _hub_kind_ }
				# [levant-iraq] per-corridor tag, same contract as the segment site: derived at the
				# entry of run_one, the only caller of this effect. No unset-temp fallback.
				set_temp_variable = { _project_type_id = _corridor_type_id_ }
				set_temp_variable = { _project_province_id = _hub_prov_ }
				set_temp_variable = { _project_connect_id = 0 }
				set_temp_variable = { _project_priority = _hub_prio_ }
				set_temp_variable = { _project_queue_num = 1 }
				set_temp_variable = { _project_queue_max = constant:wa_ai_railway.corridor.queue_max }
				set_temp_variable = { _project_queue_max_scoped = 1 }
				set_temp_variable = { _project_target_level = 1 }
				set_temp_variable = { _project_target_state = _hub_state_ }

				set_temp_variable = { _hub_q_before_ = ROOT.WA_AI_PC_queue^num }
				WA_AI_PC_start_project = yes
				if = {
					limit = { check_variable = { ROOT.WA_AI_PC_queue^num > _hub_q_before_ } }
					set_temp_variable = { corridor_hub_admitted_ = 1 }
				}

				if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } }
					ROOT = { log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RAILWAY CORRIDOR: hub kind=[?_hub_kind_] at province [?_hub_prov_] admitted=[?corridor_hub_admitted_]" }
				}
			}
		}
	}
}
