### simulate naval training for xp (game AI never training its navy workaround)

WA_AI_train_navy = {

	@TRAINING_AVG_FUEL_CONSUMED_PER_SHIP = 32
	@TRAINING_AVG_XP_GAINED_PER_SHIP = 0.004

	set_temp_variable = { fuel_consumed = @TRAINING_AVG_FUEL_CONSUMED_PER_SHIP }
	multiply_temp_variable = { fuel_consumed = num_ships }
	multiply_temp_variable = { fuel_consumed = -1 }
	add_fuel = fuel_consumed

	set_temp_variable = { xp_gained = @TRAINING_AVG_XP_GAINED_PER_SHIP }
	multiply_temp_variable = { xp_gained = num_ships }
	navy_experience = xp_gained
}

WA_AI_calc_air_force_sizes = {

	### allied air size

	set_temp_variable = { friendly_air_force_size = num_deployed_planes }
	add_to_temp_variable = { friendly_air_force_size = num_equipment@small_fighter_airframe }

	for_each_scope_loop = { array = allies

		add_to_temp_variable = { friendly_air_force_size = num_deployed_planes }
		add_to_temp_variable = { friendly_air_force_size = num_equipment@small_fighter_airframe }
	}

	### enemies air size

	every_enemy_country = {

		set_temp_variable = { air_force_size_@THIS = num_deployed_planes }
		add_to_temp_variable = { air_force_size_@THIS = num_equipment@small_fighter_airframe }

		if = { limit = { is_in_faction = yes NOT = { is_faction_leader = yes } }

			var:faction_leader = {

				add_to_temp_variable = { faction_air_force_size_@THIS = air_force_size_@PREV }
			}
		}
		else_if = { limit = { is_faction_leader = yes }

			add_to_temp_variable = { faction_air_force_size_@THIS = air_force_size_@THIS }
			add_to_temp_array = { enemy_air_force_sizes = THIS.id }
		}
		else = { add_to_temp_array = { enemy_air_force_sizes = THIS.id } }
	}
}

WA_AI_stop_air_production_strategy = {

	if = {
		limit = {
			has_war = yes
			has_capitulated = no
		}

		### abandon air production if overwhelmed by enemy air

		if = {
			limit = {
				has_war = yes
				NOT = { has_country_flag = WA_AI_stop_air_production }
				surrender_progress > 0.3
				is_major = no
				has_deployed_air_force_size = { size < 300 type = fighter }
			}

			WA_AI_calc_air_force_sizes =  yes

			### compare

			for_each_scope_loop = { array = enemy_air_force_sizes

				if = { limit = { is_faction_leader = yes }

					if = { limit = { check_variable = { faction_air_force_size_@THIS > 800 } }

						set_temp_variable = { air_force_size_ratio = friendly_air_force_size }
						divide_temp_variable = { air_force_size_ratio = faction_air_force_size_@THIS }

						if = { limit = { check_variable = { air_force_size_ratio < 0.25 } }

							ROOT = {
								set_country_flag = WA_AI_stop_air_production
								set_country_flag = { flag = WA_AI_stop_air_production_min_duration value = 1 days = 365 }
								set_country_flag = WA_AI_anti_air_design_focus
								clear_variable = WA_AI_no_surrender_progression_months

								if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | AIR PRODUCTION: stopping air production" }
							}
						}
					}
				}
				else = {

					if = { limit = { check_variable = { air_force_size_@THIS > 200 } }

						set_temp_variable = { air_force_size_ratio = friendly_air_force_size }
						divide_temp_variable = { air_force_size_ratio = air_force_size_@THIS }

						if = { limit = { check_variable = { air_force_size_ratio < 0.25 } }

							ROOT = {
								set_country_flag = WA_AI_stop_air_production
								set_country_flag = { flag = WA_AI_stop_air_production_min_duration value = 1 days = 365 }
								set_country_flag = WA_AI_anti_air_design_focus
								clear_variable = WA_AI_no_surrender_progression_months

								if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | AIR PRODUCTION: stopping air production" }
							}
						}
					}
				}
			}
		}

		### resume air production conditions

		else_if = {
			limit = {
				has_country_flag = WA_AI_stop_air_production
				NOT = { has_country_flag = WA_AI_stop_air_production_min_duration }
			}

			# track how long it has had no surrender progress (track change instead?)
			if = { limit = { NOT = { surrender_progress > 0.05 } }

				add_to_variable = { WA_AI_no_surrender_progression_months = 1 }
			}
			else = { clear_variable = WA_AI_no_surrender_progression_months }

			# is not losing anymore or has an airforce
			if = {
				limit = {
					OR = {
						check_variable = { WA_AI_no_surrender_progression_months > 5 }
						has_deployed_air_force_size = { size > 400 type = fighter }
					}
				}

				clr_country_flag = WA_AI_stop_air_production
				if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | AIR PRODUCTION: restarting air production" }
			}

			# none of its enemies have a significant air force remaining
			else = {

				WA_AI_calc_air_force_sizes =  yes

				### compare

				for_each_scope_loop = { array = enemy_air_force_sizes

					if = { limit = { is_faction_leader = yes }

						if = { limit = { check_variable = { faction_air_force_size_@THIS > 800 } }

							set_temp_variable = { break = 1 }
						}
					}
					else = {

						if = { limit = { check_variable = { air_force_size_@THIS > 200 } }

							set_temp_variable = { break = 1 }
						}
					}
				}

				if = { limit = { check_variable = { break = 0 } }

					clr_country_flag = WA_AI_stop_air_production
					if = { limit = { has_country_flag = WA_AI_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | AIR PRODUCTION: restarting air production" }
				}
			}
		}
	}
	else = { clr_country_flag = WA_AI_stop_air_production }
}

WA_AI_defensive_front_strategy = {

	if = { limit = { has_war = yes }

		### critically low on equipment

		if = { limit = { check_variable = { WA_AI_fielded_eq_ratio < 0.6 } }

			if = { limit = { NOT = { has_country_flag = WA_AI_defensive_front_strategy } }

				log = "[GetYear] [GetMonth] | AI | [Root.GetName] | stopping attacks due to low equipment"
			}

			set_country_flag = WA_AI_defensive_front_strategy
		}
		else_if = {
			limit = {
				has_country_flag = WA_AI_defensive_front_strategy
				check_variable = { WA_AI_fielded_eq_ratio > 0.9 }
			}

			log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CANCEL stopping attacks due to low equipment"

			clr_country_flag = WA_AI_defensive_front_strategy
		}

		# Removed: the per-enemy WA_AI_defensive_front_strategy_at array (introduced in bd11fde6b, WIP)
		# was write-only - no ai_strategy block or trigger ever consumed it. Per-enemy defensive posture
		# is owned by the posture system (WA_AI_MILITARY_posture_vs_<TAG>), which has better inputs.
		# Only the country-wide flag above is consumed (WA_AI_MILITARY_DEFAULT_FRONT_control.txt,
		# WA_AI_MILITARY_posture_triggers.txt).
		clear_array = WA_AI_defensive_front_strategy_at
	}
}

WA_AI_calculate_fielded_eq_ratio = {

	### only count equipment that has a major impact in combat

	set_temp_variable = { fielded_eq_ratio = num_equipment_in_armies_k@infantry_equipment }
	divide_temp_variable = { fielded_eq_ratio = num_target_equipment_in_armies_k@infantry_equipment }
	add_to_temp_variable = { tot_fielded_eq_ratio = fielded_eq_ratio }
	add_to_temp_variable = { equipment_count = 1 }

	if = { limit = { check_variable = { num_target_equipment_in_armies_k@heavy_infantry_equipment > 1 } }

		set_temp_variable = { fielded_eq_ratio = num_equipment_in_armies_k@heavy_infantry_equipment }
		divide_temp_variable = { fielded_eq_ratio = num_target_equipment_in_armies_k@heavy_infantry_equipment }
		add_to_temp_variable = { tot_fielded_eq_ratio = fielded_eq_ratio }
		add_to_temp_variable = { equipment_count = 1 }
	}

	if = { limit = { check_variable = { num_target_equipment_in_armies_k@artillery_equipment > 1 } }

		set_temp_variable = { fielded_eq_ratio = num_equipment_in_armies_k@artillery_equipment }
		divide_temp_variable = { fielded_eq_ratio = num_target_equipment_in_armies_k@artillery_equipment }
		add_to_temp_variable = { tot_fielded_eq_ratio = fielded_eq_ratio }
		add_to_temp_variable = { equipment_count = 1 }
	}

	###

	divide_temp_variable = { tot_fielded_eq_ratio = equipment_count }

	set_variable = { WA_AI_fielded_eq_ratio = tot_fielded_eq_ratio }
}

# [overextension-brake] industry overextension hysteresis flag.
# Scope: country (THIS = ROOT = evaluated country). Called MONTHLY from
# on_monthly in common/on_actions/WA_AI_misc_on_actions.txt (is_ai gated there).
# Inputs:  WA_AI_needs_steel/aluminium/rubber/tungsten/chromium (written by
#          WA_AI_check_resource_needs above on a ~2-day cadence, clamped 0-3),
#          resource@<r> post-import balances, factory counts.
# Outputs: country flag WA_AI_industry_overextended_flag. Its LARGEST consumer is the
#          equipment family - all 8 WA_AI_EQUIPMENT_*_headroom_* triggers carry
#          NOT = { WA_AI_industry_overextended = yes }, so a standing flag stops the
#          monthly _armed/_ok latch chain and closes ~640 design sites. Also read by
#          WA_AI_industry_overextended (WA_AI_CONSTRUCTION_triggers.txt) and by
#          WA_AI_RESEARCH_needs_synth_oil/_rubber (WA_AI_RESEARCH_industry.txt);
#          WA_AI_overext_dbg_active - persistent debug counter, +1 per monthly
#          evaluation while the flag is set (savegame-visible probe).
# TRIP    (set flag):   2+ of {steel, aluminium, rubber, tungsten, chromium} at
#                       needs = 3 AND still net-negative AFTER imports; AND
#                       mil factories > civ factories; AND mil+civ > 100.
#                       All state-based - no tags, no is_major: a resource-rich
#                       country grown big never trips, any overextended one does.
# RELEASE (clear flag): fewer than 2 of those resources at needs >= 2 AND still
#                       net-negative - the same conjunction TRIP uses - held for
#                       @WA_AI_OVEREXT_RELEASE_MONTHS consecutive monthly evaluations.
# Between the two thresholds the flag keeps its previous state (hysteresis).
# WHY THE RELEASE TESTS THE BALANCE ([overextension-brake]). It used to test needs alone, and
# WA_AI_needs_<r> ratchets to 3 on the IMPORT-VOLUME leg of WA_AI_check_resource_needs,
# so an import-dependent major never falls below the bar and the flag became a one-way
# latch. WA_AI_EQUIPMENT_triggers.txt refused to read WA_AI_needs_<r> for exactly that
# reason and chose this flag as its fragility guard - and the ratchet came back through
# this release condition. Pairing the balance term cuts it where it re-entered.
# WHY THE PERSISTENCE. The balance sign moves daily while this runs monthly, so the
# paired term alone makes the flag chatter across every downstream reader. The dwell is
# what buys the hysteresis that the needs axis has and the balance axis does not.
# KNOWN RESIDUAL, deliberately not fixed here: about half the remaining state changes
# are TRIP re-sets, not releases. TRIP sits on its own 2-of-5 boundary and spikes there
# for a single month; no release-side term can damp that. Raising the TRIP bar removes
# it, but the reconstruction that showed this is unreliable at exactly trip == 2, so it
# is left alone until that is measured properly. See checklist R24.
# Consecutive monthly evaluations the release condition must hold before the flag clears.
# Protects ~640 equipment design sites from a flag that would otherwise chatter with the
# daily balance sign. Assumes this effect stays monthly; if its cadence changes, this is a
# count of evaluations, not of months. Read by this file only.
@WA_AI_OVEREXT_RELEASE_MONTHS = 3

		set_country_flag = WA_AI_industry_overextended_flag
		set_variable = { WA_AI_overext_release_streak = 0 }
	}
	else_if = {
		limit = {
			has_country_flag = WA_AI_industry_overextended_flag
			check_variable = { _overext_release_ < 2 }
			check_variable = { WA_AI_overext_release_streak = @WA_AI_OVEREXT_RELEASE_MONTHS }
		}

		clr_country_flag = WA_AI_industry_overextended_flag
		set_variable = { WA_AI_overext_release_streak = 0 }
		if = { limit = { has_country_flag = WA_AI_resources_logging } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | RESOURCES: industry overextension released" }
	}

	if = { limit = { has_country_flag = WA_AI_industry_overextended_flag }
		add_to_variable = { WA_AI_overext_dbg_active = 1 }
	}
}

WA_AI_track_submarine_losses = {

	set_temp_variable = {WA_total_subs = num_ships_with_type@submarine}
	add_to_temp_variable = {WA_total_subs = num_ships_with_type@cruiser_submarine}

	subtract_from_variable = { WA_AI_previous_subs = WA_total_subs }

	if = { limit = { check_variable = { WA_AI_previous_subs > 0 } }

		divide_variable = { WA_AI_previous_subs = WA_total_subs }
		if = { limit = { check_variable = { WA_AI_previous_subs > 0.05 } } # lost more than 5%
			set_country_flag = { flag = WA_AI_halt_sub_missions value = 1 days = 14 }
		}
	}

	set_variable = { WA_AI_previous_subs = WA_total_subs }
}

### AI Training Modifier - recalculate weekly

WA_AI_update_training_modifier = {

	# Reset
	set_variable = { WA_AI_minimum_training_level = 0 }
	# Save hygiene: WA_AI_training_time_factor was a modifier slot that never had add logic
	# (born unused in 248aa116c); clear the orphan from old saves.
	clear_variable = WA_AI_training_time_factor

	### DOCTRINE BONUS (one applies)

	# large_front_operations: +0.29 (NOT SOV, NOT historical GER before Sep 1939)
	if = {
		limit = {
			has_doctrine = large_front_operations
			NOT = { original_tag = SOV }
			NOT = {
				AND = {
					original_tag = GER
					is_historical_focus_on = yes
					date < 1939.9.1
				}
			}
		}
		add_to_variable = { WA_AI_minimum_training_level = 0.29 }
	}

	# decisive_strategy: +0.09 (USA only)
	if = {
		limit = {
			has_doctrine = decisive_strategy
			original_tag = USA
		}
		add_to_variable = { WA_AI_minimum_training_level = 0.09 }
	}

	# advanced_squad_tactics: -0.2 (SOV alt-history)
	if = {
		limit = {
			has_doctrine = advanced_squad_tactics
			original_tag = SOV
		}
		add_to_variable = { WA_AI_minimum_training_level = -0.2 }
	}

	### CONSCRIPTION BONUS (highest tier only, if/else_if chain)
	# GER historical exemption wraps entire block

	if = {
		limit = {
			NOT = {
				AND = {
					original_tag = GER
					is_historical_focus_on = yes
					date < 1939.9.1
				}
			}
		}

		if = {
			limit = { has_idea = scraping_the_barrel }
			add_to_variable = { WA_AI_minimum_training_level = 0.7 }
		}
		else_if = {
			limit = { has_idea = all_adults_serve }
			add_to_variable = { WA_AI_minimum_training_level = 0.55 }
		}
		else_if = {
			limit = { has_idea = service_by_requirement }
			add_to_variable = { WA_AI_minimum_training_level = 0.4 }
		}
		else_if = {
			limit = { has_idea = extensive_conscription }
			add_to_variable = { WA_AI_minimum_training_level = 0.25 }
		}
		else_if = {
			limit = { has_idea = limited_conscription }
			add_to_variable = { WA_AI_minimum_training_level = 0.1 }
		}
	}
}
