#	[armor-role-budget] Instrument for the runtime role_ratio budget
#	(common/scripted_effects/WA_AI_PRODUCTION_armor_budget.txt). Nothing in a savegame names a
#	role_ratio strategy by role, and the split is computed from a book that gameplay never reads
#	back, so the only way to see whether the budget is correct is to print it.
#
#	Recipe: events/wa_test_armor_budget.txt header. Fire `event wa_abg.1 SOV` from ANY tag - do NOT
#	tag into the country under test. Output: logs/game.log, "ARMOR BUDGET TEST".
#
#	Section C RE-DERIVES the open-role set from the TEMPLATE triggers and re-does the split with its
#	own arithmetic, on purpose (contract rule 4): it shares no helper with the shipped effect except
#	the budget constant, which IS the bar being measured. When C and D disagree the fault is in one
#	of the two, not hidden between them.
#
#	[armor-budget-ramp] The budget is DATE-dependent (5 / 10 / 15 / 20 / 25 by year), so `budget=`
#	changing between two runs on different years is correct, not a fault. One expected transient:
#	the rung steps on 1 January but the books are rewritten on the next MONTHLY pulse, so a run in
#	the first days of a new year can read offset-equals-budget=0. Re-fire after a month passes
#	before treating it as a failure.
#
#	READ THE VERDICT LINE AT THE BOTTOM. All four values must read 1.
############################################################################################################

	if = { limit = { date > 1939.12.31 } set_temp_variable = { _abt_budget = constant:wa_ai_production.army_composition.armor_budget_1940 } }
	if = { limit = { date > 1940.12.31 } set_temp_variable = { _abt_budget = constant:wa_ai_production.army_composition.armor_budget_1941 } }
	if = { limit = { date > 1941.12.31 } set_temp_variable = { _abt_budget = constant:wa_ai_production.army_composition.armor_budget_total } }
	set_temp_variable = { _abt_wstd = constant:wa_ai_production.army_composition.armor_weight_standard }
	set_temp_variable = { _abt_whvy = constant:wa_ai_production.army_composition.armor_weight_heavy }
	set_temp_variable = { _abt_total = 0 }
	set_temp_variable = { _abt_wsum = 0 }
	set_temp_variable = { _abt_std = 0 }
	set_temp_variable = { _abt_hvy = 0 }
	if = {
		limit = { check_variable = { _abt_n > 0 } }
		set_temp_variable = { _abt_total = _abt_budget }

		# Weight sum, counted from the open flags rather than from a shared helper.
		set_temp_variable = { _abt_nstd = 0 }
		if = { limit = { check_variable = { _abt_wl > 0 } } add_to_temp_variable = { _abt_nstd = 1 } }
		if = { limit = { check_variable = { _abt_wm > 0 } } add_to_temp_variable = { _abt_nstd = 1 } }
		if = { limit = { check_variable = { _abt_wmo > 0 } } add_to_temp_variable = { _abt_nstd = 1 } }
		set_temp_variable = { _abt_wsum = _abt_nstd }
		multiply_temp_variable = { _abt_wsum = _abt_wstd }
		if = { limit = { check_variable = { _abt_wh > 0 } } add_to_temp_variable = { _abt_wsum = _abt_whvy } }

		# Standard share, floored. Re-typed rather than shared: a rounded-up share fails multiply-back.
		set_temp_variable = { _abt_nstd = _abt_budget }
		multiply_temp_variable = { _abt_nstd = _abt_wstd }
		set_temp_variable = { _abt_std = _abt_nstd }
		divide_temp_variable = { _abt_std = _abt_wsum }
		round_temp_variable = _abt_std
		set_temp_variable = { _abt_back = _abt_std }
		multiply_temp_variable = { _abt_back = _abt_wsum }
		if = {
			limit = { check_variable = { _abt_back > _abt_nstd } }
			subtract_from_temp_variable = { _abt_std = 1 }
		}

		# Heavy share, same construction.
		set_temp_variable = { _abt_nhvy = _abt_budget }
		multiply_temp_variable = { _abt_nhvy = _abt_whvy }
		set_temp_variable = { _abt_hvy = _abt_nhvy }
		divide_temp_variable = { _abt_hvy = _abt_wsum }
		round_temp_variable = _abt_hvy
		set_temp_variable = { _abt_back = _abt_hvy }
		multiply_temp_variable = { _abt_back = _abt_wsum }
		if = {
			limit = { check_variable = { _abt_back > _abt_nhvy } }
			subtract_from_temp_variable = { _abt_hvy = 1 }
		}
	}

	set_temp_variable = { _abt_xl = 0 }
	set_temp_variable = { _abt_xm = 0 }
	set_temp_variable = { _abt_xh = 0 }
	set_temp_variable = { _abt_xmo = 0 }
	if = { limit = { check_variable = { _abt_wl > 0 } } set_temp_variable = { _abt_xl = _abt_std } }
	if = { limit = { check_variable = { _abt_wm > 0 } } set_temp_variable = { _abt_xm = _abt_std } }
	if = { limit = { check_variable = { _abt_wmo > 0 } } set_temp_variable = { _abt_xmo = _abt_std } }
	if = { limit = { check_variable = { _abt_wh > 0 } } set_temp_variable = { _abt_xh = _abt_hvy } }

	set_temp_variable = { _abt_rem = _abt_total }
	subtract_from_temp_variable = { _abt_rem = _abt_xl }
	subtract_from_temp_variable = { _abt_rem = _abt_xm }
	subtract_from_temp_variable = { _abt_rem = _abt_xh }
	subtract_from_temp_variable = { _abt_rem = _abt_xmo }
	if = { limit = { check_variable = { _abt_wmo > 0 } } add_to_temp_variable = { _abt_xmo = _abt_rem } }
	else_if = { limit = { check_variable = { _abt_wm > 0 } } add_to_temp_variable = { _abt_xm = _abt_rem } }
	else_if = { limit = { check_variable = { _abt_wl > 0 } } add_to_temp_variable = { _abt_xl = _abt_rem } }
	else_if = { limit = { check_variable = { _abt_wh > 0 } } add_to_temp_variable = { _abt_xh = _abt_rem } }

	# [light-support-conversion] Same tank-park light floor as the shipped effect, RE-TYPED
	# (contract rule 4) INCLUDING the gate - spelled from its six terms rather than calling
	# WA_AI_PRODUCTION_should_floor_tank_park_budget, so a gate defect shows as a C/D mismatch
	# instead of reproducing in the expect walk.
	set_temp_variable = { _abt_floor = 0 }
	if = {
		limit = {
			check_variable = { _abt_wl > 0 }
			WA_AI_PRODUCTION_army_composition_is_enabled = yes
			WA_AI_DIFFICULTY_is_historical = yes
			WA_AI_CONFIG_pursues_historical_tank_park = yes
			NOT = { any_enemy_country = { is_major = yes } }
			NOT = { WA_AI_PRODUCTION_light_support_armor_at_fielded_cap = yes }
			WA_AI_TEMPLATES_use_light_support_armor_templates = yes
			check_variable = { _abt_xl < constant:wa_ai_production.army_composition.tank_park_light_floor }
		}
		set_temp_variable = { _abt_floor = constant:wa_ai_production.army_composition.tank_park_light_floor }
		subtract_from_temp_variable = { _abt_floor = _abt_xl }
		add_to_temp_variable = { _abt_total = _abt_floor }
		set_temp_variable = { _abt_xl = constant:wa_ai_production.army_composition.tank_park_light_floor }
	}

	log = "  expect: budget=[?_abt_budget|.0]  open-roles=[?_abt_n|.0]  weight-sum=[?_abt_wsum|.0](std [?_abt_wstd|.0] / heavy [?_abt_whvy|.0])  std-share=[?_abt_std|.0]  heavy-share=[?_abt_hvy|.0]  leftover=[?_abt_rem|.0]  park-floor-add=[?_abt_floor|.0]"
	log = "  expect: light=[?_abt_xl|.0] medium=[?_abt_xm|.0] heavy=[?_abt_xh|.0] modern=[?_abt_xmo|.0] infantry-offset=[?_abt_total|.0]"

	############################################################
	# D. What the shipped effect actually applied - its own books
	############################################################

	set_temp_variable = { _abt_bl = 0 }
	if = { limit = { has_variable = WA_AI_ARMOR_BUDGET_light } set_temp_variable = { _abt_bl = WA_AI_ARMOR_BUDGET_light } }
	set_temp_variable = { _abt_bm = 0 }
	if = { limit = { has_variable = WA_AI_ARMOR_BUDGET_medium } set_temp_variable = { _abt_bm = WA_AI_ARMOR_BUDGET_medium } }
	set_temp_variable = { _abt_bh = 0 }
	if = { limit = { has_variable = WA_AI_ARMOR_BUDGET_heavy } set_temp_variable = { _abt_bh = WA_AI_ARMOR_BUDGET_heavy } }
	set_temp_variable = { _abt_bmo = 0 }
	if = { limit = { has_variable = WA_AI_ARMOR_BUDGET_modern } set_temp_variable = { _abt_bmo = WA_AI_ARMOR_BUDGET_modern } }
	set_temp_variable = { _abt_binf = 0 }
	if = { limit = { has_variable = WA_AI_ARMOR_BUDGET_inf } set_temp_variable = { _abt_binf = WA_AI_ARMOR_BUDGET_inf } }

	set_temp_variable = { _abt_bsum = _abt_bl }
	add_to_temp_variable = { _abt_bsum = _abt_bm }
	add_to_temp_variable = { _abt_bsum = _abt_bh }
	add_to_temp_variable = { _abt_bsum = _abt_bmo }

	log = "  books : light=[?_abt_bl|.0] medium=[?_abt_bm|.0] heavy=[?_abt_bh|.0] modern=[?_abt_bmo|.0]  sum=[?_abt_bsum|.0]  infantry-offset=[?_abt_binf|.0]"

	############################################################
	# E. Verdicts - all three must read 1
	############################################################

	# V1 - the constancy claim: whatever the number of open roles, the tank rows sum to the infantry
	# offset. A 0 here means infantry is paying more (or less) than the tanks receive.
	set_temp_variable = { _abt_v1 = 0 }
	if = { limit = { check_variable = { _abt_bsum = _abt_binf } } set_temp_variable = { _abt_v1 = 1 } }

	# V2 - the budget claim: with at least one open role the offset IS the budget, and with none it
	# is zero. A 0 here with V1 at 1 means the split is internally consistent but the wrong size.
	set_temp_variable = { _abt_v2 = 0 }
	if = { limit = { check_variable = { _abt_binf = _abt_total } } set_temp_variable = { _abt_v2 = 1 } }

	# V3 - the reconciliation claim: the books match an independent re-derivation from the template
	# layer. A 0 here is the interesting failure - a slot trigger and its template trigger disagree,
	# or a reconcile tick was missed.
	set_temp_variable = { _abt_v3 = 0 }
	if = {
		limit = {
			check_variable = { _abt_bl = _abt_xl }
			check_variable = { _abt_bm = _abt_xm }
			check_variable = { _abt_bh = _abt_xh }
			check_variable = { _abt_bmo = _abt_xmo }
		}
		set_temp_variable = { _abt_v3 = 1 }
	}

	# V4 - [modern-chassis-tier] the chassis-tier claim: the carried template value sits in exactly
	# one band, and that band is the one the latch asked for. A 0 with the medium role open is the
	# interesting failure - either the mirror is missing the value the ladder produced (regenerate
	# with tools/gen_ai_medium_modern_mirror.py), or the latch and the offset disagree.
	set_temp_variable = { _abt_v4 = 1 }
	if = {
		limit = { check_variable = { _abt_tm > 0 } }
		set_temp_variable = { _abt_v4 = 0 }
		if = {
			limit = {
				check_variable = { _abt_bandmed = 1 }
				check_variable = { _abt_bandmod = 0 }
				check_variable = { _abt_modowns = 0 }
			}
			set_temp_variable = { _abt_v4 = 1 }
		}
		if = {
			limit = {
				check_variable = { _abt_bandmed = 0 }
				check_variable = { _abt_bandmod = 1 }
				check_variable = { _abt_modowns = 1 }
			}
			set_temp_variable = { _abt_v4 = 1 }
		}
	}

	log = "  VERDICT: sum-equals-offset=[?_abt_v1|.0]  offset-equals-budget=[?_abt_v2|.0]  books-match-independent-split=[?_abt_v3|.0]  chassis-tier-consistent=[?_abt_v4|.0]   (all four must read 1)"
