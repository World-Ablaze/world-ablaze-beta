# The Subjects (Shinmin), one of Japan's domestic factions (dynamic modifier JAP_shinmin).
# Attitude: 0 is neutral, each step up or down is worth 0.05% weekly war support. Kept between -5 and 5.
# Influence: eight levels like the other factions, Total (7) at the start. The more of it, the less political power:
# Moderate (3) is neutral, each level above costs 0.05 political power, each level below gives 0.05.
JAP_shinmin_attitude_up_effect = {
	custom_effect_tooltip = JAP_shinmin_attitude_up_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_attitude < 5 } }
			add_to_variable = { JAP_shinmin_attitude = 1 }
			add_to_variable = { JAP_shinmin_war_support_weekly = 0.0005 }
		}
		JAP_faction_decision_days_update = yes
	}
}

JAP_shinmin_attitude_down_effect = {
	custom_effect_tooltip = JAP_shinmin_attitude_down_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_attitude > -5 } }
			add_to_variable = { JAP_shinmin_attitude = -1 }
			add_to_variable = { JAP_shinmin_war_support_weekly = -0.0005 }
		}
		JAP_faction_decision_days_update = yes
	}
}

JAP_shinmin_influence_up_effect = {
	custom_effect_tooltip = JAP_shinmin_influence_up_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_influence < 7 } }
			add_to_variable = { JAP_shinmin_influence = 1 }
			add_to_variable = { JAP_shinmin_political_power_gain = -0.05 }
		}
		JAP_faction_decision_days_update = yes
	}
}

JAP_shinmin_influence_down_effect = {
	custom_effect_tooltip = JAP_shinmin_influence_down_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_influence > 0 } }
			add_to_variable = { JAP_shinmin_influence = -1 }
			add_to_variable = { JAP_shinmin_political_power_gain = 0.05 }
		}
		JAP_faction_decision_days_update = yes
	}
}
