# The Subjects (Shinmin), one of Japan's domestic factions. One value, the attitude: 0 is neutral, and each step up or down
# is worth 0.05% weekly war support and 0.05 political power (dynamic modifier JAP_shinmin). Kept between -5 and 5.
JAP_shinmin_attitude_up_effect = {
	custom_effect_tooltip = JAP_shinmin_attitude_up_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_attitude < 5 } }
			add_to_variable = { JAP_shinmin_attitude = 1 }
			add_to_variable = { JAP_shinmin_war_support_weekly = 0.0005 }
			add_to_variable = { JAP_shinmin_political_power_gain = 0.05 }
		}
	}
}

JAP_shinmin_attitude_down_effect = {
	custom_effect_tooltip = JAP_shinmin_attitude_down_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_shinmin_attitude > -5 } }
			add_to_variable = { JAP_shinmin_attitude = -1 }
			add_to_variable = { JAP_shinmin_war_support_weekly = -0.0005 }
			add_to_variable = { JAP_shinmin_political_power_gain = -0.05 }
		}
	}
}
