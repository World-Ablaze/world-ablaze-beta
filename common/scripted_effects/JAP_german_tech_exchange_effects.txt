# German technology exchange (JAP_gte_technology_exchange decision). Country scope: JAP.
#
# A round trip of timed missions (JAP_german_tech_exchange.txt). Outbound: Asia, then either
# Red Sea and Mediterranean (via Suez) or Cape and Atlantic. On reaching Germany, Germany gets
# a random naval research bonus and navy experience (jap_gte.4) and the return route is chosen
# again. Return: the same legs in reverse; reaching Japan gives a random armour research bonus
# and army experience (jap_gte.2). Each leg rolls when its timer runs out: 95% to go on if every
# point of that leg is held, 70% if any is missing, so six rolls end up near 74% / 12% overall.
# Losing a roll ends the trip (jap_gte.3). Leg days follow sea-lane distance at one common speed:
#   via Suez 47 days each way, round the Cape 79 days each way.
# Japan also gains 30 political power on reaching Germany.
# The resolve effects are shared by the missions' timeout_effect and the debug decisions.

# Departure: the outbound route depends on the Suez Canal now.
JAP_gte_depart = {
	hidden_effect = {
		set_country_flag = JAP_gte_in_transit
		if = {
			limit = { JAP_gte_held_suez_canal = yes }
			set_country_flag = JAP_gte_via_suez
		}
		else = {
			clr_country_flag = JAP_gte_via_suez
		}
	}
	activate_mission = JAP_gte_out_asia
}

# Arrival in Germany: reward Germany, then head home by whichever route is open now.
JAP_gte_arrive_germany = {
	if = {
		limit = {
			country_exists = GER
			GER = { has_capitulated = no }
			NOT = { has_war_with = GER }
		}
		add_political_power = 30
		GER = { country_event = { id = jap_gte.4 } }
		if = {
			limit = { JAP_gte_held_suez_canal = yes }
			activate_mission = JAP_gte_ret_mediterranean
		}
		else = {
			activate_mission = JAP_gte_ret_atlantic
		}
	}
	else = {
		country_event = { id = jap_gte.3 }
	}
}

JAP_gte_out_asia_resolve = {
	if = {
		limit = {
			JAP_gte_held_ceylon = yes
			JAP_gte_held_singapore = yes
			JAP_gte_held_taiwan = yes
		}
		random_list = {
			95 = {
				if = {
					limit = { has_country_flag = JAP_gte_via_suez }
					activate_mission = JAP_gte_out_red_sea
				}
				else = {
					activate_mission = JAP_gte_out_cape
				}
			}
			5 = { country_event = { id = jap_gte.3 } }
		}
	}
	else = {
		random_list = {
			70 = {
				if = {
					limit = { has_country_flag = JAP_gte_via_suez }
					activate_mission = JAP_gte_out_red_sea
				}
				else = {
					activate_mission = JAP_gte_out_cape
				}
			}
			30 = { country_event = { id = jap_gte.3 } }
		}
	}
}

# Removes whichever leg is under way (debug decisions).
JAP_gte_remove_legs = {
	if = { limit = { has_active_mission = JAP_gte_out_asia } remove_mission = JAP_gte_out_asia }
	if = { limit = { has_active_mission = JAP_gte_out_red_sea } remove_mission = JAP_gte_out_red_sea }
