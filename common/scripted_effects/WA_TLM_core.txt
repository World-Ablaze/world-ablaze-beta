############################################################################################################
#	Clock - global.WA_TLM_clock = months elapsed since 1936.1
#	Scope: any country (called from the per-country monthly pulse; the timed global
#	flag guarantees exactly one increment per calendar month).
############################################################################################################
WA_TLM_tick_clock = {
	if = { limit = { NOT = { has_global_flag = WA_TLM_clock_ticked } }
		set_global_flag = { flag = WA_TLM_clock_ticked value = 1 days = 20 }
		add_to_variable = { global.WA_TLM_clock = 1 }
	}
}

############################################################################################################
#	Init - PRESERVING zero-init: each metric is created at 0 only if absent, then
#	the version is stamped. Scope: country. Called at startup for every country and
#	lazily from the monthly sampler for tags created mid-game (civil wars, releases).
#
#	The has_variable guards are load-bearing, not defensive noise: counters write at
#	event sites at ANY cadence, so a mid-game tag can accumulate counter values
#	BEFORE its first monthly pulse runs this init (e.g. the weekly-written
#	WA_TLM_pc_aging_* pair) - an unconditional zero-init would wipe them. Same
#	mechanism preserves ring-buffer history across a version-bump re-init on
#	resumed campaigns. On a fresh start nothing exists and the guards are no-ops.
#	The zero-init is the absence contract: namespace absent = pre-TLM build (or a
#	mid-game tag inside its first month); metric 0 with _last_t 0 = never sampled;
#	metric 0 with _last_t > 0 = real zero.
############################################################################################################
WA_TLM_init_country = {
	set_variable = { WA_TLM_version = @WA_TLM_VERSION }

	# composition (pilot, consumer: checklist R6 / F8)
	if = { limit = { NOT = { has_variable = WA_TLM_comp_div_total } } set_variable = { WA_TLM_comp_div_total = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_armor } } set_variable = { WA_TLM_comp_armor = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_mech } } set_variable = { WA_TLM_comp_mech = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_armor_mech_pct } } set_variable = { WA_TLM_comp_armor_mech_pct = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_last_t } } set_variable = { WA_TLM_comp_last_t = 0 } }
	# ring buffers: no clear - arrays create themselves on first add_to_array, and
	# clearing on re-init is exactly the history wipe the guards exist to prevent

	# PC overtake lane (Fix 41, consumer: checklist R26; write sites in
	# WA_AI_PC_assign_factories, WA_AI_CONSTRUCTION_PRIORITY_core.txt)
	if = { limit = { NOT = { has_variable = WA_TLM_pc_aging_grants } } set_variable = { WA_TLM_pc_aging_grants = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_aging_reval_cancels } } set_variable = { WA_TLM_pc_aging_reval_cancels = 0 } }

	# PC termination ledger (standing, consumer: any "why did the queue shrink" question).
	# One counter per way a project can leave WA_AI_PC_queue, each written at its own decision
	# site and NOT by the monthly sampler below:
	#   pc_built_n      WA_AI_PC_add_finished_building_by_id  (a building was spawned)
	#   pc_refused_n    WA_AI_PC_complete_project_by_id       (fully paid for, denied the building)
	#   pc_sweep_n      WA_AI_PC_update_project_progress      (30-week stall sweep)
	#   pc_orphan_n     WA_AI_PC_update_project_progress      (target_state 0 / cost <= 0)
	#   pc_stale_n      WA_AI_PC_validate_queued_provincial_projects
	#   pc_lost_n       on_state_control_changed              (ground changed hands)
	#   pc_peace_n      on_peace                              (Fix 5/19 railway purge)
	#   pc_aging_reval_cancels                                (above, Fix 41 overtake lane)
	# Together these are EVERY caller of WA_AI_PC_end_project_by_id and WA_AI_PC_cancel_projects
	# outside the WA_TEST_ harness, so their sum is the total number of departures. A gap
	# between that sum and the observed queue shrinkage means a new exit path was added
	# without its counter.
	# pc_built_by_type is INDEXED by building type code (1..16) and is deliberately NOT
	# zero-initialised: sixteen init lines per country to say "never built one" is the wrong
	# trade when pc_built_n beside it is the family's presence witness. An absent index means
	# this country has never completed that type.
	if = { limit = { NOT = { has_variable = WA_TLM_pc_built_n } } set_variable = { WA_TLM_pc_built_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_refused_n } } set_variable = { WA_TLM_pc_refused_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_refused_ctrl_n } } set_variable = { WA_TLM_pc_refused_ctrl_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_sweep_n } } set_variable = { WA_TLM_pc_sweep_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_orphan_n } } set_variable = { WA_TLM_pc_orphan_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_stale_n } } set_variable = { WA_TLM_pc_stale_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_lost_n } } set_variable = { WA_TLM_pc_lost_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_peace_n } } set_variable = { WA_TLM_pc_peace_n = 0 } }

	# PC queue + allocation gauges (standing, consumer: the civ-factory share questions).
	# Written by WA_TLM_sample_pc below. pc_civs_avail is the one reading that exists nowhere
	# else: the engine's num_of_civilian_factories_available_for_projects is not serialized
	# into a savegame, so without this metric the allocation base can only be guessed at.
	if = { limit = { NOT = { has_variable = WA_TLM_pc_queue } } set_variable = { WA_TLM_pc_queue = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_civs } } set_variable = { WA_TLM_pc_civs = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_civs_avail } } set_variable = { WA_TLM_pc_civs_avail = 0 } }

	if = { limit = { NOT = { has_variable = WA_TLM_pc_alloc_base } } set_variable = { WA_TLM_pc_alloc_base = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_share_pct } } set_variable = { WA_TLM_pc_share_pct = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_avail_share_pct } } set_variable = { WA_TLM_pc_avail_share_pct = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_last_t } } set_variable = { WA_TLM_pc_last_t = 0 } }

	# WA_AI_PC_start_project and cleared by WA_AI_PC_end_project_by_id. Indexed, so no init

############################################################################################################
#	Priority-construction sample (standing; §3.8 criteria 2 and 3).
#
#	What this can and cannot see, and why the split falls where it does. The QUEUE itself is
#	fully readable from a savegame - it is a country-scope array plus ~11 parallel indexed
#	families - and `savegame.py pc TAG FILE...` decodes it, per project, with the building
#	type and priority band by name. Duplicating that here as telemetry would be pure cost.
#
#	Two things the save cannot give up, and they are all this sampler covers:
#	  (a) num_of_civilian_factories_available_for_projects. The engine never serializes it.
#	      A save can approximate total civ industry by summing industrial_complex levels over
#	      controlled states, but that is an upper bound on num_of_civilian_factories and says
#	      nothing about how much of it the vanilla construction queue had already absorbed -
#	      which IS the number WA_AI_PC_assign_factories multiplies by @AI_PC_ALLOC_FRACTION.
#	      "PC is using 7% of civ industry" and "PC took its full 40% of what it was offered"
#	      are both true of the same month and only the second one is actionable.
#	  (b) A trend inside one save (the ring buffer). Monthly saves already carry a queue
#	      snapshot each, so the multi-save sweep works - it just costs a ~100 MB streaming
#	      parse per save per question.
#
#	Cost: eight variable reads, two divides, and quarterly three array pushes for majors.
#	Cheaper than the composition ladder next door (26 has_army_size evaluations). Nothing is
#	written on the weekly PC pulse itself - the termination counters live on paths that only
#	run when a project actually leaves the queue, never inside the per-project queue loops,
#	which are the system's hot path.
#
#	Token risk: none. num_of_civilian_factories and
#	num_of_civilian_factories_available_for_projects are both read by
#	WA_AI_PC_assign_factories itself (WA_AI_CONSTRUCTION_PRIORITY_core.txt), so a bad token
#	here would have broken the allocator years ago. F9 boot check is still owed and is
#	correspondingly cheap: on any 1936 AI country, wa_tlm_pc_civs > 0 and wa_tlm_pc_last_t
#	present.
#
#	Scope: country, is_ai gated by the caller.
############################################################################################################
WA_TLM_sample_pc = {
	# tlm: pc_queue / pc_civs / pc_civs_avail / pc_assigned - gauges, all AI countries -
	# consumer: standing (queue depth and the civ-factory share questions)
	set_variable = { WA_TLM_pc_queue = WA_AI_PC_queue^num }
	set_variable = { WA_TLM_pc_civs = num_of_civilian_factories }
	set_variable = { WA_TLM_pc_civs_avail = num_of_civilian_factories_available_for_projects }
	set_variable = { WA_TLM_pc_assigned = WA_AI_PC_assigned_factories_total }

	# tlm: pc_alloc_base - gauge. The pool WA_AI_PC_assign_factories reconstructs before it
	# applies the fraction: currently-free civs PLUS the ones PC is already holding (the
	# allocator adds its own assignments back precisely because they are not "free" until it
	# releases them). This is the honest denominator for "share of AVAILABLE civs".
	# It deliberately does NOT model the @AI_PC_STABLE_BASE_FRACTION floor or the
	# @AI_PC_ALLOC_HARD_CAP_FRACTION ceiling: both are functions of pc_civs, which is sampled
	# beside it, so an analyst can apply them - and baking them in would hide the case where
	# the floor is what is actually funding the queue.
	set_variable = { WA_TLM_pc_alloc_base = WA_TLM_pc_civs_avail }
	add_to_variable = { WA_TLM_pc_alloc_base = WA_TLM_pc_assigned }

	# tlm: pc_share_pct - derived gauge, 100 * assigned / civs. Share of the country's whole
	# civilian industry that this shadow construction system is consuming.
	set_temp_variable = { _tlm_pcs_ = WA_TLM_pc_assigned }
	multiply_temp_variable = { _tlm_pcs_ = 100 }
	if = { limit = { check_variable = { WA_TLM_pc_civs > 0 } }
		divide_temp_variable = { _tlm_pcs_ = WA_TLM_pc_civs }
	}
	else = {
		set_temp_variable = { _tlm_pcs_ = 0 }
	}
	set_variable = { WA_TLM_pc_share_pct = _tlm_pcs_ }

	# tlm: pc_avail_share_pct - derived gauge, 100 * assigned / alloc_base. THE diagnostic
	# number of this family. Read it against the fraction actually in force that month
	# (@AI_PC_ALLOC_FRACTION = 40, or WA_AI_PC_override_max_factories_factor x100 while its
	# country flag is live - `savegame.py pc` prints which):
	#   at the fraction        -> PC took everything it was allowed; the queue is pool-bound
	#   well under it          -> the fill could not place factories: no eligible project
	#                             (hostile-controlled targets, Fix 34) or an empty queue
	#   over it                -> the stable-base floor is carrying the queue, i.e. the
	#                             vanilla construction queue has absorbed the free civs
	set_temp_variable = { _tlm_pca_ = WA_TLM_pc_assigned }
	multiply_temp_variable = { _tlm_pca_ = 100 }
	if = { limit = { check_variable = { WA_TLM_pc_alloc_base > 0 } }
		divide_temp_variable = { _tlm_pca_ = WA_TLM_pc_alloc_base }
	}
	else = {
		set_temp_variable = { _tlm_pca_ = 0 }
	}
	set_variable = { WA_TLM_pc_avail_share_pct = _tlm_pca_ }

	# tlm: pc_last_t - sample stamp (absence contract, doc 3.5). Written with the widest
	# gauges of the family - all AI countries - never inside the major gate below, so a
	# minor's genuine "queue empty, zero factories" does not read as "never sampled".
	set_variable = { WA_TLM_pc_last_t = global.WA_TLM_clock }

	# Quarterly ring buffer on its OWN axis (depth 44 = 11 game-years). It must not share
	# WA_TLM_hist_t: that axis is pushed by the composition sampler, and although the gate
	# below happens to be the same trigger today, a shared axis makes the two families'
	# gates a permanent coupling - any later divergence desynchronises index i for every
	# series on it (doc 4, risk ii). The gauges above stay ungated; only the history is
	# major-only, which is where the save-weight is.
	if = { limit = { WA_AI_CONFIG_is_major_country = yes }
		set_temp_variable = { _tlm_pcq_ = global.WA_TLM_clock }
		modulo_temp_variable = { _tlm_pcq_ = 3 }
		if = { limit = { check_variable = { _tlm_pcq_ = 0 } }
			add_to_array = { array = WA_TLM_pc_hist_t value = global.WA_TLM_clock }
			add_to_array = { array = WA_TLM_pc_queue_hist value = WA_TLM_pc_queue }
			add_to_array = { array = WA_TLM_pc_share_pct_hist value = WA_TLM_pc_share_pct }
			add_to_array = { array = WA_TLM_pc_avail_share_pct_hist value = WA_TLM_pc_avail_share_pct }
			if = { limit = { check_variable = { WA_TLM_pc_hist_t^num > 44 } }
				remove_from_array = { array = WA_TLM_pc_hist_t index = 0 }
				remove_from_array = { array = WA_TLM_pc_queue_hist index = 0 }
				remove_from_array = { array = WA_TLM_pc_share_pct_hist index = 0 }
				remove_from_array = { array = WA_TLM_pc_avail_share_pct_hist index = 0 }
			}
		}
	}
}

#	      a readable form in the save;
#	  (b) an INDEPENDENT reading of "the escorts are sitting in port", via
#	      ships_in_state_ports, which does not depend on the empirically-derived
#	      mission-id map the `navy` command relies on. If the two disagree, the
#	      mission map is what is wrong.
#
#	Scope: country, is_ai gated by the caller. Cheap gauges run for anyone with a
#	hull; the per-state port ladder is gated to the six naval majors (doc 3.4).
############################################################################################################
WA_TLM_sample_navy = {
	if = { limit = { has_navy_size = { size > 0 } }

		# tlm: nav_ships / nav_screens / nav_subs / nav_convoys - gauges, any AI navy
		# - consumer: R36 (standing).
		#
		# num_ships_with_type@ takes a SUB-UNIT DEF TYPE or one of the four engine
		# aggregates carrier / capital / screen / submarine (dynamic_variables_
		# documentation.md). It does NOT take `screen_ship` - that is the sub-unit
		# CATEGORY name from common/units/ship_*.txt - and `convoy` is not a ship at
		# all (convoys are equipment; there is no common/units/ship_convoy.txt).
		# Both wrong tokens returned a silent 0 on every country through v4, which
		# also forced nav_port_pct and its whole ring buffer to the else = 0 branch.
		# Verified in-save on campaign 02bd4445: @submarine and num_ships are exact
		# to the hull, @screen_ship and @convoy read 0 on ENG/USA/GER/JAP.
		#
		# nav_screens is summed from the three concrete hull types rather than the
		# @screen aggregate on purpose: destroyer + frigate + light_cruiser is the
		# exact set carrying `type = screen_ship` in this mod, and it is also what
		# savegame.py's `navy` command counts as `screens=` (_NAVAL_SCREENS), so
		# R36's "nav_screens must equal the command's screens= exactly" cross-check
		# stays an identity instead of depending on an engine aggregate's membership.
		set_variable = { WA_TLM_nav_ships = num_ships }
		set_variable = { WA_TLM_nav_screens = num_ships_with_type@destroyer }
		add_to_variable = { WA_TLM_nav_screens = num_ships_with_type@frigate }
		add_to_variable = { WA_TLM_nav_screens = num_ships_with_type@light_cruiser }
		set_variable = { WA_TLM_nav_subs = num_ships_with_type@submarine }

		# tlm: r64_csubs - gauge, any AI with a hull - consumer: R41 (Fix 64, SOV routed from the
		# naval_submarine role to the cruiser_submarine role). Read against WA_TLM_nav_subs, which
		# is the other half of the same question: the fix passes when SOV's csubs climb while its
		# subs stay flat at the 1936 OOB.
		# `cruiser_submarine` is a sub-unit def type (common/units/ship_cruiser_submarine.txt) and
		# is listed as a legal num_ships_with_type@ target in doc 5.
		# SETTLED 2026-08-14 on campaign be18f9c7: nav_subs above IS inclusive of these hulls, so
		# this metric is a SHARE of nav_subs, not an independent series. Both sub-unit defs carry
		# `type = { submarine }` and `submarine` is simultaneously a def name and one of the four
		# engine aggregates - the aggregate wins. Measured against hull-resolved fleets: SOV
		# nav_subs 101 = 93 cruiser + 8 plain, GER 18 = 14 + 4, r64_csubs matching the cruiser
		# count exactly in both. Plain hulls = nav_subs - r64_csubs; never read nav_subs as the
		# plain-boat count.
		# SECOND SIGNAL (doc 7.5), and it is free: GER has been routed to the cruiser_submarine
		# role since before this fix, so GER is the built-in control. GER > 0 with SOV = 0 means
		# the token is fine and the fix failed; BOTH 0 means the token is bad and neither reading
		# may be scored - the v4 nav_screens failure mode.
		set_variable = { WA_TLM_r64_csubs = num_ships_with_type@cruiser_submarine }

		# tlm: nav_cv_f / nav_cv_n / nav_cv_hulls - gauges, any AI with a hull - consumer:
		# R43 (Fix 66, the carrier-plane saturation brake). These are the EXACT three
		# quantities Fix 66's gate compares, sampled at the same values the gate sees, and
		# they exist to settle one question the outcome alone cannot answer.
		#
		# Fix 66 fails SAFE: an unresolvable token reads 0, 0 never exceeds a bar, the brake
		# never trips and behaviour is unchanged - which is indistinguishable from "the brake
		# armed and the AI over-built anyway" if you only look at the plane counts. These
		# three separate the cases outright:
		#   nav_cv_f = nav_cv_n = 0 on a carrier major in a war year  -> the token is bad,
		#     the brake was never live, and NOTHING about Fix 66 may be scored (this is the
		#     v4 nav_screens failure mode repeating, on the same class of token).
		#   nav_cv_f / _n populated                                   -> the gate saw real
		#     numbers; compare them against the bars for nav_cv_hulls to say whether the
		#     brake SHOULD have tripped, then read the production lines to say whether it did.
		# nav_cv_hulls is also the independent check on the hull banding: it is the same
		# `type = carrier` aggregate the gate bands on. SETTLED 2026-08-14 on campaign be18f9c7 -
		# the aggregate DOES count light carriers: JAP 21 = 16 fleet + 5 light, ENG 20 = 18 + 2,
		# USA 52 = 52 + 0. So a light deck bands at the weight of a fleet deck and the per-hull
		# bars run loose for a light-heavy fleet - it under-brakes, which is the safe direction.
		# The plane tokens are archetype-spelled, the idiom already live at
		# common/decisions/ITA.txt:957 on this very archetype. UNVERIFIED nonetheless.
		set_variable = { WA_TLM_nav_cv_f = num_deployed_planes_with_type@cv_small_fighter_airframe }
		set_variable = { WA_TLM_nav_cv_n = num_deployed_planes_with_type@cv_small_naval_bomber_airframe }
		set_variable = { WA_TLM_nav_cv_hulls = num_ships_with_type@carrier }
		# convoy is an equipment archetype, so it is read with num_equipment@, the
		# idiom used for every other archetype in this repo (WA_AI_DIVISION_CREATOR_
		# effects.txt, 100_wa_on_actions.txt). UNVERIFIED IN-GAME until the F9 boot
		# check below - if it reads 0 on ENG in a war year while nav_ships is
		# non-zero, the token filter failed again and the metric must be dropped,
		# not scored. Same discipline as the composition ladder's boot-check note.
		set_variable = { WA_TLM_nav_convoys = num_equipment@convoy }

		# tlm: nav_conv_threat / nav_conv_ws - gauges, any AI navy - consumer: R36
		# (standing). Both tokens ARE valid variable reads (dynamic_variables_
		# documentation.md lists both, auto-generated from the same-named triggers);
		# both engine values are 0-1, stored x100 so the save shows integers-ish.
		# conv_threat = how dangerous our convoy routes are.
		# conv_ws = the war-support malus our sunk convoys have already cost us.
		# READING NOTE for conv_ws, which read 0 on all four majors on 02bd4445
		# after 11 500 sinkings: that is a REAL zero, not a bad token. The engine
		# penalty is a rate, not a lifetime tally - it accrues at most
		# CONVOYS_BEING_RAIDED_WAR_SUPPORT_PENALTY_SCALE / WEEKLY (-0.01) per week
		# from convoys being raided RIGHT NOW and decays 0.025/week
		# (05_defines.lua:134-137), capped at -0.3 total. Observable range is
		# therefore 0-30 in this metric's units, and it returns to ~0 within a
		# couple of months of the raiding stopping. Cumulative losses are not
		# visible here; use nav_conv_killed (the raider side) and the save's
		# convoy efficiency instead.
		set_variable = { WA_TLM_nav_conv_threat = convoy_threat }
		multiply_variable = { WA_TLM_nav_conv_threat = 100 }
		set_variable = { WA_TLM_nav_conv_ws = has_convoys_war_support }
		multiply_variable = { WA_TLM_nav_conv_ws = 100 }

		# tlm: nav_conv_killed - gauge (engine-side cumulative), any AI navy -
		# consumer: R36 (standing). Convoys THIS country destroyed: the raider side
		# of the same ledger, so an escort failure can be read from both ends.
		set_variable = { WA_TLM_nav_conv_killed = convoys_destroyed }

		# tlm: nav_last_t - sample stamp (absence contract, doc 3.5). Written with the
		# widest gauge of the family, never inside the naval-major gate.
		set_variable = { WA_TLM_nav_last_t = global.WA_TLM_clock }

		if = { limit = { WA_AI_CONFIG_MILITARY_is_major_naval = yes }

			# tlm: nav_port_screens - banded gauge, naval majors - consumer: R36
			# (standing). Screens sitting in our own naval-base states, summed over
			# states with a per-state floor ladder (ships_in_state_ports is a trigger,
			# not a getter). Reads LOW by construction: a state holding 39 screens
			# books 20. That is the honest direction - it cannot manufacture an
			# idle-fleet problem that is not there.
			set_variable = { WA_TLM_nav_port_screens = 0 }
			every_owned_state = {
				limit = {
					is_coastal = yes
					naval_base > 0
					is_controlled_by = PREV
				}
				# Country scope; `state = PREV` then points back at the state, the
				# idiom vanilla uses in common/decisions/ITA.txt (human torpedo raid).
				PREV = {
					if = { limit = { ships_in_state_ports = { type = screen_ship size > 79 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 80 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 39 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 40 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 19 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 20 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 9 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 10 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 4 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 5 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 0 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 1 }
					}
				}
			}

			# tlm: nav_port_pct - derived gauge, 100 * port_screens / screens -
			# consumer: R36 (standing). This is THE headline number: what share of the
			# escort force is parked at home this month.
			set_temp_variable = { _tlm_pp_ = WA_TLM_nav_port_screens }
			multiply_temp_variable = { _tlm_pp_ = 100 }
			if = { limit = { check_variable = { WA_TLM_nav_screens > 0 } }
				divide_temp_variable = { _tlm_pp_ = WA_TLM_nav_screens }
			}
			else = {
				set_temp_variable = { _tlm_pp_ = 0 }
			}
			set_variable = { WA_TLM_nav_port_pct = _tlm_pp_ }

			# Quarterly ring buffer on its OWN axis. It must not share WA_TLM_hist_t:
			# that axis is pushed under WA_AI_CONFIG_is_major_country, a different
			# country set, and any gate mismatch desynchronises index i from the axis
			# for every series on it (doc 4, risk ii). Depth 44 = 11 game-years.
			set_temp_variable = { _tlm_nq_ = global.WA_TLM_clock }
			modulo_temp_variable = { _tlm_nq_ = 3 }
			if = { limit = { check_variable = { _tlm_nq_ = 0 } }
				add_to_array = { array = WA_TLM_nav_hist_t value = global.WA_TLM_clock }
				add_to_array = { array = WA_TLM_nav_port_pct_hist value = WA_TLM_nav_port_pct }
				if = { limit = { check_variable = { WA_TLM_nav_hist_t^num > 44 } }
					remove_from_array = { array = WA_TLM_nav_hist_t index = 0 }
					remove_from_array = { array = WA_TLM_nav_port_pct_hist index = 0 }
				}
			}
		}
	}
}

#	Standard-construction-queue sample (standing, family sq; Fix 88). One gauge: the breadth K the
#	queue functions would use this month = clamp(ceil((civs - PC-assigned) / MAX_CIV_FACTORIES_PER_LINE),
#	1, 20) - the same arithmetic as WA_AI_queue_breadth_prepare (queue_functions), duplicated here
#	because that effect writes a TEMP and telemetry must never be read by gameplay. If the two
#	constants there change, change them here (20 / 20).
#	Everything else in the family is a counter written at its event site (adders / selection /
#	on_state_control_changed) - see the v17 note at the top of this file.
WA_TLM_sample_sq = {
	if = { limit = { is_ai = yes }
		set_temp_variable = { _tlm_sq_k_ = num_of_civilian_factories }
		subtract_from_temp_variable = { _tlm_sq_k_ = WA_AI_PC_assigned_factories_total }
		divide_temp_variable = { _tlm_sq_k_ = 20 }
		clamp_temp_variable = { var = _tlm_sq_k_ min = 1 max = 20 }
		set_variable = { WA_TLM_sq_k = _tlm_sq_k_ }
	}
}
