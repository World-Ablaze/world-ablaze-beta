# Version stamp. Bump when the registered metric set changes; the sampler re-inits on
# mismatch so new metrics exist at 0 even on campaigns resumed on a newer build.
#
# READING RULE for the whole log: every entry is ADDITIVE - no existing metric ever changes
# meaning - so a save at a LOWER version simply lacks that family. Score it NOT CHECKED, never
# FAILED. One exception: at wa_tlm_version = 4 nav_screens and nav_convoys read invalid engine
# tokens and returned a silent 0, which also forced nav_port_pct and every nav_port_pct_hist
# sample to 0. Those are ARTEFACTS on a v4 save and must not be scored; from v5 they are
# measurements. The metric SET is identical either way - the v5 bump exists only so the two can
# be told apart.
#
# Each family's ownership, write site and reading rule live at its init block below; this log is
# only "which build first carried it".
#	v6  post_*                       offensive posture, level-1 exchange rates
#	v7  r64_csubs                    cruiser-submarine hulls
#	v8  r8_uk_air_*                  UK air-base build ORDER
#	v9  nav_cv_f / _n / _hulls       carrier-aviation saturation
#	v10 r44_freeze_*                 scripted-landing invasion freeze, west/east split
#	v11 r74_ally_rail_*              RETIRED - write site, inits and registry rows gone, names free
#	v12 r31_lrvar_t                  long-range La-5FN variant-creation stamp
#	v13 r48_prebuild_synth_*         RETIRED with the proactive rubber lane, names free
#	v14 pc_*                         PC termination ledger + allocation gauges + queued_t
#	v15 r8_air_lane_*                air-basing reservation lane, funded weeks
#	v16 r51_local_hold_*             posture front-local hold band

#	v17 sq_*                         standard construction queue (adds / tiers / K)

@WA_TLM_VERSION = 35

############################################################################################################
#	Clock - global.WA_TLM_clock = months elapsed since 1936.1
#	Scope: any country (called from the per-country monthly pulse; the timed global
#	flag guarantees exactly one increment per calendar month).
############################################################################################################
WA_TLM_tick_clock = {
	if = { limit = { NOT = { has_global_flag = WA_TLM_clock_ticked } }
		set_global_flag = { flag = WA_TLM_clock_ticked value = 1 days = 20 }
		add_to_variable = { global.WA_TLM_clock = 1 }
	}
}

#	Init - PRESERVING zero-init: each metric is created at 0 only if absent, then the version is
#	stamped. Scope: country. Called at startup for every country and lazily from the monthly
#	sampler for tags created mid-game (civil wars, releases).
#	The has_variable guards are load-bearing, not defensive noise: counters write at event sites at
#	ANY cadence, so a mid-game tag can accumulate values BEFORE its first monthly pulse runs this
#	init, and the same guards preserve ring-buffer history across a version-bump re-init. An
#	unconditional zero-init would wipe both. On a fresh start the guards are no-ops.
#	Absence contract: namespace absent = pre-TLM build (or a mid-game tag in its first month);
#	metric 0 with _last_t 0 = never sampled; metric 0 with _last_t > 0 = a real zero.

############################################################################################################
#	Priority-construction sample (standing; doc 3.8 criteria 2 and 3). Scope: country, is_ai gated by
#	the caller.
#
#	The QUEUE itself is fully readable from a savegame and `savegame.py pc TAG FILE...` decodes it per
#	project, so duplicating it here would be pure cost. This sampler covers only what a save cannot
#	give up:
#	  (a) num_of_civilian_factories_available_for_projects - the engine never serializes it. A save
#	      can sum industrial_complex levels, but that is an upper bound on total civ industry and says
#	      nothing about how much the vanilla construction queue had already absorbed, which IS the
#	      number WA_AI_PC_assign_factories multiplies by constant:wa_ai_pc.alloc.fraction. "PC is
#	      using 7% of civ industry" and "PC took its full 40% of what it was offered" can both be
#	      true of the same month, and only the second is actionable.
#	  (b) a trend inside ONE save (the ring buffer), so one save answers the question.
#
#	Cost: eight variable reads, two divides, quarterly three array pushes for majors. Nothing is
#	written on the weekly PC pulse itself - the termination counters live on paths that only run when
#	a project actually leaves the queue, never inside the per-project loops, which are the hot path.
#	Token risk: none - WA_AI_PC_assign_factories itself reads both factory tokens.
############################################################################################################
WA_TLM_sample_pc = {
	# tlm: pc_queue / pc_civs / pc_civs_avail / pc_assigned - gauges, all AI countries -
	# consumer: standing (queue depth and the civ-factory share questions)
	set_variable = { WA_TLM_pc_queue = WA_AI_PC_queue^num }
	set_variable = { WA_TLM_pc_civs = num_of_civilian_factories }
	set_variable = { WA_TLM_pc_civs_avail = num_of_civilian_factories_available_for_projects }
	set_variable = { WA_TLM_pc_assigned = WA_AI_PC_assigned_factories_total }

	# tlm: pc_alloc_base - gauge. The pool WA_AI_PC_assign_factories reconstructs before it applies the
	# fraction: currently-free civs PLUS the ones PC is already holding (they are not "free" until it
	# releases them). The honest denominator for "share of AVAILABLE civs". It deliberately does NOT
	# apply constant:wa_ai_pc.alloc.stable_base_fraction or .hard_cap_fraction: both are functions of
	# pc_civs, sampled beside it, and baking them in would hide the case where the floor is what is
	# actually funding the queue.
	set_variable = { WA_TLM_pc_alloc_base = WA_TLM_pc_civs_avail }
	add_to_variable = { WA_TLM_pc_alloc_base = WA_TLM_pc_assigned }

	# tlm: pc_share_pct - derived gauge, 100 * assigned / civs. Share of the country's whole
	# civilian industry that this shadow construction system is consuming.
	set_temp_variable = { _tlm_pcs_ = WA_TLM_pc_assigned }
	multiply_temp_variable = { _tlm_pcs_ = 100 }
	if = { limit = { check_variable = { WA_TLM_pc_civs > 0 } }
		divide_temp_variable = { _tlm_pcs_ = WA_TLM_pc_civs }
	}
	else = {
		set_temp_variable = { _tlm_pcs_ = 0 }
	}
	set_variable = { WA_TLM_pc_share_pct = _tlm_pcs_ }

	# tlm: pc_avail_share_pct - derived gauge, 100 * assigned / alloc_base. THE diagnostic
	# number of this family. Read it against the fraction actually in force that month
	# (constant:wa_ai_pc.alloc.fraction = 40, or WA_AI_PC_override_max_factories_factor x100 while its
	# country flag is live - `savegame.py pc` prints which):
	#   at the fraction        -> PC took everything it was allowed; the queue is pool-bound
	#   well under it          -> the fill could not place factories: no eligible project
	#                             (hostile-controlled targets, [pc-queue]) or an empty queue
	#   over it                -> the stable-base floor is carrying the queue, i.e. the
	#                             vanilla construction queue has absorbed the free civs
	set_temp_variable = { _tlm_pca_ = WA_TLM_pc_assigned }
	multiply_temp_variable = { _tlm_pca_ = 100 }
	if = { limit = { check_variable = { WA_TLM_pc_alloc_base > 0 } }
		divide_temp_variable = { _tlm_pca_ = WA_TLM_pc_alloc_base }
	}
	else = {
		set_temp_variable = { _tlm_pca_ = 0 }
	}
	set_variable = { WA_TLM_pc_avail_share_pct = _tlm_pca_ }

	# tlm: pc_last_t - sample stamp (absence contract, doc 3.5). Written with the widest
	# gauges of the family - all AI countries - never inside the major gate below, so a
	# minor's genuine "queue empty, zero factories" does not read as "never sampled".
	set_variable = { WA_TLM_pc_last_t = global.WA_TLM_clock }

	# Quarterly ring buffer on its OWN axis (depth 44 = 11 game-years). It must not share WA_TLM_hist_t:
	# that axis is pushed by the composition sampler, and a shared axis makes the two families' gates a
	# permanent coupling - any later divergence desynchronises index i for every series on it (doc 4,
	# risk ii). Only the history is major-gated; the gauges above stay ungated.
	if = { limit = { WA_AI_CONFIG_is_major_country = yes }
		set_temp_variable = { _tlm_pcq_ = global.WA_TLM_clock }
		modulo_temp_variable = { _tlm_pcq_ = 3 }
		if = { limit = { check_variable = { _tlm_pcq_ = 0 } }
			add_to_array = { array = WA_TLM_pc_hist_t value = global.WA_TLM_clock }
			add_to_array = { array = WA_TLM_pc_queue_hist value = WA_TLM_pc_queue }
			add_to_array = { array = WA_TLM_pc_share_pct_hist value = WA_TLM_pc_share_pct }
			add_to_array = { array = WA_TLM_pc_avail_share_pct_hist value = WA_TLM_pc_avail_share_pct }
			if = { limit = { check_variable = { WA_TLM_pc_hist_t^num > 44 } }
				remove_from_array = { array = WA_TLM_pc_hist_t index = 0 }
				remove_from_array = { array = WA_TLM_pc_queue_hist index = 0 }
				remove_from_array = { array = WA_TLM_pc_share_pct_hist index = 0 }
				remove_from_array = { array = WA_TLM_pc_avail_share_pct_hist index = 0 }
			}
		}
	}
}

#	Naval / convoy-war sample (fills the R36 gap). Scope: country, is_ai gated by the caller. The
#	cheap gauges run for anyone with a hull; the per-state port ladder is gated to the naval majors.
#
#	A task force's MISSION is not exposed to script - no trigger reads it - so "how many escorts are
#	actually escorting" is only answerable from the save (`savegame.py navy TAG FILE...`). This
#	sampler therefore covers the two things the save does NOT hand over cheaply:
#	  (a) the OUTCOME of escorting - engine-computed convoy threat, the convoy-loss war-support
#	      malus, and cumulative convoy kills; none are stored in a readable form in the save;
#	  (b) an INDEPENDENT reading of "the escorts are sitting in port" via ships_in_state_ports, which
#	      does not depend on the empirically-derived mission-id map the `navy` command relies on. If
#	      the two disagree, the mission map is what is wrong.
############################################################################################################
WA_TLM_sample_navy = {
	if = { limit = { has_navy_size = { size > 0 } }

		# tlm: nav_ships / nav_screens / nav_subs / nav_convoys - gauges, any AI navy - consumer: R36
		# (standing).
		# num_ships_with_type@ takes a SUB-UNIT DEF TYPE or one of the four engine aggregates carrier /
		# capital / screen / submarine (dynamic_variables_documentation.md). It does NOT take `screen_ship`
		# - that is the sub-unit CATEGORY name from common/units/ship_*.txt - and `convoy` is not a ship at
		# all. Both wrong tokens returned a silent 0 on every country, which is the v4 artefact above.
		# nav_screens is summed from the three concrete hull types rather than the @screen aggregate on
		# purpose: destroyer + frigate + light_cruiser is the exact set carrying `type = screen_ship` in
		# this mod and also what savegame.py's `navy` command counts as `screens=`, so R36's "nav_screens
		# must equal screens= exactly" stays an identity instead of depending on an aggregate's membership.
		set_variable = { WA_TLM_nav_ships = num_ships }
		set_variable = { WA_TLM_nav_screens = num_ships_with_type@destroyer }
		add_to_variable = { WA_TLM_nav_screens = num_ships_with_type@frigate }
		add_to_variable = { WA_TLM_nav_screens = num_ships_with_type@light_cruiser }
		set_variable = { WA_TLM_nav_subs = num_ships_with_type@submarine }

		# tlm: r64_csubs - gauge, any AI with a hull - consumer: R41 ([equipment-selection], SOV routed from
		# the naval_submarine role to the cruiser_submarine role). `cruiser_submarine` is a sub-unit def type
		# (common/units/ship_cruiser_submarine.txt) and a legal num_ships_with_type@ target.
		# nav_subs above IS inclusive of these hulls - both sub-unit defs carry `type = { submarine }` and
		# `submarine` is simultaneously a def name and an engine aggregate, and the aggregate wins - so this
		# is a SHARE of nav_subs, not an independent series. Plain hulls = nav_subs - r64_csubs; never read
		# nav_subs as the plain-boat count.
		# Built-in control (doc 7.5), and it is free: GER has been routed to the cruiser_submarine role since
		# before this fix, so GER > 0 with SOV = 0 means the token is fine and the fix failed, while BOTH 0
		# means the token is bad and neither reading may be scored - the v4 nav_screens failure mode.
		set_variable = { WA_TLM_r64_csubs = num_ships_with_type@cruiser_submarine }

		# tlm: nav_cv_f / nav_cv_n / nav_cv_hulls - gauges, any AI with a hull - consumer: R43
		# ([equipment-selection], the carrier-plane saturation brake). These are the EXACT three quantities
		# the gate compares, sampled at the same values the gate sees, because the gate fails SAFE: an
		# unresolvable token reads 0, 0 never exceeds a bar, the brake never trips and behaviour is
		# unchanged - indistinguishable from "the brake armed and the AI over-built anyway" if you only look
		# at plane counts. So nav_cv_f = nav_cv_n = 0 on a carrier major in a war year means the token is bad,
		# the brake was never live, and NOTHING about the brake may be scored (the v4 failure mode on the
		# same class of token); populated means the gate saw real numbers and nav_cv_hulls says whether it
		# SHOULD have tripped.
		# nav_cv_hulls is the same `type = carrier` aggregate the gate bands on, and it DOES count light
		# carriers - so a light deck bands at the weight of a fleet deck and the per-hull bars run loose for
		# a light-heavy fleet: it under-brakes, which is the safe direction.
		# The plane tokens are archetype-spelled (idiom already live at common/decisions/ITA.txt on this very
		# archetype). UNVERIFIED nonetheless.
		set_variable = { WA_TLM_nav_cv_f = num_deployed_planes_with_type@cv_small_fighter_airframe }
		set_variable = { WA_TLM_nav_cv_n = num_deployed_planes_with_type@cv_small_naval_bomber_airframe }
		set_variable = { WA_TLM_nav_cv_hulls = num_ships_with_type@carrier }
		# convoy is an equipment archetype, so it is read with num_equipment@ - the idiom used for every
		# other archetype in this repo. UNVERIFIED IN-GAME: if it reads 0 on a war-year major while nav_ships
		# is non-zero, the token filter failed again and the metric must be dropped, not scored.
		set_variable = { WA_TLM_nav_convoys = num_equipment@convoy }

		# tlm: nav_conv_threat / nav_conv_ws - gauges, any AI navy - consumer: R36 (standing). Both tokens
		# ARE valid variable reads (dynamic_variables_documentation.md lists both, auto-generated from the
		# same-named triggers); both engine values are 0-1, stored x100 so the save shows integers-ish.
		# conv_threat = how dangerous our convoy routes are. conv_ws = the war-support malus our sunk
		# convoys have already cost us.
		# READING NOTE for conv_ws: a 0 after heavy sinkings is a REAL zero, not a bad token. The engine
		# penalty is a RATE, not a lifetime tally - it accrues from convoys being raided RIGHT NOW, decays
		# weekly and is capped (CONVOYS_BEING_RAIDED_WAR_SUPPORT_PENALTY_SCALE and neighbours in
		# 05_defines.lua), so the observable range is 0-30 in this metric's units and it returns to ~0 within
		# a couple of months of the raiding stopping. Cumulative losses are not visible here; use
		# nav_conv_killed (the raider side) and the save's convoy efficiency instead.
		set_variable = { WA_TLM_nav_conv_threat = convoy_threat }
		multiply_variable = { WA_TLM_nav_conv_threat = 100 }
		set_variable = { WA_TLM_nav_conv_ws = has_convoys_war_support }
		multiply_variable = { WA_TLM_nav_conv_ws = 100 }

		# tlm: nav_conv_killed - gauge (engine-side cumulative), any AI navy -
		# consumer: R36 (standing). Convoys THIS country destroyed: the raider side
		# of the same ledger, so an escort failure can be read from both ends.
		set_variable = { WA_TLM_nav_conv_killed = convoys_destroyed }

		# tlm: nav_last_t - sample stamp (absence contract, doc 3.5). Written with the
		# widest gauge of the family, never inside the naval-major gate.
		set_variable = { WA_TLM_nav_last_t = global.WA_TLM_clock }

		if = { limit = { WA_AI_CONFIG_MILITARY_is_major_naval = yes }

			# tlm: nav_port_screens - banded gauge, naval majors - consumer: R36 (standing). Screens sitting in
			# our own naval-base states, summed over states with a per-state floor ladder (ships_in_state_ports
			# is a trigger, not a getter). Reads LOW by construction - a state holding 39 screens books 20 -
			# which is the honest direction: it cannot manufacture an idle-fleet problem that is not there.
			set_variable = { WA_TLM_nav_port_screens = 0 }
			every_owned_state = {
				limit = {
					is_coastal = yes
					naval_base > 0
					is_controlled_by = PREV
				}
				# Country scope; `state = PREV` then points back at the state, the
				# idiom vanilla uses in common/decisions/ITA.txt (human torpedo raid).
				PREV = {
					if = { limit = { ships_in_state_ports = { type = screen_ship size > 79 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 80 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 39 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 40 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 19 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 20 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 9 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 10 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 4 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 5 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 0 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 1 }
					}
				}
			}

			# tlm: nav_port_pct - derived gauge, 100 * port_screens / screens -
			# consumer: R36 (standing). This is THE headline number: what share of the
			# escort force is parked at home this month.
			set_temp_variable = { _tlm_pp_ = WA_TLM_nav_port_screens }
			multiply_temp_variable = { _tlm_pp_ = 100 }
			if = { limit = { check_variable = { WA_TLM_nav_screens > 0 } }
				divide_temp_variable = { _tlm_pp_ = WA_TLM_nav_screens }
			}
			else = {
				set_temp_variable = { _tlm_pp_ = 0 }
			}
			set_variable = { WA_TLM_nav_port_pct = _tlm_pp_ }

			# Quarterly ring buffer on its OWN axis, depth 44 (= 11 game-years). It must not share WA_TLM_hist_t:
			# that axis is pushed under WA_AI_CONFIG_is_major_country, a different country set, and any gate
			# mismatch desynchronises index i from the axis for every series on it (doc 4, risk ii).
			set_temp_variable = { _tlm_nq_ = global.WA_TLM_clock }
			modulo_temp_variable = { _tlm_nq_ = 3 }
			if = { limit = { check_variable = { _tlm_nq_ = 0 } }
				add_to_array = { array = WA_TLM_nav_hist_t value = global.WA_TLM_clock }
				add_to_array = { array = WA_TLM_nav_port_pct_hist value = WA_TLM_nav_port_pct }
				if = { limit = { check_variable = { WA_TLM_nav_hist_t^num > 44 } }
					remove_from_array = { array = WA_TLM_nav_hist_t index = 0 }
					remove_from_array = { array = WA_TLM_nav_port_pct_hist index = 0 }
				}
			}
		}
	}
}

WA_TLM_sample_sq = {

		subtract_from_temp_variable = { _tlm_sq_k_ = WA_AI_PC_assigned_factories_total }
		divide_temp_variable = { _tlm_sq_k_ = 20 }
		clamp_temp_variable = { var = _tlm_sq_k_ min = 1 max = 20 }
		set_variable = { WA_TLM_sq_k = _tlm_sq_k_ }
	}
}
