############################################################################################################
#	WA_TLM - save-visible telemetry for AI systems
#
#	Design + registry + honesty rules: documentation/WA_TLM_TELEMETRY_SYSTEM.md
#	Retrieval: savegame.py `tlm TAG FILE...` (regex ^wa_tlm_ on the country's variables)
#
#	RESERVED NAMESPACE: everything prefixed WA_TLM_ is telemetry. It is WRITE-ONLY
#	from the mod's point of view - no gameplay/AI logic may ever read a WA_TLM_
#	variable. Saves lowercase variable names: the save-side namespace is wa_tlm_.
#
#	Wiring:
#	  on_startup  (WA_AI_startup_on_actions.txt) - global clock = 0, WA_TLM_init_country for every country
#	  on_monthly  (WA_AI_misc_on_actions.txt)    - WA_TLM_tick_clock + WA_TLM_monthly_sample, is_ai gated
############################################################################################################

# Bump when the registered metric set changes; the sampler re-inits on mismatch so
# new metrics exist at 0 even on campaigns resumed on a newer build.
@WA_TLM_VERSION = 4

############################################################################################################
#	Clock - global.WA_TLM_clock = months elapsed since 1936.1
#	Scope: any country (called from the per-country monthly pulse; the timed global
#	flag guarantees exactly one increment per calendar month).
############################################################################################################
WA_TLM_tick_clock = {
	if = { limit = { NOT = { has_global_flag = WA_TLM_clock_ticked } }
		set_global_flag = { flag = WA_TLM_clock_ticked value = 1 days = 20 }
		add_to_variable = { global.WA_TLM_clock = 1 }
	}
}

############################################################################################################
#	Init - PRESERVING zero-init: each metric is created at 0 only if absent, then
#	the version is stamped. Scope: country. Called at startup for every country and
#	lazily from the monthly sampler for tags created mid-game (civil wars, releases).
#
#	The has_variable guards are load-bearing, not defensive noise: counters write at
#	event sites at ANY cadence, so a mid-game tag can accumulate counter values
#	BEFORE its first monthly pulse runs this init (e.g. the weekly-written
#	WA_TLM_pc_aging_* pair) - an unconditional zero-init would wipe them. Same
#	mechanism preserves ring-buffer history across a version-bump re-init on
#	resumed campaigns. On a fresh start nothing exists and the guards are no-ops.
#	The zero-init is the absence contract: namespace absent = pre-TLM build (or a
#	mid-game tag inside its first month); metric 0 with _last_t 0 = never sampled;
#	metric 0 with _last_t > 0 = real zero.
############################################################################################################
WA_TLM_init_country = {
	set_variable = { WA_TLM_version = @WA_TLM_VERSION }

	# composition (pilot, consumer: checklist R6 / F8)
	if = { limit = { NOT = { has_variable = WA_TLM_comp_div_total } } set_variable = { WA_TLM_comp_div_total = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_armor } } set_variable = { WA_TLM_comp_armor = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_mech } } set_variable = { WA_TLM_comp_mech = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_armor_mech_pct } } set_variable = { WA_TLM_comp_armor_mech_pct = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_comp_last_t } } set_variable = { WA_TLM_comp_last_t = 0 } }
	# ring buffers: no clear - arrays create themselves on first add_to_array, and
	# clearing on re-init is exactly the history wipe the guards exist to prevent

	# PC overtake lane (Fix 41, consumer: checklist R26; write sites in
	# WA_AI_PC_assign_factories, WA_AI_CONSTRUCTION_PRIORITY_core.txt)
	if = { limit = { NOT = { has_variable = WA_TLM_pc_aging_grants } } set_variable = { WA_TLM_pc_aging_grants = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_pc_aging_reval_cancels } } set_variable = { WA_TLM_pc_aging_reval_cancels = 0 } }

	# ai_equipment design-context gates (Fix 47, consumers: checklist R31/R32;
	# write sites in WA_AI_EQUIPMENT_update_context_flags, WA_AI_EQUIPMENT_effects.txt)
	if = { limit = { NOT = { has_variable = WA_TLM_r47_lrange_n } } set_variable = { WA_TLM_r47_lrange_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_r47_lrange_first_t } } set_variable = { WA_TLM_r47_lrange_first_t = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_r47_lrange_last_t } } set_variable = { WA_TLM_r47_lrange_last_t = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_r47_alu_large_n } } set_variable = { WA_TLM_r47_alu_large_n = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_r47_alu_large_first_t } } set_variable = { WA_TLM_r47_alu_large_first_t = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_r47_alu_large_last_t } } set_variable = { WA_TLM_r47_alu_large_last_t = 0 } }

	# naval / convoy war (standing, consumer: checklist R36 + any convoy-war question)
	if = { limit = { NOT = { has_variable = WA_TLM_nav_ships } } set_variable = { WA_TLM_nav_ships = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_screens } } set_variable = { WA_TLM_nav_screens = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_subs } } set_variable = { WA_TLM_nav_subs = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_convoys } } set_variable = { WA_TLM_nav_convoys = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_conv_threat } } set_variable = { WA_TLM_nav_conv_threat = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_conv_ws } } set_variable = { WA_TLM_nav_conv_ws = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_conv_killed } } set_variable = { WA_TLM_nav_conv_killed = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_port_screens } } set_variable = { WA_TLM_nav_port_screens = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_port_pct } } set_variable = { WA_TLM_nav_port_pct = 0 } }
	if = { limit = { NOT = { has_variable = WA_TLM_nav_last_t } } set_variable = { WA_TLM_nav_last_t = 0 } }
	# WA_TLM_nav_hist_t / WA_TLM_nav_port_pct_hist: arrays, no init (see the comment
	# above - clearing on re-init is the history wipe the guards exist to prevent)
}

############################################################################################################
#	Monthly sampler - dispatch. Scope: country, is_ai gated by the caller.
############################################################################################################
WA_TLM_monthly_sample = {
	if = { limit = { NOT = { check_variable = { WA_TLM_version = @WA_TLM_VERSION } } }
		WA_TLM_init_country = yes
	}
	WA_TLM_sample_composition = yes
	WA_TLM_sample_navy = yes
}

############################################################################################################
#	Composition sample (pilot - fills the R6 probe gap: no save-visible army
#	composition metric existed; divisions reference templates by numeric id only).
#
#	Script has no numeric divisions-by-type getter, so armor/mech counts are
#	BANDED: an ascending has_army_size threshold ladder stores the highest floor
#	passed. has_army_size counts divisions by dominant type = "fielded share" in
#	R6's sense. type = infantry/motorized precedent: common/decisions/GER_factions.txt:155.
#	Boot-check note (F9): if armor/mechanized were invalid type tokens the ladder
#	would read 0 on countries with known tanks - do not score R6 off a reading
#	that fails that sanity cross-check on the first instrumented campaign.
############################################################################################################
WA_TLM_sample_composition = {
	# tlm: comp_div_total - gauge, all AI countries - consumer: R6/F8 (standing)
	set_variable = { WA_TLM_comp_div_total = num_divisions }

	# tlm: comp_last_t - sample stamp (absence contract, doc 3.5). Written with the
	# WIDEST gauge of the family, not inside the major gate - otherwise a minor's
	# genuine zero reads as "never sampled" (found in the 1936.4 smoke test: BHU).
	set_variable = { WA_TLM_comp_last_t = global.WA_TLM_clock }

	# The ladders and the ring buffer are major-gated (cost discipline, doc 3.4);
	# R6 scores GER/SOV, and the archetype covers every candidate.
	if = { limit = { WA_AI_CONFIG_is_major_country = yes }

		# tlm: comp_armor - banded gauge - consumer: R6 (standing)
		set_temp_variable = { _tlm_armor_ = 0 }
		if = { limit = { has_army_size = { size > 4 type = armor } } set_temp_variable = { _tlm_armor_ = 5 } }
		if = { limit = { has_army_size = { size > 9 type = armor } } set_temp_variable = { _tlm_armor_ = 10 } }
		if = { limit = { has_army_size = { size > 14 type = armor } } set_temp_variable = { _tlm_armor_ = 15 } }
		if = { limit = { has_army_size = { size > 19 type = armor } } set_temp_variable = { _tlm_armor_ = 20 } }
		if = { limit = { has_army_size = { size > 29 type = armor } } set_temp_variable = { _tlm_armor_ = 30 } }
		if = { limit = { has_army_size = { size > 39 type = armor } } set_temp_variable = { _tlm_armor_ = 40 } }
		if = { limit = { has_army_size = { size > 59 type = armor } } set_temp_variable = { _tlm_armor_ = 60 } }
		if = { limit = { has_army_size = { size > 79 type = armor } } set_temp_variable = { _tlm_armor_ = 80 } }
		if = { limit = { has_army_size = { size > 119 type = armor } } set_temp_variable = { _tlm_armor_ = 120 } }
		if = { limit = { has_army_size = { size > 159 type = armor } } set_temp_variable = { _tlm_armor_ = 160 } }
		if = { limit = { has_army_size = { size > 239 type = armor } } set_temp_variable = { _tlm_armor_ = 240 } }
		if = { limit = { has_army_size = { size > 319 type = armor } } set_temp_variable = { _tlm_armor_ = 320 } }
		set_variable = { WA_TLM_comp_armor = _tlm_armor_ }

		# tlm: comp_mech - banded gauge - consumer: R6 (standing)
		set_temp_variable = { _tlm_mech_ = 0 }
		if = { limit = { has_army_size = { size > 4 type = mechanized } } set_temp_variable = { _tlm_mech_ = 5 } }
		if = { limit = { has_army_size = { size > 9 type = mechanized } } set_temp_variable = { _tlm_mech_ = 10 } }
		if = { limit = { has_army_size = { size > 14 type = mechanized } } set_temp_variable = { _tlm_mech_ = 15 } }
		if = { limit = { has_army_size = { size > 19 type = mechanized } } set_temp_variable = { _tlm_mech_ = 20 } }
		if = { limit = { has_army_size = { size > 29 type = mechanized } } set_temp_variable = { _tlm_mech_ = 30 } }
		if = { limit = { has_army_size = { size > 39 type = mechanized } } set_temp_variable = { _tlm_mech_ = 40 } }
		if = { limit = { has_army_size = { size > 59 type = mechanized } } set_temp_variable = { _tlm_mech_ = 60 } }
		if = { limit = { has_army_size = { size > 79 type = mechanized } } set_temp_variable = { _tlm_mech_ = 80 } }
		if = { limit = { has_army_size = { size > 119 type = mechanized } } set_temp_variable = { _tlm_mech_ = 120 } }
		if = { limit = { has_army_size = { size > 159 type = mechanized } } set_temp_variable = { _tlm_mech_ = 160 } }
		if = { limit = { has_army_size = { size > 239 type = mechanized } } set_temp_variable = { _tlm_mech_ = 240 } }
		if = { limit = { has_army_size = { size > 319 type = mechanized } } set_temp_variable = { _tlm_mech_ = 320 } }
		set_variable = { WA_TLM_comp_mech = _tlm_mech_ }

		# tlm: comp_armor_mech_pct - derived gauge, 100 * (armor + mech) / total - consumer: R6 (standing)
		set_temp_variable = { _tlm_am_ = _tlm_armor_ }
		add_to_temp_variable = { _tlm_am_ = _tlm_mech_ }
		multiply_temp_variable = { _tlm_am_ = 100 }
		if = { limit = { check_variable = { WA_TLM_comp_div_total > 0 } }
			divide_temp_variable = { _tlm_am_ = WA_TLM_comp_div_total }
		}
		else = {
			set_temp_variable = { _tlm_am_ = 0 }
		}
		set_variable = { WA_TLM_comp_armor_mech_pct = _tlm_am_ }

		# Quarterly ring buffer, shared time axis, depth 44 (= 11 game-years).
		# Push value and axis on the SAME tick or every later pairing shifts (doc 4, risk ii).
		set_temp_variable = { _tlm_q_ = global.WA_TLM_clock }
		modulo_temp_variable = { _tlm_q_ = 3 }
		if = { limit = { check_variable = { _tlm_q_ = 0 } }
			add_to_array = { array = WA_TLM_hist_t value = global.WA_TLM_clock }
			add_to_array = { array = WA_TLM_comp_armor_mech_pct_hist value = WA_TLM_comp_armor_mech_pct }
			if = { limit = { check_variable = { WA_TLM_hist_t^num > 44 } }
				remove_from_array = { array = WA_TLM_hist_t index = 0 }
				remove_from_array = { array = WA_TLM_comp_armor_mech_pct_hist index = 0 }
			}
		}
	}
}

############################################################################################################
#	Naval / convoy-war sample (fills the R36 gap).
#
#	What this can and cannot see. A task force's MISSION is not exposed to script -
#	no trigger reads it - so "how many escorts are actually escorting" is only
#	answerable from the save (`savegame.py navy TAG FILE...`, which brace-parses
#	units -> fleet -> task_force -> mission). This sampler therefore covers the two
#	things the save does NOT hand over cheaply:
#	  (a) the OUTCOME of escorting - engine-computed convoy threat, the convoy-loss
#	      war-support malus, and cumulative convoy kills; none of these are stored in
#	      a readable form in the save;
#	  (b) an INDEPENDENT reading of "the escorts are sitting in port", via
#	      ships_in_state_ports, which does not depend on the empirically-derived
#	      mission-id map the `navy` command relies on. If the two disagree, the
#	      mission map is what is wrong.
#
#	Scope: country, is_ai gated by the caller. Cheap gauges run for anyone with a
#	hull; the per-state port ladder is gated to the six naval majors (doc 3.4).
############################################################################################################
WA_TLM_sample_navy = {
	if = { limit = { has_navy_size = { size > 0 } }

		# tlm: nav_ships / nav_screens / nav_subs / nav_convoys - gauges, any AI navy
		# - consumer: R36 (standing). screen_ship is the sub-unit category shared by
		# destroyer/frigate/light_cruiser (common/units/ship_*.txt), i.e. the escort hulls.
		set_variable = { WA_TLM_nav_ships = num_ships }
		set_variable = { WA_TLM_nav_screens = num_ships_with_type@screen_ship }
		set_variable = { WA_TLM_nav_subs = num_ships_with_type@submarine }
		set_variable = { WA_TLM_nav_convoys = num_ships_with_type@convoy }

		# tlm: nav_conv_threat / nav_conv_ws - gauges, any AI navy - consumer: R36
		# (standing). Both engine values are 0-1; stored x100 so the save shows
		# integers-ish. conv_threat = how dangerous our convoy routes are,
		# conv_ws = the war-support malus our sunk convoys have already cost us.
		set_variable = { WA_TLM_nav_conv_threat = convoy_threat }
		multiply_variable = { WA_TLM_nav_conv_threat = 100 }
		set_variable = { WA_TLM_nav_conv_ws = has_convoys_war_support }
		multiply_variable = { WA_TLM_nav_conv_ws = 100 }

		# tlm: nav_conv_killed - gauge (engine-side cumulative), any AI navy -
		# consumer: R36 (standing). Convoys THIS country destroyed: the raider side
		# of the same ledger, so an escort failure can be read from both ends.
		set_variable = { WA_TLM_nav_conv_killed = convoys_destroyed }

		# tlm: nav_last_t - sample stamp (absence contract, doc 3.5). Written with the
		# widest gauge of the family, never inside the naval-major gate.
		set_variable = { WA_TLM_nav_last_t = global.WA_TLM_clock }

		if = { limit = { WA_AI_CONFIG_MILITARY_is_major_naval = yes }

			# tlm: nav_port_screens - banded gauge, naval majors - consumer: R36
			# (standing). Screens sitting in our own naval-base states, summed over
			# states with a per-state floor ladder (ships_in_state_ports is a trigger,
			# not a getter). Reads LOW by construction: a state holding 39 screens
			# books 20. That is the honest direction - it cannot manufacture an
			# idle-fleet problem that is not there.
			set_variable = { WA_TLM_nav_port_screens = 0 }
			every_owned_state = {
				limit = {
					is_coastal = yes
					naval_base > 0
					is_controlled_by = PREV
				}
				# Country scope; `state = PREV` then points back at the state, the
				# idiom vanilla uses in common/decisions/ITA.txt (human torpedo raid).
				PREV = {
					if = { limit = { ships_in_state_ports = { type = screen_ship size > 79 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 80 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 39 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 40 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 19 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 20 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 9 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 10 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 4 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 5 }
					}
					else_if = { limit = { ships_in_state_ports = { type = screen_ship size > 0 state = PREV } }
						add_to_variable = { WA_TLM_nav_port_screens = 1 }
					}
				}
			}

			# tlm: nav_port_pct - derived gauge, 100 * port_screens / screens -
			# consumer: R36 (standing). This is THE headline number: what share of the
			# escort force is parked at home this month.
			set_temp_variable = { _tlm_pp_ = WA_TLM_nav_port_screens }
			multiply_temp_variable = { _tlm_pp_ = 100 }
			if = { limit = { check_variable = { WA_TLM_nav_screens > 0 } }
				divide_temp_variable = { _tlm_pp_ = WA_TLM_nav_screens }
			}
			else = {
				set_temp_variable = { _tlm_pp_ = 0 }
			}
			set_variable = { WA_TLM_nav_port_pct = _tlm_pp_ }

			# Quarterly ring buffer on its OWN axis. It must not share WA_TLM_hist_t:
			# that axis is pushed under WA_AI_CONFIG_is_major_country, a different
			# country set, and any gate mismatch desynchronises index i from the axis
			# for every series on it (doc 4, risk ii). Depth 44 = 11 game-years.
			set_temp_variable = { _tlm_nq_ = global.WA_TLM_clock }
			modulo_temp_variable = { _tlm_nq_ = 3 }
			if = { limit = { check_variable = { _tlm_nq_ = 0 } }
				add_to_array = { array = WA_TLM_nav_hist_t value = global.WA_TLM_clock }
				add_to_array = { array = WA_TLM_nav_port_pct_hist value = WA_TLM_nav_port_pct }
				if = { limit = { check_variable = { WA_TLM_nav_hist_t^num > 44 } }
					remove_from_array = { array = WA_TLM_nav_hist_t index = 0 }
					remove_from_array = { array = WA_TLM_nav_port_pct_hist index = 0 }
				}
			}
		}
	}
}
