# Promotes an imperial protectorate to nominal statehood (subject scope)
JAP_grant_statehood_effect = {
	if = {
		limit = { original_tag = IPP }
		set_cosmetic_tag = PHI_imperial_state
	}
	else_if = {
		limit = { original_tag = IPM }
		set_cosmetic_tag = MAL_imperial_state
	}
	else_if = {
		limit = { original_tag = IPI }
		set_cosmetic_tag = INS_imperial_state
	}
	else_if = {
		limit = { original_tag = IPR }
		set_cosmetic_tag = RAJ_imperial_state
	}
	else_if = {
		limit = { original_tag = IPA }
		set_cosmetic_tag = AST_imperial_state
	}
	else_if = {
		limit = { original_tag = IPN }
		set_cosmetic_tag = NZL_imperial_state
	}
	else_if = {
		limit = { original_tag = IPB }
		set_cosmetic_tag = BRM_imperial_state
	}
	every_owned_state = {
		add_compliance = 25
	}
	every_owned_state = {
		limit = {
			NOT = { is_core_of = PREV }
		}
		add_core_of = PREV
	}
	OVERLORD = {
		set_autonomy = {
			target = PREV
			autonomous_state = autonomy_wtt_imperial_state
		}
	}
}
