############################################################################################################
#	WA_TEST - fuel-silo breadth-first queue ([silo-breadth]), console-driven.
#
#	Fired by wa_silo.1 / .2 in events/wa_test_silo.txt. Read the console recipe there.
#
#	WHAT THIS PROVES AND WHAT IT CANNOT. Three legs:
#	  A  availability (read-only): for the top states of WA_AI_shared_slot_scores, the per-state
#	     verdicts of WA_AI_available_SILO and WA_AI_queue_depth_ok_SILO against the built level and
#	     the committed sum - a state at the silo cap must read available = 0.
#	  B  burst (side-effectful, wa_silo.2): 8 consecutive WA_AI_queue_SILO calls in one execution,
#	     the same shape as the caller's event burst. Logs which state each pick landed in and its
#	     selection tier. PASS = picks spread over >= 2 states on any country with >= 2 eligible states.
#	  C  caller gate: the caller (events/WA_AI_construction.txt, "# Silos") renders a TEMP variable
#	     into a meta_trigger `fuel_silo < [x]`. Meta renders can lose temp context and a malformed
#	     trigger gates TRUE, so leg C re-renders the same shape beside a known-true and a known-false
#	     control.
#	  ENGINE RESIDUAL, stated honestly: leg B counts adds the way the committed sum does - script
#	  cannot read the engine construction queue, so whether the engine ACCEPTED each add is only
#	  visible in the UI / a later save. The campaign probe (WORK.md subject) owns built levels.
#
#	SIDE EFFECTS OF A RUN: wa_silo.2 queues up to 8 REAL fuel-silo constructions on ROOT (script
#	cannot cancel them - cancel in the construction UI, or run on a throwaway save) and increments
#	WA_TLM_sq_adds_n / _adds_by_type^8 - do not score this save's sq telemetry after testing on it.
############################################################################################################

# ROOT = THIS = the country under test. Read-only: legs A and C.
WA_TEST_SILO_report = {

	log = "=========================================================================================="
	log = "[GetYear].[GetMonth] | SILO TEST | [Root.GetName]"

	### CONTEXT HEADER - printed FIRST, on every run. harness-contract: v1 (wa-testing SKILL).
	### Without these lines a poisoned call site reads every country-valued trigger false while
	### [Root.GetName] still prints the right name beside it ("Two call sites, one effect" lesson).
	### The fifth check is a KNOWN-FALSE control: it must read 0.
	log = "  who   : THIS = [This.GetName]   ROOT = [Root.GetName]   FROM = [From.GetName]"
	set_temp_variable = { _ws_e1_ = 0 }
	set_temp_variable = { _ws_e2_ = 0 }
	set_temp_variable = { _ws_e3_ = 0 }
	set_temp_variable = { _ws_e4_ = 0 }
	set_temp_variable = { _ws_e5_ = 0 }
	if = { limit = { always = yes } set_temp_variable = { _ws_e1_ = 1 } }
	if = { limit = { tag = ROOT } set_temp_variable = { _ws_e2_ = 1 } }
	if = { limit = { tag = THIS } set_temp_variable = { _ws_e3_ = 1 } }
	if = { limit = { ROOT = { always = yes } } set_temp_variable = { _ws_e4_ = 1 } }
	if = { limit = { NOT = { always = yes } } set_temp_variable = { _ws_e5_ = 1 } }
	log = "  scope : always=[?_ws_e1_|.0]  I-am-ROOT=[?_ws_e2_|.0]  I-am-THIS=[?_ws_e3_|.0]  ROOT-scope-usable=[?_ws_e4_|.0]  control-false=[?_ws_e5_|.0]"
	log = "            (anything but 1 1 1 1 0 here: STOP. Nothing below is a measurement. Re-fire from events/wa_test_scope_isolation.txt - tag GER then event wa_iso.3 - before editing this effect.)"
	set_temp_variable = { _ws_tech_ = 0 }
	if = { limit = { has_tech = fuel_silos } set_temp_variable = { _ws_tech_ = 1 } }
	log = "  tech  : fuel_silos = [?_ws_tech_|.0]   (0 = the queue is gated off for this country; legs A/C still measure, leg B will pick nothing)"

	### LEG A - availability walk over the shared-slot score array (top 8). Scores are computed on
	### demand exactly as the queue functions do.
	if = { limit = { check_variable = { WA_AI_has_shared_slot_scores = 0 } }
		WA_AI_get_shared_slot_scores = yes
	}
	log = "  legA  : shared_slot_scores size = [?WA_AI_shared_slot_scores^num|.0]   (0 = no scored state, every leg below is vacuous)"
	set_temp_variable = { _ws_rank_ = 0 }
	for_each_scope_loop = { array = WA_AI_shared_slot_scores break = _ws_stop_
		set_temp_variable = { _ws_avail_ = 0 }
		set_temp_variable = { _ws_depth_ok_ = 0 }
		set_temp_variable = { _ws_shared_ = 0 }
		set_temp_variable = { _ws_flag_ = 0 }
		if = { limit = { WA_AI_available_SILO = yes } set_temp_variable = { _ws_avail_ = 1 } }
		if = { limit = { WA_AI_queue_depth_ok_SILO = yes } set_temp_variable = { _ws_depth_ok_ = 1 } }
		if = { limit = { WA_AI_available_SHARED = yes } set_temp_variable = { _ws_shared_ = 1 } }
		if = { limit = { has_state_flag = WA_AI_committed_SILO_recent } set_temp_variable = { _ws_flag_ = 1 } }
		log = "  legA  : #[?_ws_rank_|.0] [This.GetName] ([This.GetID])  built = [?building_level@fuel_silo|.0]  committed = [?WA_AI_committed_SILO|.0]  ttl-flag = [?_ws_flag_|.0]  avail = [?_ws_avail_|.0]  depth_ok = [?_ws_depth_ok_|.0]  shared_ok = [?_ws_shared_|.0]"
		if = { limit = { check_variable = { building_level@fuel_silo > 5 } check_variable = { _ws_avail_ = 1 } }
			log = "  legA  : FAIL - state above is at the silo cap (6) and still reads available; the cap test is not enforcing."
		}
		add_to_temp_variable = { _ws_rank_ = 1 }
		if = { limit = { check_variable = { _ws_rank_ > 7 } } set_temp_variable = { _ws_stop_ = 1 } }
	}
	log = "  legA  : PASS = every state with built = 6 reads avail = 0, and at least one state reads avail = 1 on a country that can still build silos."

	### LEG C - the caller's gate shape: TEMP need rendered into meta_trigger `fuel_silo < [x]`,
	### between a known-true and a known-false control. Need formula copied from the caller
	### (num_of_military_factories - resource_produced@oil, / 6).
	set_temp_variable = { _ws_need_ = num_of_military_factories }
	subtract_from_temp_variable = { _ws_need_ = resource_produced@oil }
	divide_temp_variable = { _ws_need_ = 6 }
	set_temp_variable = { _ws_c1_ = 0 }
	set_temp_variable = { _ws_c2_ = 0 }
	set_temp_variable = { _ws_c3_ = 0 }
	if = {
		limit = {
			meta_trigger = {
				text = { fuel_silo < [x] }
				x = "[?_ws_need_|.0]"
			}
		}
		set_temp_variable = { _ws_c1_ = 1 }
	}
	if = {
		limit = {
			meta_trigger = {
				text = { fuel_silo < [x] }
				x = "9999"
			}
		}
		set_temp_variable = { _ws_c2_ = 1 }
	}
	if = {
		limit = {
			meta_trigger = {
				text = { fuel_silo < [x] }
				x = "0"
			}
		}
		set_temp_variable = { _ws_c3_ = 1 }
	}
	log = "  legC  : need(temp) = [?_ws_need_|.0]   gate(fuel_silo < need) = [?_ws_c1_|.0]   known-true(< 9999) = [?_ws_c2_|.0]   known-false(< 0) = [?_ws_c3_|.0]"
	log = "  legC  : PASS = controls read 1 / 0 and the gate verdict is consistent with the need value against the country's silo count. Controls 1/1 or 0/0 = the meta render is broken - the caller's gate (events/WA_AI_construction.txt '# Silos') cannot be trusted either way."
	log = "  next  : event wa_silo.2 runs the burst leg (SIDE-EFFECTFUL - queues real silos)."
}

# ROOT = THIS = the country under test. Leg B: side-effectful burst.
WA_TEST_SILO_burst = {

	log = "=========================================================================================="
	log = "[GetYear].[GetMonth] | SILO TEST BURST | [Root.GetName]"

	### CONTEXT HEADER - harness-contract: v1. Same rules and STOP rule as the main report.
	log = "  who   : THIS = [This.GetName]   ROOT = [Root.GetName]   FROM = [From.GetName]"
	set_temp_variable = { _ws_b1_ = 0 }
	set_temp_variable = { _ws_b2_ = 0 }
	set_temp_variable = { _ws_b5_ = 0 }
	if = { limit = { always = yes } set_temp_variable = { _ws_b1_ = 1 } }
	if = { limit = { tag = ROOT } set_temp_variable = { _ws_b2_ = 1 } }
	if = { limit = { NOT = { always = yes } } set_temp_variable = { _ws_b5_ = 1 } }
	log = "  scope : always=[?_ws_b1_|.0]  I-am-ROOT=[?_ws_b2_|.0]  control-false=[?_ws_b5_|.0]   (anything but 1 1 0: STOP, see WA_TEST_SILO_report)"

	### 8 queue calls in ONE execution = one caller burst: WA_AI_bf_k and the picked array live
	### across the calls exactly as they would in the real event. The while loop MUST use its own
	### break variable: the queue sets the plain `break` temp to 1 on every add (downstream
	### contract) and while_loop_effect's default break variable IS `break`
	### (effects_documentation.md, while_loop_effect) - the default would end the burst at pick 0.
	set_temp_variable = { _ws_i_ = 0 }
	while_loop_effect = {
		limit = { check_variable = { _ws_i_ < 8 } }
		break = _ws_burst_stop_
		set_temp_variable = { _wa_bf_target_found_ = 0 }
		WA_AI_queue_SILO = yes
		if = { limit = { check_variable = { _wa_bf_target_found_ = 1 } }
			log = "  legB  : pick [?_ws_i_|.0] -> [?_wa_bf_target_.GetName] ([?_wa_bf_target_.GetID])  tier = [?_wa_bf_tier_|.0]  k = [?WA_AI_bf_k|.0]"
		}
		else = {
			log = "  legB  : pick [?_ws_i_|.0] -> none (nothing eligible - tech off, no scored state, or every candidate at cap)"
		}
		add_to_temp_variable = { _ws_i_ = 1 }
	}
	log = "  legB  : PASS = picks land on >= 2 distinct states (tiers 1-3) on a country with >= 2 eligible states; the pre-[silo-breadth] behaviour was every pick on the same top state. Adds are counted, not verified builds - check the construction UI, and do not score this save's sq telemetry."
}
