# The two semicircle icons have 21 frames; the frame follows the share of seats backing the government.
JAP_recalc_diet_icon_frames = {
	set_temp_variable = { diet_temp = JAP_peers_support }
	divide_temp_variable = { diet_temp = JAP_peers_total }
	multiply_temp_variable = { diet_temp = 21 }
	if = {
		limit = { check_variable = { diet_temp > 0 } }
		clamp_temp_variable = {
			var = diet_temp
			min = 1
			max = 21
		}
	}
	else = {
		set_temp_variable = { diet_temp = 1 }
	}
	set_variable = { JAP_peers_icon_frame = diet_temp }
	set_temp_variable = { diet_temp = JAP_representatives_support }
	divide_temp_variable = { diet_temp = JAP_representatives_total }
	multiply_temp_variable = { diet_temp = 21 }
	if = {
		limit = { check_variable = { diet_temp > 0 } }
		clamp_temp_variable = {
			var = diet_temp
			min = 1
			max = 21
		}
	}
	else = {
		set_temp_variable = { diet_temp = 1 }
	}
	set_variable = { JAP_representatives_icon_frame = diet_temp }
}
