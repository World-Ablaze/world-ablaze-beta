# The Imperial Diet.
# House of Representatives: 466 seats split between the parties. JAP_diet_<party> is a party's seats and
# JAP_diet_<party>_gov is 1 while that party backs the government; JAP_diet_others is whatever is left of
# JAP_representatives_total. JAP_representatives_support is the sum of the seats of the parties behind the government,
# which is what the conditions in common/scripted_triggers/JAP_diet_triggers.txt read.
# House of Peers: no parties, JAP_peers_support out of JAP_peers_total (400).
# The player has no way to move seats directly: elections and the focus tree do.
# After changing any seat or _gov variable, call JAP_diet_recalc.

JAP_peers_small_support = {
	custom_effect_tooltip = JAP_peers_small_support_tt
	add_to_variable = { JAP_peers_support = 15 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

JAP_peers_small_opposition = {
	custom_effect_tooltip = JAP_peers_small_opposition_tt
	add_to_variable = { JAP_peers_support = -10 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

JAP_peers_medium_support = {
	custom_effect_tooltip = JAP_peers_medium_support_tt
	add_to_variable = { JAP_peers_support = 30 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

JAP_peers_medium_opposition = {
	custom_effect_tooltip = JAP_peers_medium_opposition_tt
	add_to_variable = { JAP_peers_support = -20 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

JAP_peers_large_support = {
	custom_effect_tooltip = JAP_peers_large_support_tt
	add_to_variable = { JAP_peers_support = 70 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

JAP_peers_large_opposition = {
	custom_effect_tooltip = JAP_peers_large_opposition_tt
	add_to_variable = { JAP_peers_support = -50 }
	clamp_variable = { var = JAP_peers_support max = JAP_peers_total min = 0 }
}

# The seats at the start of 1936: the Seiyukai holds a majority against the Okada cabinet (Minseito + Showakai = 152).
JAP_diet_reset_start = {
	set_variable = { JAP_diet_yokusankai = 0 }
	set_variable = { JAP_diet_yokusankai_gov = 0 }
	set_variable = { JAP_diet_minseito = 127 }
	set_variable = { JAP_diet_minseito_gov = 1 }
	set_variable = { JAP_diet_showakai = 25 }
	set_variable = { JAP_diet_showakai_gov = 1 }
	set_variable = { JAP_diet_kokumin_domei = 20 }
	set_variable = { JAP_diet_kokumin_domei_gov = 0 }
	set_variable = { JAP_diet_others_gov = 0 }
	set_variable = { JAP_diet_seiyukai = 242 }
	set_variable = { JAP_diet_seiyukai_gov = 0 }
	JAP_diet_recalc = yes
}

# The general election of 20 February 1936. Minseito 205 + Showakai 20 = 225 behind the cabinet, nine short of a majority.
JAP_diet_election_1936 = {
	custom_effect_tooltip = JAP_diet_election_1936_tt
	hidden_effect = {
		set_variable = { JAP_diet_yokusankai = 0 }
		set_variable = { JAP_diet_minseito = 205 }
		set_variable = { JAP_diet_showakai = 20 }
		set_variable = { JAP_diet_kokumin_domei = 15 }
		set_variable = { JAP_diet_seiyukai = 174 }
		JAP_diet_recalc = yes
	}
}

# The parties dissolve into the Imperial Rule Assistance Association. Everyone joins but 40 members, who stay with the Others.
JAP_diet_form_yokusankai = {
	custom_effect_tooltip = JAP_diet_form_yokusankai_tt
	hidden_effect = {
		set_variable = { JAP_diet_minseito = 0 }
		set_variable = { JAP_diet_showakai = 0 }
		set_variable = { JAP_diet_kokumin_domei = 0 }
		set_variable = { JAP_diet_seiyukai = 0 }
		set_variable = { JAP_diet_yokusankai = JAP_representatives_total }
		subtract_from_variable = { JAP_diet_yokusankai = 40 }
		set_variable = { JAP_diet_yokusankai_gov = 1 }
		JAP_diet_recalc = yes
	}
	JAP_peers_large_support = yes
}

# Rebuilds everything that is derived from the seats: the Others, the support of the government, the strength of the
# Diet's decision and the coloured seats of the panel (JAP_diet_seat_frames, one frame number per seat; the grid of
# interface/jap_diet_scripted_gui.gui fills row by row, 47 to a row, so the seats are laid down column by column here
# to keep each party in one block).
JAP_diet_recalc = {
	set_variable = { JAP_diet_others = JAP_representatives_total }
	subtract_from_variable = { JAP_diet_others = JAP_diet_yokusankai }
	subtract_from_variable = { JAP_diet_others = JAP_diet_minseito }
	subtract_from_variable = { JAP_diet_others = JAP_diet_showakai }
	subtract_from_variable = { JAP_diet_others = JAP_diet_kokumin_domei }
	subtract_from_variable = { JAP_diet_others = JAP_diet_seiyukai }
	clamp_variable = { var = JAP_diet_others min = 0 }

	set_variable = { JAP_representatives_support = 0 }
	set_temp_variable = { diet_temp = JAP_diet_yokusankai }
	multiply_temp_variable = { diet_temp = JAP_diet_yokusankai_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }
	set_temp_variable = { diet_temp = JAP_diet_minseito }
	multiply_temp_variable = { diet_temp = JAP_diet_minseito_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }
	set_temp_variable = { diet_temp = JAP_diet_showakai }
	multiply_temp_variable = { diet_temp = JAP_diet_showakai_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }
	set_temp_variable = { diet_temp = JAP_diet_kokumin_domei }
	multiply_temp_variable = { diet_temp = JAP_diet_kokumin_domei_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }
	set_temp_variable = { diet_temp = JAP_diet_others }
	multiply_temp_variable = { diet_temp = JAP_diet_others_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }
	set_temp_variable = { diet_temp = JAP_diet_seiyukai }
	multiply_temp_variable = { diet_temp = JAP_diet_seiyukai_gov }
	add_to_variable = { JAP_representatives_support = diet_temp }

	# strength of JAP_diet_budget_modifier: 10% at a full house
	set_variable = { JAP_diet_budget_bonus = JAP_representatives_support }
	divide_variable = { JAP_diet_budget_bonus = JAP_representatives_total }
	multiply_variable = { JAP_diet_budget_bonus = 0.1 }

	clear_temp_array = diet_order
	for_loop_effect = {
		end = JAP_diet_yokusankai
		add_to_temp_array = { diet_order = 1 }
	}
	for_loop_effect = {
		end = JAP_diet_minseito
		add_to_temp_array = { diet_order = 2 }
	}
	for_loop_effect = {
		end = JAP_diet_showakai
		add_to_temp_array = { diet_order = 3 }
	}
	for_loop_effect = {
		end = JAP_diet_kokumin_domei
		add_to_temp_array = { diet_order = 4 }
	}
	for_loop_effect = {
		end = JAP_diet_others
		add_to_temp_array = { diet_order = 5 }
	}
	for_loop_effect = {
		end = JAP_diet_seiyukai
		add_to_temp_array = { diet_order = 6 }
	}
	clear_array = JAP_diet_seat_frames
	for_loop_effect = {
		end = JAP_representatives_total
		value = diet_i
		set_temp_variable = { diet_col = diet_i }
		modulo_temp_variable = { diet_col = 47 }
		set_temp_variable = { diet_row = diet_i }
		subtract_from_temp_variable = { diet_row = diet_col }
		divide_temp_variable = { diet_row = 47 }
		round_temp_variable = diet_row
		if = {
			limit = { check_variable = { diet_col < 43 } }
			set_temp_variable = { diet_k = diet_col }
			multiply_temp_variable = { diet_k = 10 }
		}
		else = {
			set_temp_variable = { diet_k = diet_col }
			subtract_from_temp_variable = { diet_k = 43 }
			multiply_temp_variable = { diet_k = 9 }
			add_to_temp_variable = { diet_k = 430 }
		}
		add_to_temp_variable = { diet_k = diet_row }
		set_temp_variable = { diet_frame = diet_order^diet_k }
		add_to_array = { JAP_diet_seat_frames = diet_frame }
	}
}
