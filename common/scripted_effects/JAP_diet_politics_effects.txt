# What the political tree does to the House of Representatives: who backs the government, and the elections.
# The machinery is in JAP_diet_effects.txt (JAP_diet_<party>, JAP_diet_<party>_gov, JAP_diet_recalc).

# A cabinet with ministers from both big parties (Hirota, the first Konoe, Yonai).
JAP_diet_gov_national_unity = {
	custom_effect_tooltip = JAP_diet_gov_national_unity_tt
	hidden_effect = {
		set_variable = { JAP_diet_minseito_gov = 1 }
		set_variable = { JAP_diet_seiyukai_gov = 1 }
		set_variable = { JAP_diet_showakai_gov = 1 }
		set_variable = { JAP_diet_kokumin_domei_gov = 1 }
		set_variable = { JAP_diet_seiyukai_seito_gov = 1 }
		JAP_diet_recalc = yes
	}
}

# The Seiyukai turns on the cabinet.
JAP_diet_gov_seiyukai_leaves = {
	custom_effect_tooltip = JAP_diet_gov_seiyukai_leaves_tt
	hidden_effect = {
		set_variable = { JAP_diet_seiyukai_gov = 0 }
		JAP_diet_recalc = yes
	}
}

# 30 April 1939: the Seiyukai splits. The numbers are those of the dissolution of July 1940 - the Reform wing 97 and the Unity wing 10
# stay the Seiyukai of the panel, the Orthodox wing (Kuhara, Hatoyama) has 66 and no minister.
JAP_diet_seiyukai_split = {
	custom_effect_tooltip = JAP_diet_seiyukai_split_tt
	hidden_effect = {
		set_variable = { JAP_diet_seiyukai = 107 }
		set_variable = { JAP_diet_seiyukai_seito = 66 }
		set_variable = { JAP_diet_seiyukai_seito_gov = 0 }
		JAP_diet_recalc = yes
	}
}

# Saito Takao is expelled from the House.
JAP_diet_expel_saito = {
	custom_effect_tooltip = JAP_diet_expel_saito_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { JAP_diet_minseito > 0 } }
			subtract_from_variable = { JAP_diet_minseito = 1 }
		}
		JAP_diet_recalc = yes
	}
}
