# [refineries] (+ [pc-queue] for CIC/MIC/NIC): how long the per-state WA_AI_committed_<X> sum (built + queued levels) is trusted
# after the last add. When no add happened on the state for this long, nothing can still be queued:
# WA_AI_available_<X> falls back to the built level alone and the next add rebases the sum to the
# built level. This is the only path by which a CANCELLED construction (state lost, queue cleared) is
# forgotten - script has no completion/cancellation hook. Timelines (cap 15, built 10, 5 queued ->
# committed 15, state excluded):
#   completion : line finishes -> built 11, committed 15 unchanged, still excluded. Correct, no pulse needed.
#   cancelled  : the 5 lines vanish at t1; at t1+TTL the flag lapses -> `built < cap` alone decides ->
#                the next add rebases committed to 10, then 11, flag renewed. Correct re-fill.
#   unfunded   : the 5 lines are still alive at t1+TTL (seen: 12+ months unfunded) -> same rebase to
#                10 -> add 11 refused by the engine (real 15+1 > cap), committed 11 ... 15, then excluded
#                again. At most cap - built = 5 refused adds, each costing that pulse's pick for this
#                type on this one state - the queue loop breaks on the first add whether or not the
#                engine accepted it. Bounded, one state, only after 1000 days without an add.
# Too short = the unfunded case above happens more often; too long = a cancelled state stays excluded
# that much longer. Err long.
@WA_AI_QUEUED_REFINERY_TTL_DAYS = 1000

WA_AI_add_CIC = {
	if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: WA_AI_add_CIC in [This.GetID] [This.GetName]" }
	# [refineries] / [pc-queue] maintain WA_AI_committed_CIC = built + queued so WA_AI_available_CIC can enforce the
	# state cap and WA_AI_queue_depth_ok_ the per-line depth. Counted only when the add can actually land
	# (same test the queue loops use), so the hard-coded state sites in WA_AI_construction.txt that spam
	# this on a full state do not inflate it. Rebase to the built level: unconditional when the TTL flag
	# lapsed (nothing can still be queued from before), upward clamp only while it is live (levels that
	# rose outside this adder - focus, PC instant build, the engine's own construction AI).
	if = { limit = { WA_AI_available_CIC = yes }
		# tlm: WA_TLM_sq_rebase_down_n (standing, family sq) - the TTL lapsed and the sum was above the built
		# level: a queue that was cancelled/lost is being forgotten. Verified effect (the set below runs).
		if = {
			limit = {
				NOT = { has_state_flag = WA_AI_committed_CIC_recent }
				check_variable = { WA_AI_committed_CIC > building_level@industrial_complex }
			}
			ROOT = { add_to_variable = { WA_TLM_sq_rebase_down_n = 1 } }
		}
		if = {
			limit = {
				OR = {
					NOT = { has_state_flag = WA_AI_committed_CIC_recent }
					check_variable = { WA_AI_committed_CIC < building_level@industrial_complex }
				}
			}
			set_variable = { WA_AI_committed_CIC = building_level@industrial_complex }
		}
		add_to_variable = { WA_AI_committed_CIC = 1 }
		set_state_flag = { flag = WA_AI_committed_CIC_recent value = 1 days = @WA_AI_QUEUED_REFINERY_TTL_DAYS }
		# tlm: WA_TLM_sq_adds_n / _adds_by_type^0 / _first_t / _last_t (standing, family sq) - a counted
		# add = availability passed and add_building_construction issued; script cannot read the engine
		# queue, so this is the same evidence level as the committed sum itself. Consumers: R49, R53.
		ROOT = {
			add_to_variable = { WA_TLM_sq_adds_n = 1 }
			add_to_variable = { WA_TLM_sq_adds_by_type^0 = 1 }
			if = { limit = { check_variable = { WA_TLM_sq_first_t = 0 } } set_variable = { WA_TLM_sq_first_t = global.WA_TLM_clock } }
			set_variable = { WA_TLM_sq_last_t = global.WA_TLM_clock }
		}
	}
	add_building_construction = { type = industrial_complex level = 1 instant_build = no }
}
