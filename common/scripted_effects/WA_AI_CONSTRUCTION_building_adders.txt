# Fix 81: how long a queued refinery level is presumed to still be in the construction queue.
# When no add happened on the state for this long, WA_AI_reconcile_queued_refineries zeroes the
# counter - the only path by which a CANCELLED construction (state lost, queue cleared) is
# forgotten, since script has no completion/cancellation hook. Must comfortably exceed the
# slowest plausible refinery build (10-12.5k IC at WA construction speed). Too short = the
# stall this fix removes comes back briefly on that state; too long = a state whose queue was
# cancelled stays excluded that much longer. Err long.
@WA_AI_QUEUED_REFINERY_TTL_DAYS = 1000

WA_AI_add_CIC = {
	if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: WA_AI_add_CIC in [This.GetID] [This.GetName]" }
	add_building_construction = { type = industrial_complex level = 1 instant_build = no }
}

WA_AI_add_REF = {
	if = { limit = { ROOT = { has_country_flag = WA_AI_construction_logging } } log = "[GetYear] [GetMonth] | AI | [Root.GetName] | CONSTRUCTION: WA_AI_add_REF in [This.GetID] [This.GetName]" }
	# Fix 81: count the queued level so WA_AI_available_REF can enforce built + queued < state cap.
	# Counted only when the add can actually land (same test the queue loops use), so the
	# hard-coded state sites in WA_AI_construction.txt that spam this on a full state do not inflate it.
	WA_AI_reconcile_queued_refineries = yes
	if = { limit = { WA_AI_available_REF = yes }
		if = { limit = { NOT = { check_variable = { WA_AI_queued_REF > 0 } } } # first queued level: baseline the built count
			set_variable = { WA_AI_last_level_REF = building_level@synthetic_refinery }
		}
		add_to_variable = { WA_AI_queued_REF = 1 }
		set_state_flag = { flag = WA_AI_queued_REF_recent value = 1 days = @WA_AI_QUEUED_REFINERY_TTL_DAYS }
	}
	add_building_construction = { type = synthetic_refinery level = 1 instant_build = no }
}
