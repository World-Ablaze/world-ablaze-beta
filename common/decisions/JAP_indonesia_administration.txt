# Merging the three administrations of the occupied Dutch East Indies.
# Which decision appears depends on the course the Greater East Asia Conference set.
JAP_southern_expansion = {

	# Independence course: the administrations become the State of Indonesia at once
	JAP_recognise_east_indies_independence = {
		icon = generic_nationalism
		allowed = {
			tag = JAP
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = JAP_indonesian_administrations_in_hand_tt
				NOT = {
					any_country = {
						JAP_is_indonesian_military_administration = yes
						NOT = { is_subject_of = ROOT }
					}
				}
			}
		}
		visible = {
			has_completed_focus = JAP_develop_the_south
			NOT = { country_exists = IPI }
			JAP_has_indonesian_military_administration = yes
		}
		cost = 50
		fire_only_once = yes
		ai_will_do = {
			factor = 200
		}
		complete_effect = {
			custom_effect_tooltip = JAP_recognise_east_indies_independence_tt
			hidden_effect = {
				JAP_unify_indonesian_military_administrations_effect = yes
				IPI = {
					JAP_grant_statehood_effect = yes
				}
			}
		}
	}

	# Exploitation course: the administrations are merged but stay a protectorate
	JAP_unify_east_indies_administration = {
		icon = generic_nationalism
		allowed = {
			tag = JAP
		}
		available = {
			custom_trigger_tooltip = {
				tooltip = JAP_indonesian_administrations_in_hand_tt
				NOT = {
					any_country = {
						JAP_is_indonesian_military_administration = yes
						NOT = { is_subject_of = ROOT }
					}
				}
			}
		}
		visible = {
			has_completed_focus = JAP_exploit_the_south
			NOT = { country_exists = IPI }
			JAP_has_indonesian_military_administration = yes
		}
		cost = 50
		fire_only_once = yes
		ai_will_do = {
			factor = 200
		}
		complete_effect = {
			custom_effect_tooltip = JAP_unify_east_indies_administration_tt
			hidden_effect = {
				JAP_unify_indonesian_military_administrations_effect = yes
			}
		}
	}
}
