# The Bullet Train Plan (focus JAP_ind_bullet_train_plan): repeating decisions that build railway, section by section.
# Built like the Atlantic Wall (common/decisions/GER.txt, GER_atlantik_wall_west_france and _start): the _start decision switches
# the running one on and off; the running one ties up 15 civilian factories, builds one railway level when it ends, costs one
# step of economic fatigue, and starts again by itself, 4 runs in all. The days do not grow from run to run.
# The last three sections are the army tree's Greater East Asia Trunk Railway (JAP_island_defence_revision) carried further;
# what the overland routes carry home follows their railway level (common/scripted_effects/JAP_corridor_effects.txt).
economy_decisions = {

	JAP_bullet_train_home_start = {
		icon = generic_construction

		allowed = {
			original_tag = JAP
		}

		visible = {
			has_completed_focus = JAP_ind_bullet_train_plan
			check_variable = { JAP_bullet_train_home_count < 4 }
		}

		available = {
			282 = { is_controlled_by = ROOT }
			529 = { is_controlled_by = ROOT }
			if = {    # not needed to stop
				limit = { NOT = { has_decision = JAP_bullet_train_home } }
				num_of_civilian_factories_available_for_projects > 14
			}
		}

		highlight_states = {
			highlight_state_targets = {
				state = 282
				state = 532
				state = 531
				state = 529
			}
		}

		cost = 0
		on_map_mode = map_and_decisions_view
		fire_only_once = no

		complete_effect = {
			if = {
				limit = { NOT = { has_decision = JAP_bullet_train_home } }
				custom_override_tooltip = {
					tooltip = {
						localization_key = JAP_bullet_train_start_tt
					}
					activate_decision = JAP_bullet_train_home
				}
				set_country_flag = JAP_bullet_train_home
			}
			else = {
				custom_override_tooltip = {
					tooltip = {
						localization_key = JAP_bullet_train_stop_tt
					}
					remove_decision_on_cooldown = JAP_bullet_train_home
				}
				clr_country_flag = JAP_bullet_train_home
			}
		}

		ai_will_do = {
			factor = 10
			modifier = {
				factor = 0
				has_decision = JAP_bullet_train_home    # the AI never stops it
			}
			modifier = {
				factor = 0
				num_of_civilian_factories_available_for_projects < 30
			}
		}
	}
