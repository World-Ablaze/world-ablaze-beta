JAP_tokyo_express = {

	# Detaches the Combined Fleet's administration into a semi-autonomous naval subject (IJN),
	# based on states 633 (Marshall Islands) and 684 (Caroline Islands).
	# Pattern follows the debug_jr_establish_* / real IPB establishment block in JAP.txt and
	# zzz_JAP_rework_debug.txt: release -> configure new tag -> puppet + set_autonomy from JAP ->
	# add_to_war into JAP's ongoing wars.
	JAP_establish_ijn = {
		icon = generic_naval

		visible = {
			original_tag = JAP
			NOT = { country_exists = IJN }
		}

		available = {
			owns_state = 633
			owns_state = 684
			controls_state = 633
			controls_state = 684
		}

		cost = 25

		complete_effect = {
			release = IJN
			IJN = {
				transfer_state = 633
				transfer_state = 684
				set_state_controller = 633
				set_state_controller = 684
				set_capital = { state = 684 }
				create_country_leader = {
					name = "Imperial General Headquarters, Navy"
					desc = ""
					picture = "gfx//leaders//JAP//Portrait_Japan_Mitsumasa_Yonai.dds"
					expire = "1965.1.1"
					ideology = fascism_ideology
					traits = { }
				}
			}
			if = {
				limit = { JAP = { has_government = fascism } }
				IJN = {
					set_politics = {
						ruling_party = fascism
						elections_allowed = no
					}
				}
			}
			else_if = {
				limit = { JAP = { has_government = democratic } }
				IJN = {
					set_politics = {
						ruling_party = democratic
						elections_allowed = no
					}
				}
			}
			else_if = {
				limit = { JAP = { has_government = communism } }
				IJN = {
					set_politics = {
						ruling_party = communism
						elections_allowed = no
					}
				}
			}
			else = {
				IJN = {
					set_politics = {
						ruling_party = neutrality
						elections_allowed = no
					}
				}
			}
			JAP = {
				puppet = IJN
				set_autonomy = {
					target = IJN
					autonomous_state = autonomy_integrated_puppet
				}
			}
			every_country = {
				limit = {
					has_war_with = JAP
				}
				IJN = {
					add_to_war = { targeted_alliance = JAP enemy = PREV hostility_reason = asked_to_join }
				}
			}
		}
	}

	# Prototype "Tokyo Express" run into the Bismarck Sea. Detaches 8 destroyers to IJN and
	# grants the 4 relevant states a stacking supply bonus (5% per surviving destroyer, up to
	# 40%) for as many 7-day ticks as JAP_kantaiha_influence dictated at activation.
	JAP_tokyo_express_bismarck = {
		icon = generic_naval

		visible = {
			original_tag = JAP
			IJN = { is_subject_of = ROOT }
		}

		available = {
			NOT = { has_country_flag = JAP_te_active }
			check_variable = { num_ships_with_type@destroyer > 7 }
			check_variable = { JAP_kantaiha_influence > 0 }
		}

		cost = 25

		complete_effect = {
			# destroy+create instead of transfer_ship so the flotilla spawns in the reserve
			# fleet at IJN's capital port (684 Caroline Islands = Truk), right in front of
			# the target sea zone.
			destroy_ships = {
				type = destroyer
				count = 8
			}
			IJN = {
				create_ship = { type = jap_destroyer_hull_1 equipment_variant = "Minekaze Class" creator = JAP amount = 8 }
			}
			set_country_flag = JAP_te_active
			IJN = {
				set_country_flag = IJN_te_target_bismarck
			}
			set_variable = { JAP_te_rounds = JAP_kantaiha_influence }
			# The flotilla was just spawned at full strength, so seed the band directly
			# instead of sampling (create_ship registration timing is unverified).
			set_variable = { JAP_te_band = 8 }
			JAP_te_push_supply_effect = yes
			737 = { add_dynamic_modifier = { modifier = JAP_te_supply } }
			1004 = { add_dynamic_modifier = { modifier = JAP_te_supply } }
			634 = { add_dynamic_modifier = { modifier = JAP_te_supply } }
			1002 = { add_dynamic_modifier = { modifier = JAP_te_supply } }
			activate_mission = JAP_te_tick_mission
		}
	}

	# Self-restarting timer mission, modelled directly on JAP_southern_development_fund_mission
	# (common/decisions/JAP.txt, category JAP_southern_expansion): ticks every 7 days, counts
	# down JAP_te_rounds, and re-activates itself while rounds remain and IJN still has destroyers.
	JAP_te_tick_mission = {
		icon = generic_naval

		allowed = {
			always = no
		}

		available = {
			always = no
		}

		activation = {
			always = no
		}

		days_mission_timeout = 7
		is_good = yes
		fire_only_once = no

		timeout_effect = {
			subtract_from_variable = { JAP_te_rounds = 1 }
			# Weekly sampling: the buff follows last week's surviving destroyer count,
			# so ships sunk mid-week only lower it on this tick.
			JAP_te_sample_band_effect = yes
			JAP_te_push_supply_effect = yes
			if = {
				limit = {
					check_variable = { JAP_te_rounds > 0 }
					check_variable = { JAP_te_band > 0 }
				}
				activate_mission = JAP_te_tick_mission
			}
			else = {
				# Rounds exhausted, or the whole flotilla was sunk (band = 0 -> the run failed).
				IJN = {
					transfer_navy = { target = JAP }
					clr_country_flag = IJN_te_target_bismarck
				}
				clr_country_flag = JAP_te_active
				737 = { remove_dynamic_modifier = { modifier = JAP_te_supply } }
				1004 = { remove_dynamic_modifier = { modifier = JAP_te_supply } }
				634 = { remove_dynamic_modifier = { modifier = JAP_te_supply } }
				1002 = { remove_dynamic_modifier = { modifier = JAP_te_supply } }
			}
		}

		complete_effect = {
		}
	}
}
