JAP_tokyo_express = {

	# Detaches the Combined Fleet's administration into a semi-autonomous naval subject (IJN),
	# based on states 633 (Marshall Islands) and 684 (Caroline Islands).
	# Pattern follows the debug_jr_establish_* / real IPB establishment block in JAP.txt and
	# zzz_JAP_rework_debug.txt: release -> configure new tag -> puppet + set_autonomy from JAP ->
	# add_to_war into JAP's ongoing wars.
	JAP_establish_ijn = {
		icon = generic_naval

		visible = {
			original_tag = JAP
			NOT = { country_exists = IJN }
		}

		available = {
			owns_state = 633
			owns_state = 684
			controls_state = 633
			controls_state = 684
		}

		cost = 25

		complete_effect = {
			release = IJN
			IJN = {
				transfer_state = 633
				transfer_state = 684
				set_state_controller = 633
				set_state_controller = 684
				set_capital = { state = 684 }
				create_country_leader = {
					name = "Imperial General Headquarters, Navy"
					desc = ""
					picture = "gfx//leaders//JAP//Portrait_Japan_Mitsumasa_Yonai.dds"
					expire = "1965.1.1"
					ideology = fascism_ideology
					traits = { }
				}
				# Provisional admiral: one more copy out of WA's AI cloning facility
				# (JAP_isoroku_yamamoto_ai in common/characters/JAP.txt), stats verbatim.
				# create_navy_leader prefixes gfx/leaders/<TAG>/ to the picture on its own,
				# so this needs a bare filename plus a copy of the .dds in gfx/leaders/IJN/.
				create_navy_leader = {
					name = JAP_isoroku_yamamoto
					picture = "Portrait_Japan_Isoroku_Yamamoto.dds"
					traits = { bold aviation_enthusiast superior_tactician veteran_captain spotter air_controller coordinated_strike }
					skill = 5
					attack_skill = 5
					defense_skill = 2
					maneuvering_skill = 3
					coordination_skill = 6
				}
				# Fixed fascist 100% regardless of JAP's government (user ruling 2026-07-14).
				set_politics = {
					ruling_party = fascism
					elections_allowed = no
				}
			}
			JAP = {
				puppet = IJN
				# integrated_puppet is disallowed for fascist governments AND for JAP as
				# overlord (common/autonomous_states/integrated_puppet.txt), so use the
				# imperial protectorate level, proven with the IP* subjects.
				set_autonomy = {
					target = IJN
					autonomous_state = autonomy_wtt_imperial_protectorate
				}
			}
			# Popularity and party name go AFTER puppet/set_autonomy: the puppeting step can
			# reshuffle the subject's politics, which kept overwriting an earlier 100%.
			IJN = {
				# set_popularities is silently ignored unless every ideology is listed
				# and the total is exactly 100.
				set_popularities = {
					fascism = 100
					democratic = 0
					communism = 0
					neutrality = 0
				}
				set_party_name = {
					ideology = fascism
					long_name = IJN_fascism_party_long
					name = IJN_fascism_party
				}
				# No fuel income of its own -> hidden fuel_gain idea (invisible on the UI).
				add_ideas = IJN_te_fuel_stockpile
			}
			every_country = {
				limit = {
					has_war_with = JAP
				}
				IJN = {
					add_to_war = { targeted_alliance = JAP enemy = PREV hostility_reason = asked_to_join }
				}
			}
		}
	}

	# "Tokyo Express" run, generalized to every adopted sea region (SE Asia, Indian Ocean,
	# Bismarck/Solomons). Launched from the MAP, not the decision panel, following the
	# economy_lift_great_depression_state pattern (_economy_fatigue.txt): state_target +
	# on_map_mode. Clicking a coastal state runs the express into that state's home sea
	# region and buffs every state touching that region (5% supply per surviving destroyer,
	# up to 40%) for as many 7-day ticks as JAP_kantaiha_influence dictated at activation.
	# State->region data is seeded on startup by the auto-generated
	# JAP_te_seed_regions_effect (_modtools/Generate-TokyoExpressRegions.py).
	JAP_tokyo_express_run = {
		icon = generic_naval

		state_target = yes
		on_map_mode = map_only

		# DEBUG: every seeded state shows the icon unconditionally (user ruling 2026-07-15).
		# Before release, restrict to states ROOT controls (see PHI_guerrillas.txt for the
		# CONTROLLER pattern).
		target_trigger = {
			FROM = { has_state_flag = JAP_te_eligible }
		}

		visible = {
			original_tag = JAP
		}

		available = {
			IJN = { is_subject_of = ROOT }
			NOT = { has_country_flag = JAP_te_active }
			check_variable = { num_ships_with_type@destroyer > 7 }
			check_variable = { JAP_kantaiha_influence > 0 }
		}

		# AI Japan never map-clicks this on its own (manual/scripted use only for now).
		ai_will_do = {
			factor = 0
		}

		cost = 25

		fire_only_once = no

		complete_effect = {
			# The clicked state's dominant sea region becomes the run target; the generated
			# ai_strategy blocks in ai_strategy/IJN.txt key off this global variable.
			set_variable = { global.JAP_te_region_active = FROM.JAP_te_home_region }
			# Real ships change hands (no destroy+create swap): 8 transfer_ship calls, each
			# preferring a named Minekaze-class boat from the 1936 OOB so second-line destroyers
			# get requisitioned before the modern fleet. A sunk name falls back to whatever
			# destroyer the engine picks. Ships keep their position and sail to the zone
			# themselves under the 7-day flank speed modifier below (ship teleports do not
			# exist in the engine; teleport_armies is army-only, tested 2026-07-14).
			transfer_ship = { prefer_name = "Minekaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Sawakaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Okikaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Yakaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Nadakaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Akikaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Hakaze" type = destroyer target = IJN }
			transfer_ship = { prefer_name = "Nokaze" type = destroyer target = IJN }
			IJN = {
				add_dynamic_modifier = { modifier = IJN_te_flank_speed days = 7 }
			}
			set_country_flag = JAP_te_active
			set_variable = { JAP_te_rounds = JAP_kantaiha_influence }
			# The flotilla was just spawned at full strength, so seed the band directly
			# instead of sampling (create_ship registration timing is unverified).
			set_variable = { JAP_te_band = 8 }
			JAP_te_apply_supply_effect = yes
			activate_mission = JAP_te_tick_mission
		}
	}

	# Self-restarting timer mission, modelled directly on JAP_southern_development_fund_mission
	# (common/decisions/JAP.txt, category JAP_southern_expansion): ticks every 7 days, counts
	# down JAP_te_rounds, and re-activates itself while rounds remain and IJN still has destroyers.
	JAP_te_tick_mission = {
		icon = generic_naval

		allowed = {
			always = no
		}

		available = {
			always = no
		}

		activation = {
			always = no
		}

		days_mission_timeout = 7
		is_good = yes
		fire_only_once = no

		timeout_effect = {
			subtract_from_variable = { JAP_te_rounds = 1 }
			# Weekly sampling: the buff follows last week's surviving destroyer count,
			# so ships sunk mid-week only lower it on this tick.
			JAP_te_sample_band_effect = yes
			JAP_te_push_supply_effect = yes
			if = {
				limit = {
					check_variable = { JAP_te_rounds > 0 }
					check_variable = { JAP_te_band > 0 }
				}
				activate_mission = JAP_te_tick_mission
			}
			else = {
				# Rounds exhausted, or the whole flotilla was sunk (band = 0 -> the run failed).
				IJN = {
					transfer_navy = { target = JAP }
				}
				clr_country_flag = JAP_te_active
				JAP_te_remove_supply_effect = yes
			}
		}

		complete_effect = {
		}
	}
}
