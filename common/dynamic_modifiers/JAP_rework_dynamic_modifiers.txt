# Tokyo Express (Bismarck Sea prototype): single stacking supply bonus while the run is active.
# The strength lives in JAP_te_supply_var on the state itself (pattern: vanilla
# SOV_increased_metals_production) and is refreshed by JAP_te_tick_mission from last week's
# surviving IJN destroyer count, so losses show up on the next tick rather than instantly.
# local_supplies_for_controller is engine-guaranteed to feed the state controller only,
# and the enable check restricts it to JAP-controlled states, so the bonus is JAP-only.
# Tokyo Express: 7-day burst applied to IJN right after the destroyers change hands, so the
# requisitioned flotilla races to the target sea zone instead of crawling there (pseudo-warp;
# the engine has no ship teleport). Added with days = 7, so it expires on its own.
IJN_te_flank_speed = { #COUNTRY MODIFIER
	enable = { always = yes }
	icon = GFX_JAP_te_supply_icon

	naval_speed_factor = 50
}

JAP_te_supply = { #STATE MODIFIER
	# Conditions do NOT belong in enable: trigger blocks there are silently discarded
	# and the modifier stays always-on (observed in-game 2026-07-15 - every South China
	# Sea coast lit up, Chinese Guangdong included). Attachment is managed entirely by
	# JAP_te_refresh_supply_effect on the WEEKLY tick: only Japan-controlled states get
	# the modifier, but a state lost mid-week keeps it until the next tick - the landed
	# stockpile is CAPTURED, feeding the new controller (user ruling 2026-07-15; also
	# why there is deliberately no remove_trigger).
	enable = { always = yes }

	icon = GFX_JAP_te_supply_icon

	local_supplies_for_controller = JAP_te_supply_var
}
