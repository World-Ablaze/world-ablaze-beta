# Tokyo Express (Bismarck Sea prototype): stacking supply relief while the run is active,
# banded by how many destroyers IJN currently has assigned to it.
# NOTE: cross-scope check_variable = { var = IJN.num_ships_with_type@destroyer ... } is
# untested syntax, see implementation report.
JAP_te_supply_1 = { #STATE MODIFIER
	enable = {
		CONTROLLER = {
			OR = {
				tag = JAP
				is_subject_of = JAP
			}
		}
		JAP = { has_country_flag = JAP_te_active }
		check_variable = { var = IJN.num_ships_with_type@destroyer value = 0 compare = greater_than }
		check_variable = { var = IJN.num_ships_with_type@destroyer value = 2 compare = less_than }
	}
	icon = GFX_JAP_te_supply_icon

	supply_consumption_factor = -0.05
}
