# Tokyo Express (Bismarck Sea prototype): single stacking supply bonus while the run is active.
# The strength lives in JAP_te_supply_var on the state itself (pattern: vanilla
# SOV_increased_metals_production) and is refreshed by JAP_te_tick_mission from last week's
# surviving IJN destroyer count, so losses show up on the next tick rather than instantly.
# local_supplies_for_controller is engine-guaranteed to feed the state controller only,
# and the enable check restricts it to JAP-controlled states, so the bonus is JAP-only.
JAP_te_supply = { #STATE MODIFIER
	enable = {
		CONTROLLER = { tag = JAP }
		JAP = { has_country_flag = JAP_te_active }
	}
	icon = GFX_JAP_te_supply_icon

	local_supplies_for_controller = JAP_te_supply_var
}
