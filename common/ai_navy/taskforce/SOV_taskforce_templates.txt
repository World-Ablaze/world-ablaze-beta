# [sov-surface-navy] a small navy over four seas needs SMALL groups: each coast_defense objective takes
# one fleet, and the generic 5-20 destroyer patrol group folds every SOV destroyer into one, so one
# coast gets a fleet. Generic PatrolReconForce_1 / NavalInvasionSupport_1 exclude SOV so its hulls form
# these. Gone if `imgui show ai_navy` shows every SOV coast_defense with a fleet without these templates.

SOV_PatrolReconForce_1 = {
	allowed = {
		original_tag = SOV
	}
	ai_will_do = {
		factor = 5
	}
	mission = { naval_patrol }
	min_composition = {
		destroyer = {
			amount = 5
		}
	}
	optimal_composition = {
		destroyer = {
			amount = 10
		}
	}
}
