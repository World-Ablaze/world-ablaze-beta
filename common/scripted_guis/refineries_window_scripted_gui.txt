	refinery_window= {
		context_type = player_context
		window_name = "refinery_window_container"
		parent_window_token = trade_tab
		# [refineries] The panel is player bookkeeping; for an AI country its numbers are only
		# maintained in a -debug launch, so outside debug it must not render on one at all.
		visible = {
			WA_REFINERY_panel_is_live = yes
		}


		triggers = {

		}

		effects = {
			force_refresh_button_click = {
				# [refineries] player_context resolves to the selected country, so this button is
				# reachable on an AI one under `observe`. recalculate_* only writes panel variables
				# and is safe anywhere. manage_* is the PLAYER refinery controller and its
				# lack-of-resources loop trips on resource@iron < 0 whatever the target holds, so one
				# click would mass-close an AI's mills against its own controller. Keep the gate.
				recalculate_steel_mills = yes
				recalculate_alu_mills = yes
				recalculate_synth = yes
				if = {
					limit = { is_ai = no }
					manage_steel_mills = yes
					manage_alu_mills = yes
					manage_synth = yes
					calculate_resource_deficite = yes
				}
			}
		}
	}

}
scripted_gui = {
	steel_mill_assignment_window = {
		context_type = player_context

		window_name = "steel_count_container"
		parent_window_name = refinery_window_containerin
		visible = {
			WA_REFINERY_panel_is_live = yes
		}


		triggers = {
			# [refineries] open_*_target is a PLAYER setpoint and the only variable in this system
			# that states INTENT rather than realised capacity; the AI controller in
			# common/decisions/_economy_fatigue.txt never writes it, so for an AI it still holds its
			# on_startup value. Hide it rather than print a lie, and never write reality into it -
			# see documentation/WA_REFINERY_CONTROLLER_REVIEW.md sections 4 and 6.
			factories_used_txt_visible = { is_ai = no }
			factories_add_1x_click_enabled = { is_ai = no }
			factories_sub_1x_click_enabled = { is_ai = no }
		}
		effects = {
			factories_sub_1x_click = {
				subtract_from_variable = {
					open_steel_mill_target = 1
				}
				clamp_variable = {
					var = open_steel_mill_target
					min = 0
				}
			}
			factories_sub_1x_control_click = {
				subtract_from_variable = {
					open_steel_mill_target = 5
				}
				clamp_variable = {
					var = open_steel_mill_target
					min = 0
				}
			}
			factories_sub_1x_shift_click = {
				subtract_from_variable = {
					open_steel_mill_target = 10
				}
				clamp_variable = {
					var = open_steel_mill_target
					min = 0
				}
			}
			factories_add_1x_click = {
				add_to_variable = {
					open_steel_mill_target = 1
				}
				clamp_variable = {
					var = open_steel_mill_target
					max = total_steel_mill 
				}
			}
			factories_add_1x_control_click = {
				add_to_variable = {
					open_steel_mill_target = 5
				}
				clamp_variable = {
					var = open_steel_mill_target
					max = total_steel_mill 
				}
			}
			factories_add_1x_shift_click = {
				add_to_variable = {
					open_steel_mill_target = 10
				}
				clamp_variable = {
					var = open_steel_mill_target
					max = total_steel_mill 
				}
			}
		}
	}
}
