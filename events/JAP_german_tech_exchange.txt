add_namespace = jap_gte

# Outcomes of the German technology exchange, fired by the leg rolls in
# JAP_german_tech_exchange_effects.txt. The trigger drops a copy fired after a debug reset.

# Success: back in Japan. Repeatable; the first success also unlocks the stronger research-column focuses.
country_event = {
	id = jap_gte.2
	title = jap_gte.2.t
	desc = jap_gte.2.desc
	picture = GFX_report_event_generic_research_lab
	is_triggered_only = yes

	trigger = {
		has_country_flag = JAP_gte_in_transit
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = JAP_gte_in_transit
		}
	}

	option = {
		name = jap_gte.2.a
		random_list = {
			25 = { add_tech_bonus = { name = JAP_gte_bonus bonus = 1.0 uses = 1 category = cat_light_armor } }
			25 = { add_tech_bonus = { name = JAP_gte_bonus bonus = 1.0 uses = 1 category = cat_medium_armor } }
			25 = { add_tech_bonus = { name = JAP_gte_bonus bonus = 1.0 uses = 1 category = cat_heavy_armor } }
			25 = { add_tech_bonus = { name = JAP_gte_bonus bonus = 1.0 uses = 1 category = cat_medium_td category = cat_medium_spg } }
		}
		army_experience = 10
		# Same influence step as JAP_fifty_division_plan_timeout_effect (cap 7, dockyard factor moves with it).
		custom_effect_tooltip = JAP_kantaiha_influence_up_tt
		hidden_effect = {
			if = {
				limit = { check_variable = { JAP_kantaiha_influence < 7 } }
				add_to_variable = { JAP_kantaiha_influence = 1 }
				add_to_variable = { JAP_kantaiha_production_speed_dockyard_factor = 0.04 }
			}
		}
		if = {
			limit = { NOT = { has_country_flag = JAP_gte_done } }
			custom_effect_tooltip = JAP_gte_success_tt
		}
		hidden_effect = { set_country_flag = JAP_gte_done }
	}
}

# Failure: costs 30 political power; the decision can be taken again after 30 days.
country_event = {
	id = jap_gte.3
	title = jap_gte.3.t
	desc = jap_gte.3.desc
	picture = GFX_report_event_merchant_ship_01
	is_triggered_only = yes

	trigger = {
		has_country_flag = JAP_gte_in_transit
	}

	immediate = {
		hidden_effect = {
			clr_country_flag = JAP_gte_in_transit
			set_country_flag = { flag = JAP_gte_cooldown days = 30 value = 1 }
		}
	}

	option = {
		name = jap_gte.3.a
		add_political_power = -30
		custom_effect_tooltip = JAP_gte_failure_tt
	}
}
