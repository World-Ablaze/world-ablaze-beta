add_namespace = jap_pol_auto

# Focuses of the political tree that are taken for the player on their historical date and cannot be taken by
# hand, built like the army's (events/JAP_army_autonomy.txt): started by
# common/on_actions/JAP_political_autonomy_on_actions.txt, the focus shines with an in-progress flag for 21 days,
# then it is completed with a side message.

# the Hiranuma cabinet, 5 January 1939
country_event = {
	id = jap_pol_auto.1
	hidden = yes
	is_triggered_only = yes
	immediate = {
		activate_shine_on_focus = JAP_pol_hiranuma_cabinet
		set_country_flag = { flag = JAP_pol_hiranuma_cabinet_in_progress_flag days = 21 value = 1 }
		country_event = { id = jap_pol_auto.2 days = 21 }
	}
	option = {
		name = jap_pol_auto.1.a
	}
}

country_event = {
	id = jap_pol_auto.2
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { has_shine_effect_on_focus = JAP_pol_hiranuma_cabinet }
			complete_national_focus = {
				focus = JAP_pol_hiranuma_cabinet
				use_side_message = yes
				originator_name = JAP_court_originator
			}
			clr_country_flag = JAP_pol_hiranuma_cabinet_in_progress_flag
		}
	}
	option = {
		name = jap_pol_auto.2.a
	}
}
