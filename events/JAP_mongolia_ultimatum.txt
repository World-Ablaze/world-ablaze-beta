add_namespace = jap_pol_mongolia

# Ultimatum to Outer Mongolia (focus JAP_pol_ultimatum_to_outer_mongolia): the northern counterpart of the ultimatum to Thailand
# (jap_armor.835 and 836 in wa_jap_events.txt), built the same way.
country_event = {
	id = jap_pol_mongolia.1
	title = jap_pol_mongolia.1.t
	desc = jap_pol_mongolia.1.d
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	# Leave the Soviet side for Japan's. The AI does not while it is the Soviet Union's subject or ally.
	option = {
		name = jap_pol_mongolia.1.a
		ai_chance = {
			factor = 10
			modifier = {
				factor = 0
				OR = {
					is_subject_of = SOV
					is_in_faction_with = SOV
				}
			}
		}
		if = {
			limit = { is_subject_of = SOV }
			SOV = { set_autonomy = { target = ROOT autonomy_state = autonomy_free } }
		}
		if = {
			limit = { is_in_faction_with = SOV }
			SOV = { remove_from_faction = ROOT }
		}
		if = {
			limit = { JAP = { is_faction_leader = yes } }
			JAP = { add_to_faction = ROOT }
		}
		if = {
			limit = {
				JAP = { has_idea = JAP_co_prosperity_sphere_leader }
				NOT = { has_idea = JAP_co_prosperity_sphere_member }
			}
			add_ideas = JAP_co_prosperity_sphere_member
		}
	}

	option = {
		name = jap_pol_mongolia.1.b
		ai_chance = {
			factor = 90
		}
		JAP = {
			create_wargoal = {
				type = annex_everything
				target = ROOT
			}
			country_event = { id = jap_pol_mongolia.2 days = 1 }
		}
	}
}
