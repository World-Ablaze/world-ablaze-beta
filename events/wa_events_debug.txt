country_event = {
	id = wa_debug.9 # The unique id. nuke_dropped is the namespace, which can be used for better organization, and is used in localisation.

	hidden = yes # Optional : if set to "yes" no one will see the event

	is_triggered_only = yes

	immediate = {
		# debug mirror of the (Fix 90b-deleted) WA_AI_PC_can_afford_project arithmetic. Reads the
		# allocator's script constants (constant:wa_ai_pc.alloc.*) directly since 2026-08-16 - it was a
		# hand-kept `@` copy that had drifted to 0.35 against the allocator's 0.40.

		# check that the ai can afford to assign factories to a new project
		set_temp_variable = { max_factories_on_projects = ROOT.num_of_civilian_factories_available_for_projects }
		add_to_temp_variable = { max_factories_on_projects = ROOT.WA_AI_PC_assigned_factories_total }
		multiply_temp_variable = { max_factories_on_projects = constant:wa_ai_pc.alloc.fraction }

		set_temp_variable = { _pc_floor_max = ROOT.num_of_civilian_factories }
		multiply_temp_variable = { _pc_floor_max = constant:wa_ai_pc.alloc.stable_base_fraction }
		multiply_temp_variable = { _pc_floor_max = constant:wa_ai_pc.alloc.fraction }
		if = {
			limit = {
				ROOT = { num_of_civilian_factories > 19 has_capitulated = no surrender_progress < 0.3 }
				check_variable = { max_factories_on_projects < _pc_floor_max }
			}
			set_temp_variable = { max_factories_on_projects = _pc_floor_max }
		}

		set_temp_variable = { min_factories_on_projects = ROOT.WA_AI_PC_assigned_factories_total }
		add_to_temp_variable = { min_factories_on_projects = constant:wa_ai_pc.alloc.max_civs_per_project }

		if = {
			limit = {
				tag = SWE
			}
			log = "tag = SWE min_factories_on_projects = [?min_factories_on_projects] max_factories_on_projects [?max_factories_on_projects]"
		}

	}

	option = {
		trigger = {
			check_variable = { min_factories_on_projects < max_factories_on_projects }
		}
		name = "can invest"
	}

	option = {
		trigger = {
			check_variable = { min_factories_on_projects > max_factories_on_projects }
		}
		name = "can't invest"
	}

}
