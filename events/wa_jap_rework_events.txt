# Rice requisition lapses after Vichy's collapse (grace period expired)
country_event = {
	id = jap_rework.1
	title = jap_rework.1.t
	desc = jap_rework.1.d
	picture = GFX_report_event_generic_sign_treaty2

	is_triggered_only = yes

	option = {
		name = jap_rework.1.a
		clr_country_flag = JAP_rice_requisition_grace
		if = {
			limit = {
				NOT = {
					IPV = {
						is_subject_of = ROOT
					}
				}
			}
			remove_ideas = JAP_indochinese_rice_requisition
		}
	}
}

# The Great Philippine Uprising - fires once US-led forces return to the archipelago from 1944
country_event = {
	id = jap_rework.2
	hidden = yes

	mean_time_to_happen = { days = 2 }

	fire_only_once = yes

	trigger = {
		tag = PHI
		date > 1944.1.1
		PHI = { has_capitulated = yes }
		has_war_with = JAP
		NOT = { has_country_flag = PHI_uprising }
		any_state = {
			is_core_of = PHI
			CONTROLLER = {
				OR = {
					tag = USA
					is_in_faction_with = USA
				}
				has_war_with = JAP
			}
		}
	}

	immediate = {
		set_country_flag = PHI_uprising
		load_oob = "PHI_Uprising"
		every_state = {
			limit = {
				is_core_of = PHI
				has_dynamic_modifier = { modifier = PHI_organized_guerrilla_activity }
			}
			add_resistance = 25
		}
	}

	option = {}
}
