#		event wa_doc.3 GER      the whole SHIPPED gift as ger_armor.1001 fires it (all tier-1
#		                        tracks, schwerpunkt, army XP -80), then report. Ignores the gate.

#	Output: logs/game.log under "LAND DOCTRINE TEST". Read the `scope :` line FIRST - anything but
#	1 1 1 1 0 means STOP (see the effect header). Then the VERDICT line.

#		                                     After wa_doc.3: t1 done=1 on all three lines, arm t2 sub=1)

#		any other tag                       (is-GER=0, gate-open=0; the report still reads its tracks)

	}

# The whole shipped gift on ROOT (mirror of ger_armor.1001's immediate), then report.

	hidden = yes
	is_triggered_only = yes
	fire_only_once = no

		}
		WA_TEST_DOC_report = yes
	}
