# Expiry of the naval-invasion attack penalty applied by WA_AI_DIVISION_spawn_invasion.
# Fired ON the invasion target 14 days after the landing: ROOT = target, FROM = invader.
# Deliberately ungated (no is_ai / historical check): this event only ever REMOVES a penalty,
# and any trigger that can fail would strand -25% attack on FROM for the rest of the campaign.
country_event = {
	id = WA_AI_invasions.1
	hidden = yes

	is_triggered_only = yes

	trigger = {
		FROM = {
			has_relation_modifier = {
				target = ROOT
				modifier = WA_AI_invasions_modifier
			}
		}
	}

	immediate = {
		FROM = {
			remove_relation_modifier = {
				target = ROOT
				modifier = WA_AI_invasions_modifier
			}
			# Probe instrumentation (campaign checklist R14) — mirrors the increment in
			# WA_AI_DIVISION_spawn_invasion. _active must fall back to 0 between landings.
			add_to_variable = { WA_AI_invasions_dbg_removes = 1 }
			subtract_from_variable = { WA_AI_invasions_dbg_active = 1 }
		}
	}
}

	hidden = yes

	is_triggered_only = yes

	trigger = {
		is_ai = yes

	}

	immediate = {
		# Get target country tag
		1018 = { # Innlandet
			CONTROLLER = {
				set_temp_variable = { _target_country = id }
			}
		}

		if = {
			limit = {
				# Triggers for invasion target country and states
				var:_target_country = {
					has_war_with = ROOT
					controls_state = 1018 # Innlandet
				}

				# PLAYER ONLY seazone check
				if = {
					limit = {
						var:_target_country = { is_ai = no }
						NOT = { has_country_flag = { flag = WA_AI_invasions.3_flag days > 179 } }
					}
					NOT = { has_enemy_naval_control = 250 } # Skagerrak
				}

				# Triggers for invasion launch country
				surrender_progress < 0.1

				# Triggers for 'invasion launch states'
				99 = { is_controlled_by_ROOT_or_ally = yes } # Jutland

				# Other misc triggers for overall game state
				# Allies control africa? Italy?

			}
			# Account for checking for more defenses with any_province_building_level?
			set_temp_variable = { _divisions_per_province = 2 }
			WA_AI_DIVISION_adjust_invasion_for_difficulty = yes

			set_temp_variable = { _division_template = 7 }

			set_temp_variable = { _invasion_state = 1020 } #Agder
			clear_temp_array = _invasion_state_provinces
			add_to_temp_array = { _invasion_state_provinces = 9296 }
			WA_AI_DIVISION_spawn_invasion = yes

			set_temp_variable = { _invasion_state = 1017 } #Rogaland

			set_temp_variable = { _invasion_state = 142 } # Vestlandet
			clear_temp_array = _invasion_state_provinces
			add_to_temp_array = { _invasion_state_provinces = 11100 }
			add_to_temp_array = { _invasion_state_provinces = 3096 }
			add_to_temp_array = { _invasion_state_provinces = 122 }
			WA_AI_DIVISION_spawn_invasion = yes

			set_temp_variable = { _invasion_state = 143 } #Trondelag
			clear_temp_array = _invasion_state_provinces
			add_to_temp_array = { _invasion_state_provinces = 3022 }
			add_to_temp_array = { _invasion_state_provinces = 107 }
			WA_AI_DIVISION_spawn_invasion = yes

			set_temp_variable = { _invasion_state = 144 } # Nord-Norge
			clear_temp_array = _invasion_state_provinces
			add_to_temp_array = { _invasion_state_provinces = 192 }

			set_temp_variable = { _division_template = 2 }

			set_temp_variable = { _invasion_state = 110 } #Ostlandet
			clear_temp_array = _invasion_state_provinces

	id = WA_AI_invasions.46
	hidden = yes

	is_triggered_only = yes

	trigger = {
		is_ai = yes
		WA_AI_DIFFICULTY_is_historical = yes
	}

	immediate = {
		# Get target country tag
		459 = { # Algiers
			CONTROLLER = {
				set_temp_variable = { _target_country = id }
			}
		}

		if = {
			limit = {
				# Triggers for invasion target country and states
				var:_target_country = {
					has_war_with = ROOT
					controls_state = 459 # Algiers
				}

				# PLAYER ONLY seazone check
				if = {
					limit = {
						var:_target_country = { is_ai = no }
						NOT = { has_country_flag = { flag = WA_AI_invasions.46_flag days > 179 } }
					}
					NOT = { has_enemy_naval_control = 48 } # African Coast
				}

				# Triggers for invasion launch country
				surrender_progress < 0.1

			}
			WA_AI_DIVISION_adjust_invasion_for_difficulty = yes

			# Gibraltar condition - only land from Med if Gibraltar is held by allies
			if = {
				limit = { NOT = { any_enemy_country = { controls_state = 118 } } }
				set_temp_variable = { _divisions_per_province = 2 }
				WA_AI_DIVISION_adjust_invasion_for_difficulty = yes
				set_temp_variable = { _invasion_state = 459 } # Algiers
				clear_temp_array = _invasion_state_provinces
				add_to_temp_array = { _invasion_state_provinces = 7132  } # (Algiers)
				add_to_temp_array = { _invasion_state_provinces = 4116  }
				add_to_temp_array = { _invasion_state_provinces = 11923 }
				add_to_temp_array = { _invasion_state_provinces = 1145  }
				add_to_temp_array = { _invasion_state_provinces = 11939 }
				add_to_temp_array = { _invasion_state_provinces = 9964  }
			set_temp_variable = { _divisions_per_province = 1 }
			WA_AI_DIVISION_adjust_invasion_for_difficulty = yes
			set_temp_variable = { _division_template = 1 }
			WA_AI_DIVISION_spawn_invasion = yes

			set_temp_variable = { _division_template = 4 }
			WA_AI_DIVISION_spawn_invasion = yes

			set_temp_variable = { _division_template = 1 }
			}
		}
		else_if = {
			limit = {
				# Only refire if the invasion target is still valid
				var:_target_country = {
					has_war_with = ROOT
					controls_state = 459 # Algiers
				}
			}
			country_event = { id = WA_AI_invasions.46 days = 7 }
		}

		# Seazone control delay mechanic
		if = {
			limit = {
				has_enemy_naval_control = 48 # African Coast
				NOT = { has_country_flag = WA_AI_invasions.46_flag }
			}
			set_country_flag = WA_AI_invasions.46_flag
		}
		else_if = {
			limit = {
				has_country_flag = { flag = WA_AI_invasions.46_flag days > 179 }
			}
			clr_country_flag = WA_AI_invasions.46_flag
		}
	}
}
