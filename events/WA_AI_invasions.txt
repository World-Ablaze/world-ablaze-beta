country_event = {
	id = WA_AI_invasions.1
	hidden = yes

	is_triggered_only = yes

	immediate = {
		# THIS = ROOT = invader (event fired on self by WA_AI_DIVISION_spawn_invasion)
		for_each_loop = {
			array = WA_AI_invasions_pending_targets
			value = _pending_target
			# _active counts pairs inside their 14-day window; decrement unconditionally —
			# even if the target was annexed meanwhile (modifier GC'd with the relation),
			# this pair's window is over. Keeps active = adds - expired pairs, returning to
			# 0 outside landing windows regardless of annexations.
			subtract_from_variable = { WA_AI_invasions_dbg_active = 1 }
			if = {
				limit = {
					meta_trigger = {
						text = {
							has_relation_modifier = {
								target = [COUNTRY_TAG]
								modifier = WA_AI_invasions_modifier
							}
						}
						COUNTRY_TAG = "[?_pending_target.GetTag]"
						# debug = yes
					}
				}
				meta_effect = {
					text = {
						remove_relation_modifier = {
							target = [COUNTRY_TAG]
							modifier = WA_AI_invasions_modifier
						}
					}
					COUNTRY_TAG = "[?_pending_target.GetTag]"
					# debug = yes
				}
				# Probe instrumentation (campaign checklist R14) — mirrors the increment in
				# WA_AI_DIVISION_spawn_invasion. removes counts actual modifier removals, so
				# adds - removes = pairs whose target died before expiry (accepted residual).
				add_to_variable = { WA_AI_invasions_dbg_removes = 1 }
			}
		}
		clear_array = WA_AI_invasions_pending_targets
	}
}

# Mulberry dismantling: removes the two artificial D-Day harbours placed by
# WA_AI_invasions.101 (Mulberry A, province 3579) and .102 (Mulberry B, province
# 13851) 120 days after each was built. Both spawners schedule this event, so it
# can fire twice - set_building_level to 0 is idempotent, the second firing is a
# no-op on the province already cleared and removes the other harbour if built.
country_event = {
	id = WA_AI_invasions.100
	hidden = yes

	is_triggered_only = yes

	immediate = {
		15 = { # Lower Normandy
			set_building_level = {
				type = naval_base
				province = { id = 3579 }
				level = 0
				instant_build = yes
			}
		}
		1016 = { # Upper Normandy
			set_building_level = {
				type = naval_base
				province = { id = 13851 }
				level = 0
				instant_build = yes
			}
		}
	}
}

# Mulberry A placement. Scheduled by WA_AI_invasions.2 at the landing; waits for
# state 15 to be ROOT/ally-controlled so the Nero-decree demolition (which fires
# synchronously on the state-control flip, once per state) has already resolved
# before the harbour exists. Retries every 3 days, gives up 45 days after the
# landing (beachhead failed). Level 5 = 25 supply (NAVAL_FLOW_PER_LEVEL 5.0).
country_event = {
	id = WA_AI_invasions.101
	hidden = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { has_country_flag = WA_AI_invasions.101_flag }
			}
			set_country_flag = WA_AI_invasions.101_flag
		}

		if = {
			limit = {
				15 = { is_controlled_by_ROOT_or_ally = yes } # Lower Normandy
			}
			15 = {
				set_building_level = {
					type = naval_base
					province = { id = 3579 }
					level = 5
					instant_build = yes
				}
			}
			country_event = { id = WA_AI_invasions.100 days = 120 }
		}
		else_if = {
			limit = {
				NOT = { has_country_flag = { flag = WA_AI_invasions.101_flag days > 45 } }
			}
			country_event = { id = WA_AI_invasions.101 days = 3 }
		}
	}
}
