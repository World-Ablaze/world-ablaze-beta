add_namespace = usa_armor

# Drops the start-state skeleton "Infantry Paper Division" template (and its units)
# a few days into the game. AI only - the player keeps the paper divisions.
# Date check is start-agnostic: on later bookmarks it is already true, so the
# event still fires within a few days of the campaign starting.
country_event = {
	id = usa_armor.002
	hidden = yes

	mean_time_to_happen = { days = 1 }

	fire_only_once = yes

	trigger = {
		tag = USA
		is_ai = yes
		date > 1936.1.5
		has_template = "Infantry Paper Division"
	}

	immediate = {
		delete_unit_template_and_units = {
			division_template = "Infantry Paper Division"
			disband = yes
		}
	}

	option = {}

}
