add_namespace = jap_navy_auto

# Second Stage operations the Combined Fleet takes on its own, built like the army's
# (events/JAP_army_autonomy.txt). They are started by common/on_actions/JAP_navy_autonomy_on_actions.txt:
# the focus shines with an in-progress flag for its 21 days, then it is completed with a side message.

# Indian Ocean Raid
country_event = {
	id = jap_navy_auto.1
	hidden = yes
	is_triggered_only = yes
	immediate = {
		activate_shine_on_focus = JAP_operation_c
		set_country_flag = { flag = JAP_operation_c_in_progress_flag days = 21 value = 1 }
		country_event = { id = jap_navy_auto.2 days = 21 }
	}
	option = {
		name = jap_navy_auto.1.a
	}
}

country_event = {
	id = jap_navy_auto.2
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { has_shine_effect_on_focus = JAP_operation_c }
			complete_national_focus = {
				focus = JAP_operation_c
				use_side_message = yes
				originator_name = JAP_combined_fleet_originator
			}
			clr_country_flag = JAP_operation_c_in_progress_flag
		}
	}
	option = {
		name = jap_navy_auto.2.a
	}
}
