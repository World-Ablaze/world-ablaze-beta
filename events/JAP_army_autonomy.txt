add_namespace = jap_army_auto

# Focuses the army takes on its own, built like the vanilla German inner circle
# (GER_inner_circle_todt.01/.02): the focus starts shining with an in-progress flag for the
# focus time, then it is completed with a side message naming who took it.

# Manchukuo-Soviet Border Dispute Guidelines: the Kwantung Army, after Lake Khasan or from April 1939
country_event = {
	id = jap_army_auto.1
	hidden = yes
	fire_only_once = yes
	trigger = {
		original_tag = JAP
		NOT = { has_completed_focus = JAP_border_conflict_guidelines }
		NOT = { has_shine_effect_on_focus = JAP_border_conflict_guidelines }
		OR = {
			has_global_flag = JAP_khasan_incident_over
			date > 1939.4.24
		}
	}
	immediate = {
		activate_shine_on_focus = JAP_border_conflict_guidelines
		set_country_flag = { flag = JAP_border_conflict_guidelines_in_progress_flag days = 21 value = 1 }
		country_event = { id = jap_army_auto.2 days = 21 }
	}
	option = {
		name = jap_army_auto.1.a
	}
}

country_event = {
	id = jap_army_auto.2
	hidden = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { has_shine_effect_on_focus = JAP_border_conflict_guidelines }
			complete_national_focus = {
				focus = JAP_border_conflict_guidelines
				use_side_message = yes
				originator_name = JAP_kwantung_army_originator
			}
			clr_country_flag = JAP_border_conflict_guidelines_in_progress_flag
		}
	}
	option = {
		name = jap_army_auto.2.a
	}
}
