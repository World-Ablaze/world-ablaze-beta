add_namespace = jap_khasan

# The first Soviet border conflict seen from Tokyo. The Soviet focus
# SOV_far_east_fortification gives Japan the 400-day mission
# SOV_border_conflict_warning_JAP and, when the Soviet mission
# SOV_danger_of_border_friction runs out, a border war starts between Hamgyung
# (846) and Vladivostok (408). jap_khasan.1 warns a human Japan about 60 days
# before that, so that divisions can be brought up to Hamgyung. Nothing here
# changes the Soviet side or the conflict itself.

# Hidden: note the day the mission appeared, so that the warning can be timed
country_event = {
	id = jap_khasan.0
	hidden = yes

	fire_only_once = yes

	trigger = {
		tag = JAP
		has_active_mission = SOV_border_conflict_warning_JAP
		NOT = { has_country_flag = JAP_khasan_countdown }
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		set_country_flag = JAP_khasan_countdown
	}
}

# Warning, about 60 days before the Soviet mission runs out (400 - 1 - 339)
country_event = {
	id = jap_khasan.1
	title = jap_khasan.1.t
	desc = jap_khasan.1.desc
	picture = GFX_report_event_soviet_soldiers

	fire_only_once = yes

	trigger = {
		tag = JAP
		is_ai = no
		has_active_mission = SOV_border_conflict_warning_JAP
		has_country_flag = {
			flag = JAP_khasan_countdown
			days > 338
		}
		NOT = { has_war_with = SOV }
		846 = {
			is_controlled_by = JAP
			has_border_war = no
		}
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = jap_khasan.1.a
	}
}
