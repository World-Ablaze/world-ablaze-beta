#	CONVOY ARSENAL - console harness for the [convoys] chain.
#
#	Effects: common/scripted_effects/WA_TEST_convoy_arsenal.txt
#
#	CONSOLE, the short way:
#	    event wa_ca.2
#	  Fires the report on every AI major at war, each in its OWN scope. Then read logs/game.log and
#	  search for "CONVOY ARSENAL TEST".
#
#	CONSOLE, one country:
#	    tag GER      (the report runs in the scope of the country under test)
#	    event wa_ca.1
#
#	WHY ITS OWN FILE AND ITS OWN NAMESPACE (2026-08-20). These two events used to be wa_test.310 /
#	wa_test.311 in events/wa_events_test.txt. The SAME report, fired from that file, read every
#	country-valued trigger as false - `tag = GER` while playing Germany, `tag = ROOT`, `has_war_with
#	= ROOT` - while `always`, `is_major` and `num_of_*_factories` beside them read correctly. Fired
#	from a different event file, byte-identical effect, everything read correctly. error.log was
#	clean, both files were BOM-free and brace-balanced, and the id was declared once. THE CAUSE IS
#	UNKNOWN. The harness lives here because a probe whose readings depend on which file fired it is
#	worthless, not because the cause was found. Full reproduction in
#	.claude/skills/wa-lessons-learned/references/lessons-log.md, "Two call sites, one effect".
#	If these readings ever go strange again, re-fire the report from events/wa_test_scope_isolation.txt
#	(`tag GER` then `event wa_iso.3`) BEFORE editing a line of the effect.
#
#	WHAT GOOD LOOKS LIKE. Measured out of campaign 3d68a183 by the out-of-save replica
#	(.claude/skills/wa-savegame-analysis/scripts/seashare.py), 1941.6 / 1943.6 / 1945.6:
#	    GER  0.079 / 0.056 / 0.029      ITA  same        SOV  0.000 flat      -> ARSENAL 0
#	    JAP  0.360 / 0.356 / 0.357      USA  0.019* / 0.440 / 0.487           -> ARSENAL 1
#	    ENG  0.519 / 0.788 / 0.814                     (* USA not yet in the Allies at 1941.6)
#	So on a mid-war save the two groups should land far apart, land side well under the 0.25 bar and
#	maritime side well over it. THE SEPARATION IS THE TEST, not any single number: if GER and ENG come
#	out close together, the classifier is wrong and no value of coalition_sea_share_bar repairs it -
#	that is checklist R79 leg 1, and this event is how to see it without waiting for a campaign.
#
#	READING A ZERO. "ARSENAL = 0" is only the Fix-115 behaviour when `share>bar=0`; the gate line
#	prints all five terms so a 0 coming from `dockyards>80` or `partner_short` is not mistaken for it.
#	The per-member "side" block is an INDEPENDENT walk - if its ACROSS SEA rows do not add up to the
#	`unreachable` figure on the gauge line, the shipped effect and the test disagree and one of them
#	is wrong; that is a finding, not noise.
#
#	CAVEAT worth knowing before you read a surprise: land access is a capital-to-capital A* through
#	FRIENDLY territory with a 75-iteration cap, so a member behind a collapsing front, or simply very
#	far overland, reads ACROSS SEA. That is the classifier's known edge (R79 regression watch), not a
#	bug in this test.
#
#	Nothing here writes gameplay state except the one call to the shipped monthly effect, which only
#	sets its own gauge variable.

add_namespace = wa_ca

country_event = {
	id = wa_ca.1

	hidden = yes
	is_triggered_only = yes

	immediate = {
		WA_TEST_CA_report = yes
	}

	option = {}
}

country_event = {
	id = wa_ca.2

	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "########## CONVOY ARSENAL TEST - sweep of every AI major at war ##########"
		# The report prints which context it ran in, so a sweep reading can never be confused with a
		# single-country one.
		set_global_flag = WA_TEST_CA_nested_sweep
		every_country = {
			limit = {
				exists = yes
				is_ai = yes
				is_major = yes
				has_war = yes
			}
			# Fired per country ON that country, never called in a loop: the report resolves its
			# coalition against ROOT, so a loop would score everyone against the firing country.
			country_event = { id = wa_ca.1 hours = 0 }
		}
		clr_global_flag = WA_TEST_CA_nested_sweep
	}

	option = {}
}
