# Default Army HQ templates, gated on TAOG DLC.
#
# Line-battalion rule: a country that fields motorized / mechanized / armored
# divisions gets a motorized HQ, everyone else gets an infantry HQ. The
# classification lives in WA_AI_CONFIG_DIVISIONS_uses_motorized_hq - do not
# hardcode the battalion type per tag here, gate on the trigger instead.
# `division_template` cannot contain an `if`, so each country spells out both
# variants; only the line battalion differs between them.
#
# To add a country-specific HQ template, add a new `if` block below (both
# variants) and add the tag to the generic fallback's NOT list.
every_possible_country = {
	limit = { has_dlc = "Thunder at Our Gates" }

	# USA
	if = {
		limit = { tag = USA }
		if = {
			limit = { WA_AI_CONFIG_DIVISIONS_uses_motorized_hq = yes }
			division_template = {
				name = "Field Army Headquarters"
				is_army_hq = yes

				regiments = {
					hq_motorized = { x = 0 y = 0 }
					hq_motorized = { x = 0 y = 1 }
				}

				support = {
					hq_support_company = { x = 0 y = 0 }
					hq_naval_liaison = { x = 0 y = 1 }
					hq_air_liaison = { x = 0 y = 2 }
				}
			}
		}
		else = {
			division_template = {
				name = "Field Army Headquarters"
				is_army_hq = yes

				regiments = {
					hq_infantry = { x = 0 y = 0 }
					hq_infantry = { x = 0 y = 1 }
